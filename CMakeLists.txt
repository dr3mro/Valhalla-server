cmake_minimum_required(VERSION 3.30...4.2)

set(CMAKE_TOOLCHAIN_FILE "${CMAKE_BINARY_DIR}/generators/conan_toolchain.cmake")

# Set the project name and version
set(Target "server")

project(${Target})

# Use CCACHE if available
find_program(CCACHE_PROGRAM ccache)
if(CCACHE_PROGRAM)
  set(CMAKE_C_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
  set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
endif()

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# set C standard
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED True)

# Set C++ flags
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -pedantic")
set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} -g")
set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} -O2")

# set C flags
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -pedantic")
set(CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG} -g")
set(CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE} -O2")

add_compile_options(-fdiagnostics-color=always -fdiagnostics-show-option)

# Cargo-like diagnostics: carets, ranges, unicode art, template trees
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 13)
  add_compile_options(-fdiagnostics-text-art-charset=unicode
                      -fdiagnostics-show-caret -fdiagnostics-show-labels)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-fcolor-diagnostics -fdiagnostics-show-template-tree
                      -fdiagnostics-fixit-info -fcaret-diagnostics-max-lines=0)
endif()

# set(CMAKE_CXX_CLANG_TIDY clang-tidy -checks=-*,readability-*)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Copy compile_commands.json to project root so clangd/LSP can find it
add_custom_target(copy_compile_commands ALL
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        ${CMAKE_BINARY_DIR}/compile_commands.json
        ${CMAKE_SOURCE_DIR}/build/compile_commands.json
    COMMENT "Copying compile_commands.json to project build dir"
)


include_directories("${CMAKE_CURRENT_SOURCE_DIR}/src")

# Suppress developer warnings and deprecated warnings
set(CMAKE_SUPPRESS_DEVELOPER_WARNINGS
    1
    CACHE INTERNAL "No dev warnings")
set(CMAKE_WARN_DEPRECATED OFF)
set(CMAKE_COLOR_DIAGNOSTICS ON)

# Get the current Git tag using the script.
# The result is a compiler flag like -DGIT_TAG="v1.2.3"; extract just the value.
execute_process(
  COMMAND ${CMAKE_SOURCE_DIR}/get_tag.sh
  OUTPUT_VARIABLE GIT_TAG_MACRO
  OUTPUT_STRIP_TRAILING_WHITESPACE)
string(STRIP "${GIT_TAG_MACRO}" GIT_TAG_MACRO)

# Extract the bare string value from -DGIT_TAG="<value>" so we can write it
# into a single generated source file instead of a global compile definition.
# This way only version.cpp recompiles when the tag changes.
if(GIT_TAG_MACRO MATCHES "GIT_TAG=\"([^\"]+)\"")
  set(GIT_TAG_VALUE "${CMAKE_MATCH_1}")
else()
  set(GIT_TAG_VALUE "unknown")
endif()
configure_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/src/version.cpp.in"
  "${CMAKE_CURRENT_BINARY_DIR}/version.cpp"
  @ONLY)
add_compile_options(-ftemplate-backtrace-limit=0)

# Find all source files.
# cmake_configure_depends re-runs cmake automatically when the file list changes,
# so adding or removing a .cpp file is picked up without a manual reconfigure.
file(GLOB_RECURSE SRC "src/*.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src")

# Collect the main() translation unit separately so that the rest of the code
# can be linked into the test executable as well.
set(SRC_MAIN "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
list(FILTER SRC EXCLUDE REGEX "src/main\\.cpp$")

# Core library: everything except main(). Tests link against this.
# version.cpp is generated by configure_file above.
add_library(server_core STATIC ${SRC} "${CMAKE_CURRENT_BINARY_DIR}/version.cpp")

# Executable: main() + core. Output name/path unchanged (build/<type>/server).
add_executable(${Target} ${SRC_MAIN})
target_link_libraries(${Target} PRIVATE server_core)

# Define all your packages in a list
set(CONAN_PACKAGES
    cppcodec
    picojson
    fmt
    libpqxx
    jwt-cpp
    jsoncons
    Catch2
    xxHash
    libsodium
    CURL
    ZLIB
    c-ares
    spdlog
    Trantor
    Drogon)

# Find all packages at once
foreach(PACKAGE ${CONAN_PACKAGES})
  find_package(${PACKAGE} CONFIG REQUIRED COMPONENTS INCLUDE_DIRS LIBRARIES)
  list(APPEND PACKAGE_INCLUDES ${${PACKAGE}_INCLUDE_DIRS})
  list(APPEND PACKAGE_LIBS ${${PACKAGE}_LIBRARIES})
endforeach()

# Add includes (PUBLIC so the unit tests inherit them via server_core)
target_include_directories(server_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src
                                              ${PACKAGE_INCLUDES})

# API docs page (docs/api.html) embedded into the binary as a constexpr string_view.
# The header ${CMAKE_BINARY_DIR}/generated/api_html.hpp (included by src/api/v2/routes/doc.hpp) is created now so it
# exists for tooling, and refreshed on every build by the valhalla_docs target below.
include(tools/cmake/embed_html.cmake)
embed_html(
  INPUT docs/api.html
  OUTPUT ${CMAKE_BINARY_DIR}/generated/api_html.hpp
  VARIABLE kApiHtml
  NAMESPACE api::v2::docs)
target_include_directories(server_core PUBLIC ${CMAKE_BINARY_DIR}/generated)

# Before compiling the server, regenerate docs/ with tools/docs/build.sh (needs bash, npm, python3; skipped with a
# note if any is missing, in which case the committed docs/api.html is embedded). Disable: -DVALHALLA_BUILD_DOCS=OFF.
option(VALHALLA_BUILD_DOCS "Regenerate docs/api.html from tools/docs on every build" ON)
if(VALHALLA_BUILD_DOCS)
  add_custom_target(
    valhalla_docs
    COMMAND ${CMAKE_COMMAND} -DSOURCE_DIR=${CMAKE_SOURCE_DIR} -DBINARY_DIR=${CMAKE_BINARY_DIR} -P
            ${CMAKE_SOURCE_DIR}/tools/cmake/build_docs.cmake
    BYPRODUCTS ${CMAKE_BINARY_DIR}/generated/api_html.hpp
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    COMMENT "Generating API docs (tools/docs/build.sh) and embedding docs/api.html"
    VERBATIM)
  # server_core (and so the server and the tests) cannot start compiling until the docs are generated and embedded.
  add_dependencies(server_core valhalla_docs)
endif()

include(tools/conan/variables.cmake)

# Link libraries (PUBLIC: static lib deps are propagated to consumers)
target_link_libraries(server_core PUBLIC ${PACKAGE_LIBS})

# Enable AddressSanitizer only for Debug build type on Linux for now.
# Applied globally so the server binary, server_core and the unit tests all
# get instrumented (ASan/UBSan/LSan enforced on test runs on Linux Debug).
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  if(LINUX)
    message("Enabling AddressSanitizer for Debug builds on Linux")
    add_compile_options(-fsanitize=address -fsanitize=undefined -fsanitize=leak
                        -fno-sanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address -fsanitize=undefined -fsanitize=leak
                     -fno-sanitize=thread -fno-omit-frame-pointer)
  endif()
endif()

# Unit tests. ON by default; CI/Docker release builds may pass -DENABLE_TESTS=OFF.
option(ENABLE_TESTS "Build the Catch2 unit test suite and register it with CTest" ON)
if(ENABLE_TESTS)
  # enable_testing() here creates build/<type>/CTestTestfile.cmake at the root,
  # which ctest needs to descend into tests/. (tools/conan/variables.cmake sets
  # BUILD_TESTING OFF above, which would otherwise suppress it.)
  enable_testing()
  add_subdirectory(tests)
else()
  message(STATUS "Tests disabled (ENABLE_TESTS=OFF)")
endif()
