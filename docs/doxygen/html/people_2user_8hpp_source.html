1/**
2 * @file user.hpp
3 * @brief Convenience alias for the id-only payload used by get-services
4 * lookups.
5 * @details This header defines the User entity for the "users" table. It
6 * inherits all the validated Client behaviour (create/update with validation,
7 * suspend/activate, username checks) and adds one small query,
8 * getSqlGetServicesStatement, which simply selects the user's own id. Despite
9 * the method name, it does not join the organisation tables the way Provider's
10 * version does.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * Controllers construct User whenever they need to work with an ordinary user
14 * account, reusing the safe Client builders for CRUD and suspension. The added
15 * id-lookup statement provides a minimal get-services answer for users. Without
16 * this class there would be no User-specific table binding for the account
17 * endpoints.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
21 * - SQL parameter binding: put $1, $2, ... markers in the SQL and pass the real
22 * values separately; see https://en.cppreference.com/w/cpp or the libpqxx
23 * documentation.
24 * - Type alias (using Data_t = ...): a short local name for a longer qualified
25 * type; no new type is created.
26 *
27 * @author Amr Osman <dr3mro@gmail.com>
28 * @date 2026-11-16
29 */

103 /**
104 * @brief Builds a query that selects the user id from the users table.
105 * @details Reads the id from the Data_t payload into client_id and binds it
106 * as $1. If the payload has no id, calling value() throws and is caught,
107 * yielding std::nullopt.
108 * @param[out] params Cleared and filled with the bound user id.
109 * @return SELECT id FROM users WHERE id = $1, or std::nullopt on failure.
110 */
111 // Builds a minimal id-lookup statement.
112 std::optional<std::string> getSqlGetServicesStatement(pqxx::params& params) {
113 try {
114 client_id = std::get<Data_t>(getData())
115 .get_id(); // Pull the user id from the id-only
116 // payload into the cached member.
117 params = {}; // Empty the caller's container so the id becomes $1.
118 params.append(client_id.value()); // value() throws if the id is absent;
119 // the catch below handles that.
120 // A simple query whose
121 // single parameter is
122 // the bound id.
123 // Missing id or another error.
124 return std::string("SELECT id FROM users WHERE id = $1");
125 } catch (const std::exception& e) {
126 Message::ErrorMessage(
127 fmt::format("Failed to create query for GetServicesStatement {}.",
128 tablename)); // Log the table.
129 Message::CriticalMessage(e.what()); // Log details.
130 return std::nullopt; // Signal failure.
131 }
132 }
