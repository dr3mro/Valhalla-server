1/**
2 * @file common.hpp
3 * @brief Safely parse a numeric id query parameter, replying 400 on failure.
4 * @details This header holds the small pieces of code that almost every v2
5 * route reuses. First it defines the filter/middleware name macros (RATELIMIT,
6 * AUTH, ELAPSED, INSECURE, SECURE) that controllers pass to Drogon's
7 * METHOD_ADD. It also provides parseIdParam, which turns an "id" query
8 * parameter into a number without letting an exception crash the request.
9 * Finally it provides executeControllerMethod, the shared helper that
10 * calls a controller, adapts its numeric reply into an HTTP JSON response, and
11 * moves slow database work off the IO thread.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * Every REST controller in this API includes this header. It is the common
15 * plumbing between the HTTP layer (Drogon) and the business layer (the
16 * controllers/entities). The route controllers call parseIdParam and
17 * executeControllerMethod; those in turn use Helper for JSON responses,
18 * DbExecutor for background work and GateKeeper-provided attributes for
19 * identity. Without this file each of the seven controllers would have to
20 * duplicate dozens of lines of tricky error handling and threading.
21 *
22 * CONCEPTS TO LEARN (for beginners):
23 * - HTTP filters/middleware: https://drogon.docsforge.com/master/filter/
24 * - Query parameters: values appended to a URL after "?" (e.g. ?id=42).
25 * - C++ templates: https://en.cppreference.com/w/cpp/language/templates
26 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
27 *
28 * @author Amr Osman <dr3mro@gmail.com>
29 * @date 2026-11-16
30 */

97 /**
98 * @brief Safely parse a numeric id query parameter, replying 400 on
99 * failure.
100 * @details Calling std::stoll directly on req->getParameter(name) would
101 * throw std::invalid_argument or std::out_of_range while building the
102 * handler's argument list, i.e. before executeControllerMethod's try/catch
103 * is entered; the exception then escaped Drogon's dispatch as a raw,
104 * empty-body 500. This helper contains the throw and emits a normal JSON
105 * 400 instead. On failure the callback has already been invoked with the
106 * error response.
107 * @param[in] req Request whose query parameter is read.
108 * @param[in] name Parameter name to parse as an integer id.
109 * @param[in,out] callback Response callback; invoked with a 400 when
110 * parsing fails.
111 * @return The parsed id, or std::nullopt on failure (callback already
112 * answered).
113 * @post If std::nullopt is returned the caller must return early without
114 * calling executeControllerMethod, otherwise a second response would be
115 * sent.
116 */
117 // "inline" lets this live in a header without
118 // duplicate-definition errors.
119 // req supplies the parameter; callback is answered on
120 // error.
121 inline std::optional<int64_t> parseIdParam(
122 const drogon::HttpRequestPtr& req,
123 const std::string& name,
124 std::function<void(const drogon::HttpResponsePtr&)>& callback) {
125 const std::string& raw =
126 req->getParameter(name); // Read the raw text value (empty string
127 // when the parameter is missing).
128 // Guard the conversion because stoll throws on bad input.
129 try {
130 // Convert text like "42" into a 64-bit integer;
131 // throws if it is not a valid number.
132 // Catches invalid_argument (not a
133 // number) and out_of_range (too big).
134 return std::stoll(raw);
135 } catch (const std::exception&) {
136 Helper::errorResponse(drogon::k400BadRequest,
137 fmt::format("Missing or invalid '{}' parameter.",
138 name),
139 std::move(callback)); // Send a friendly JSON 400
140 // and consume the callback.
141 return std::nullopt; // Signal failure so the caller returns early
142 // (the response is already sent).
143 }
144 }

146 /**
147 * @brief Dispatch a controller method for a keyed entity type, off the IO
148 * thread when needed.
149 * @details Looks up @p key in @p registry (an unordered_map of std::variant
150 * controller pointers); a missing key is answered with 400 and no
151 * controller is invoked. A model callback (int status + content) is adapted
152 * to HTTP responses: OK -> success, INTERNAL_SERVER_ERROR -> failure (500
153 * body), anything else -> error with that numeric status. Requester
154 * identity is read from the "clientID"/"clientGroup" attributes that the
155 * Auth filter inserts on SECURE routes (INSECURE routes such as client
156 * create/login also use this helper). When executor.ioFastPath() is enabled
157 * the call is first tried on the IO thread inside a DbProbe: work that
158 * needs no connection completes there, whereas the first WouldBlock is
159 * caught and the whole call is re-queued on the DB executor from the start.
160 * Request, arguments and callback are captured by value because queued work
161 * finishes after this function returns on a worker thread. A full executor
162 * queue is answered 503 without running the controller;
163 * DatabaseBusy becomes 503, and any other std::exception becomes a 500
164 * failure response.
165 * @tparam Func Member-function pointer type (e.g. &ControllerBase::Create).
166 * @tparam Registry Map type from entity-type key to a std::variant of
167 * controller pointers.
168 * @tparam Args Types of the extra arguments forwarded to the controller
169 * method.
170 * @param[in] executor Pool that runs blocking permission checks and
171 * queries.
172 * @param[in] registry Entity-type -> controller variant map.
173 * @param[in] key Registry key selected from the request path.
174 * @param[in] method Controller member function to invoke.
175 * @param[in] req Request supplying attributes and body; captured by value
176 * for async work.
177 * @param[in,out] callback Response callback; consumed exactly once on every
178 * path.
179 * @param[in] args Extra arguments forwarded to the controller (e.g. body,
180 * parsed id).
181 */
182 // Generic over the method, the map and any extra
183 // arguments.
184 // "static" at namespace scope
185 // gives this helper internal
186 // linkage.
187 template <typename Func, typename Registry, typename... Args>
188 static void executeControllerMethod(
189 DbExecutor& executor,
190 const Registry& registry,
191 const std::string_view key,
192 Func method,
193 const drogon::HttpRequestPtr& req,
194 std::function<void(const drogon::HttpResponsePtr&)>&& callback,
195 Args&&... args) {
196 auto ctl = registry.find(key); // Look up the controller variant for
197 // this URL's type segment.
198 // An unknown type key cannot be dispatched.
199 if (ctl == registry.end()) {
200 Helper::errorResponse(drogon::k400BadRequest,
201 fmt::format("Type mapping not found for: {}", key),
202 std::move(
203 callback)); // Tell the client the type is
204 // unknown (400) and finish.
205 return; // Nothing else to do; the callback was already answered.
206 }
207
208 // Whole dispatch is guarded so no exception escapes to Drogon as
209 // an empty 500.
210 try {
211 // std::variant holds one of several controller pointer
212 // types; visit picks the right one.
213 // This lambda is called with whichever
214 // controller type is stored.
215 std::visit(
216 [&](const auto& controller) {
217 // Adapts the controller's numeric status callback to an
218 // HTTP response.
219 // The controller replies with an
220 // int code and a JSON string.
221 CALLBACK_ mcb = [callback](int code, const std::string& content) {
222 using Reply = std::function<void(
223 const drogon::HttpResponsePtr&)>; // Short alias for the
224 // callback type we must
225 // pass to Helper.
226 // Translate the numeric result into the
227 // matching JSON HTTP reply.
228 switch (code) {
229 // Controller succeeded.
230 case api::v2::Http::Status::OK:
231 Helper::successResponse(content,
232 Reply(callback)); // Send 200 with
233 // the body as-is.
234 break; // Leave the switch.
235 case api::v2::Http::Status::
236 INTERNAL_SERVER_ERROR: // Controller hit an
237 // internal error.
238 Helper::failureResponse(content,
239 Reply(
240 callback)); // Send a tagged 500
241 // failure body.
242 break; // Leave the switch.
243 default: // Any other code is a specific error
244 // (e.g. 400, 403, 503).
245 Helper::errorResponse(static_cast<drogon::HttpStatusCode>(
246 code),
247 content,
248 Reply(
249 callback)); // Reuse that number as
250 // the HTTP status.
251 break; // Leave the switch.
252 }
253 };
254
255 // Read what the handler needs from the request now; the
256 // rest runs on a worker.
257 Requester
258 requester(req->getAttributes()->get<uint64_t>("clientID"),
259 req->getAttributes()->get<std::string>(
260 "clientGroup")); // Auth filter stored these;
261 // build the identity used for
262 // permission checks.
263
264 auto packed = std::make_tuple(std::decay_t<Args>(
265 args)...); // Bundle the extra arguments so we can
266 // store/copy them for async work.
267
268 // Fast path: run the call here first. A request that is
269 // answered from memory (bad body, refused by the permission
270 // cache) never queues; the first query throws WouldBlock
271 // before anything was written, and the request goes to the
272 // executor and runs again from the start.
273 // Is the cheap in-thread path
274 // enabled?
275 if (executor.ioFastPath()) {
276 // The probe call may throw; handle each case
277 // below.
278 try {
279 DbProbe probe; // Guard object that throws WouldBlock
280 // if a real DB connection is needed.
281 CALLBACK_ probe_cb =
282 mcb; // its own copy: the real one must still
283 // be there if the probe gives up
284 std::apply(
285 [&](auto&... unpacked) {
286 std::invoke(method,
287 controller.get(),
288 std::move(probe_cb),
289 std::move(requester),
290 unpacked...);
291 },
292 packed); // Try the controller right here on
293 // the IO thread.
294 return; // It completed without needing the
295 // database, so we are done.
296 } catch (const WouldBlock&) // The probe says a
297 // connection is required.
298 // needs the database: queue it below
299 // Some other error happened on
300 // the fast path.
301 {
302 } catch (const std::exception& e) {
303 Helper::failureResponse(e.what(),
304 std::move(
305 callback)); // Reply 500 with the
306 // exception text.
307 return; // Done; do not also queue the call.
308 }
309 }
310
311 // Queue the call on the DB workers when the fast path was
312 // disabled or gave up; runOnDbExecutor answers 503 itself
313 // if the bounded queue is full.
314 // Hand the response adapter to the
315 // executor wrapper.
316 // Capture everything by
317 // value: this runs later
318 // on another thread.
319 runOnDbExecutor(executor,
320 std::move(mcb),
321 [method,
322 target = controller.get(),
323 req,
324 callback,
325 requester = std::move(requester),
326 packed =
327 std::move(packed)](CALLBACK_&& cb) mutable {
328 // The queued controller call can also fail.
329 try {
330 std::apply(
331 [&](auto&... unpacked) {
332 // Run the real controller method
333 // now that DB access is allowed.
334 // The connection pool is exhausted.
335 std::invoke(method,
336 target,
337 std::move(cb),
338 std::move(requester),
339 unpacked...);
340 },
341 packed);
342 } catch (const DatabaseBusy& e) {
343 // Exhausted connection pool -> retryable 503
344 // (Helper adds Retry-After).
345 // Tell the client to
346 // try again shortly.
347 // Any other failure.
348 Helper::
349 errorResponse(drogon::
350 k503ServiceUnavailable,
351 e.what(),
352 std::move(callback));
353 } catch (const std::exception& e) {
354 // Reply 500 with
355 // the error text.
356 // Pass the selected controller variant to
357 // std::visit.
358 // Backstop for errors while setting up the
359 // visit.
360 Helper::failureResponse(e.what(),
361 std::move(callback));
362 }
363 });
364 },
365 ctl->second);
366 } catch (const std::exception& e) {
367 Helper::failureResponse(e.what(),
368 std::move(
369 callback)); // Always answer exactly once.
370 }
371 }
