1/**
2 * @file common.hpp
3 * @brief Safely parse a numeric id query parameter, replying 400 on failure.
4 * @details This header holds the small pieces of code that almost every v2 route reuses. First it
5 * defines the filter/middleware name macros (RATELIMIT, AUTH, ELAPSED, INSECURE, SECURE)
6 * that controllers pass to Drogon's METHOD_ADD. It also provides parseIdParam, which turns
7 * an "id" query parameter into a number without letting an exception crash the request.
8 * Finally it provides executeControllerMethod, the shared helper that calls a controller,
9 * adapts its numeric reply into an HTTP JSON response, and moves slow database work off the
10 * IO thread.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * Every REST controller in this API includes this header. It is the common plumbing between the
14 * HTTP layer (Drogon) and the business layer (the controllers/entities). The route controllers call
15 * parseIdParam and executeControllerMethod; those in turn use Helper for JSON responses, DbExecutor
16 * for background work and GateKeeper-provided attributes for identity. Without this file each of the
17 * seven controllers would have to duplicate dozens of lines of tricky error handling and threading.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - HTTP filters/middleware: https://drogon.docsforge.com/master/filter/
21 * - Query parameters: values appended to a URL after "?" (e.g. ?id=42).
22 * - C++ templates: https://en.cppreference.com/w/cpp/language/templates
23 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

72 /**
73 * @brief Safely parse a numeric id query parameter, replying 400 on failure.
74 * @details Calling std::stoll directly on req->getParameter(name) would throw
75 * std::invalid_argument or std::out_of_range while building the handler's argument
76 * list, i.e. before executeControllerMethod's try/catch is entered; the exception
77 * then escaped Drogon's dispatch as a raw, empty-body 500. This helper contains the
78 * throw and emits a normal JSON 400 instead. On failure the callback has already been
79 * invoked with the error response.
80 * @param[in] req Request whose query parameter is read.
81 * @param[in] name Parameter name to parse as an integer id.
82 * @param[in,out] callback Response callback; invoked with a 400 when parsing fails.
83 * @return The parsed id, or std::nullopt on failure (callback already answered).
84 * @post If std::nullopt is returned the caller must return early without calling
85 * executeControllerMethod, otherwise a second response would be sent.
86 */
87 inline std::optional<int64_t> parseIdParam( // "inline" lets this live in a header without duplicate-definition errors.
88 const drogon::HttpRequestPtr& req, const std::string& name, std::function<void(const drogon::HttpResponsePtr&)>& callback) // req supplies the parameter; callback is answered on error.
89 {
90 const std::string& raw = req->getParameter(name); // Read the raw text value (empty string when the parameter is missing).
91 try // Guard the conversion because stoll throws on bad input.
92 {
93 return std::stoll(raw); // Convert text like "42" into a 64-bit integer; throws if it is not a valid number.
94 }
95 catch (const std::exception&) // Catches invalid_argument (not a number) and out_of_range (too big).
96 {
97 Helper::errorResponse(drogon::k400BadRequest, fmt::format("Missing or invalid '{}' parameter.", name), std::move(callback)); // Send a friendly JSON 400 and consume the callback.
98 return std::nullopt; // Signal failure so the caller returns early (the response is already sent).
99 }
100 }

102 /**
103 * @brief Dispatch a controller method for a keyed entity type, off the IO thread when needed.
104 * @details Looks up @p key in @p registry (an unordered_map of std::variant controller pointers);
105 * a missing key is answered with 400 and no controller is invoked. A model callback (int
106 * status + content) is adapted to HTTP responses: OK -> success, INTERNAL_SERVER_ERROR ->
107 * failure (500 body), anything else -> error with that numeric status. Requester identity
108 * is read from the "clientID"/"clientGroup" attributes that the Auth filter inserts on
109 * SECURE routes (INSECURE routes such as client create/login also use this helper).
110 * When executor.ioFastPath() is enabled the call is
111 * first tried on the IO thread inside a DbProbe: work that needs no connection completes
112 * there, whereas the first WouldBlock is caught and the whole call is re-queued on the DB
113 * executor from the start. Request, arguments and callback are captured by value because
114 * queued work finishes after this function returns on a worker thread. A full executor
115 * queue is answered 503 without running the controller; DatabaseBusy becomes 503, and any
116 * other std::exception becomes a 500 failure response.
117 * @tparam Func Member-function pointer type (e.g. &ControllerBase::Create).
118 * @tparam Registry Map type from entity-type key to a std::variant of controller pointers.
119 * @tparam Args Types of the extra arguments forwarded to the controller method.
120 * @param[in] executor Pool that runs blocking permission checks and queries.
121 * @param[in] registry Entity-type -> controller variant map.
122 * @param[in] key Registry key selected from the request path.
123 * @param[in] method Controller member function to invoke.
124 * @param[in] req Request supplying attributes and body; captured by value for async work.
125 * @param[in,out] callback Response callback; consumed exactly once on every path.
126 * @param[in] args Extra arguments forwarded to the controller (e.g. body, parsed id).
127 */
128 template <typename Func, typename Registry, typename... Args> // Generic over the method, the map and any extra arguments.
129 static void executeControllerMethod(DbExecutor& executor, const Registry& registry, const std::string_view key, Func method, const drogon::HttpRequestPtr& req,
130 std::function<void(const drogon::HttpResponsePtr&)>&& callback, Args&&... args) // "static" at namespace scope gives this helper internal linkage.
131 {
132 auto ctl = registry.find(key); // Look up the controller variant for this URL's type segment.
133 if (ctl == registry.end()) // An unknown type key cannot be dispatched.
134 {
135 Helper::errorResponse(drogon::k400BadRequest, fmt::format("Type mapping not found for: {}", key), std::move(callback)); // Tell the client the type is unknown (400) and finish.
136 return; // Nothing else to do; the callback was already answered.
137 }
138
139 try // Whole dispatch is guarded so no exception escapes to Drogon as an empty 500.
140 {
141 std::visit( // std::variant holds one of several controller pointer types; visit picks the right one.
142 [&](const auto& controller) // This lambda is called with whichever controller type is stored.
143 {
144 // Adapts the controller's numeric status callback to an HTTP response.
145 CALLBACK_ mcb = [callback](int code, const std::string& content) // The controller replies with an int code and a JSON string.
146 {
147 using Reply = std::function<void(const drogon::HttpResponsePtr&)>; // Short alias for the callback type we must pass to Helper.
148 switch (code) // Translate the numeric result into the matching JSON HTTP reply.
149 {
150 case api::v2::Http::Status::OK: // Controller succeeded.
151 Helper::successResponse(content, Reply(callback)); // Send 200 with the body as-is.
152 break; // Leave the switch.
153 case api::v2::Http::Status::INTERNAL_SERVER_ERROR: // Controller hit an internal error.
154 Helper::failureResponse(content, Reply(callback)); // Send a tagged 500 failure body.
155 break; // Leave the switch.
156 default: // Any other code is a specific error (e.g. 400, 403, 503).
157 Helper::errorResponse(static_cast<drogon::HttpStatusCode>(code), content, Reply(callback)); // Reuse that number as the HTTP status.
158 break; // Leave the switch.
159 }
160 };
161
162 // Read what the handler needs from the request now; the rest runs on a worker.
163 Requester requester(req->getAttributes()->get<uint64_t>("clientID"), req->getAttributes()->get<std::string>("clientGroup")); // Auth filter stored these; build the identity used for permission checks.
164
165 auto packed = std::make_tuple(std::decay_t<Args>(args)...); // Bundle the extra arguments so we can store/copy them for async work.
166
167 // Fast path: run the call here first. A request that is answered from memory (bad body, refused by the
168 // permission cache) never queues; the first query throws WouldBlock before anything was written, and
169 // the request goes to the executor and runs again from the start.
170 if (executor.ioFastPath()) // Is the cheap in-thread path enabled?
171 {
172 try // The probe call may throw; handle each case below.
173 {
174 DbProbe probe; // Guard object that throws WouldBlock if a real DB connection is needed.
175 CALLBACK_ probe_cb = mcb; // its own copy: the real one must still be there if the probe gives up
176 std::apply([&](auto&... unpacked) { std::invoke(method, controller.get(), std::move(probe_cb), std::move(requester), unpacked...); }, packed); // Try the controller right here on the IO thread.
177 return; // It completed without needing the database, so we are done.
178 }
179 catch (const WouldBlock&) // The probe says a connection is required.
180 {
181 // needs the database: queue it below
182 }
183 catch (const std::exception& e) // Some other error happened on the fast path.
184 {
185 Helper::failureResponse(e.what(), std::move(callback)); // Reply 500 with the exception text.
186 return; // Done; do not also queue the call.
187 }
188 }
189
190 // Queue the call on the DB workers when the fast path was disabled or gave up;
191 // runOnDbExecutor answers 503 itself if the bounded queue is full.
192 runOnDbExecutor(executor, std::move(mcb), // Hand the response adapter to the executor wrapper.
193 [method, target = controller.get(), req, callback, requester = std::move(requester), packed = std::move(packed)](CALLBACK_&& cb) mutable // Capture everything by value: this runs later on another thread.
194 {
195 try // The queued controller call can also fail.
196 {
197 std::apply([&](auto&... unpacked) { std::invoke(method, target, std::move(cb), std::move(requester), unpacked...); }, packed); // Run the real controller method now that DB access is allowed.
198 }
199 catch (const DatabaseBusy& e) // The connection pool is exhausted.
200 {
201 // Exhausted connection pool -> retryable 503 (Helper adds Retry-After).
202 Helper::errorResponse(drogon::k503ServiceUnavailable, e.what(), std::move(callback)); // Tell the client to try again shortly.
203 }
204 catch (const std::exception& e) // Any other failure.
205 {
206 Helper::failureResponse(e.what(), std::move(callback)); // Reply 500 with the error text.
207 }
208 });
209 },
210 ctl->second); // Pass the selected controller variant to std::visit.
211 }
212 catch (const std::exception& e) // Backstop for errors while setting up the visit.
213 {
214 Helper::failureResponse(e.what(), std::move(callback)); // Always answer exactly once.
215 }
216 }
