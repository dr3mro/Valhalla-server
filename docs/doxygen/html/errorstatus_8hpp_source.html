1/**
2 * @file errorstatus.hpp
3 * @brief Maps caught exceptions to the HTTP status reported to API clients.
4 * @details Provides one small helper, statusForException, that looks at an
5 * exception which was caught and decides which HTTP status code the
6 * client should receive. A "database busy" problem maps to 503 (try
7 * again later); every other problem maps to 500 (generic server
8 * error). The function is inline and header-only, so there is no separate .cpp
9 * file and no linking needed.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * When a controller fails while handling a request it must still send an HTTP
13 * answer. The CRITICALMESSAGERESPONSE macro (in global.hpp) calls this helper
14 * to choose that answer's status. Doing the choice in one place keeps every
15 * error response consistent: clients always get 503 specifically when the
16 * database was too busy, and 500 for anything else. Without it, each catch
17 * block would have to repeat the classification logic.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - Exceptions in C++: https://en.cppreference.com/w/cpp/error/exception
21 * - dynamic_cast: https://en.cppreference.com/w/cpp/language/dynamic_cast
22 * - inline functions: https://en.cppreference.com/w/cpp/language/inline
23 *
24 * @author Amr Osman <dr3mro@gmail.com>
25 * @date 2026-11-16
26 */

41 // 503 when no database connection was available (the client should retry),
42 // 500 for any other failure.
43 /**
44 * @brief Selects the HTTP status corresponding to a caught exception.
45 * @details A DatabaseBusy means no database connection was available, so
46 * the client is told 503 and may retry; all other failures are 500. Used by
47 * the CRITICALMESSAGERESPONSE macro.
48 * @param[in] e The caught exception to classify.
49 * @return Status::SERVICE_UNAVAILABLE for DatabaseBusy, otherwise
50 * Status::INTERNAL_SERVER_ERROR.
51 * @note Classification uses dynamic_cast, so it relies on the exception
52 * being polymorphic (as all std::exception-derived types are).
53 */
54 // Takes the exception by const reference: no
55 // copy, and it is only inspected.
56 inline Status statusForException(const std::exception& e) {
57 // dynamic_cast checks at runtime whether e is really a DatabaseBusy; if
58 // so, "retry later" (503), else generic 500.
59 return dynamic_cast<const DatabaseBusy*>(&e) != nullptr
60 ? Status::SERVICE_UNAVAILABLE
61 : Status::INTERNAL_SERVER_ERROR; // 503 if it is a
62 // DatabaseBusy, otherwise
63 // 500, so the caller can
64 // choose the right reply.
65 }
66} // namespace api::v2::Http
