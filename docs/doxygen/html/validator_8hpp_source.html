1/**
2 * @file validator.hpp
3 * @brief Stateless collection of checks applied to client JSON before it reaches the database.
4 * @details This header declares the Validator class: a toolbox of static functions that check
5 * JSON sent by a client against the shape of a database table. It also declares the nested
6 * Rule type, which lets a caller describe exceptions to the usual checks. None of these
7 * functions talks to the database directly; the checks run on an in-memory copy of the
8 * schema, so they are cheap enough to run on every incoming request.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This is the gatekeeper between the HTTP layer and the database. Entity controllers (for
12 * clients, staff, appointments and so on) call these helpers with the parsed request body
13 * before they build any SQL. The helpers read the cached schema owned by DatabaseSchema.
14 * Without this file, badly shaped or mistyped client data would flow straight into SQL
15 * queries and cause errors or security problems, so every create, update and read endpoint
16 * depends on it.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - Bit flags / bitmasks: https://www.learncpp.com/cpp-tutorial/bitwise-operators/
20 * - Regular expressions (regex): https://en.cppreference.com/w/cpp/regex
21 * - JSON schema validation: https://json-schema.org/
22 *
23 * @author Amr Osman <dr3mro@gmail.com>
24 * @date 2026-11-16
25 */

39/**
40 * @class Validator
41 * @brief Stateless collection of checks applied to client JSON before it reaches the database.
42 *
43 * @details Entity controllers call these helpers with the parsed request body, the target
44 * table name and a Rule describing per-key exceptions. Validation is performed against a
45 * cached snapshot of the database schema (DatabaseSchema), not against a live query, so the
46 * helpers are cheap enough for the request path. Every entry point is static and reports
47 * failures by filling an api::v2::Http::Error (code + message) and returning false rather
48 * than throwing; exceptions from the JSON layer are caught and converted to 400 responses.
49 *
50 * @note All members are static; Validator carries no state and needs no lifetime management.
51 * @warning The cached schema must have been populated before these helpers are used,
52 * otherwise table lookups fail and the validators return false.
53 * @see Rule, DatabaseSchema, api::v2::Http::Error
54 */
55class Validator
56{

71 /**
72 * @struct Rule
73 * @brief A set of validation exceptions keyed by the actions they relax or assert.
74 *
75 * @details Holds one or more bitmask Action flags and, for each flag group, the set of
76 * column names it applies to. A check passes only when the relevant flag is present in
77 * @ref action and a matching flag group contains the key being checked. Rules are
78 * typically static const objects in controller code, composed with the bitwise operators.
79 * @note Copying/moving is implicit; the key sets own their strings.
80 */
81 struct Rule
82 {
83 /**
84 * @brief Bitmask of validation relaxations/assertions that make up a rule.
85 * @details Values are powers of two and may be combined with operator|.
86 */
87 enum Action : std::uint8_t
88 {
89 NONE = 0, // No flags set: the "do nothing" value used when testing flags.
90 IGNORE_IF_MISSING_FROM_SCHEMA = 1U << 0U, // Ignore if key is missing from database schema
91 IGNORE_IF_NOT_NULLABLE_IN_SCHEMA = 1U << 1U, // Ignore if key is not nullable in database schema
92 ASSERT_IMMUTABLE = 1U << 2U, // Assert that key is immutable
93 ASSERT_NOT_PRESENT = 1U << 3U, // Assert that key is not present in json
94 };
95
96 /** @brief Combined action flags that this rule activates. */
97 Action action; // NOLINT - the on/off flags, kept public so callers can compose rules.
98 /** @brief Maps an action flag (or group of flags) to the keys it applies to. */
99 std::unordered_map<Action, std::unordered_set<std::string>> keys; // NOLINT - each flag group owns the keys it relaxes.
100
101 /**
102 * @brief Tests whether a given action applies to a given key.
103 * @param[in] check Action flag being tested.
104 * @param[in] key Column/key name being validated.
105 * @return True only when @p check is set in this rule and @p key belongs to a flag
106 * group that intersects @p check.
107 */
108 bool check(const Action& check, const std::string& key) const // NOLINT
109 {
110 bool isMatch = hasMatch(check); // True when this rule's flags include the flag we are testing.
111 bool isKey = hasKey(check, key); // True when the key sits in a matching flag group.
112 return isMatch && isKey; // Both must hold, so return their logical AND to the caller.
113 }
114
115 // constructor for rule
116 /**
117 * @brief Constructs a rule combining @p _action with the given key set.
118 * @param[in] _action Action flags (usually a bitwise OR of several).
119 * @param[in] _keys Keys the flags apply to; taken by value into @ref keys.
120 * @pre At least one Action flag should be set, otherwise the rule can never match.
121 */
122 explicit Rule(Action _action, const std::unordered_set<std::string>&& _keys) : action(_action) { this->keys[action] = _keys; }
123
124 /** @brief Rules require an explicit action and key set. */
125 Rule() = delete;
126
127 // Overload | operator
128 /** @brief Combines two action flags into one bitmask. */
129 friend Action operator|(Action first, Action second)
130 {
131 return static_cast<Action>(static_cast<std::uint8_t>(first) | static_cast<std::uint8_t>(second)); // OR the raw bytes, then treat the result as an Action again.
132 }
133
134 // Overload & operator
135 /** @brief Returns the intersection of two action bitmasks. */
136 friend Action operator&(Action first, Action second)
137 {
138 return static_cast<Action>(static_cast<std::uint8_t>(first) & static_cast<std::uint8_t>(second)); // AND the raw bytes to keep only shared flags.
139 }
140
141 // Overload |= operator
142 /** @brief Adds @p second's flags to @p first in place. */
143 friend Action& operator|=(Action& first, Action second)
144 {
145 first = first | second; // Reuse operator| and store the combined flags back into first.
146 return first; // Return the updated object so calls can be chained.
147 }
148
149 // Overload &= operator
150 /** @brief Keeps only the flags common to @p first and @p second. */
151 friend Action& operator&=(Action& first, Action second)
152 {
153 first = first & second; // Reuse operator& and store the narrowed flags back into first.
154 return first; // Return the updated object so calls can be chained.
155 }
156
157 // Invert operator
158 /** @brief Inverts all eight bits; include NONE filtering at the call site. */
159 friend Action operator~(Action first) { return static_cast<Action>(~static_cast<std::uint8_t>(first)); }
160
161 // Check if any flag is set
162 /** @brief Convenience conversion: true when at least one action flag is set. */
163 explicit operator bool() const { return static_cast<int>(action) != 0; }
164
165 private:
166 // check if rule is set
167 /** @brief True when the rule's combined action intersects @p check. */
168 bool hasMatch(const Action& check) const { return (action & check) != NONE; } // NOLINT
169 /** @brief True when a stored flag group intersects @p check. */
170 bool hasMatch(const Action& key, const Action& check) const { return (key & check) != NONE; } // NOLINT
171
172 // check if key exists in rule for and action
173 /** @brief True when @p element appears in a key set whose flags intersect @p check. */
174 bool hasKey(const Action& check, const std::string& element) const // NOLINT
175 {
176 return std::ranges::find_if(this->keys, [&](const auto& key) { return (hasMatch(key.first, check)) && (key.second.contains(element)); }) !=
177 this->keys.end(); // Found a matching group means the key is allowed; reaching end() means it is not.
178 }
179 };

181 /**

187 * @param[in] rule Per-key exceptions (e.g. ignore a server-generated "id").
188 * @return True when the body satisfies all schema constraints.
189 * @details Runs null/empty check, schema lookup, per-column type and nullability checks

196 /**

198 * @param[in] tablename Target table used for the schema lookup.

219 /**
220 * @brief Applies field-specific regex rules to client-supplied entity data.
221 * @param[in] data JSON object whose fields are validated by name (username, password,
222 * phone, email, dob, gender). Fields without a rule are passed through if they are
223 * non-empty strings.
224 * @param[out] error Filled with 400 and a descriptive message on the first bad field.
