2 * @file asyncsqlbinder.cpp
3 * @brief Implementation for asyncsqlbinder: Copies each parameter's bytes into
4 * a string, or records a null when pqxx reports no value.
5 * @details This file implements the three bridge functions declared in
6 * asyncsqlbinder.hpp. toTextParams() copies parameters into optional strings,
7 * bindParams() feeds them into a Drogon SQL binder one at a time, and
8 * execSqlCoro() creates that binder and returns an awaiter. All three
9 * work from the C-style parameter view that pqxx exposes, which reports each
10 * value's bytes, length and format.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * This is the actual data-copy layer of every asynchronous database query. The
14 * entity builders still hand over pqxx::params, and this file is where those
15 * values are transformed into Drogon bind calls. Copying by the reported length
16 * keeps embedded NUL bytes intact, and putting binary data in a shared buffer
17 * keeps it alive until the query finishes rather than only until this function
18 * returns. Without these details, async queries would corrupt binary values,
19 * mishandle nulls, or read memory that has already gone away.
20 *
21 * CONCEPTS TO LEARN (for beginners):
22 * - Drogon SQL binding: https://drogon.docsforge.com/master/database/
23 * - C++20 coroutines and co_await:
24 * https://en.cppreference.com/w/cpp/language/co_await
25 * - std::shared_ptr (shared ownership of a buffer):
26 * https://en.cppreference.com/w/cpp/memory/shared_ptr
27 * - std::source_location (where a call came from):
28 * https://en.cppreference.com/w/cpp/utility/source_location
29 * - libpqxx parameters (pqxx::params): https://pqxx.org/
30 *
31 * @author Amr Osman <dr3mro@gmail.com>
32 * @date 2026-11-16
33 */

54 /**
55 * @brief Copies each parameter's bytes into a string, or records a null
56 * when pqxx reports no value.
57 * @details The source location is passed to pqxx for conversion
58 * diagnostics. Copying uses the reported length, so embedded NUL bytes
59 * survive the round trip.
60 */
61 std::vector<std::optional<std::string>> toTextParams(
62 const pqxx::params& params) {
63 std::vector<std::optional<std::string>>
64 result; // Start empty; `optional` lets us store "no value" for a
65 // SQL NULL.
66 const auto c = params.make_c_params(
67 std::source_location::current()); // Ask pqxx for its low-level
68 // view of the values; the source
69 // location helps pqxx
70 // report where a conversion failed.
71
72 result.reserve(
73 c.values.size()); // Allocate space once up front, so adding
74 // entries cannot reallocate the vector midway.
75 for (std::size_t i = 0; i < c.values.size();
76 // Walk the values in append order; stop when every parameter
77 // has been copied.
78 ++i) {
79 // A null pointer in the C view is pqxx's way
80 // of saying "this parameter is SQL NULL".
81 if (c.values[i] == nullptr) {
82 result.emplace_back(std::nullopt); // Record an empty optional rather
83 // than an empty string, so NULL
84 // stays distinguishable from "".
85 } else {
86 result.emplace_back(
87 std::string(c.values[i],
88 static_cast<std::size_t>(
89 c.lengths[i]))); // Copy exactly the reported
90 // number of bytes; the cast makes
91 // the length
92 // unsigned, so embedded zero bytes are preserved.
93 }
94 }
95 return result; // The caller gets one entry per parameter, in the
96 // original order.
97 }

99 /**
100 * @brief Emits one bind operation per parameter: NULL, binary (`format ==
101 * 1`) or text.
102 * @details Binary values are copied into a shared, heap-owned buffer and
103 * passed as a RawParameter, so the bytes remain valid for the lifetime of
104 * the binder rather than for this stack frame. Text values are copied into
105 * a string.
106 */
107 void bindParams(drogon::orm::internal::SqlBinder& binder,
108 const pqxx::params& params) {
109 const auto c = params.make_c_params(
110 std::source_location::current()); // Get the same low-level view as
111 // toTextParams(), again with the
112 // call site for diagnostics.
113
114 for (std::size_t i = 0; i < c.values.size();
115 // Bind parameters one by one, in the order the builder
116 // appended them.
117 ++i) {
118 // NULL values need their own bind form; binding an
119 // empty string would be a different value.
120 if (c.values[i] == nullptr) {
121 // Drogon's operator<< recognises nullptr and
122 // sends an SQL NULL for this placeholder.
123 // In the PostgreSQL protocol, format 1
124 // means binary data rather than text.
125 binder << nullptr;
126 } else if (c.formats[i] == 1) {
127 auto buffer = std::make_shared<std::vector<
128 char>>(c.values[i],
129 c.values[i] +
130 c.lengths[i]); // Copy the raw bytes into a heap buffer
131 // shared through a shared_ptr, so they
132 // outlive this function's stack frame.
133 binder << drogon::orm::
134 RawParameter{buffer,
135 buffer->data(),
136 static_cast<int>(buffer->size()),
137 1}; // Pass pointer, length and binary format; the
138 // shared_ptr travels with the parameter, and
139 // the cast suits Drogon's int-based API.
140 } else {
141 binder << std::string(c.values[i],
142 static_cast<std::size_t>(
143 c.lengths[i])); // Ordinary text is copied
144 // into a string and bound by
145 // value; Drogon keeps that
146 // copy until the query runs.
147 }
148 }
149 }

151 /**
152 * @brief Builds and binds the statement, then transfers the binder into an
153 * awaiter.
154 * @details The binder is moved, not copied; no query is sent until the
155 * caller co_awaits the returned object.
156 */
157 drogon::orm::internal::SqlAwaiter execSqlCoro(drogon::orm::DbClient& client,
158 const std::string& sql,
159 const pqxx::params& params) {
160 auto binder = client << sql; // Drogon's operator<< on a client creates a
161 // binder already holding the statement text.
162 bindParams(binder,
163 params); // Attach the values to that binder in placeholder
164 // order; this performs no network I/O.
165 return drogon::orm::internal::SqlAwaiter(
166 std::move(binder)); // Move ownership into an awaiter: nothing
167 // executes until the caller co_awaits it, and
168 // the bound data stays alive until then.
169 }
