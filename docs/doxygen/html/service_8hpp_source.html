1/**
2 * @file service.hpp
3 * @brief Convenience alias for the staff-invitation payload consumed by Service builders.
4 * @details This header defines the Service entity for organisation tables such as clinics,
5 * pharmacies, laboratories and radiology centres. On top of the inherited CRUD statements it adds
6 * two builders that edit the JSONB "staff" array of a service row to add or remove a team member,
7 * and a static query that PermissionManager uses to read a service's owner, admin and staff.
8 *
9 * WHY THIS FILE MATTERS IN THE SERVER:
10 * Services (organisations) need to manage teams, and permission checks need to know who owns or
11 * administers a service. Controllers use Service to build the jsonb_set UPDATEs that change the
12 * staff array and the SELECT that feeds PermissionManager. Without it there would be no safe way
13 * to add/remove staff or to authorise service-level operations.
14 *
15 * CONCEPTS TO LEARN (for beginners):
16 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
17 * - JSONB: PostgreSQL's binary JSON column type; jsonb_set changes a value at a path inside it and
18 * the -> operator walks into a JSON object.
19 * - SQL parameter binding: put $1, $2, ... markers in the SQL and pass the real values separately;
20 * see https://en.cppreference.com/w/cpp or the libpqxx documentation.
21 * - Type alias (using StaffData = ...): a short local name for a longer qualified type; no new type
22 * is created.
23 *
24 * @author Amr Osman <dr3mro@gmail.com>
25 * @date 2026-11-16
26 */

84 /**
85 * @brief Builds the UPDATE that appends a nominee id to the service's Doctors staff array.
86 * @details Uses jsonb_set on the mjson payload path {payload,providers,Doctors}, concatenating
87 * the nominee id as a JSON text value. Both the nominee id (as text) and the entity id are bound.
88 * @param[out] params Cleared and filled with the bound nominee and entity ids.
89 * @return UPDATE ... RETURNING id, or std::nullopt if the payload is not StaffData or an
90 * exception occurs.
91 */
92 std::optional<std::string> getSqlAddStaffStatement(pqxx::params& params) // Builds the "add team member" statement.
93 {
94 try
95 {
96 auto idata = std::get<StaffData>(getData()); // The payload must be StaffData; std::get throws otherwise.
97 Binder binder(params); // Fresh binder for the two parameters.
98 // Bound in a fixed order (argument evaluation order is unspecified).
99 const std::string nominee_ph = binder.bind(std::to_string(idata.nominee_id)); // Convert the id to text because the staff array holds text values; bind as $1.
100 const std::string entity_ph = binder.bind(idata.entity_id); // Bind the service row id as $2.
101 // The SQL calls jsonb_set on mjson at the path {payload,providers,Doctors}, concatenating
102 // (||) the existing array with the nominee id wrapped by to_jsonb(...::text). This adds
103 // the member without rebuilding the whole array; the WHERE id clause scopes it to one service.
104 return fmt::format(
105 "UPDATE {} SET mjson = jsonb_set(mjson, '{{payload,providers,Doctors}}', "
106 "((mjson->'payload'->'providers'->'Doctors') || to_jsonb({}::text))) WHERE id={} RETURNING id;",
107 tablename, nominee_ph, entity_ph); // Two adjacent string literals form one SQL string; the placeholders are filled in order.
108 }
109 catch (const std::exception& e) // Wrong payload type or formatting error.
110 {
111 Message::ErrorMessage("Error creating query for add member."); // Log a human-readable failure.
112 Message::CriticalMessage(e.what()); // Log the exception details.
113 return std::nullopt; // Signal failure.
114 }
115 }

117 /**
118 * @brief Builds the UPDATE that removes a nominee id from the service's Doctors staff array.
119 * @details Rebuilds the array with jsonb_agg over the elements that do not equal the bound
120 * nominee id (as JSON text). The nominee and entity ids are bound in a fixed order.
121 * @param[out] params Cleared and filled with the bound nominee and entity ids.
122 * @return UPDATE ... RETURNING id, or std::nullopt if the payload is not StaffData or an
123 * exception occurs.
124 */
125 std::optional<std::string> getSqlRemoveStaffStatement(pqxx::params& params) // Builds the "remove team member" statement.
126 {
127 try
128 {
129 auto idata = std::get<StaffData>(getData()); // Must be a StaffData payload.
130 Binder binder(params); // Fresh binder for the two parameters.
131 // Bound in a fixed order (argument evaluation order is unspecified).
132 const std::string nominee_ph = binder.bind(std::to_string(idata.nominee_id)); // Nominee id as text -> $1.
133 const std::string entity_ph = binder.bind(idata.entity_id); // Service row id -> $2.
134 // The SQL expands the Doctors array with jsonb_array_elements, keeps only elements that
135 // do NOT equal the nominee id, and re-aggregates them with jsonb_agg. jsonb_set then
136 // writes that smaller array back at the same path for the one service row.
137 return fmt::format(
138 "UPDATE {} SET mjson = jsonb_set(mjson, '{{payload,providers,Doctors}}', "
139 "(SELECT jsonb_agg(elem) FROM jsonb_array_elements(mjson->'payload'->'providers'->'Doctors') AS elem "
140 "WHERE elem <> to_jsonb({}::text))) WHERE id={} RETURNING id;",
141 tablename, nominee_ph, entity_ph); // Placeholders fill the nominee comparison and the row id.
142 }
143 catch (const std::exception& e) // Wrong payload type or formatting error.
144 {
145 Message::ErrorMessage("Error creating query for remove member."); // Log a human-readable failure.
146 Message::CriticalMessage(e.what()); // Log details.
147 return std::nullopt; // Signal failure.
148 }
149 }
150 /**
151 * @brief Builds the permission lookup for a service row by id.
152 * @details Returns the owner/admin/staff columns used by PermissionManager to authorise
153 * service-level actions. The table name and id are formatted into the query text rather than
154 * bound; the resulting text also serves as the permission cache key.
155 * @param[in] service_name Name of the service/organisation table.
156 * @param[in] service_id Primary key of the service row.
157 * @return A SELECT statement (always produced by this implementation).
158 */
159 static std::optional<std::string> getServicePermissionsQuery(const std::string& service_name, std::uint64_t service_id) // Static: called directly by the permission layer, no object needed.
160 {
161 return fmt::format("SELECT owner_id,admin_id,staff FROM {} WHERE id = '{}' LIMIT 1;", service_name, service_id); // Read who owns/administers/staffs the one service row; LIMIT 1 because ids are unique.
162 }
