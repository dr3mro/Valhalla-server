2 * @file asyncresultmapper.cpp
3 * @brief Implementation for asyncresultmapper: Converts one row into a JSON
4 * object using name- and value-based type inference.
5 * @details This file implements the row-to-JSON conversion. For each column it
6 * checks whether the value is SQL NULL, whether the column is a known boolean
7 * or identifier field, whether the text looks like JSON, and otherwise keeps
8 * the text as a string. The first-row and all-row helpers simply call the row
9 * mapper in a loop or at index zero.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * Async controller methods receive a Drogon Result and must return JSON to the
13 * HTTP layer. This file performs that translation and deliberately mirrors the
14 * older synchronous Database::executeQuery outcomes: nulls stay null, the
15 * active flag is a boolean, ids are numbers, and json/jsonb columns become
16 * nested JSON. Because parsing errors are caught and the raw text is kept, one
17 * unusual database value cannot crash a whole list response. Without this file,
18 * the async database path could not produce the same JSON contract that the
19 * rest of the API relies on.
20 *
21 * CONCEPTS TO LEARN (for beginners):
22 * - Drogon result sets, rows and fields:
23 * https://drogon.docsforge.com/master/database/
24 * - jsoncons JSON values and parsing: https://danielaparker.github.io/jsoncons/
25 * - std::string_view (a cheap, non-owning look at text):
26 * https://en.cppreference.com/w/cpp/string/basic_string_view
27 * - C++ exceptions and catch-all handlers:
28 * https://en.cppreference.com/w/cpp/language/try_catch
29 * - C++20 coroutines: https://en.cppreference.com/w/cpp/language/coroutines
30 *
31 * @author Amr Osman <dr3mro@gmail.com>
32 * @date 2026-11-16
33 */

58 /**
59 * @brief Converts one row into a JSON object using name- and value-based
60 * type inference.
61 * @details Each row gets its own object. Parse failures are swallowed and
62 * the raw text is kept, so this function never throws for a malformed
63 * integer or JSON value. Field values are read through their C-string text
64 * form.
65 */
66 jsoncons::json mapRow(const drogon::orm::Result& result,
67 drogon::orm::Result::SizeType row) {
68 jsoncons::json object; // A fresh, empty JSON object for this row;
69 // nothing from another row can leak into it.
70
71 const auto column_count =
72 result.columns(); // How many columns this result has, so we know
73 // how far to walk.
74 for (drogon::orm::Result::RowSizeType i = 0; i < column_count;
75 // Visit each column exactly once, from left to right; stop
76 // after the last column.
77 ++i) {
78 const std::string_view field_name =
79 result.columnName(i); // A non-owning view of the column's name, used
80 // for comparisons and as the JSON key.
81 const auto field = result[row][i]; // The single cell at (row, column):
82 // its value and null flag.
83
84 // SQL NULL must become JSON null, not the
85 // string "NULL" or an empty string.
86 if (field.isNull()) {
87 object[std::string(field_name)] =
88 nullptr; // Copy the name into an owning string for the
89 // key, then store a JSON null value.
90 continue; // Nothing else can be inferred from a null value, so
91 // move on to the next column.
92 }
93
94 const std::string name(
95 field_name); // Own the column name now, so it can be compared
96 // and reused as a JSON key safely.
97 const std::string value(
98 field.c_str()); // Read the cell as text; copying it keeps the
99 // data valid after the Result is touched again.
100
101 // The active flag is known to be a boolean,
102 // whatever text PostgreSQL used.
103 if (name == "active") {
104 // PostgreSQL emits t/f for booleans; accept the common truthy
105 // spellings too.
106 // The comparison produces a C++ bool, which
107 // jsoncons stores as JSON true or false.
108 // These column names
109 // mark values that the
110 // API exposes as whole
111 // numbers.
112 object[name] = (value == "t" || value == "true" || value == "1");
113 } else if (name == "id" || name.ends_with("_id") ||
114 name == "appointment_duration" ||
115 name == "appointment_payment_amount") {
116 // These columns are numbers in the API, but keep the text if it
117 // does not parse.
118 try {
119 // Convert the text to an integer; stoi throws
120 // if the text is not a valid number.
121 // Catch every exception type from the conversion,
122 // including out-of-range numbers.
123 object[name] = std::stoi(value);
124 } catch (...) {
125 // Fall back to the original text so a single
126 // odd value cannot abort the row.
127 // A value starting with a brace or bracket
128 // is probably a JSON object or array.
129 object[name] = value;
130 }
131 } else if (!value.empty() &&
132 (value.front() == '{' || value.front() == '[')) {
133 // json/jsonb arrive as text; a failed parse degrades to a plain
134 // string.
135 try {
136 // Parse the text into real nested
137 // JSON when it is well formed.
138 // Any parse error is deliberately ignored here.
139 object[name] = jsoncons::json::parse(value);
140 } catch (...) {
141 // Keep the raw text as a string instead of
142 // failing the whole request.
143 // No special rule matched, so this is an ordinary text
144 // column.
145 object[name] = value;
146 }
147 } else {
148 object[name] = value; // Store the value as a plain JSON string.
149 }
150 }
151
152 return object; // The caller receives one JSON entry per column, in
153 // column order.
154 }

156 /**
157 * @brief Maps the first row, using a null JSON value to represent an empty
158 * result.
159 * @details The emptiness check also avoids indexing into a result with no
160 * rows.
161 */
162 jsoncons::json mapResultToObject(const drogon::orm::Result& result) {
163 // A SELECT can legitimately return no rows, e.g. a
164 // lookup for a missing id.
165 if (result.empty()) {
166 return jsoncons::json(); // A default-constructed jsoncons::json is
167 // a JSON null, matching the synchronous
168 // path.
169 }
170 return mapRow(result,
171 0); // Row indexes start at zero, so this converts the first
172 // (and possibly only) row.
173 }

175 /**
176 * @brief Maps rows sequentially into an array, preserving the query's row
177 * order.
178 */
179 jsoncons::json::array mapResultToArray(const drogon::orm::Result& result) {
180 jsoncons::json::array
181 reply; // Empty array that will hold one JSON object per row.
182
183 for (drogon::orm::Result::SizeType row = 0; row < result.size();
184 // Walk rows from first to last; stop when every row has
185 // been converted.
186 ++row) {
187 reply.push_back(mapRow(result,
188 row)); // Convert the row and append it, keeping
189 // the database's order intact.
190 }
191 return reply; // An empty result simply produces an empty array, which
192 // is what list endpoints return.
193 }
