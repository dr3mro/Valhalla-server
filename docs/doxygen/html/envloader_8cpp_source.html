1/**
2 * @file envloader.cpp
3 * @brief Typed environment-variable parsing helpers used by configuration sections.
4 * @details This file implements the non-template functions declared in envloader.hpp: reading
5 * a plain string, a boolean, a duration in seconds, and a comma-separated set of names. It
6 * also contains two shared helpers: parseSet() splits text into trimmed tokens, and
7 * optimize_performance() replaces a zero "auto" count with a value based on the CPU count.
8 * Every parser returns a supplied default when the variable is missing or malformed.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This is the implementation of the configuration engine's middle layer. Configurator's
12 * sections call these functions in their constructors, so this file decides what happens when
13 * a deployment sets a variable to an unexpected value (for example a negative timeout or the
14 * word "TRUE"): it logs a clear message and falls back to a safe default rather than crashing.
15 * Without it, section constructors would each need their own error-prone parsing code.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - Environment variables: https://man7.org/linux/man-pages/man7/environ.7.html
19 * - std::chrono durations (lengths of time): https://en.cppreference.com/w/cpp/chrono/duration
20 * - std::unordered_set (a collection of unique values): https://en.cppreference.com/w/cpp/container/unordered_set
21 * - std::stringstream (reading text as if it were a stream): https://en.cppreference.com/w/cpp/io/basic_stringstream
22 *
23 * @author Amr Osman <dr3mro@gmail.com>
24 * @date 2026-11-16
25 */

51// Reads a duration in whole seconds, rejecting negative or invalid numbers.
52std::chrono::seconds EnvLoader::getEnvironmentVariable(const std::string& key, const std::chrono::seconds& defaultValue) // Interprets the value as seconds, or keeps the default on bad input.
53{
54 auto value = get(key); // Fetch the raw value through EnvVars.
55 if (!value.has_value()) // Variable absent: there is nothing to convert.
56 {
57 return defaultValue; // Hand back the safe fallback.
58 }
59 try // Parsing text to a number can fail, so recover instead of letting an exception escape.
60 {
61 auto count = std::stoll(value.value()); // Convert the text to a 64-bit integer (wider than stoi's int).
62 // Durations are stored as unsigned seconds; a negative input cannot be represented,
63 // so keep the default instead of wrapping it through the cast.
64 if (count < 0) // A negative length of time makes no sense for a timeout/duration.
65 {
66 Message::ErrorMessage("Negative duration for env var " + key + "; using default."); // Log the offending variable so it can be fixed.
67 return defaultValue; // Refuse the negative value and keep the safe default.
68 }
69 return std::chrono::seconds(static_cast<uint64_t>(count)); // Wrap the count in a type that clearly represents "this many seconds".
70 }
71 catch (const std::exception&) // std::stoll throws when the text is not numeric or is out of range.
72 {
73 Message::ErrorMessage("Invalid duration value for env var " + key + "; using default."); // Report the bad variable name.
74 return defaultValue; // Recover so malformed settings cannot stop startup.
75 }
76}

86// Splits a comma-separated string into a trimmed set of unique, non-empty tokens.
87std::unordered_set<std::string> EnvLoader::parseSet(const std::string& str) // Turns "a, b, c" into {"a","b","c"} with spaces and blanks removed.
88{
89 std::unordered_set<std::string> result; // The output set; duplicates disappear automatically because a set holds unique values only.
90 std::stringstream ss(str); // Wrap the text so getline can pull out comma-delimited pieces one at a time.
91 std::string item; // Temporary holder for each piece as it is read.
92 while (std::getline(ss, item, ',')) // Keep reading until no more comma-separated pieces remain; each pass handles one token.
93 {
94 // Trim any leading or trailing whitespace
95 item.erase(item.find_last_not_of(" \n\r\t") + 1); // Cut off trailing spaces/newlines by erasing from just after the last non-space character.
96 item.erase(0, item.find_first_not_of(" \n\r\t")); // Cut off leading spaces/newlines by erasing up to the first non-space character.
97 // Drop empty tokens so "a,,b" and a string of only whitespace yield no empty member.
98 if (!item.empty()) // Skip pieces that are empty after trimming (the gap between two commas, for example).
99 {
100 result.insert(item); // Add the cleaned token to the set.
101 }
102 }
103 return result; // Hand back all distinct, non-empty tokens to the caller.
104}

106// Replaces a zero "auto" count with a multiple of the machine's CPU count.
107void EnvLoader::optimize_performance(uint16_t& parameter, const uint8_t factor) // Grows a zero value to CPUs times factor, leaving explicit values alone.
108{
109 uint16_t ncpus = (std::thread::hardware_concurrency() > 0) ? std::thread::hardware_concurrency() : 1; // Ask the OS for the CPU count, treating an unknown count (0) as one CPU.
110 // Zero is the "let the runtime size it" sentinel; a nonzero explicit value is respected.
111 if (parameter < 1) // Only an unset/zero count needs auto-sizing; a positive value was chosen deliberately.
112 {
113 parameter = ncpus * factor; // Scale with the hardware so a bigger machine gets more workers/connections by default.
114 }
115}
