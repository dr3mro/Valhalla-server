3 * @brief Implementation for database: Takes ownership of a connection, validates it and caches its connection string.
4 * @details This file implements every helper declared in database.hpp. The constructor opens

52/**
53 * Takes ownership of a connection, validates it and caches its connection string.
54 * A failed validation is logged (critical) but does not throw; the object still
55 * exists with the supplied pointer. reconnect() later depends on the cached string.
56 * conn: Connection moved into this Database.
57 */
58Database::Database(std::shared_ptr<pqxx::connection>&& conn) : connection(std::move(conn)) // The initializer list stores the moved-in connection before the body runs.
59{
60 try // Probing the connection can throw, so it is done inside a guarded block.
61 {
62 if (connection != nullptr) // A null pointer is tolerated: there is nothing to validate in that case.
63 {
64 pqxx::nontransaction ntxn(*connection); // A nontransaction runs a statement without starting a transaction; it is scoped to this block.
65
66 ntxn.exec("SELECT 1"); // A trivial round trip that proves the link works and the server responds.
67
68 connection_info = connection->connection_string(); // Cache the credentials so reconnect() can rebuild the link later.
69 }
70 }
71 catch (const std::exception& e) // A broken link is reported, but construction still succeeds.
72 {
73 Message::CriticalMessage(fmt::format("Exception caught during database connection: {}", e.what())); // Log the exact driver error for diagnosis.
74 }
75}

82/**
83 * Checks whether a row exists using a parameterised EXISTS query.
84 * Table and column are passed through `quote_name` (identifiers cannot be bound as
85 * parameters); only the value is bound. The transaction scope is closed before the
86 * result is read so no idle transaction is left open on the connection.
87 * table: Table name.
88 * column: Column name.
89 * value: Value bound to `$1`.
90 * Returns `true` if the row exists; `false` if not, or if the query threw.
91 * Note: Query failures are swallowed and reported as `false` (see the header note).
92 */
93bool Database::checkExists(const std::string& table, const std::string& column, const std::string& value)
94{
95 try // Any SQL error is converted into a false return below instead of escaping the function.
96 {
97 std::string query = fmt::format("SELECT EXISTS (SELECT 1 FROM {} WHERE {} = $1);", connection->quote_name(table), connection->quote_name(column)); // quote_name escapes identifiers (which cannot be bound as parameters); only the value uses $1.
98 pqxx::result result; // Empty result holder, kept alive after the transaction object is destroyed.
99 {
100 pqxx::nontransaction txn(*connection); // A short-lived, non-transactional execution context.
101
102 result = txn.exec(pqxx::zview(query), pqxx::params{value}); // Send the SQL with the value bound separately, avoiding SQL injection.
103 }
104 return result[0][0].as<bool>(); // EXISTS always returns one row/column: read it as a boolean.
105 }
106 catch (const std::exception& e) // A query failure is indistinguishable from "not found" for callers.
107 {
108 Message::CriticalMessage(fmt::format("Error executing query: {}", e.what())); // Still log it so the failure is visible in the logs.
109 return false; // Report "not found" on error; use check_connection() to tell the cases apart.
110 }
111}

113/**
114 * @brief Probes the connection with a lightweight `SELECT 1` round trip.
115 * @return `true` only if the pointer is non-null, the probe succeeded and the socket is open.
116 */
117bool Database::check_connection()
118{
119 if (connection == nullptr) // There is no connection to probe, so it cannot be healthy.
120 {
121 return false; // Report unhealthy immediately.
122 }
123 try // exec() throws if the link is broken; that means "unhealthy" rather than a program error.
124 {
125 pqxx::nontransaction ntxn(*connection); // Minimal execution context for the probe.
126 ntxn.exec("SELECT 1"); // If this completes, the server answered.
127 }
128 catch (const std::exception& e) // Any driver exception means the probe failed.
129 {
130 return false; // Signal that this link needs reconnecting.
131 }
132 return connection->is_open(); // Confirm the underlying socket is still open as a final check.
133}

135/**
136 * @brief Rebuilds the wrapped connection from the string cached at construction.
137 * @return The health check result of the new connection; `false` if creation or check failed.
138 * @details The old connection is dropped when the `shared_ptr` is reassigned. The failure
139 * path is logged as critical. Not thread-safe: the caller must own the Database
140 * exclusively (see DatabaseConnectionPool::reconnect_all and the watchdog).
141 * @pre check_connection() returned false, or a reconnect was otherwise requested.
142 */
143bool Database::reconnect()
144{
145 try // Opening a new connection can throw (bad credentials, server down, timeout).
146 {
147 {
148 connection = std::make_shared<pqxx::connection>(connection_info); // Build a fresh connection from the cached string; the old one is released here.
149 }
150 return check_connection(); // Only report success if the new link also passes a real probe.
151 }
152 catch (const std::exception& e) // A failure leaves the old pointer in place; report it.
153 {
154 Message::CriticalMessage(fmt::format("Failed to reconnect to database,{}", e.what())); // Log the driver's reason for the reconnect failure.
155 return false; // Tell the caller the reconnect did not work.
156 }
157}

159/**
160 * Reads the column metadata of a table, excluding the synthetic `id` column.
161 * Maps `information_schema.columns` rows into ColumnInfo: a missing default becomes
162 * the literal "None" and `is_nullable == "YES"` becomes `true`.
163 * tableName: Table to inspect (bound as a parameter).
164 * Returns A set of columns, or `std::nullopt` if the query threw.
165 * Note: ColumnInfo's equality/hash ignore the Constraint field, so the set is keyed by
166 * name, data type and nullability.
167 */
168std::optional<std::unordered_set<api::v2::ColumnInfo>> Database::getTableSchema(const std::string& tableName)
169{
170 try // Catalog queries can fail, so the whole read is guarded.
171 {
172 pqxx::result result; // Holds the returned metadata rows.
173
174 {
175 pqxx::nontransaction ntxn(*connection); // A read-only statement needs no transaction.
176 result = ntxn.exec(pqxx::zview("SELECT column_name, data_type, column_default, is_nullable FROM "
177 "information_schema.columns WHERE table_name = $1 AND column_name != 'id';"), // information_schema is PostgreSQL's built-in catalog of table/column metadata; the id column is filtered out.
178 pqxx::params{tableName}); // The table name is bound as a value, not spliced into the SQL text.
179 }
180
181 std::unordered_set<api::v2::ColumnInfo> schema; // Collect columns in a set so duplicates collapse automatically.
182
183 for (const auto& row : result) // One iteration per column found in the catalog.
184 {
185 api::v2::ColumnInfo column; // Fill a value object from the four selected fields.
186 column.Name = row["column_name"].as<std::string>(); // Column name as text.
187 column.DataType = row["data_type"].as<std::string>(); // Database type name, e.g. "character varying".
188 column.Constraint = row["column_default"].is_null() ? "None" : row["column_default"].as<std::string>(); // No default value becomes the readable string "None".
189 column.isNullable = row["is_nullable"].as<std::string>() == "YES"; // Convert the catalog's "YES"/"NO" into a boolean.
190 schema.insert(column); // Add the finished column to the set.
191 }
192 return schema; // Return the complete set to the caller.
193 }
194 catch (const std::exception& e) // A failed query must not crash the caller.
195 {
196 Message::CriticalMessage(fmt::format("Error executing query: {}", e.what())); // Log the failure with the driver's message.
197 return std::nullopt; // Use the empty optional to signal "schema unavailable".
198 }
199}

203 * @return A set of names, or `std::nullopt` if the query threw.
204 * @details Uses a `pqxx::work` (committed implicitly when it goes out of scope) even though

212

214 pqxx::work txn(*connection); // A work transaction commits on scope exit; harmless for a read-only statement.

216 }

223 }

233/**
234 * Executes a parameterised query and converts result rows to JSON.
235 * Commits the transaction only when TransactionType is `pqxx::work`; a
236 * `pqxx::nontransaction` simply ends at scope exit. Columns are converted by
237 * PostgreSQL type OID, with unhandled types falling back to a string.
238 * Template parameter jsonType: `jsoncons::json` (returns the first row) or `jsoncons::json::array`
239 * (returns all rows); any other type is rejected by a static_assert.
240 * Template parameter TransactionType: Transaction wrapper used for the execution.
241 * query: SQL text with `$1, $2, ...` placeholders.
242 * params: Values bound to those placeholders.
243 * Returns The converted value, or `std::nullopt` if any exception was caught.
244 * Note: For the single-object specialisation an empty result yields a default-constructed
245 * JSON value, not `std::nullopt` (only exceptions produce `std::nullopt`).
246 */
247template <typename jsonType, typename TransactionType>
248std::optional<jsonType> Database::executeQuery(const std::string& query, const pqxx::params& params)
249{
250 static_assert(std::is_same_v<jsonType, jsoncons::json> || std::is_same_v<jsonType, jsoncons::json::array>, "Unsupported jsonType specialization"); // Fail at compile time if a caller passes an unsupported JSON shape.
251
252 try // Any conversion or SQL error becomes an empty optional below.
253 {
254 pqxx::result results; // Kept outside the transaction scope so rows survive after it closes.
255
256 {
257 TransactionType txn(*connection); // Uses either a work or a nontransaction, chosen by the template argument.
258
259 results = txn.exec(pqxx::zview(query), params); // Run the SQL with the parameters bound separately.
260
261 if constexpr (std::is_same_v<TransactionType, pqxx::work>) // Only a real transaction needs an explicit commit; this check happens at compile time.
262 {
263 txn.commit(); // Make the writes permanent for work transactions.
264 }
265 }
266
267 jsonType reply; // Either a JSON array to fill, or a default JSON object for the single-row case.
268
269 for (const auto& row : results) // Walk every row returned by the query.
270 {
271 // A fresh object per row, so no field can leak from one row into the next.
272 jsoncons::json object; // Build one JSON object for this row.
273
274 for (const auto& field : row) // Walk every column in the current row.
275 {
276 const std::string_view field_name = field.name(); // A view avoids copying the column name; it stays valid for the row's lifetime.
277 const unsigned int field_type = field.type(); // PostgreSQL type OID, used below to pick the right conversion.
278
279 if (field.is_null()) // SQL NULL has to become JSON null, not the string "null".
280 {
281 object[field_name] = nullptr; // Represent a SQL NULL explicitly in the JSON output.
282 }
283 else
284 {
285 switch (field_type) // Dispatch on the column's OID so values keep their proper JSON type.
286 {
287 case TEXT: // TEXT or VARCHAR: database text becomes a JSON string.
288 object[field_name] = field.as<std::string>(); // Convert the field to a C++ string, which jsoncons stores as a JSON string.
289 break; // Move on to the next column.
290
291 case INTEGER: // INTEGER: keep it a number rather than a string.
292 object[field_name] = field.as<int>(); // Convert to int so the JSON value is numeric.
293 break; // Move on to the next column.
294
295 case BOOLEAN: // BOOLEAN: keep the true/false meaning.
296 object[field_name] = field.as<bool>(); // Convert to a C++ bool, which becomes a JSON boolean.
297 break; // Move on to the next column.
298
299 case JSON: // JSON: the field holds JSON text that should stay structured.
300 case JSONB: // JSONB: binary JSON, delivered as text by libpqxx, parsed the same way.
301 object[field_name] = jsoncons::json::parse(field.as<std::string>()); // Parse the text so the result is a nested JSON value, not an escaped string.
302 break; // Move on to the next column.
303
304 default: // Handle unknown or unhandled types: fall back safely instead of throwing.
305 object[field_name] = field.c_str(); // Default to string
306 // representation, so no data is lost for unfamiliar types.
307 break; // Move on to the next column.
308 }
309 }
310 }
311 if constexpr (std::is_same_v<jsonType, jsoncons::json::array>) // For the array shape, accumulate every row at compile time.
312 {
313 reply.push_back(std::move(object)); // Transfer this row's object into the result array without copying.
314 }
315 else
316 {
317 return object; // For the single-object shape, return the first row immediately and ignore the rest.
318 }
319 }
320 return reply; // For the array shape, return all collected rows after the loop.
321 }
322 catch (const std::exception& e) // A SQL or conversion error must not propagate to the caller.
323 {
324 Message::ErrorMessage("Error executing query:"); // Print a short marker first.
325 Message::InfoMessage(query); // Log the exact SQL to make the failing statement identifiable.
326 Message::CriticalMessage(e.what()); // Log the driver/conversion error itself.
327 }
328 return std::nullopt; // Signal failure with an empty optional.
329}

331/**
332 * Executes a query in a nontransaction and returns a single scalar value.
333 * Reads the first column of the first row; an empty result becomes `std::nullopt`.
334 * Template parameter T: Target scalar type (`as<std::optional<T>>()` maps SQL NULL to `std::nullopt`).
335 * query: SQL text with placeholders.
336 * params: Values bound to those placeholders.
337 * Returns The value, or `std::nullopt` on empty result or any caught exception.
338 */
339template <typename T>
340std::optional<T> Database::doSimpleQuery(const std::string& query, const pqxx::params& params)
341{
342 try // Guard against SQL and conversion errors.
343 {
344 pqxx::result result; // Holds rows after the transaction scope ends.
345
346 {
347 pqxx::nontransaction txn(*connection); // A simple lookup needs no transaction.
348
349 result = txn.exec(pqxx::zview(query), params); // Run the query with bound parameters.
350 }
351
352 return result.empty() ? std::nullopt : result[0][0].as<std::optional<T>>(); // No rows means no value; the optional conversion also turns a SQL NULL into nullopt.
353 }
354 catch (const std::exception& e) // Convert any error into the empty result.
355 {
356 Message::ErrorMessage("Error executing query:"); // Marker line in the log.
357 Message::InfoMessage(query); // The failing SQL statement.
358 Message::CriticalMessage(e.what()); // The error text from libpqxx.
359 return std::nullopt; // Report failure to the caller.
360 // throw; // Rethrow the exception to indicate failure
361 }
