77/**
78 * Takes ownership of a connection, validates it and caches its connection
79 * string. A failed validation is logged (critical) but does not throw; the
80 * object still exists with the supplied pointer. reconnect() later depends on
81 * the cached string. conn: Connection moved into this Database.
82 */
83Database::Database(std::shared_ptr<pqxx::connection>&& conn)
84 // The initializer list stores the moved-in
85 // connection before the body runs.
86 : connection(std::move(conn)) {
87 // Probing the connection can throw, so it is done inside a guarded
88 // block.
89 try {
90 // A null pointer is tolerated: there is
91 // nothing to validate in that case.
92 if (connection != nullptr) {
93 pqxx::nontransaction ntxn(
94 *connection); // A nontransaction runs a statement without
95 // starting a transaction; it is scoped to this
96 // block.
97
98 ntxn.exec("SELECT 1"); // A trivial round trip that proves the link
99 // works and the server responds.
100
101 // Cache the credentials so
102 // reconnect() can rebuild the
103 // link later.
104 // A broken link is reported, but
105 // construction still succeeds.
106 connection_info = connection->connection_string();
107 }
108 } catch (const std::exception& e) {
109 Message::CriticalMessage(
110 fmt::format("Exception caught during database connection: {}",
111 e.what())); // Log the exact driver error for diagnosis.
112 }
113}

124/**
125 * Checks whether a row exists using a parameterised EXISTS query.
126 * Table and column are passed through `quote_name` (identifiers cannot be bound
127 * as parameters); only the value is bound. The transaction scope is closed
128 * before the result is read so no idle transaction is left open on the
129 * connection. table: Table name. column: Column name. value: Value bound to
130 * `$1`. Returns `true` if the row exists; `false` if not, or if the query
131 * threw. Note: Query failures are swallowed and reported as `false` (see the
132 * header note).
133 */
134bool Database::checkExists(const std::string& table,
135 const std::string& column,
136 const std::string& value) {
137 // Any SQL error is converted into a false return below instead of
138 // escaping the function.
139 try {
140 std::string query =
141 fmt::format("SELECT EXISTS (SELECT 1 FROM {} WHERE {} = $1);",
142 connection->quote_name(table),
143 connection->quote_name(
144 column)); // quote_name escapes identifiers (which
145 // cannot be bound as parameters); only the
146 // value uses $1.
147 // Empty result holder, kept alive after the
148 // transaction object is destroyed.
149 pqxx::result result;
150 {
151 pqxx::nontransaction txn(
152 *connection); // A short-lived, non-transactional execution
153 // context.
154
155 result =
156 txn.exec(pqxx::zview(query),
157 pqxx::params{value}); // Send the SQL with the value bound
158 // separately, avoiding SQL injection.
159 }
160 // EXISTS always returns one
161 // row/column: read it as a boolean.
162 // A query failure is indistinguishable
163 // from "not found" for callers.
164 return result[0][0].as<bool>();
165 } catch (const std::exception& e) {
166 Message::CriticalMessage(
167 fmt::format("Error executing query: {}",
168 e.what())); // Still log it so the failure is visible in
169 // the logs.
170 return false; // Report "not found" on error; use check_connection() to
171 // tell the cases apart.
172 }
173}

175/**
176 * @brief Probes the connection with a lightweight `SELECT 1` round trip.
177 * @return `true` only if the pointer is non-null, the probe succeeded and the
178 * socket is open.
179 */
180bool Database::check_connection() {
181 // There is no connection to probe, so it cannot be healthy.
182 if (connection == nullptr) {
183 return false; // Report unhealthy immediately.
184 }
185 // exec() throws if the link is broken; that means "unhealthy" rather
186 // than a program error.
187 try {
188 pqxx::nontransaction ntxn(
189 *connection); // Minimal execution context for the probe.
190 // If this completes, the server answered.
191 // Any driver exception means the probe failed.
192 ntxn.exec("SELECT 1");
193 } catch (const std::exception& e) {
194 return false; // Signal that this link needs reconnecting.
195 }
196 return connection->is_open(); // Confirm the underlying socket is still
197 // open as a final check.
198}

204 * check failed.
205 * @details The old connection is dropped when the `shared_ptr` is reassigned.
206 * The failure path is logged as critical. Not thread-safe: the caller must own
207 * the Database exclusively (see DatabaseConnectionPool::reconnect_all and the
208 * watchdog).
209 * @pre check_connection() returned false, or a reconnect was otherwise
210 * requested.
211 */
212bool Database::reconnect() {
213 // Opening a new connection can throw (bad credentials, server down,
214 // timeout).
215 try {
216 {
217 connection = std::make_shared<pqxx::connection>(
218 connection_info); // Build a fresh connection from the cached
219 // string; the old one is released here.
220 }
221 // Only report success if the new link also
222 // passes a real probe.
223 // A failure leaves the old pointer in place; report it.
224 return check_connection();
225 } catch (const std::exception& e) {
226 Message::CriticalMessage(
227 fmt::format("Failed to reconnect to database,{}",
228 e.what())); // Log the driver's reason for the reconnect
229 // failure.
230 return false; // Tell the caller the reconnect did not work.
231 }

235 * Reads the column metadata of a table, excluding the synthetic `id` column.
236 * Maps `information_schema.columns` rows into ColumnInfo: a missing default
237 * becomes the literal "None" and `is_nullable == "YES"` becomes `true`.
238 * tableName: Table to inspect (bound as a parameter).
239 * Returns A set of columns, or `std::nullopt` if the query threw.
240 * Note: ColumnInfo's equality/hash ignore the Constraint field, so the set is
241 * keyed by name, data type and nullability.
242 */
243std::optional<std::unordered_set<api::v2::ColumnInfo>> Database::getTableSchema(
244 const std::string& tableName) {
245 // Catalog queries can fail, so the whole read is guarded.
246 try {
247 pqxx::result result; // Holds the returned metadata rows.
248
249 {
250 pqxx::nontransaction ntxn(
251 *connection); // A read-only statement needs no transaction.
252 result =
253 ntxn.exec(pqxx::zview(
254 "SELECT column_name, data_type, column_default, "
255 "is_nullable FROM "
256 "information_schema.columns WHERE table_name = $1 AND "
257 "column_name != 'id';"), // information_schema is
258 // PostgreSQL's built-in
259 // catalog of
260 // table/column metadata; the id column is filtered out.
261 pqxx::params{
262 tableName}); // The table name is bound as a value, not
263 // spliced into the SQL text.
264 }
265
266 std::unordered_set<api::v2::ColumnInfo>
267 schema; // Collect columns in a set so duplicates collapse
268 // automatically.
269
270 for (const auto& row :
271 // One iteration per column found in the catalog.
272 result) {
273 api::v2::ColumnInfo
274 column; // Fill a value object from the four selected fields.
275 column.Name =
276 row["column_name"].as<std::string>(); // Column name as text.
277 column.DataType =
278 row["data_type"].as<std::string>(); // Database type name, e.g.
279 // "character varying".
280 column.Constraint =
281 row["column_default"].is_null()
282 ? "None"
283 : row["column_default"]
284 .as<std::string>(); // No default value becomes the
285 // readable string "None".
286 column.isNullable =
287 row["is_nullable"].as<std::string>() ==
288 "YES"; // Convert the catalog's "YES"/"NO" into a boolean.
289 schema.insert(column); // Add the finished column to the set.
290 }
291 // Return the complete set to the caller.
292 // A failed query must not crash the caller.
293 return schema;
294 } catch (const std::exception& e) {
295 Message::CriticalMessage(
296 fmt::format("Error executing query: {}",
297 e.what())); // Log the failure with the driver's message.
298 return std::nullopt; // Use the empty optional to signal "schema
299 // unavailable".
300 }

341/**
342 * Executes a parameterised query and converts result rows to JSON.
343 * Commits the transaction only when TransactionType is `pqxx::work`; a
344 * `pqxx::nontransaction` simply ends at scope exit. Columns are
345 * converted by PostgreSQL type OID, with unhandled types falling back to a
346 * string. Template parameter jsonType: `jsoncons::json` (returns the first row)
347 * or `jsoncons::json::array` (returns all rows); any other type is rejected by
348 * a static_assert. Template parameter TransactionType: Transaction wrapper used
349 * for the execution. query: SQL text with `$1, $2, ...` placeholders. params:
350 * Values bound to those placeholders. Returns The converted value, or
351 * `std::nullopt` if any exception was caught. Note: For the single-object
352 * specialisation an empty result yields a default-constructed JSON value, not
353 * `std::nullopt` (only exceptions produce `std::nullopt`).
354 */
355template <typename jsonType, typename TransactionType>
356std::optional<jsonType> Database::executeQuery(const std::string& query,
357 const pqxx::params& params) {
358 static_assert(std::is_same_v<jsonType, jsoncons::json> ||
359 std::is_same_v<jsonType, jsoncons::json::array>,
360 "Unsupported jsonType specialization"); // Fail at compile time
361 // if a caller passes
362 // an unsupported JSON
363 // shape.
364
365 // Any conversion or SQL error becomes an empty optional below.
366 try {
367 pqxx::result results; // Kept outside the transaction scope so rows
368 // survive after it closes.
369
370 {
371 TransactionType txn(
372 *connection); // Uses either a work or a nontransaction, chosen
373 // by the template argument.
374
375 results = txn.exec(pqxx::zview(query),
376 params); // Run the SQL with the parameters bound
377 // separately.
378
379 // Only a real transaction
380 // needs an explicit commit;
381 // this check happens at
382 // compile time.
383 if constexpr (std::is_same_v<TransactionType, pqxx::work>) {
384 txn.commit(); // Make the writes permanent for work
385 // transactions.
386 }
387 }
388
389 jsonType reply; // Either a JSON array to fill, or a default JSON
390 // object for the single-row case.
391
392 // Walk every row returned by the query.
393 for (const auto& row : results) {
394 // A fresh object per row, so no field can leak from one row into
395 // the next.
396 jsoncons::json object; // Build one JSON object for this row.
397
398 // Walk every column in the current row.
399 for (const auto& field : row) {
400 const std::string_view field_name =
401 field.name(); // A view avoids copying the column name; it
402 // stays valid for the row's lifetime.
403 const unsigned int field_type =
404 field.type(); // PostgreSQL type OID, used below to pick
405 // the right conversion.
406
407 // SQL NULL has to become JSON null, not
408 // the string "null".
409 if (field.is_null()) {
410 object[field_name] = nullptr; // Represent a SQL NULL explicitly in
411 // the JSON output.
412 } else {
413 // Dispatch on the column's OID so
414 // values keep their proper JSON type.
415 switch (field_type) {
416 case TEXT: // TEXT or VARCHAR: database text becomes a
417 // JSON string.
418 object[field_name] =
419 field.as<std::string>(); // Convert the field to a C++
420 // string, which jsoncons
421 // stores as a JSON string.
422 break; // Move on to the next column.
423
424 case INTEGER: // INTEGER: keep it a number rather than
425 // a string.
426 object[field_name] = field.as<int>(); // Convert to int so the
427 // JSON value is numeric.
428 break; // Move on to the next column.
429
430 case BOOLEAN: // BOOLEAN: keep the true/false meaning.
431 object[field_name] = field.as<bool>(); // Convert to a C++ bool,
432 // which becomes a JSON
433 // boolean.
434 break; // Move on to the next column.
435
436 case JSON: // JSON: the field holds JSON text that
437 // should stay structured.
438 case JSONB: // JSONB: binary JSON, delivered as text by
439 // libpqxx, parsed the same way.
440 object[field_name] = jsoncons::json::parse(
441 field.as<std::string>()); // Parse the text so
442 // the result is a
443 // nested JSON value,
444 // not an escaped
445 // string.
446 break; // Move on to the next column.
447
448 default: // Handle unknown or unhandled types: fall
449 // back safely instead of throwing.
450 object[field_name] = field.c_str(); // Default to string
451 // representation, so no data is lost for unfamiliar
452 // types.
453 break; // Move on to the next column.
454 }
455 }
456 }
457 // For the array
458 // shape, accumulate
459 // every row at
460 // compile time.
461 if constexpr (std::is_same_v<jsonType, jsoncons::json::array>) {
462 reply.push_back(
463 std::move(object)); // Transfer this row's object into the
464 // result array without copying.
465 } else {
466 return object; // For the single-object shape, return the first
467 // row immediately and ignore the rest.
468 }
469 }
470 // For the array shape, return all collected rows after
471 // the loop.
472 // A SQL or conversion error must not
473 // propagate to the caller.
474 return reply;
475 } catch (const std::exception& e) {
476 Message::ErrorMessage(
477 "Error executing query:"); // Print a short marker first.
478 Message::InfoMessage(query); // Log the exact SQL to make the failing
479 // statement identifiable.
480 Message::CriticalMessage(
481 e.what()); // Log the driver/conversion error itself.
482 }
483 return std::nullopt; // Signal failure with an empty optional.
484}

486/**
487 * Executes a query in a nontransaction and returns a single scalar value.
488 * Reads the first column of the first row; an empty result becomes
489 * `std::nullopt`. Template parameter T: Target scalar type
490 * (`as<std::optional<T>>()` maps SQL NULL to `std::nullopt`). query: SQL text
491 * with placeholders. params: Values bound to those placeholders. Returns The
492 * value, or `std::nullopt` on empty result or any caught exception.
493 */
494template <typename T>
495std::optional<T> Database::doSimpleQuery(const std::string& query,
496 const pqxx::params& params) {
497 // Guard against SQL and conversion errors.
498 try {
499 pqxx::result result; // Holds rows after the transaction scope ends.
500
501 {
502 pqxx::nontransaction txn(
503 *connection); // A simple lookup needs no transaction.
504
505 result = txn.exec(pqxx::zview(query),
506 params); // Run the query with bound parameters.
507 }
508
509 return result.empty() ? std::nullopt
510 // No rows means no value;
511 // the optional conversion
512 // also turns a SQL NULL into
513 // nullopt.
514 // Convert any error into the empty result.
515 : result[0][0].as<std::optional<T>>();
516 } catch (const std::exception& e) {
517 Message::ErrorMessage("Error executing query:"); // Marker line in the log.
518 Message::InfoMessage(query); // The failing SQL statement.
519 Message::CriticalMessage(e.what()); // The error text from libpqxx.
520 return std::nullopt; // Report failure to the caller.
521 // throw; // Rethrow the exception to indicate failure
522 }
523}
