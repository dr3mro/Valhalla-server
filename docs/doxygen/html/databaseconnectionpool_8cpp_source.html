1/**
2 * @file databaseconnectionpool.cpp
3 * @brief Implementation for databaseconnectionpool: Opens one pqxx connection using the configured credentials and a connect timeout.
4 * @details This file implements the pool's lifecycle. createDatabaseConnection() opens and
5 * validates a single connection; the constructor spins up all slots in parallel with
6 * retries; get_connection()/return_connection() lease and return links, measuring
7 * waits and throttling saturation warnings; reconnect_all() repairs every idle link
8 * in parallel while holding the pool lock so nothing is handed out half-reconnected.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This is where the server decides how many database connections exist and what happens when
12 * they are all busy. It translates configuration (DB_MAX_CONN, DB_ACQUIRE_TIMEOUT_MS) into
13 * real behaviour, maps a wait that expires to a DatabaseBusy exception (which the HTTP layer
14 * answers with 503), and cooperates with the watchdog to recover from database restarts.
15 * Without it there would be no connection reuse, no back-pressure, and no automatic recovery.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - std::async and std::future (run work in parallel and collect its result): https://en.cppreference.com/w/cpp/thread/async
19 * - std::future::get and wait: https://en.cppreference.com/w/cpp/thread/future
20 * - std::atomic::compare_exchange_strong (an atomic "update if unchanged"): https://en.cppreference.com/w/cpp/atomic/atomic/compare_exchange
21 * - std::chrono::steady_clock (a monotonic clock ideal for timeouts): https://en.cppreference.com/w/cpp/chrono/steady_clock
22 * - Exponential backoff (sleep longer after each failed attempt)
23 * - exit() and EXIT_FAILURE from the C standard library: https://en.cppreference.com/w/cpp/utility/program/exit
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

56/**
57 * Opens one pqxx connection using the configured credentials and a connect timeout.
58 * Returns a Database only when the connection is open; failures are logged rather
59 * than thrown so the caller can retry the whole slot.
60 * config_: Configuration holding host, dbname, user and password.
61 * Returns A validated Database, or `nullptr` on failure.
62 */
63std::shared_ptr<Database> DatabaseConnectionPool::createDatabaseConnection(const auto& config_)
64{
65 try // libpqxx throws on bad credentials or an unreachable server; the caller prefers a nullptr to an exception.
66 {
67 auto conn = std::make_shared<pqxx::connection>( // Build a connection from a libpq-style keyword/value string.
68 fmt::format("host={} dbname={} user={} password={} connect_timeout={}", config_.host, config_.name, config_.user, config_.pass, TIMEOUT).c_str()); // connect_timeout stops a dead server from blocking a worker forever.
69
70 if (conn->is_open()) // Only a successfully opened socket is worth wrapping.
71 {
72 return std::make_shared<Database>(std::move(conn)); // Hand the connection to Database, which validates it with SELECT 1.
73 }
74
75 Message::CriticalMessage("Failed to open database connection."); // Report a connection object that was created but not open.
76 }
77 catch (const std::exception& e) // Turn any driver exception into a log entry plus nullptr.
78 {
79 Message::CriticalMessage(fmt::format("Failed to create a Database connection: {}", e.what())); // Include the driver's reason.
80 }
81 return nullptr; // Signal failure so the caller can retry this slot.
82}

84/**
85 * @brief Initialises the pool from the Configurator; failure terminates the process.
86 * @details Reads DB_MAX_CONN and the acquire timeout, warns when the pool is undersized,
87 * then starts all connections concurrently. Each slot retries with exponential
88 * backoff (MAX_RETRIES attempts); if any slot cannot be created, a runtime_error is
89 * raised and the catch block exits the process with EXIT_FAILURE.
90 */
91DatabaseConnectionPool::DatabaseConnectionPool() : configurator_(Store::getObject<Configurator>()) // Resolve the shared configuration object before the body runs.
92{
93 try // Any setup failure is fatal, so it is funnelled into the exit path below.
94 {
95 const Configurator::DatabaseConfig& config = configurator_->get<Configurator::DatabaseConfig>(); // Read the DB_MAX_CONN/acquire-timeout/credential settings once.
96 acquireTimeout_ = std::chrono::milliseconds(config.acquire_timeout_ms); // Cache the timeout so get_connection() need not touch configuration later.
97
98 // Every server thread blocks on one connection while it queries, and the watchdog needs one more. A smaller pool
99 // makes requests queue (and time out with 503) even though nothing is slow.
100 const unsigned int server_threads = configurator_->get<Configurator::ServerConfig>().threads; // Number of Drogon IO threads that may query at once.
101 if (config.max_conn < server_threads + 1U) // Need one slot per thread plus one spare for the watchdog's health checks.
102 {
103 Message::WarningMessage(fmt::format("DB_MAX_CONN ({}) is below SERVER_THREADS + 1 ({}): requests may queue for a connection.", config.max_conn,
104 server_threads + 1U)); // Warn early so an operator can raise the pool size before users see 503s.
105 }
106
107 std::vector<std::future<std::shared_ptr<Database>>> futures; // One future per connection slot, used to collect the parallel results.
108
109 futures.reserve(config.max_conn); // Reserve once so push_back does not reallocate while tasks are launched.
110 for (uint16_t i = 0; i < config.max_conn; ++i) // One iteration per configured pool slot.
111 {
112 futures.push_back(std::async(std::launch::async, // Start each connection attempt on its own thread so all slots open in parallel.
113 [this, config]() -> std::shared_ptr<Database> // Capture the pool and config by value; the task outlives this loop iteration.
114 {
115 unsigned int retryCount = 0; // Counts failed attempts for the backoff and the warning message.
116 while (retryCount < MAX_RETRIES) // Stop after the configured number of attempts.
117 {
118 auto conn = createDatabaseConnection(config); // Try once to open and validate a connection.
119 if (conn) // A non-null pointer means the slot is ready.
120 {
121 return conn; // Hand the finished connection back to the constructor.
122 }
123 std::this_thread::sleep_for(std::chrono::seconds(1U << retryCount)); // Exponential backoff: 1s, 2s, 4s... gives the DB time to become ready.
124 Message::WarningMessage(fmt::format("Failed to create a database connection. Attempt {}/{}...", ++retryCount, MAX_RETRIES)); // Log the attempt number, then loop.
125 }
126 return nullptr; // All attempts failed for this slot.
127 }));
128 }
129
130 // Wait for all futures
131 for (auto& future : futures) // Walk the collected results in slot order.
132 {
133 auto conn = future.get(); // Blocks only on completion of the individual task
134 if (conn != nullptr) // This slot was created successfully.
135 {
136 pool_.release(conn); // Put the new connection into the idle set so it can be leased at once.
137 Message::InitMessage(fmt::format("Connection {}/{} created successfully.", pool_.idle(), config.max_conn)); // Report progress using the number now idle.
138 }
139 else
140 {
141 Message::ErrorMessage("Failed to establish one or more connections after maximum retries."); // One bad slot is enough to abort startup.
142 throw std::runtime_error("Database connection pool initialization failure."); // Jump to the catch below, which exits the process.
143 }
144 }
145 }
146 catch (const std::exception& e) // Catch the runtime_error above (or any other setup failure).
147 {
148 std::string what = e.what(); // Copy the error text so it can be formatted into the log messages.
149 Message::CriticalMessage(fmt::format("Failed to initialize database connection pool: {}", what)); // Log the detailed reason.
150 Message::ErrorMessage("Fatal Error!\nExiting..."); // Tell the operator this is not recoverable.
151 exit(EXIT_FAILURE); /*NOLINT*/ // Terminate: a partially built pool would serve broken requests.
152 }
153}

155/**
156 * @brief Leases an idle connection, waiting at most the configured acquire timeout.
157 * @return A `shared_ptr` to be returned with return_connection() when work is done.
158 * @throws DatabaseBusy If no connection became free before the timeout expired.
159 * @details Measures the wait with a steady clock and emits at most one saturation warning
160 * per second using an atomic compare-exchange on the last-warning timestamp.
161 */
162std::shared_ptr<Database> DatabaseConnectionPool::get_connection()
163{
164 const auto timeout = acquireTimeout_; // Snapshot the timeout read at startup; it never changes at runtime.
165 const auto started = std::chrono::steady_clock::now(); // A monotonic start time for measuring the wait.
166
167 std::shared_ptr<Database> db_ptr = pool_.acquire(timeout); // Block until a connection is free or the timeout elapses.
168
169 const auto waited_ms = std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::steady_clock::now() - started).count(); // Convert the elapsed time to milliseconds for logging/comparison.
170
171 if (db_ptr == nullptr) // No connection became free in time.
172 {
173 Message::CriticalMessage(fmt::format("Timeout ({} ms) while waiting for a database connection", timeout.count())); // Log how long callers waited.
174 throw DatabaseBusy(); // The HTTP layer maps this exception to a retryable 503.
175 }
176
177 if (waited_ms >= SLOW_WAIT_WARNING_MS) // Anything slower than the threshold suggests pool saturation.
178 {
179 // At most one warning per second, so a saturated pool does not flood the log.
180 const std::int64_t now_ms = std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::steady_clock::now().time_since_epoch()).count(); // A stable wall-independent timestamp for rate limiting.
181 std::int64_t last = lastSlowWaitWarningMs_.load(); // Read the last warning time atomically.
182 if (now_ms - last >= 1000 && lastSlowWaitWarningMs_.compare_exchange_strong(last, now_ms)) // Only the thread that successfully updates the timestamp logs, at most once per second.
183 {
184 Message::WarningMessage(fmt::format("Waited {} ms for a database connection: the pool is saturated (see DB_MAX_CONN).", waited_ms)); // Point operators at the setting that likely needs raising.
185 }
186 }
187
188 return db_ptr; // Give the caller the leased connection; they must return it later.
189}

197/**
198 * @brief Reconnects all currently idle connections in parallel.
199 * @details Uses forEachIdle to hold the pool lock while the reconnect tasks run and until
200 * every task has been waited on, so no idle connection is handed out half-way
201 * through a reconnect. Busy (already leased) connections are not touched here.
202 * @note Each reconnect's success or failure (including exceptions) is logged; the method
203 * itself does not throw for an individual connection failure.
204 */
205void DatabaseConnectionPool::reconnect_all()
206{
207 const Configurator::DatabaseConfig& config = configurator_->get<Configurator::DatabaseConfig>(); // Read the configured pool size to reserve the future vector.
208
209 std::vector<std::future<void>> futures; // One future per idle connection being reconnected.
210 futures.reserve(config.max_conn); // Avoid reallocations while launching the reconnect tasks.
211
212 // The pool lock is held until every reconnect finished, so no idle connection can be handed out (and used) while it
213 // is being reconnected.
214 pool_.forEachIdle( // Holds the pool lock across both callbacks: nothing idle is leased during the repair.
215 [&futures](std::shared_ptr<Database>& connection) // Called once for each idle connection still in the pool.
216 {
217 futures.push_back(std::async(std::launch::async, // Start each reconnect on its own thread so all idle links are refreshed together.
218 [&connection]()
219 {
220 try // A failed reconnect of one link must not abort the others.
221 {
222 if (connection->reconnect()) // Ask the Database to rebuild its link from the cached string.
223 {
224 Message::InfoMessage(fmt::format("Database connection {} is re-established successfully", static_cast<void*>(connection.get()))); // Log success with the object address for tracing.
225 }
226 else
227 {
228 Message::ErrorMessage(fmt::format("Database connection {} failed to be re-established", static_cast<void*>(connection.get()))); // Log a non-throwing reconnect failure.
229 }
230 }
231 catch (const std::exception& e) // Catch unexpected driver exceptions so the loop continues.
232 {
233 Message::CriticalMessage(fmt::format("Reconnect exception: {}", e.what())); // Record the exception text for diagnosis.
234 }
235 }));
236 },
237 [&futures]()
238 {
239 for (auto& future : futures) // Visit every launched reconnect task.
240 {
241 future.wait(); // Block until it finishes, still under the pool lock, before any link can be leased.
242 }
243 });
