1/**
2 * @file databaseconnectionpool.cpp
3 * @brief Implementation for databaseconnectionpool: Opens one pqxx connection
4 * using the configured credentials and a connect timeout.
5 * @details This file implements the pool's lifecycle.
6 * createDatabaseConnection() opens and validates a single connection; the
7 * constructor spins up all slots in parallel with retries;
8 * get_connection()/return_connection() lease and return links, measuring waits
9 * and throttling saturation warnings; reconnect_all() repairs every idle link
10 * in parallel while holding the pool lock so nothing is handed out
11 * half-reconnected.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * This is where the server decides how many database connections exist and what
15 * happens when they are all busy. It translates configuration (DB_MAX_CONN,
16 * DB_ACQUIRE_TIMEOUT_MS) into real behaviour, maps a wait that expires to a
17 * DatabaseBusy exception (which the HTTP layer answers with 503), and
18 * cooperates with the watchdog to recover from database restarts. Without it
19 * there would be no connection reuse, no back-pressure, and no automatic
20 * recovery.
21 *
22 * CONCEPTS TO LEARN (for beginners):
23 * - std::async and std::future (run work in parallel and collect its result):
24 * https://en.cppreference.com/w/cpp/thread/async
25 * - std::future::get and wait: https://en.cppreference.com/w/cpp/thread/future
26 * - std::atomic::compare_exchange_strong (an atomic "update if unchanged"):
27 * https://en.cppreference.com/w/cpp/atomic/atomic/compare_exchange
28 * - std::chrono::steady_clock (a monotonic clock ideal for timeouts):
29 * https://en.cppreference.com/w/cpp/chrono/steady_clock
30 * - Exponential backoff (sleep longer after each failed attempt)
31 * - exit() and EXIT_FAILURE from the C standard library:
32 * https://en.cppreference.com/w/cpp/utility/program/exit
33 *
34 * @author Amr Osman <dr3mro@gmail.com>
35 * @date 2026-11-16
36 */

79/**
80 * Opens one pqxx connection using the configured credentials and a connect
81 * timeout. Returns a Database only when the connection is open; failures are
82 * logged rather than thrown so the caller can retry the whole slot. config_:
83 * Configuration holding host, dbname, user and password. Returns A validated
84 * Database, or `nullptr` on failure.
85 */
86std::shared_ptr<Database> DatabaseConnectionPool::createDatabaseConnection(
87 const auto& config_) {
88 // libpqxx throws on bad credentials or an unreachable server; the
89 // caller prefers a nullptr to an exception.
90 try {
91 auto conn = std::make_shared<pqxx::connection>( // Build a connection from
92 // a libpq-style
93 // keyword/value string.
94 fmt::format("host={} dbname={} user={} password={} connect_timeout={}",
95 config_.host,
96 config_.name,
97 config_.user,
98 config_.pass,
99 TIMEOUT)
100 .c_str()); // connect_timeout stops a dead server from blocking
101 // a worker forever.
102
103 // Only a successfully opened socket is worth
104 // wrapping.
105 if (conn->is_open()) {
106 return std::make_shared<Database>(
107 std::move(conn)); // Hand the connection to Database, which
108 // validates it with SELECT 1.
109 }
110
111 // Report a connection
112 // object that was created
113 // but not open.
114 // Turn any driver exception into a log entry plus nullptr.
115 Message::CriticalMessage("Failed to open database connection.");
116 } catch (const std::exception& e) {
117 Message::CriticalMessage(
118 fmt::format("Failed to create a Database connection: {}",
119 e.what())); // Include the driver's reason.
120 }
121 return nullptr; // Signal failure so the caller can retry this slot.
122}

124/**
125 * @brief Initialises the pool from the Configurator; failure terminates the
126 * process.
127 * @details Reads DB_MAX_CONN and the acquire timeout, warns when the pool is
128 * undersized, then starts all connections concurrently. Each slot retries with
129 * exponential backoff (MAX_RETRIES attempts); if any slot cannot be created, a
130 * runtime_error is raised and the catch block exits the process with
131 * EXIT_FAILURE.
132 */
133DatabaseConnectionPool::DatabaseConnectionPool()
134 // Resolve the shared configuration
135 // object before the body runs.
136 : configurator_(Store::getObject<Configurator>()) {
137 // Any setup failure is fatal, so it is funnelled into the exit path
138 // below.
139 try {
140 const Configurator::DatabaseConfig& config = configurator_->get<
141 Configurator::
142 DatabaseConfig>(); // Read the
143 // DB_MAX_CONN/acquire-timeout/credential
144 // settings once.
145 acquireTimeout_ = std::chrono::milliseconds(
146 config.acquire_timeout_ms); // Cache the timeout so get_connection()
147 // need not touch configuration later.
148
149 // Every server thread blocks on one connection while it queries, and
150 // the watchdog needs one more. A smaller pool makes requests queue (and
151 // time out with 503) even though nothing is slow.
152 const unsigned int server_threads =
153 configurator_->get<Configurator::ServerConfig>()
154 .threads; // Number of Drogon IO threads that may query at
155 // once.
156 // Need one slot per thread plus one spare for
157 // the watchdog's health checks.
158 if (config.max_conn < server_threads + 1U) {
159 Message::WarningMessage(
160 fmt::format("DB_MAX_CONN ({}) is below SERVER_THREADS + 1 ({}): "
161 "requests "
162 "may queue for a connection.",
163 config.max_conn,
164 server_threads +
165 1U)); // Warn early so an operator can raise
166 // the pool size before users see 503s.
167 }
168
169 std::vector<std::future<std::shared_ptr<Database>>>
170 futures; // One future per connection slot, used to collect the
171 // parallel results.
172
173 futures.reserve(config.max_conn); // Reserve once so push_back does not
174 // reallocate while tasks are launched.
175 for (uint16_t i = 0; i < config.max_conn;
176 // One iteration per configured pool slot.
177 ++i) {
178 // Start each connection attempt on
179 // its own thread so all slots open in
180 // parallel.
181 // Capture the pool and
182 // config by value; the
183 // task outlives this
184 // loop iteration.
185 futures.push_back(
186 std::async(std::launch::async,
187 [this, config]() -> std::shared_ptr<Database> {
188 unsigned int retryCount =
189 0; // Counts failed attempts for the
190 // backoff and the warning message.
191 // Stop after the configured
192 // number of attempts.
193 while (retryCount < MAX_RETRIES) {
194 auto conn = createDatabaseConnection(
195 config); // Try once to open and
196 // validate a connection.
197 // A non-null pointer means the slot is
198 // ready.
199 if (conn) {
200 return conn; // Hand the finished connection back
201 // to the constructor.
202 }
203 std::this_thread::sleep_for(std::chrono::seconds(
204 1U << retryCount)); // Exponential backoff: 1s,
205 // 2s, 4s... gives the DB time
206 // to become ready.
207 Message::WarningMessage(
208 fmt::format("Failed to create a database "
209 "connection. Attempt "
210 "{}/{}...",
211 ++retryCount,
212 MAX_RETRIES)); // Log the attempt
213 // number, then loop.
214 }
215 return nullptr; // All attempts failed for this slot.
216 }));
217 }
218
219 // Wait for all futures
220 // Walk the collected results in slot order.
221 for (auto& future : futures) {
222 auto conn =
223 future.get(); // Blocks only on completion of the individual task
224 // This slot was created successfully.
225 if (conn != nullptr) {
226 pool_.release(conn); // Put the new connection into the idle
227 // set so it can be leased at once.
228 Message::InitMessage(
229 fmt::format("Connection {}/{} created successfully.",
230 pool_.idle(),
231 config.max_conn)); // Report progress using the number
232 // now idle.
233 } else {
234 Message::ErrorMessage(
235 "Failed to establish one or more connections after maximum "
236 "retries."); // One bad slot is enough to abort startup.
237 // Jump to the catch below, which exits the process.
238 // Catch the runtime_error above (or any
239 // other setup failure).
240 throw std::runtime_error(
241 "Database connection pool initialization failure.");
242 }
243 }
244 } catch (const std::exception& e) {
245 std::string what = e.what(); // Copy the error text so it can be
246 // formatted into the log messages.
247 Message::CriticalMessage(
248 fmt::format("Failed to initialize database connection pool: {}",
249 what)); // Log the detailed reason.
250 Message::ErrorMessage(
251 "Fatal Error!\nExiting..."); // Tell the operator this is not
252 // recoverable.
253 exit(EXIT_FAILURE);
254 /*NOLINT*/ // Terminate: a partially built pool would serve broken
255 // requests.
256 }
257}

259/**
260 * @brief Leases an idle connection, waiting at most the configured acquire
261 * timeout.
262 * @return A `shared_ptr` to be returned with return_connection() when work is
263 * done.
264 * @throws DatabaseBusy If no connection became free before the timeout expired.
265 * @details Measures the wait with a steady clock and emits at most one
266 * saturation warning per second using an atomic compare-exchange on the
267 * last-warning timestamp.
268 */
269std::shared_ptr<Database> DatabaseConnectionPool::get_connection() {
270 const auto timeout =
271 acquireTimeout_; // Snapshot the timeout read at startup; it never
272 // changes at runtime.
273 const auto started =
274 std::chrono::steady_clock::now(); // A monotonic start time for
275 // measuring the wait.
276
277 std::shared_ptr<Database> db_ptr = pool_.acquire(
278 timeout); // Block until a connection is free or the timeout elapses.
279
280 const auto waited_ms = std::chrono::duration_cast<std::chrono::milliseconds>(
281 std::chrono::steady_clock::now() - started)
282 .count(); // Convert the elapsed time to
283 // milliseconds for logging/comparison.
284
285 // No connection became free in time.
286 if (db_ptr == nullptr) {
287 Message::CriticalMessage(
288 fmt::format("Timeout ({} ms) while waiting for a database connection",
289 timeout.count())); // Log how long callers waited.
290 throw DatabaseBusy(); // The HTTP layer maps this exception to a
291 // retryable 503.
292 }
293
294 // Anything slower than the threshold
295 // suggests pool saturation.
296 if (waited_ms >= SLOW_WAIT_WARNING_MS) {
297 // At most one warning per second, so a saturated pool does not flood
298 // the log.
299 const std::int64_t now_ms =
300 std::chrono::duration_cast<std::chrono::milliseconds>(
301 std::chrono::steady_clock::now().time_since_epoch())
302 .count(); // A stable wall-independent timestamp for rate
303 // limiting.
304 std::int64_t last = lastSlowWaitWarningMs_
305 .load(); // Read the last warning time atomically.
306 // Only the thread that
307 // successfully updates the
308 // timestamp logs, at most once
309 // per second.
310 if (now_ms - last >= 1000 &&
311 lastSlowWaitWarningMs_.compare_exchange_strong(last, now_ms)) {
312 Message::WarningMessage(
313 fmt::format("Waited {} ms for a database connection: the pool is "
314 "saturated "
315 "(see DB_MAX_CONN).",
316 waited_ms)); // Point operators at the setting that
317 // likely needs raising.
318 }
319 }
320
321 return db_ptr; // Give the caller the leased connection; they must return
322 // it later.
323}

334/**
335 * @brief Reconnects all currently idle connections in parallel.
336 * @details Uses forEachIdle to hold the pool lock while the reconnect tasks run
337 * and until every task has been waited on, so no idle connection is handed out
338 * half-way through a reconnect. Busy (already leased) connections are not
339 * touched here.
340 * @note Each reconnect's success or failure (including exceptions) is logged;
341 * the method itself does not throw for an individual connection failure.
342 */
343void DatabaseConnectionPool::reconnect_all() {
344 const Configurator::DatabaseConfig& config = configurator_->get<
345 Configurator::DatabaseConfig>(); // Read the configured pool size to
346 // reserve the future vector.
347
348 std::vector<std::future<void>>
349 futures; // One future per idle connection being reconnected.
350 futures.reserve(config.max_conn); // Avoid reallocations while launching
351 // the reconnect tasks.
352
353 // The pool lock is held until every reconnect finished, so no idle
354 // connection can be handed out (and used) while it is being reconnected.
355 // Holds the pool lock across both callbacks: nothing
356 // idle is leased during the repair.
357 // Called once for each idle connection still
358 // in the pool.
359 pool_.forEachIdle(
360 [&futures](std::shared_ptr<Database>& connection) {
361 futures.push_back(std::async(
362 std::launch::async, // Start each reconnect on its own thread
363 // so all idle links are refreshed
364 // together.
365 [&connection]() {
366 // A failed reconnect of one link must not abort the
367 // others.
368 try {
369 // Ask the Database to
370 // rebuild its link from
371 // the cached string.
372 if (connection->reconnect()) {
373 Message::InfoMessage(
374 fmt::format("Database connection {} is re-established "
375 "successfully",
376 static_cast<void*>(
377 connection
378 .get()))); // Log success with
379 // the object address
380 // for tracing.
381 } else {
382 Message::ErrorMessage(
383 fmt::format("Database connection {} failed to be "
384 // Log a non-throwing
385 // reconnect failure.
386 // Catch unexpected driver
387 // exceptions so the loop continues.
388 "re-established",
389 static_cast<void*>(connection.get())));
390 }
391 } catch (const std::exception& e) {
392 Message::CriticalMessage(
393 fmt::format("Reconnect exception: {}",
394 e.what())); // Record the exception text for
395 // diagnosis.
396 }
397 }));
398 },
399 [&futures]() {
400 // Visit every launched reconnect task.
401 for (auto& future : futures) {
402 future.wait(); // Block until it finishes, still under the pool
403 // lock, before any link can be leased.
404 }
405 });
