1/**
2 * @file elapsedtime.hpp
3 * @brief Timing middleware that logs how long a request took to produce its
4 * response.
5 * @details This file defines the ElapsedTime middleware, used in both the
6 * INSECURE and SECURE chains. When a request arrives it records a start time
7 * using a steady clock. It then passes the request down the chain and installs
8 * a callback that runs on the way back out, when the downstream handler has
9 * produced its response. That callback measures the elapsed milliseconds and
10 * writes a coloured log line containing the time, client IP, HTTP method,
11 * status code and path, then forwards the response unchanged.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * This middleware is the server's stopwatch. It is the only place that reports
15 * how long each endpoint actually took, which is how developers spot slow
16 * routes or database bottlenecks. It sits around the rest of the chain, so it
17 * measures the full cost of rate limiting, authentication and the handler. It
18 * changes nothing about the response; without it the server would still work
19 * but you would be blind to performance problems.
20 *
21 * CONCEPTS TO LEARN (for beginners):
22 * - HTTP middleware: https://drogon.docsforge.com/master/filter/
23 * - std::chrono steady clock (a clock that never jumps backwards):
24 * https://en.cppreference.com/w/cpp/chrono/steady_clock
25 * - Lambdas and captures: https://en.cppreference.com/w/cpp/language/lambda
26 * - CORS (unrelated here, but part of the HTTP filter world):
27 * https://developer.mozilla.org/en-US/docs/Web/HTTP/CORS
28 *
29 * @author Amr Osman <dr3mro@gmail.com>
30 * @date 2026-11-16
31 */

56 /**
57 * @class ElapsedTime
58 * @brief Timing middleware that logs how long a request took to produce its
59 * response.
60 * @details Part of both the INSECURE and SECURE chains. It records a
61 * steady_clock start, passes the request onward, and measures the interval
62 * in the return-path callback when the downstream handler's response
63 * propagates back out. The result is logged with request IP, method, status
64 * and path; the response is then forwarded unchanged. Stateless and safe
65 * for concurrent requests: all per-request state lives in the callback
66 * capture.
67 */
68 // Inherit Drogon's middleware template
69 // so it can be registered by name.
70 class ElapsedTime : public drogon::HttpMiddleware<ElapsedTime> {
71 public:
72 ElapsedTime() = default; // do not omit constructor: Drogon needs a default
73 // constructor to create the middleware.
74
75 /**
76 * @brief Start the timer and install a return-path callback that logs
77 * elapsed milliseconds.
78 * @details Invoked by Drogon on its IO thread. The callback passed to
79 * @p nextCb runs when the response comes back from later
80 * middleware/handlers, so the measurement covers the whole downstream
81 * chain. The response is forwarded untouched to @p mcb.
82 * @param[in] req Request whose peer address, method and path are
83 * included in the log line.
84 * @param[in,out] nextCb Chain callback given the measuring wrapper.
85 * @param[in,out] mcb Middleware callback that receives the final
86 * response.
87 */
88 // Drogon calls invoke to run this middleware.
89 void invoke(const drogon::HttpRequestPtr& req,
90 drogon::MiddlewareNextCallback&& nextCb,
91 drogon::MiddlewareCallback&& mcb) override {
92 // Store the start time for this request
93 std::chrono::steady_clock::time_point start_time =
94 std::chrono::steady_clock::now(); // Snapshot the monotonic
95 // clock; captured by the
96 // callback below.
97
98 // will run while getting out from this middleware
99 // Continue down the chain, giving it our
100 // measuring callback to call on the way back.
101 // Capture the final callback, the start time and
102 // the request for later.
103 std::move(nextCb)([mcb = std::move(mcb), start_time, req](
104 const drogon::HttpResponsePtr& resp) {
105 // Calculate the elapsed time for this request
106 auto end_time =
107 std::chrono::steady_clock::now(); // Take a second clock
108 // reading when the response
109 // returns.
110 auto elapsed_ms = std::chrono::duration_cast<std::chrono::milliseconds>(
111 end_time - start_time)
112 .count(); // Difference converted to whole
113 // milliseconds for the log.
114 // Log the elapsed time
115 LOG_INFO << fmt::format(
116 ": {}{}{}{}{}",
117 fmt::styled(fmt::format("[{} ms] ", elapsed_ms),
118 fmt::fg(fmt::color::orange)), // Build one line:
119 // elapsed time in
120 // orange.
121 fmt::styled(fmt::format("[{}]", req->getPeerAddr().toIp()),
122 fmt::fg(fmt::color::yellow)), // Client IP in yellow.
123 fmt::styled(fmt::format("[{}]", req->getMethodString()),
124 fmt::fg(fmt::color::red)), // HTTP method in red.
125 fmt::styled(fmt::format("[{}]",
126 static_cast<int>(resp->getStatusCode())),
127 fmt::fg(fmt::color::magenta)), // Response status
128 // in magenta.
129 fmt::styled(fmt::format("[{}]", req->path()),
130 fmt::fg(fmt::color::cyan))); // Request path in cyan,
131 // then write to the log.
132 std::move(mcb)(resp); // Forward the unchanged response to
133 // the previous middleware/route.
134 });
135 }
136 };
