1/**
2 * @file entity.hpp
3 * @brief Short names for the CRUD value types and the variant they live in (see api::v2::Types).
4 * @details This header defines the Entity class, the workhorse that turns a requested operation
5 * (create, read, update, delete or search) into a safe SQL statement. It first introduces short
6 * aliases for the value types defined in types.hpp so callers do not have to write the long
7 * namespace over and over. Entity stores one of those values plus a table name, and exposes five
8 * builders - one per operation - that produce SQL with $1, $2, ... placeholders.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * Entity is the concrete implementation behind the Base interface. Request controllers construct
12 * an Entity for the operation they need and call getSql*Statement() to obtain SQL that the database
13 * layer executes. More specialised entities (Client, Service, Case, Appointment) inherit from this
14 * class and override only what they must. Without it, every controller would have to know how to
15 * spell CREATE/READ/UPDATE/DELETE/SEARCH SQL for every table.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - std::variant: https://en.cppreference.com/w/cpp/utility/variant
19 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
20 * - SQL parameter binding: put $1, $2, ... markers in the SQL and pass the real values separately;
21 * see https://en.cppreference.com/w/cpp or the libpqxx documentation.
22 * - A "type alias" (using X = Y) gives a long type a short local name; no new type is created.
23 *
24 * @author Amr Osman <dr3mro@gmail.com>
25 * @date 2026-11-16
26 */

59/**
60 * @class Entity
61 * @brief Concrete Base implementation that turns a CRUD value type into SQL statements.
62 * @details An Entity pairs one api::v2::Types::EntityType alternative with the database table
63 * (also used as the permission "group") it belongs to. The five Base builders are thin adapters
64 * that create a fresh Binder and delegate to the build* helpers below; each helper owns the
65 * details of one statement shape. Instances are created for each request and are immutable after
66 * construction (both the variant and the table name are const), except for the id cached by
67 * check_id_exists(). All parameter values go through Binder::bind, so they never appear in the
68 * SQL text.
69 * @note Neither copy nor move is allowed; construct the entity in place for the request that uses it.
70 * @see Base, api::v2::Types::EntityType
71 * @note For a beginner: the five getSql*Statement functions are the public "adapters"; each makes
72 * a Binder and calls one private-ish build* helper. Splitting them this way keeps the public API
73 * small while letting each SQL shape be written and read on its own.
74 */

174 /**
175 * @brief Asks the database whether the id carried by the payload already exists in this table.
176 * @tparam T The variant alternative whose get_id() should be consulted.
177 * @return false when the payload has no id or the database reports no matching row; otherwise
178 * the database result. Database failures collapse to false via value_or(false).
179 * @note Logs an error message when no id is present. Uses the shared DatabaseController.
180 * @throws std::bad_variant_access if T is not the alternative currently held by the variant.

182 template <typename T>
183 bool check_id_exists() // Templated so each payload type can be asked for its id uniformly.
184 {
185 std::optional<uint64_t> _id = std::get<T>(getData()).get_id(); // Pull the id out of the chosen variant alternative; it may be absent.
186 if (!_id.has_value()) // No id means we cannot check anything useful.
187 {
188 Message::ErrorMessage(fmt::format("Failed to create Sql search statement for table {}.", tablename)); // Log why the check is abandoned.

192 return result.value_or(false); // value_or turns a missing/failed result into false rather than throwing.

197 /**
198 * @brief Builds the INSERT statement from a Create_t payload.
199 * @details Iterates the JSON object range, binding every value and using the (sorted) keys as
200 * column names. An explicit id from the caller is inserted; otherwise "id" is erased so the
201 * database assigns it, which also prevents a client from choosing its own primary key.
202 * @tparam Binder Binder-like type providing bind().
203 * @param[in,out] binder Appends values and returns their placeholders.
204 * @return INSERT ... RETURNING *, or std::nullopt if an exception occurs.
205 */
206 template <typename Binder>
207 std::optional<std::string> buildCreateStatement(Binder& binder) // The helper is a template so it works with Binder or a test double.
208 {
209 try // Everything below can throw (JSON access, formatting); one catch keeps callers safe.
210 {
211 std::vector<std::string> keys_arr; // Column names collected from the JSON object.
212 std::vector<std::string> values_arr; // Matching placeholders ($1, $2, ...) in the same order.
213
214 jsoncons::json data_json = std::get<Create_t>(data).get_data(); // Take a copy of the payload so we can edit it without changing the entity.
215 // With an id given (e.g. a patient's health record shares the patient id) it is inserted. Without one the
216 // column is left out, so the database assigns it, and any "id" the client sent is dropped.
217 if (const auto create_id = std::get<Create_t>(data).get_id(); create_id.has_value()) // If the caller supplied an explicit primary key...
218 {
219 data_json["id"] = create_id.value(); // Keep it so the INSERT writes that exact id.
220 }
221 else
222 {
223 data_json.erase("id"); // Otherwise remove any client-sent id so the database generates one (a client must not pick its own key).
224 }
225
226 for (auto& iterator : data_json.object_range()) // Walk each JSON member; object_range may not preserve order, so keys and values are collected in lockstep.
227 {
228 keys_arr.push_back(iterator.key()); // The JSON key becomes a column name.
229 values_arr.push_back(binder.bind(iterator.value().template as<std::string>())); // Bind the value as text and store the returned placeholder.
230 }
231
232 return fmt::format("INSERT INTO {} ({}) VALUES ({}) RETURNING *;", tablename, fmt::join(keys_arr, ","), fmt::join(values_arr, ",")); // RETURNING * gives all new columns back to the caller.
233 }
234 catch (const std::exception& e) // Catch any error from JSON access or formatting.
235 {
236 Message::ErrorMessage(fmt::format("Failed to create Sql create statement for table {}.", tablename)); // Log which table failed.
237 Message::CriticalMessage(e.what()); // Log the low-level reason for debugging.
238 return std::nullopt; // Signal failure instead of returning half-built SQL.
239 }
240 }

242 /**
243 * @brief Builds a single-row SELECT against the table's _safe view.
244 * @details An optional schema set limits the selected columns; an empty set selects *. The
245 * id is bound. A missing id is a caller error: an error is logged and no statement is returned.
246 * @tparam Binder Binder-like type providing bind().
247 * @param[in,out] binder Appends the id and returns its placeholder.
248 * @return SELECT ... LIMIT 1, or std::nullopt when no id is available or an exception occurs.
249 */
250 template <typename Binder>
251 std::optional<std::string> buildReadStatement(Binder& binder) // Builds the "read one row" SELECT.
252 {
253 try
254 {
255 auto user_id = std::get<Read_t>(data).get_id(); // The id to look up (may be absent).
256 auto schema = std::get<Read_t>(data).get_data(); // The set of column names the caller wants back.
257
258 std::string columns = schema.empty() ? "*" : fmt::format("{}", fmt::join(schema, ", ")); // No requested columns means "all columns"; otherwise join the names with commas.
259
260 if (!user_id.has_value()) // Reading without an id would return the wrong row, so refuse.
261 {
262 Message::ErrorMessage(fmt::format("Failed to read data from table {}. No id provided.", tablename)); // Log the caller mistake.
263 return std::nullopt; // No statement: the caller turns this into an error response.
264 }
265 return fmt::format("SELECT {} FROM {}_safe WHERE id={} LIMIT 1;", columns, tablename, binder.bind(user_id.value())); // Read from the "_safe" view (a filtered view), bind the id, and LIMIT 1 because ids are unique.
266 }
267 catch (const std::exception& e)
268 {
269 Message::ErrorMessage(fmt::format("Failed to create Sql read statement for table {}.", tablename)); // Log the failing table.
270 Message::CriticalMessage(e.what()); // Log the exception details.
271 return std::nullopt; // Report failure without throwing at the caller.
272 }
273 }

275 /**
276 * @brief Builds the UPDATE statement from an Update_t payload.
277 * @details Each JSON member becomes one bound "column = $n" SET clause and the id is bound
278 * as the final parameter. A missing id aborts the build rather than producing an unscoped UPDATE.
279 * @tparam Binder Binder-like type providing bind().
280 * @param[in,out] binder Appends SET values and the id, returning their placeholders.
281 * @return UPDATE ... RETURNING *, or std::nullopt when no id is available or an exception occurs.
282 */
283 template <typename Binder>
284 std::optional<std::string> buildUpdateStatement(Binder& binder) // Builds the UPDATE ... SET ... WHERE id=... statement.
285 {
286 try
287 {
288 jsoncons::json payload = std::get<Update_t>(data).get_data(); // The fields to change, as JSON.
289
290 auto _id = std::get<Update_t>(data).get_id(); // The row id from the payload.
291
292 if (!_id.has_value()) // An UPDATE without an id would change every row - a serious bug - so stop here.
293 {
294 Message::ErrorMessage(fmt::format("Failed to update data from table {}. No id provided.", tablename)); // Log the missing id.
295 return std::nullopt; // No statement is safer than an unscoped UPDATE.
296 }
297
298 std::vector<std::string> set_clauses; // One "column = $n" string per field to change.
299
300 for (auto it = payload.object_range().begin(); it != payload.object_range().end(); ++it) // Walk every JSON member to build its SET clause.
301 {
302 set_clauses.push_back(fmt::format("{} = {}", it->key(), binder.bind(it->value().template as<std::string>()))); // Column name is text from the validated payload; the value is bound safely.
303 }
304
305 return fmt::format("UPDATE {} SET {} WHERE id={} RETURNING *;", tablename, fmt::join(set_clauses, ", "), binder.bind(_id.value())); // Bind the id LAST so its placeholder number follows the SET values.
306 }
307 catch (const std::exception& e)
308 {
309 Message::ErrorMessage(fmt::format("Failed to create Sql update statement for table {}.", tablename)); // Log the failing table.
310 Message::CriticalMessage(e.what()); // Log details.
311 return std::nullopt; // Signal failure.
312 }
313 }

343 /**
344 * @brief Builds the paginated ILIKE search statement for the table's _safe view.
345 * @details The keyword is wrapped in %...% for substring matching and bound together with a
346 * limit of (requested limit + 1) and the offset; the extra row lets callers detect a further
347 * page. filter, order_by and direction are SQL identifiers and are written into the text, so
348 * they must already have been validated by Search_t.
349 * @tparam Binder Binder-like type providing bind().
350 * @param[in,out] binder Appends keyword, limit and offset and returns their placeholders.
351 * @return The SELECT text, or std::nullopt if an exception occurs.
352 * @note Bindings are made into named locals first because argument evaluation order is
353 * unspecified; the placeholder numbering must follow a fixed order.
354 */
355 template <typename Binder>
356 std::optional<std::string> buildSearchStatement(Binder& binder) // Builds the paginated, case-insensitive search SELECT.
357 {
358 try
359 {
360 Search_t searchdata = std::get<Search_t>(getData()); // The parsed and validated search parameters.
361 // Bound in a fixed order (argument evaluation order is unspecified).
362 const std::string keyword_ph = binder.bind(fmt::format("%{}%", searchdata.keyword)); // Wrap the keyword in % so ILIKE matches anywhere; bind it as $1.
363 const std::string limit_ph = binder.bind(searchdata.limit + 1); // Ask for one extra row to detect another page; bound as $2.
364 const std::string offset_ph = binder.bind(searchdata.offset); // Number of rows to skip; bound as $3.
365 return fmt::format(
366 "SELECT * FROM {}_safe WHERE {}::text ILIKE {} ORDER BY {} {} "
367 "LIMIT {} OFFSET {};",
368 tablename, searchdata.filter, keyword_ph, searchdata.order_by, searchdata.direction, limit_ph, offset_ph); // filter/order_by/direction are pre-validated identifiers written into the text; values use $n placeholders.
369 }
370 catch (const std::exception& e)
371 {
372 Message::ErrorMessage(fmt::format("Failed to create Sql search statement for table {}.", tablename)); // Log the failing table.
373 Message::CriticalMessage(e.what()); // Log details.
374 return std::nullopt; // Signal failure.
375 }
376 }
