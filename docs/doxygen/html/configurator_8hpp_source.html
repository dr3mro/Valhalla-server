1/**
2 * @file configurator.hpp
3 * @brief Environment-driven configuration sections and typed accessors for the server.
4 * @details This header defines Configurator, a container that owns one settings object for
5 * each part of the server. Every settings object reads its values from environment variables
6 * when it is created, using hard-coded defaults from the Defaults namespace when a variable
7 * is missing or invalid. Callers retrieve a settings object by asking for its type, for
8 * example get<Configurator::ServerConfig>(). Most values are logged on startup with secrets
9 * hidden, and a few counts are automatically sized from the machine's CPU count.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * This is the single home for ALL runtime settings. The Store owns one Configurator and builds
13 * it before the Drogon IO threads start; then the server, database, gatekeeper and worker
14 * components pull their settings from it through get<T>(). Nothing else reads environment
15 * variables directly for these options. Without this file, settings would be scattered and
16 * inconsistent, and there would be no central place to print or override them. Note that
17 * values are captured once at construction, so changing the environment later has no effect.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - Environment variables: https://man7.org/linux/man-pages/man7/environ.7.html
21 * - C++ concepts and constraints: https://en.cppreference.com/w/cpp/language/constraints
22 * - Function templates: https://en.cppreference.com/w/cpp/language/function_template
23 * - std::chrono durations (lengths of time): https://en.cppreference.com/w/cpp/chrono/duration
24 * - std::unordered_set (a collection of unique values): https://en.cppreference.com/w/cpp/container/unordered_set
25 *
26 * @author Amr Osman <dr3mro@gmail.com>
27 * @date 2026-11-16
28 */

57/**
58 * @class Configurator
59 * @brief Owning aggregate of all runtime configuration sections.
60 *
61 * @details Holds one instance per configuration domain (server, database, DoS detection,
62 * tokens, frontend, email sender, caches and workers) plus a testing section. Every section
63 * derives from EnvLoader and reads its values from the process environment in its
64 * constructor, falling back to the constants in the @ref Defaults namespace when a variable
65 * is absent or invalid. The Configurator is intended to be a single process-wide object
66 * owned by Store and fully constructed before the Drogon IO threads start; sections are
67 * retrieved by type through get<T>().
68 *
69 * @note Values are snapshotted at construction time, so later changes to the environment do
70 * not affect an already-built Configurator.
71 * @warning get() returns a non-const reference and performs no locking. Treat sections as
72 * read-only once construction is complete; unsynchronized concurrent mutation is unsafe.
73 * @see EnvLoader, Defaults, Store
74 */

104 /**
105 * @struct DOSDetectorConfig
106 * @brief Thresholds and durations used by the DoS detector / rate limiter.
107 * @details All fields are read from the environment in the constructor; @ref
108 * Defaults::DosDetector supplies the fallbacks. Durations are whole seconds and the
109 * whitelist/blacklist are comma-separated sets of client fingerprints.
110 */
111 using DOSDetectorConfig = struct DOSDetectorConfig : public EnvLoader // Declares a struct and an alias of the same name, inheriting the env helpers.
112 {
113 /** @brief Maximum requests accepted within one @ref period before limiting applies. */
114 uint32_t max_requests; // How many requests one client may make per period. 32-bit unsigned is plenty and cannot be negative.
115 /** @brief Interval (seconds) between sweeps of expired rate-limit/ban entries. */
116 uint32_t clean_freq; // Seconds between cleanups; unsigned because a negative interval is meaningless.
117 /** @brief Upper bound on the number of distinct client fingerprints retained. */
118 uint32_t max_fingerprints; // Caps how many unique clients are tracked, limiting memory use.
119 /** @brief Length (seconds) of the sliding counting window for @ref max_requests. */
120 std::chrono::seconds period; // A duration type makes the unit explicit; the detector counts requests over this window.
121 /** @brief Duration (seconds) a client stays rate-limited. */
122 std::chrono::seconds ratelimit_duration; // How long excess requests are refused before the client is allowed again.
123 /** @brief Duration (seconds) a banned client stays blocked. */
124 std::chrono::seconds ban_duration; // Longer penalty than rate limiting; used for serious abusers.
125 /** @brief Fingerprints exempt from limiting; parsed from a comma-separated value. */
126 std::unordered_set<std::string> whitelist; // A set gives fast "is this client exempt?" checks and ignores duplicates.
127 /** @brief Fingerprints always subject to limiting; parsed from a comma-separated value. */
128 std::unordered_set<std::string> blacklist; // Clients here are always limited; a set keeps membership checks quick.
129
130 /**
131 * @brief Reads each DOS limit from its environment variable, using Defaults on absence
132 * or parse failure.
133 * @note Duration variables use EnvLoader's seconds overload, which falls back to the
134 * default for negative or unparsable values.
135 */
136 DOSDetectorConfig() // Initialises every field by reading its variable, with a Defaults fallback.
137 : max_requests(getEnvironmentVariable("MAX_REQUESTS", Defaults::DosDetector::MAX_REQUESTS_)), // MAX_REQUESTS overrides the request cap.
138 clean_freq(getEnvironmentVariable("CLN_FRQ", Defaults::DosDetector::CLN_FRQ_)), // CLN_FRQ = cleanup frequency in seconds.
139 max_fingerprints(getEnvironmentVariable("MAX_FPS", Defaults::DosDetector::MAX_FPS_)), // MAX_FPS caps tracked client fingerprints.
140 period(getEnvironmentVariable("PERIOD", std::chrono::seconds(Defaults::DosDetector::PERIOD_))), // PERIOD is wrapped as seconds so it matches the field type.
141 ratelimit_duration(getEnvironmentVariable("RL_DURATION", std::chrono::seconds(Defaults::DosDetector::RL_DURATION_))), // RL_DURATION = rate-limit length in seconds.
142 ban_duration(getEnvironmentVariable("BAN_DURATION", std::chrono::seconds(Defaults::DosDetector::BAN_DURATION_))), // BAN_DURATION = ban length in seconds.
143 whitelist(getEnvironmentVariable("WHITELIST")), // Comma-separated list of exempt fingerprints; no default means an empty set.
144 blacklist(getEnvironmentVariable("BLACKLIST")) // Comma-separated list of always-limited fingerprints; empty when unset.
145 {
146 }

148 /** @brief Logs all DOS limits and both fingerprint sets. */
149 void printValues() const override // Required by EnvLoader; prints this section's effective values.
150 {
151 Message::ConfMessage("------------------------------------------------"); // Decorative separator line for readable output.
152 Message::ConfMessage("---------------DOS Detector Config--------------"); // Section title.
153 Message::ConfMessage("------------------------------------------------"); // Decorative separator line.
154 Message::ConfMessage(fmt::format("Max Requests: {}", max_requests)); // Show the per-window request cap.
155 Message::ConfMessage(fmt::format("Period: {} seconds", period.count())); // .count() returns the raw number of seconds for printing.
156 Message::ConfMessage(fmt::format("Max FPS: {}", max_fingerprints)); // Show the fingerprint-tracking limit.
157 Message::ConfMessage(fmt::format("Rate Limit Duration: {} seconds", ratelimit_duration.count())); // Show how long rate limiting lasts.
158 Message::ConfMessage(fmt::format("Ban Duration: {} seconds", ban_duration.count())); // Show how long a ban lasts.
159 Message::ConfMessage(fmt::format("Cleanup Frequency: {} seconds", clean_freq)); // Show how often expired entries are swept.
160 Message::ConfMessage(fmt::format("Whitelist: {}", fmt::join(whitelist, ", "))); // fmt::join prints the whole set as one comma-separated string.
161 Message::ConfMessage(fmt::format("Blacklist: {}", fmt::join(blacklist, ", "))); // Same for the blacklist.
162 }
