1/**
2 * @file requester.hpp
3 * @brief Caller identity used by the permission layer.
4 * @details Defines the Requester value object, which carries two pieces of
5 * information about who made an API request: an opaque numeric id and
6 * a group name. It offers simple getters, setters and matching helpers such as
7 * isAdmin() or idMatch(). A GroupId enum lists known groups, but the actual
8 * checks compare group names as plain strings.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * The gatekeeper and permission manager need to know who is calling before they
12 * allow an action. Controllers copy a Requester into the work they dispatch to
13 * other threads, so the checks still see the original caller's identity. The
14 * isX helpers (isAdmin, isProvider, ...) let permission rules read like plain
15 * English. Without this object there would be no consistent way to carry caller
16 * identity from the HTTP layer down to the permission checks.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - Classes and access specifiers (public/private):
20 * https://en.cppreference.com/w/cpp/language/access
21 * - special member functions (= default / = delete):
22 * https://en.cppreference.com/w/cpp/language/member_functions#Special_member_functions
23 * - [[nodiscard]] attribute:
24 * https://en.cppreference.com/w/cpp/language/attributes/nodiscard
25 *
26 * @author Amr Osman <dr3mro@gmail.com>
27 * @date 2026-11-16
28 */

39 /**
40 * @struct Requester
41 * @brief Value object describing who issued a request.
42 * @details Holds an opaque numeric id and a group name. It is copied into
43 * lambdas dispatched to worker threads and consulted by the
44 * permission manager, so instances are cheap value types with
45 * defaulted copy/move operations.
46 * @note Group checks are literal string comparisons against hardcoded names
47 * such as "admins", "users", "providers", "staff" and "owners"
48 * (isSuper matches "Super" with a capital S). The GroupId enumeration
49 * is declared but is not used to validate or store `group`.
50 */
51 // A named struct exposed through a
52 // "using" alias so both spellings work.
53 using Requester = struct Requester {
54 private:
55 /**
56 * @brief Opaque identity of the requester, compared by idMatch.
57 */
58 uint64_t id; // The caller's numeric identity; a fixed 64-bit unsigned
59 // type gives a wide, portable range.
60 /**
61 * @brief Group name, compared literally by groupMatch and the isX
62 * helpers.
63 */
64 std::string group; // The caller's group as text, e.g. "admins";
65 // compared exactly, so spelling/case matter.
66
67 public:
68 /**
69 * @enum GroupId
70 * @brief Named requester groups.
71 * @details Values are Users, Providers, Owners, Admin, Super and Staff.
72 * This enumerator is informational: group membership is stored
73 * and checked as a string, not as a GroupId.
74 */
75 // A scoped enum (accessed as
76 // GroupId::Users) stored compactly as
77 // 8-bit integers.
78 enum class GroupId : uint8_t {
79 Users, // A normal user account.
80 Providers, // A healthcare provider account.
81 Owners, // An owner account.
82 Admin, // An administrator account.
83 Super, // A super-user account (note: the check uses the string
84 // "Super", capital S).
85 Staff // A staff account.
86 };
