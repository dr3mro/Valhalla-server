1/**
2 * @file communicate.cpp
3 * @brief Implementation for communicate.
4 * @details Implements Communicate::sendRequest, which allocates a cURL handle
5 * and header list, builds the URL, chooses GET or POST through a small lookup
6 * table of lambdas, performs the transfer and cleans up.
7 * handleResponse registers WriteCallback so incoming bytes are appended to a
8 * string, and WriteCallback does that append. Failures are logged and returned
9 * as empty optionals instead of thrown exceptions.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * This is where the actual outbound HTTP work happens. Callers get a simple
13 * "send this request, give me the body" interface while all the messy libcurl
14 * setup, method selection, resource cleanup and error logging stay here.
15 * Because every failure is converted to std::nullopt, callers do not need
16 * try/catch blocks of their own. Without this file the service-to-service
17 * calling code would have to be repeated and carefully cleaned up at every call
18 * site.
19 *
20 * CONCEPTS TO LEARN (for beginners):
21 * - libcurl easy API: https://curl.se/libcurl/c/curl_easy_init.html
22 * - std::function:
23 * https://en.cppreference.com/w/cpp/utility/functional/function
24 * - Lambdas with captures ([&curl, &data]):
25 * https://en.cppreference.com/w/cpp/language/lambda
26 * - std::optional::value_or:
27 * https://en.cppreference.com/w/cpp/utility/optional/value_or
28 *
29 * @author Amr Osman <dr3mro@gmail.com>
30 * @date 2026-11-16
31 */

44// Returns the response body, or nothing on any
45// failure.
46// Address, verb and body of the request to send.
47std::optional<std::string> Communicate::sendRequest(
48 const std::string& server,
49 const int port,
50 const std::string& path,
51 const drogon::HttpMethod& method,
52 const std::string& data) {
53 // Both libcurl resources are allocated before the try block so a single
54 // cleanup path in the catch covers failures at any later stage.
55 CURL* curl = nullptr; // The cURL handle; starts null so cleanup code can
56 // tell whether it was created.
57 struct curl_slist* headers =
58 nullptr; // Linked list of custom HTTP headers; starts empty.
59 headers =
60 curl_slist_append(headers,
61 "Content-Type: application/json"); // Adds one header
62 // saying the body
63 // is JSON.
64 std::string response; // Will hold the body received from the server.
65 std::string url; // Will hold the assembled "server:port/path" address.
66
67 try {
68 curl = curl_easy_init(); // Creates a fresh libcurl handle for this one
69 // request.
70 // libcurl can fail to initialise; check before
71 // using the handle.
72 if (curl == nullptr) {
73 throw std::runtime_error(
74 "Failed to initialize cURL."); // Turn the failure into an
75 // exception for the single
76 // catch block.
77 }
78
79 // Using fmt::format to construct the URL string
80 url = fmt::format("{}:{}{}",
81 server,
82 port,
83 path); // Joins server, port and path into a URL like
84 // "host:8080/api".
85 curl_easy_setopt(curl,
86 CURLOPT_URL,
87 url.c_str()); // Tells cURL which URL to request; c_str
88 // gives the C-style string it needs.
89 curl_easy_setopt(curl,
90 CURLOPT_HTTPHEADER,
91 headers); // Attaches the JSON content-type header to the
92 // request.
93
94 // Map of methods to their respective cURL options
95 // Lambda for POST: captures curl and data by
96 // reference so it can configure them.
97 const std::map<drogon::HttpMethod, std::function<void()>> methodMap =
98 {{drogon::HttpMethod::Post,
99 [&curl, &data]() {
100 curl_easy_setopt(curl,
101 CURLOPT_POST,
102 1L); // Mark this as a POST request.
103 curl_easy_setopt(curl,
104 CURLOPT_POSTFIELDS,
105 data.c_str()); // Attach the request body; data
106 // must outlive the call.
107 }},
108 {drogon::HttpMethod::Get, [&curl]() {
109 curl_easy_setopt(curl, CURLOPT_HTTPGET, 1L);
110 }}}; // Lambda for GET: mark it as a GET request (no body).
111
112 auto method_opt =
113 methodMap.find(method); // Look up the requested verb in the map.
114 // Only call the lambda if the verb was actually found.
115 if (method_opt != methodMap.end()) {
116 method_opt->second(); // Call the appropriate method
117 } else {
118 throw std::invalid_argument(
119 "Unsupported HTTP method: " +
120 drogon::to_string(method)); // Reject verbs other than GET/POST.
121 }
122
123 // Handle response and potential errors. An empty body is treated as
124 // failure even when the transfer itself reported success.
125 response = handleResponse(curl).value_or(
126 ""); // Run the transfer; value_or turns an empty optional into an
127 // empty string.
128 // An empty body means failure even if cURL
129 // itself reported success.
130 if (response.empty()) {
131 throw std::runtime_error(
132 "Failed to get a valid response."); // Escalate to the catch
133 // block for cleanup and
134 // logging.
135 }
136
137 // Release resources on the success path; the catch block covers the
138 // failure path.
139 curl_easy_cleanup(
140 curl); // Free the cURL handle now that the transfer is done.
141 // Free the linked list of custom headers.
142 // Catches every exception thrown above,
143 // including from handleResponse's callers.
144 curl_slist_free_all(headers);
145 } catch (const std::exception& e) {
146 Message::ErrorMessage(fmt::format("Failed to send request to {}.",
147 url)); // Log which URL failed.
148 Message::CriticalMessage(
149 e.what()); // Log the underlying error text at critical level.
150
151 // Only free the handle if it was actually created.
152 if (curl != nullptr) {
153 curl_easy_cleanup(curl); // Free the cURL handle on the failure path too.
154 }
155 curl_slist_free_all(headers); // Always free the header list.
156 return std::nullopt; // Signal failure to the caller without throwing.
157 }
158
159 return response; // Success: hand the response body back to the caller.
160}
161// Performs the transfer configured on curl and returns the body.
162std::optional<std::string> Communicate::handleResponse(CURL* curl) {
163 std::string response; // Accumulates the bytes received from the server.
164
165 try {
166 curl_easy_setopt(curl,
167 CURLOPT_WRITEFUNCTION,
168 WriteCallback); // Register the function libcurl should
169 // call with incoming data.
170 curl_easy_setopt(curl,
171 CURLOPT_WRITEDATA,
172 &response); // Pass our string as the user data given to
173 // that callback.
174
175 CURLcode res = curl_easy_perform(
176 curl); // Actually send the request; this blocks until it finishes.
177 // Any non-OK cURL code means the transfer failed.
178 if (res != CURLE_OK) {
179 // Convert the code to a
180 // readable message and throw.
181 // Catches transfer errors so callers never see an exception.
182 throw std::runtime_error("cURL request failed: " +
183 std::string(curl_easy_strerror(res)));
184 }
185 } catch (const std::exception& e) {
186 Message::ErrorMessage(fmt::format(
187 "Failed to handle request.")); // Log a generic failure notice.
188 Message::CriticalMessage(e.what()); // Log the specific error text.
189 response.clear(); // Clear the response to indicate failure
190 }
191
192 return response; // Returns the body (possibly emptied by the catch block).
193}

195// Called by libcurl for each chunk of received data.
196std::optional<size_t> Communicate::WriteCallback(void* contents,
197 size_t size,
198 size_t nmemb,
199 std::string* response) {
200 size_t totalSize = size * nmemb; // libcurl passes item size and count;
201 // their product is the byte count.
202 response->append(static_cast<char*>(contents),
203 totalSize); // Append those bytes to the receiving string.
204 return totalSize; // Report how many bytes were handled; libcurl expects
205 // this exact count.
206}
