1/**
2 * @file communicate.cpp
3 * @brief Implementation for communicate.
4 * @details Implements Communicate::sendRequest, which allocates a cURL handle and
5 * header list, builds the URL, chooses GET or POST through a small lookup
6 * table of lambdas, performs the transfer and cleans up. handleResponse
7 * registers WriteCallback so incoming bytes are appended to a string,
8 * and WriteCallback does that append. Failures are logged and returned as
9 * empty optionals instead of thrown exceptions.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * This is where the actual outbound HTTP work happens. Callers get a simple
13 * "send this request, give me the body" interface while all the messy libcurl
14 * setup, method selection, resource cleanup and error logging stay here. Because
15 * every failure is converted to std::nullopt, callers do not need try/catch blocks
16 * of their own. Without this file the service-to-service calling code would have
17 * to be repeated and carefully cleaned up at every call site.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - libcurl easy API: https://curl.se/libcurl/c/curl_easy_init.html
21 * - std::function: https://en.cppreference.com/w/cpp/utility/functional/function
22 * - Lambdas with captures ([&curl, &data]): https://en.cppreference.com/w/cpp/language/lambda
23 * - std::optional::value_or: https://en.cppreference.com/w/cpp/utility/optional/value_or
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

38std::optional<std::string> Communicate::sendRequest( // Returns the response body, or nothing on any failure.
39 const std::string& server, const int port, const std::string& path, const drogon::HttpMethod& method, const std::string& data) // Address, verb and body of the request to send.
40{
41 // Both libcurl resources are allocated before the try block so a single
42 // cleanup path in the catch covers failures at any later stage.
43 CURL* curl = nullptr; // The cURL handle; starts null so cleanup code can tell whether it was created.
44 struct curl_slist* headers = nullptr; // Linked list of custom HTTP headers; starts empty.
45 headers = curl_slist_append(headers, "Content-Type: application/json"); // Adds one header saying the body is JSON.
46 std::string response; // Will hold the body received from the server.
47 std::string url; // Will hold the assembled "server:port/path" address.
48
49 try
50 {
51 curl = curl_easy_init(); // Creates a fresh libcurl handle for this one request.
52 if (curl == nullptr) // libcurl can fail to initialise; check before using the handle.
53 {
54 throw std::runtime_error("Failed to initialize cURL."); // Turn the failure into an exception for the single catch block.
55 }
56
57 // Using fmt::format to construct the URL string
58 url = fmt::format("{}:{}{}", server, port, path); // Joins server, port and path into a URL like "host:8080/api".
59 curl_easy_setopt(curl, CURLOPT_URL, url.c_str()); // Tells cURL which URL to request; c_str gives the C-style string it needs.
60 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers); // Attaches the JSON content-type header to the request.
61
62 // Map of methods to their respective cURL options
63 const std::map<drogon::HttpMethod, std::function<void()>> methodMap = {{drogon::HttpMethod::Post,
64 [&curl, &data]() // Lambda for POST: captures curl and data by reference so it can configure them.
65 {
66 curl_easy_setopt(curl, CURLOPT_POST, 1L); // Mark this as a POST request.
67 curl_easy_setopt(curl, CURLOPT_POSTFIELDS, data.c_str()); // Attach the request body; data must outlive the call.
68 }},
69 {drogon::HttpMethod::Get, [&curl]() { curl_easy_setopt(curl, CURLOPT_HTTPGET, 1L); }}}; // Lambda for GET: mark it as a GET request (no body).
70
71 auto method_opt = methodMap.find(method); // Look up the requested verb in the map.
72 if (method_opt != methodMap.end()) // Only call the lambda if the verb was actually found.
73 {
74 method_opt->second(); // Call the appropriate method
75 }
76 else
77 {
78 throw std::invalid_argument("Unsupported HTTP method: " + drogon::to_string(method)); // Reject verbs other than GET/POST.
79 }
80
81 // Handle response and potential errors. An empty body is treated as
82 // failure even when the transfer itself reported success.
83 response = handleResponse(curl).value_or(""); // Run the transfer; value_or turns an empty optional into an empty string.
84 if (response.empty()) // An empty body means failure even if cURL itself reported success.
85 {
86 throw std::runtime_error("Failed to get a valid response."); // Escalate to the catch block for cleanup and logging.
87 }
88
89 // Release resources on the success path; the catch block covers the
90 // failure path.
91 curl_easy_cleanup(curl); // Free the cURL handle now that the transfer is done.
92 curl_slist_free_all(headers); // Free the linked list of custom headers.
93 }
94 catch (const std::exception& e) // Catches every exception thrown above, including from handleResponse's callers.
95 {
96 Message::ErrorMessage(fmt::format("Failed to send request to {}.", url)); // Log which URL failed.
97 Message::CriticalMessage(e.what()); // Log the underlying error text at critical level.
98
99 if (curl != nullptr) // Only free the handle if it was actually created.
100 {
101 curl_easy_cleanup(curl); // Free the cURL handle on the failure path too.
102 }
103 curl_slist_free_all(headers); // Always free the header list.
104 return std::nullopt; // Signal failure to the caller without throwing.
105 }
106
107 return response; // Success: hand the response body back to the caller.
108}
109std::optional<std::string> Communicate::handleResponse(CURL* curl) // Performs the transfer configured on curl and returns the body.
110{
111 std::string response; // Accumulates the bytes received from the server.
112
113 try
114 {
115 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, WriteCallback); // Register the function libcurl should call with incoming data.
116 curl_easy_setopt(curl, CURLOPT_WRITEDATA, &response); // Pass our string as the user data given to that callback.
117
118 CURLcode res = curl_easy_perform(curl); // Actually send the request; this blocks until it finishes.
119 if (res != CURLE_OK) // Any non-OK cURL code means the transfer failed.
120 {
121 throw std::runtime_error("cURL request failed: " + std::string(curl_easy_strerror(res))); // Convert the code to a readable message and throw.
122 }
123 }
124 catch (const std::exception& e) // Catches transfer errors so callers never see an exception.
125 {
126 Message::ErrorMessage(fmt::format("Failed to handle request.")); // Log a generic failure notice.
127 Message::CriticalMessage(e.what()); // Log the specific error text.
128 response.clear(); // Clear the response to indicate failure
129 }
130
131 return response; // Returns the body (possibly emptied by the catch block).
132}

134std::optional<size_t> Communicate::WriteCallback(void* contents, size_t size, size_t nmemb, std::string* response) // Called by libcurl for each chunk of received data.
135{
136 size_t totalSize = size * nmemb; // libcurl passes item size and count; their product is the byte count.
137 response->append(static_cast<char*>(contents), totalSize); // Append those bytes to the receiving string.
138 return totalSize; // Report how many bytes were handled; libcurl expects this exact count.
139}
