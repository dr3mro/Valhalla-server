1/**
2 * @file ratelimit.hpp
3 * @brief Drogon filter that screens requests for DoS/rate-limit violations before authentication.
4 * @details This file defines the RateLimit filter, the first element of both the INSECURE and SECURE
5 * chains. It collects facts about each request (IP address, HTTP method, path, a hash of the
6 * headers and the body), bundles them into a DOSDetector::Request and asks GateKeeper for a
7 * verdict. Allowed and whitelisted requests continue to the next filter; blacklisted, rate
8 * limited, banned or internally-failed requests are stopped immediately with an HTTP error.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This filter is the server's first line of defence. Because Drogon runs it before the Auth filter and
12 * before any handler, abusive clients are rejected cheaply without spending time on login checks or
13 * database queries. It calls GateKeeper, which forwards to DOSDetector; without it the API would have
14 * no protection against floods of requests and every client would reach the expensive code paths.
15 *
16 * CONCEPTS TO LEARN (for beginners):
17 * - HTTP filters/middleware: https://drogon.docsforge.com/master/filter/
18 * - Denial-of-service (DoS) and rate limiting: limiting how many requests one client may make.
19 * - HTTP 429 Too Many Requests: https://developer.mozilla.org/en-US/docs/Web/HTTP/Status/429
20 * - std::shared_ptr: https://en.cppreference.com/w/cpp/memory/shared_ptr
21 *
22 * @author Amr Osman <dr3mro@gmail.com>
23 * @date 2026-11-16
24 */

45 /**
46 * @class RateLimit
47 * @brief Drogon filter that screens requests for DoS/rate-limit violations before authentication.
48 * @details First element of both the INSECURE and SECURE chains. It builds a DOSDetector::Request
49 * from the peer IP, method, path, hashed headers and body, and asks GateKeeper for a
50 * verdict. ALLOWED/WHITELISTED continue the chain; blacklist, rate limit, ban and internal
51 * error short-circuit with a corresponding HTTP status. The GateKeeper and DOSDetector
52 * are shared service-locator objects held for the filter's lifetime.
53 * @note Unknown verdicts fall through to continuing the chain (default branch).
54 */
55 class RateLimit : public drogon::HttpFilter<RateLimit> // Inherit Drogon's filter template so it can be registered by name.
56 {
57 public:
58 /** @brief Obtain the shared GateKeeper and DOSDetector from the service locator. */
59 RateLimit() // Constructor fetches the two shared objects once, when the filter is created.
60 : gateKeeper(Store::getObject<GateKeeper>()), dos_detector(Store::getObject<DOSDetector>()) // Look up both singletons from the Store.
61 {
62 }
63 /**
64 * @brief Classify the request and continue the chain or send a rejection response.
65 * @details Invoked by Drogon on its IO thread; verdicts map as: BLACKLISTED -> 403,
66 * RATELIMITED -> 429, BANNED -> 503, ERROR -> 500. ALLOWED/WHITELISTED and any
67 * unrecognised status invoke @p fccb; rejected requests invoke @p fcb only.
68 * @param[in] req Request supplying peer IP, method, path, headers and body for classification.
69 * @param[in,out] fcb Filter callback used to send the rejection response.
70 * @param[in,out] fccb Chain callback invoked when the request is permitted.
71 */
72 void doFilter(const drogon::HttpRequestPtr& req, drogon::FilterCallback&& fcb, drogon::FilterChainCallback&& fccb) override // Drogon calls this on every request passing through this filter.
73 {
74 DOSDetector::Request request = {.ip = req->peerAddr().toIp(), // Designated initializers fill each field by name: caller IP address.
75 .method = req->methodString(), // HTTP verb as text (GET, POST, ...).
76 .path = req->path(), // URL path being requested.
77 .header_hash = DOSDetector::hashHeaders(req->getHeaders()), // Condense all headers into one number to spot repeated patterns cheaply.
78 .body = req->body()}; // Raw request body, used by the detector's heuristics.
79
80 // Ask the gatekeeper for a verdict; the detector details are encapsulated there.
81 DOSDetector::Status status = gateKeeper->isDosAttack(request); // GateKeeper forwards to DOSDetector and returns an enum verdict.
82
83 // Permit verdicts continue the chain; every rejection sends one response and stops here.
84 switch (status) // Map each verdict to "continue" or a specific error reply.
85 {
86 case DOSDetector::Status::ALLOWED: // Normal, permitted traffic.
87 case DOSDetector::Status::WHITELISTED: // Explicitly trusted caller.
88 fccb(); // Continue to the next filter/handler.
89 break; // Leave the switch.
90 case DOSDetector::Status::BLACKLISTED: // Caller is on the deny list.
91 reply("IP is blacklisted.", drogon::k403Forbidden, std::move(fcb)); // Send 403 Forbidden.
92 break; // Leave the switch.
93 case DOSDetector::Status::RATELIMITED: // Caller sent too many requests.
94 reply("Too many requests.", drogon::k429TooManyRequests, std::move(fcb)); // Send 429 with a short message.
95 break; // Leave the switch.
96 case DOSDetector::Status::BANNED: // Caller is temporarily banned.
97 reply("IP is banned.", drogon::k503ServiceUnavailable, std::move(fcb)); // Send 503 Service Unavailable.
98 break; // Leave the switch.
99 case DOSDetector::Status::ERROR: // The detector itself failed.
100 reply("Server error.", drogon::k500InternalServerError, std::move(fcb)); // Send 500 Internal Server Error.
101 break; // Leave the switch.
102 default: // Any future/unknown verdict.
103 fccb(); // Be permissive by default so new codes do not break all traffic.
104 break; // Leave the switch.
105 }
106 }
107
108 private:
109 std::shared_ptr<GateKeeper> gateKeeper; /**< Used by doFilter via isDosAttack(). */
110 std::shared_ptr<DOSDetector> dos_detector; /**< Held for its lifetime only; doFilter never dereferences it. */
111 /**
112 * @brief Send a small plain-text rejection response with the given status.
113 * @param[in] msg Response body text.
114 * @param[in] code HTTP status code to set.
115 * @param[in] cb Filter callback invoked with the response.
116 */
117 void reply(const std::string&& msg, drogon::HttpStatusCode code, const drogon::FilterCallback&& cb) // Small helper to avoid repeating response-building code.
118 {
119 auto resp = drogon::HttpResponse::newHttpResponse(); // Build an empty HTTP reply object.
120 resp->setBody(msg); // Put the short message in the body (plain text, not JSON).
121 resp->setStatusCode(code); // Set the chosen HTTP status.
122 cb(resp); // Send the response, which stops the chain (the chain callback is never called).
123 }
124 };
