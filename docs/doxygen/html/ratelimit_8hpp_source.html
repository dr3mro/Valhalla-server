1/**
2 * @file ratelimit.hpp
3 * @brief Drogon filter that screens requests for DoS/rate-limit violations
4 * before authentication.
5 * @details This file defines the RateLimit filter, the first element of both
6 * the INSECURE and SECURE chains. It collects facts about each request (IP
7 * address, HTTP method, path, a hash of the headers and the body), bundles them
8 * into a DOSDetector::Request and asks GateKeeper for a verdict. Allowed and
9 * whitelisted requests continue to the next filter; blacklisted, rate limited,
10 * banned or internally-failed requests are stopped immediately with an HTTP
11 * error.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * This filter is the server's first line of defence. Because Drogon runs it
15 * before the Auth filter and before any handler, abusive clients are rejected
16 * cheaply without spending time on login checks or database queries. It calls
17 * GateKeeper, which forwards to DOSDetector; without it the API would have no
18 * protection against floods of requests and every client would reach the
19 * expensive code paths.
20 *
21 * CONCEPTS TO LEARN (for beginners):
22 * - HTTP filters/middleware: https://drogon.docsforge.com/master/filter/
23 * - Denial-of-service (DoS) and rate limiting: limiting how many requests one
24 * client may make.
25 * - HTTP 429 Too Many Requests:
26 * https://developer.mozilla.org/en-US/docs/Web/HTTP/Status/429
27 * - std::shared_ptr: https://en.cppreference.com/w/cpp/memory/shared_ptr
28 *
29 * @author Amr Osman <dr3mro@gmail.com>
30 * @date 2026-11-16
31 */

60 /**
61 * @class RateLimit
62 * @brief Drogon filter that screens requests for DoS/rate-limit violations
63 * before authentication.
64 * @details First element of both the INSECURE and SECURE chains. It builds
65 * a DOSDetector::Request from the peer IP, method, path, hashed headers and
66 * body, and asks GateKeeper for a verdict. ALLOWED/WHITELISTED continue the
67 * chain; blacklist, rate limit, ban and internal error short-circuit with a
68 * corresponding HTTP status. The GateKeeper and DOSDetector are shared
69 * service-locator objects held for the filter's lifetime.
70 * @note Unknown verdicts fall through to continuing the chain (default
71 * branch).
72 */
73 class RateLimit
74 // Inherit Drogon's filter
75 // template so it can be
76 // registered by name.
77 : public drogon::HttpFilter<RateLimit> {
78 public:
79 /** @brief Obtain the shared GateKeeper and DOSDetector from the service
80 * locator. */
81 RateLimit() // Constructor fetches the two shared objects once, when
82 // the filter is created.
83 // Look up both singletons
84 // from the Store.
85 : gateKeeper(Store::getObject<GateKeeper>()),
86 dos_detector(Store::getObject<DOSDetector>()) {
87 }
88 /**
89 * @brief Classify the request and continue the chain or send a
90 * rejection response.
91 * @details Invoked by Drogon on its IO thread; verdicts map as:
92 * BLACKLISTED -> 403, RATELIMITED -> 429, BANNED -> 503, ERROR -> 500.
93 * ALLOWED/WHITELISTED and any unrecognised status invoke @p fccb;
94 * rejected requests invoke @p fcb only.
95 * @param[in] req Request supplying peer IP, method, path, headers and
96 * body for classification.
97 * @param[in,out] fcb Filter callback used to send the rejection
98 * response.
99 * @param[in,out] fccb Chain callback invoked when the request is
100 * permitted.
101 */
102 // Drogon calls this on every request passing through this
103 // filter.
104 void doFilter(const drogon::HttpRequestPtr& req,
105 drogon::FilterCallback&& fcb,
106 drogon::FilterChainCallback&& fccb) override {
107 DOSDetector::Request request =
108 {.ip = req->peerAddr().toIp(), // Designated initializers fill each
109 // field by name: caller IP address.
110 .method =
111 req->methodString(), // HTTP verb as text (GET, POST, ...).
112 .path = req->path(), // URL path being requested.
113 .header_hash = DOSDetector::hashHeaders(
114 req->getHeaders()), // Condense all headers into one number
115 // to spot repeated patterns cheaply.
116 .body = req->body()}; // Raw request body, used by the
117 // detector's heuristics.
118
119 // Ask the gatekeeper for a verdict; the detector details are
120 // encapsulated there.
121 DOSDetector::Status status = gateKeeper->isDosAttack(
122 request); // GateKeeper forwards to DOSDetector and returns an
123 // enum verdict.
124
125 // Permit verdicts continue the chain; every rejection sends one
126 // response and stops here.
127 // Map each verdict to "continue" or a specific
128 // error reply.
129 switch (status) {
130 case DOSDetector::Status::ALLOWED: // Normal, permitted
131 // traffic.
132 case DOSDetector::Status::WHITELISTED: // Explicitly trusted
133 // caller.
134 fccb(); // Continue to the next filter/handler.
135 break; // Leave the switch.
136 case DOSDetector::Status::BLACKLISTED: // Caller is on the deny
137 // list.
138 reply("IP is blacklisted.",
139 drogon::k403Forbidden,
140 std::move(fcb)); // Send 403 Forbidden.
141 break; // Leave the switch.
142 case DOSDetector::Status::RATELIMITED: // Caller sent too many
143 // requests.
144 reply("Too many requests.",
145 drogon::k429TooManyRequests,
146 std::move(fcb)); // Send 429 with a short message.
147 break; // Leave the switch.
148 case DOSDetector::Status::BANNED: // Caller is temporarily
149 // banned.
150 reply("IP is banned.",
151 drogon::k503ServiceUnavailable,
152 std::move(fcb)); // Send 503 Service Unavailable.
153 break; // Leave the switch.
154 case DOSDetector::Status::ERROR: // The detector itself failed.
155 reply("Server error.",
156 drogon::k500InternalServerError,
157 std::move(fcb)); // Send 500 Internal Server Error.
158 break; // Leave the switch.
159 default: // Any future/unknown verdict.
160 fccb(); // Be permissive by default so new codes do not
161 // break all traffic.
162 break; // Leave the switch.
163 }
164 }
165
166 private:
167 std::shared_ptr<GateKeeper>
168 gateKeeper; /**< Used by doFilter via isDosAttack(). */
169 std::shared_ptr<DOSDetector>
170 dos_detector; /**< Held for its lifetime only; doFilter never
171 dereferences it. */
172 /**
173 * @brief Send a small plain-text rejection response with the given
174 * status.
175 * @param[in] msg Response body text.
176 * @param[in] code HTTP status code to set.
177 * @param[in] cb Filter callback invoked with the response.
178 */
179 // Small helper to avoid repeating response-building code.
180 void reply(const std::string&& msg,
181 drogon::HttpStatusCode code,
182 const drogon::FilterCallback&& cb) {
183 auto resp =
184 drogon::HttpResponse::newHttpResponse(); // Build an empty HTTP
185 // reply object.
186 resp->setBody(msg); // Put the short message in the body (plain
187 // text, not JSON).
188 resp->setStatusCode(code); // Set the chosen HTTP status.
189 cb(resp); // Send the response, which stops the chain (the chain
190 // callback is never called).
191 }
192 };
