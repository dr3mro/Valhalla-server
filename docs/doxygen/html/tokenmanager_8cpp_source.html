32 * @brief Signs a JWT from the client's identity and writes it back to @p clientLoginData.
33 * @return True on success; false when signing throws (reported as a critical message).
34 * @see TokenManager::generateToken
35 */
36bool TokenManager::generateToken(std::optional<Types::ClientLoginData>& clientLoginData) // Signs a new token and stores it in clientLoginData; true on success.
37{
38 try // Signing or reading optional claims can throw, so guard it.
39 {
40 auto tokenManagerParameters = getTokenManagerParameters(); // Fetch issuer, type, validity and secret from the base class's configuration.
41 // Create JWT token with payload
42 std::optional<std::string> token = // Will hold the signed token if creation succeeds.
43 jwt::create<jwt::traits::kazuho_picojson>() // Start building a JWT using the chosen JSON traits.
44 .set_issuer(std::string(tokenManagerParameters.issuer)) // Record who issued the token.
45 .set_type(std::string(tokenManagerParameters.type)) // Record the token type.
46 .set_subject(clientLoginData->username.value()) // The subject is the client's username.
47 .set_id(std::to_string(clientLoginData->clientId.value())) // The token id is the numeric client id as text.
48 .set_issued_at(std::chrono::system_clock::now()) // Stamp the current time as the issue time.
49 .set_expires_at(std::chrono::system_clock::now() + std::chrono::minutes{tokenManagerParameters.validity}) // Set expiry a configured number of minutes from now.
50 .set_payload_claim("ip_address", jwt::basic_claim<jwt::traits::kazuho_picojson>(clientLoginData->ip_address.value())) // Embed the client's IP address as a custom claim.
51 .set_payload_claim("llodt", jwt::basic_claim<jwt::traits::kazuho_picojson>(clientLoginData->lastLogoutTime.value())) // Embed the last logout time so a later logout can be detected.
52 .set_payload_claim("group", jwt::basic_claim<jwt::traits::kazuho_picojson>(clientLoginData->group.value())) // Embed the client's group (user or provider).
53 .sign(jwt::algorithm::hs256{std::string(tokenManagerParameters.secret)}); // Sign with the shared secret using HS256 so the token cannot be forged.
54
55 if (token.has_value()) // Only store a token that was actually produced.
56 {
57 clientLoginData->token = token.value(); // Write the signed token back for the caller to send to the client.
58 return true; // Token generation succeeded.
59 }
60 }
61 catch (const std::exception& e) // Runs when a claim is missing or signing fails.
62 {
63 Message::CriticalMessage(fmt::format("Error generating token:{}", e.what())); // Log a serious error with the exception text.
64 }
65 return false; // Token generation failed.
66}
