135 try {
136 jwt::decoded_jwt<jwt::traits::kazuho_picojson> decodedToken =
137 jwt::decode<jwt::traits::kazuho_picojson>(

139 // a structured object.
140 // First check that the claims can be read

143 message =

145 message); // Add context to the decode failure reason.
146 return false; // Token generation failed.

149 // Token is valid
150 // Then verify the signature and
