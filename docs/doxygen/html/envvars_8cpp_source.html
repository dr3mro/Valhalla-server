1/**
2 * @file envvars.cpp
3 * @brief Process-environment get/set wrappers used by configuration sections.
4 * @details This file contains the two small functions declared in envvars.hpp.
5 * One reads a name/value pair from the operating system's environment; the
6 * other writes (or overwrites) one. Both operate on plain text and leave any
7 * type conversion to the class above them. Errors while writing are reported
8 * and swallowed rather than thrown.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This is the lowest level of the configuration stack: the only code that
12 * touches the C library functions std::getenv and setenv directly. EnvLoader
13 * calls get() every time a configuration section is built, so this file is what
14 * actually makes environment-driven configuration possible. If it were missing,
15 * no other file could discover the deployment's settings, and the server could
16 * only rely on hard-coded defaults.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - Environment variables: https://man7.org/linux/man-pages/man7/environ.7.html
20 * - std::getenv / std::setenv:
21 * https://en.cppreference.com/w/cpp/utility/program/getenv
22 * - std::optional (a value that may be missing):
23 * https://en.cppreference.com/w/cpp/utility/optional
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

43// Returns the raw value or an empty optional when
44// the variable is not set.
45std::optional<std::string> EnvVars::get(const std::string& key) {
46 // Retrieve the value associated with the environment variable key
47 const char* _key = key.c_str(); // The C API needs a null-terminated char
48 // array, so convert the C++ string to one.
49 const char* value =
50 std::getenv(_key); // Ask the OS for the value; it returns nullptr when
51 // the name does not exist.
52
53 // If the value exists, copy it into a std::string that outlives this call
54 // A non-null pointer means the variable is present;
55 // otherwise it is absent.
56 if (value != nullptr) {
57 return std::string(
58 value); // Copy the C string into a C++ string so the caller owns
59 // the memory after getenv's buffer may change.
60 }
61
62 // Otherwise, return an empty optional
63 return std::nullopt; // Signal "not set" explicitly, which is clearer than
64 // returning an empty string.
65}

67// Stores value under key; used by tests to fake
68// settings, not on the request path.
69void EnvVars::set(const std::string& key, const std::string& value) {
70 // Overwrite = 1 so repeated calls replace the existing value rather than
71 // being ignored.
72 // The OS returns 0 on success; any other result means the write
73 // failed (bad name, no memory, etc.).
74 if (setenv(key.c_str(), value.c_str(), 1) != 0) {
75 Message::ErrorMessage(
76 fmt::format("Failed to set environment variable {}.",
77 key)); // Log the failure with the variable name; do not
78 // throw so callers keep running.
79 }
80}
