1/**
2 * @file banner.hpp
3 * @brief Header-only helpers for printing the startup banner, configuration and ASCII logo.
4 *
5 * @details This header defines the Banner class, a small set of static functions that print the startup information to
6 * the terminal. The functions print a labelled value, a decorative separator line, the project's ASCII-art
7 * logo in a random colour, and the logo plus a final flush. Everything lives in the header, so no separate
8 * .cpp file or linking step is needed to use it.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This file is the server's "face" at startup. Server::print_banner() calls these helpers to show the effective
12 * configuration, version, port, thread count and API documentation address before Drogon begins handling requests.
13 * The banner is useful for confirming that environment variables were read correctly and for spotting a wrong port or
14 * database setting immediately. Without it the server would still run, but operators would have no quick on-screen
15 * summary of which settings took effect.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - Pseudo-random number generation (random_device, mt19937, distributions): https://en.cppreference.com/w/cpp/numeric/random
19 * - Raw string literals (R"(...)"): https://en.cppreference.com/w/cpp/language/string_literal
20 * - Variadic templates and parameter packs (Args&&...): https://en.cppreference.com/w/cpp/language/parameter_pack
21 * - The fmt formatting library: https://fmt.dev/latest/index.html
22 *
23 * @author Amr Osman <dr3mro@gmail.com>
24 * @date 2026-11-16
25 */

100 /**
101 * @brief Prints one label/value line with the label and value coloured independently.
102 *
103 * @tparam T Type of VALUE; must be formattable by fmt.
104 * @param[in] LABEL Left-hand label, padded to a fixed width.
105 * @param[in] VALUE Value rendered with fmt::format("{}", VALUE).
106 * @param[in] CLR1 Colour of the label.
107 * @param[in] CLR2 Colour of the value.
108 */
109 template <typename T> // T is the type of the value; it only needs to be printable by fmt.
110 static void print(const std::string& LABEL, const T& VALUE, fmt::color CLR1, fmt::color CLR2) // Print a label/value pair with a colour for each part.
111 {
112 fmt::print(fmt::emphasis::bold | fg(CLR1), "{:12} : ", LABEL); // Print the bold, coloured label padded to a fixed width.
113 fmt::print(fg(CLR2), "{} \n", fmt::format("{}", VALUE)); // Print the value in its own colour, then move to the next line.
114 }

116 /**
117 * @brief Prints a bold red separator line used to delimit banner sections.
118 *
119 * @tparam Args Extra arguments forwarded to fmt::print; call sites normally pass none.
120 * @param[in] args Arguments forwarded to fmt::print after the separator format string.
121 */
122 template <typename... Args> // A parameter pack lets callers optionally supply extra values for fmt.
123 static void print_line(Args&&... args) // Print a bold red separator between banner sections.
124 {
125 fmt::print(fmt::emphasis::bold | fg(fmt::color::red), "{}\n\n", // Print in bold red, followed by two newlines for spacing.
126 "______________________________________________________________________________________", std::forward<Args>(args)...); // Separator line.
127 }
