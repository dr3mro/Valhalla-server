1/**
2 * @file banner.hpp
3 * @brief Header-only helpers for printing the startup banner, configuration and
4 * ASCII logo.
5 *
6 * @details This header defines the Banner class, a small set of static
7 * functions that print the startup information to the terminal. The functions
8 * print a labelled value, a decorative separator line, the project's ASCII-art
9 * logo in a random colour, and the logo plus a final flush. Everything
10 * lives in the header, so no separate .cpp file or linking step is needed to
11 * use it.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * This file is the server's "face" at startup. Server::print_banner() calls
15 * these helpers to show the effective configuration, version, port, thread
16 * count and API documentation address before Drogon begins handling requests.
17 * The banner is useful for confirming that environment variables were read
18 * correctly and for spotting a wrong port or database setting immediately.
19 * Without it the server would still run, but operators would have no quick
20 * on-screen summary of which settings took effect.
21 *
22 * CONCEPTS TO LEARN (for beginners):
23 * - Pseudo-random number generation (random_device, mt19937, distributions):
24 * https://en.cppreference.com/w/cpp/numeric/random
25 * - Raw string literals (R"(...)"):
26 * https://en.cppreference.com/w/cpp/language/string_literal
27 * - Variadic templates and parameter packs (Args&&...):
28 * https://en.cppreference.com/w/cpp/language/parameter_pack
29 * - The fmt formatting library: https://fmt.dev/latest/index.html
30 *
31 * @author Amr Osman <dr3mro@gmail.com>
32 * @date 2026-11-16
33 */

244 /**
245 * @brief Prints one label/value line with the label and value coloured
246 * independently.
247 *
248 * @tparam T Type of VALUE; must be formattable by fmt.
249 * @param[in] LABEL Left-hand label, padded to a fixed width.
250 * @param[in] VALUE Value rendered with fmt::format("{}", VALUE).
251 * @param[in] CLR1 Colour of the label.
252 * @param[in] CLR2 Colour of the value.
253 */
254 // T is the type of the value; it only needs to be
255 // printable by fmt.
256 // Print a label/value pair
257 // with a colour for each part.
258 template <typename T>
259 static void print(const std::string& LABEL,
260 const T& VALUE,
261 fmt::color CLR1,
262 fmt::color CLR2) {
263 fmt::print(fmt::emphasis::bold | fg(CLR1),
264 "{:12} : ",
265 LABEL); // Print the bold, coloured label padded to a fixed
266 // width.
267 fmt::print(fg(CLR2),
268 "{} \n",
269 fmt::format("{}",
270 VALUE)); // Print the value in its own colour, then
271 // move to the next line.
272 }

274 /**
275 * @brief Prints a bold red separator line used to delimit banner sections.
276 *
277 * @tparam Args Extra arguments forwarded to fmt::print; call sites normally
278 * pass none.
279 * @param[in] args Arguments forwarded to fmt::print after the separator
280 * format string.
281 */
282 // A parameter pack lets callers optionally
283 // supply extra values for fmt.
284 // Print a bold red
285 // separator between
286 // banner sections.
287 template <typename... Args>
288 static void print_line(Args&&... args) {
289 fmt::print(fmt::emphasis::bold | fg(fmt::color::red),
290 "{}\n\n", // Print in bold red, followed by two newlines for
291 // spacing.
292 "_______________________________________________________________"
293 "___"
294 "____________________",
295 std::forward<Args>(args)...); // Separator line.
296 }
