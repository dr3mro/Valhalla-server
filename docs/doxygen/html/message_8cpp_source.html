1/**
2 * @file message.cpp
3 * @brief Implementation for message.
4 * @details Implements the Message facade. The five severity helpers forward to a
5 * private MessageImpl that writes a coloured, single-line record through
6 * trantor::Logger. The two startup helpers use a second MessageImpl that
7 * prints directly to stdout. get_color holds the severity-to-colour map,
8 * and removeNewLine flattens embedded newlines.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This is the implementation behind nearly every log line the server produces.
12 * Code throughout the project calls Message::InfoMessage/ErrorMessage/etc.; those
13 * calls end up here, get coloured and cleaned, and flow into the configured
14 * logger. Keeping the formatting in one place means all log output looks the same
15 * and startup banners are clearly distinguished from runtime records.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - Function overloading: https://en.cppreference.com/w/cpp/language/overload_resolution
19 * - std::unordered_map: https://en.cppreference.com/w/cpp/container/unordered_map
20 * - std::ranges algorithms (std::ranges::replace): https://en.cppreference.com/w/cpp/algorithm/ranges/replace
21 *
22 * @author Amr Osman <dr3mro@gmail.com>
23 * @date 2026-11-16
24 */

46std::string Message::get_color(trantor::Logger::LogLevel level) // Takes the severity and returns the matching ANSI escape sequence.
47{
48 static const std::unordered_map<trantor::Logger::LogLevel, std::string> color_map = { // A read-only map built once, pairing each level with its colour.
49 {trantor::Logger::LogLevel::kTrace, "\033[38;5;140m"}, // Soft Purple (for subtle debug messages)
50 {trantor::Logger::LogLevel::kDebug, "\033[38;5;140m"}, // Soft Purple (for subtle debug messages)
51 {trantor::Logger::LogLevel::kInfo, "\033[1;34m"}, // Bold Blue (clear and calm for info)
52 {trantor::Logger::LogLevel::kWarn, "\033[1;93m"}, // Bright Yellow (attention-grabbing for warnings)
53 {trantor::Logger::LogLevel::kError, "\033[1;91m"}, // Bright Red (strong and clear for errors)
54 {trantor::Logger::LogLevel::kFatal, "\033[38;5;196m"} // messages)
55 };
56 return color_map.at(level); // .at looks up the level and throws std::out_of_range if it is somehow missing.
57}

59void Message::MessageImpl(const std::string& _status_message, trantor::Logger::LogLevel _level) // Implementation shared by the severity helpers.
60{
61 // Route through Trantor so these records reach the configured spdlog sinks.
62 trantor::Logger(_level).stream() << fmt::format("{}{}{}", get_color(_level), removeNewLine(_status_message), "\033[0m"); // Builds "colour + cleaned text + reset" and streams it into the trantor log at the chosen level.
63}
64void Message::MessageImpl(const std::string& _status_message, const std::string& _type) /*NOLINT */ // Implementation shared by the two startup notices.
65{
66 // Startup notices bypass the logger and go straight to stdout with a fixed prefix.
67 fmt::print("{}{}{}{}{}\n", // Prints five coloured pieces followed by a newline.
68 fmt::format(fg(fmt::color::indian_red), // Start the first piece in red.
69 "[Project Valhalla]"), // The server name in red
70 fmt::format(fg(fmt::color::white), " ["), // the left square bracket
71 fmt::format("{}", fmt::format(fg(fmt::color::green), "{}",
72 _type)), // The message type in green
73 fmt::format(fg(fmt::color::white), "] "), // the right square bracket
74 removeNewLine(_status_message)); // The message
75}

77std::string Message::removeNewLine(const std::string& _str) // Takes text that might span lines.
78{
79 // Keep each record on a single line so log prefixes stay meaningful.
80 std::string status_message = _str; // Work on a copy so the caller's original string is untouched.
81 std::ranges::replace(status_message, '\n', ' '); // Replace every newline with a space to flatten multi-line messages.
82 return status_message; // Returns the single-line version for logging/printing.
83}
