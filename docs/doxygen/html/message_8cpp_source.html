1/**
2 * @file message.cpp
3 * @brief Implementation for message.
4 * @details Implements the Message facade. The five severity helpers forward to
5 * a private MessageImpl that writes a coloured, single-line record through
6 * trantor::Logger. The two startup helpers use a second MessageImpl
7 * that prints directly to stdout. get_color holds the severity-to-colour map,
8 * and removeNewLine flattens embedded newlines.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This is the implementation behind nearly every log line the server produces.
12 * Code throughout the project calls Message::InfoMessage/ErrorMessage/etc.;
13 * those calls end up here, get coloured and cleaned, and flow into the
14 * configured logger. Keeping the formatting in one place means all log output
15 * looks the same and startup banners are clearly distinguished from runtime
16 * records.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - Function overloading:
20 * https://en.cppreference.com/w/cpp/language/overload_resolution
21 * - std::unordered_map:
22 * https://en.cppreference.com/w/cpp/container/unordered_map
23 * - std::ranges algorithms (std::ranges::replace):
24 * https://en.cppreference.com/w/cpp/algorithm/ranges/replace
25 *
26 * @author Amr Osman <dr3mro@gmail.com>
27 * @date 2026-11-16
28 */

69std::string Message::get_color(trantor::Logger::LogLevel level) {
70 static const std::unordered_map<trantor::Logger::LogLevel, std::string>
71 color_map = {
72 // A read-only map built once, pairing each level with its colour.
73 {
74 trantor::Logger::LogLevel::kTrace,
75 "\033[38;5;140m",
76 }, // Soft Purple (for subtle debug messages)
77 {
78 trantor::Logger::LogLevel::kDebug,
79 "\033[38;5;140m",
80 }, // Soft Purple (for subtle debug messages)
81 {
82 trantor::Logger::LogLevel::kInfo,
83 "\033[1;34m",
84 }, // Bold Blue (clear and calm for info)
85 {
86 trantor::Logger::LogLevel::kWarn,
87 "\033[1;93m",
88 }, // Bright Yellow (attention-grabbing for
89 // warnings)
90 {
91 trantor::Logger::LogLevel::kError,
92 "\033[1;91m",
93 }, // Bright Red (strong and clear for errors)
94 {
95 trantor::Logger::LogLevel::kFatal,
96 "\033[38;5;196m",
97 },
98 };
99 return color_map.at(level); // .at looks up the level and throws
100 // std::out_of_range if it is somehow missing.
101}

103void Message::MessageImpl(const std::string& _status_message,
104 trantor::Logger::LogLevel
105 // Implementation shared by the severity helpers.
106 _level) {
107 // Route through Trantor so these records reach the configured spdlog sinks.
108 trantor::Logger(_level).stream()
109 << fmt::format("{}{}{}",
110 get_color(_level),
111 removeNewLine(_status_message),
112 "\033[0m"); // Builds "colour + cleaned text + reset" and
113 // streams it into the trantor log at the
114 // chosen level.
115}

139// Takes text that might span lines.
140std::string Message::removeNewLine(const std::string& _str) {
141 // Keep each record on a single line so log prefixes stay meaningful.
142 std::string status_message =
143 _str; // Work on a copy so the caller's original string is untouched.
144 std::ranges::replace(status_message,
145 '\n',
146 ' '); // Replace every newline with a space to flatten
147 // multi-line messages.
148 return status_message; // Returns the single-line version for
149 // logging/printing.
150}
