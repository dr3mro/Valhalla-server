1/**
2 * @file watchdog.cpp
3 * @brief Implementation for watchdog: Starts the monitoring thread that periodically checks a pooled connection.
4 * @details This file implements WatchDog. The constructor guards against a pre-existing thread
5 * and then launches a lambda that loops while the atomic flag is set. Each cycle
6 * leases a connection with an RAII handler, checks it, and on failure reconnects it
7 * and asks the pool to reconnect all idle links. DatabaseBusy and other exceptions are
8 * caught and followed by an interval sleep. The destructor clears the flag and joins
9 * the thread before the watchdog object disappears.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * It is the server's automatic recovery mechanism for the synchronous database path. A
13 * database restart or a dropped socket is detected here within one second and repaired without
14 * any request failing permanently. Because the monitor leases its connection through a
15 * DatabaseHanndler, it also participates in the same pool and back-pressure rules as regular
16 * requests. Without this file, stale connections would keep failing until manual intervention.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - std::thread and lambdas as thread entry points: https://en.cppreference.com/w/cpp/thread/thread
20 * - std::atomic (the stop flag shared between threads): https://en.cppreference.com/w/cpp/atomic/atomic
21 * - RAII with std::unique_ptr (automatic cleanup): https://en.cppreference.com/w/cpp/memory/unique_ptr
22 * - Exceptions and catch blocks: https://en.cppreference.com/w/cpp/language/try_catch
23 * - std::this_thread::sleep_for (pause a thread): https://en.cppreference.com/w/cpp/thread/sleep_for
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

44/**
45 * @brief Starts the monitoring thread that periodically checks a pooled connection.
46 * @details The loop leases a connection through a DatabaseHanndler (so it is returned on
47 * scope exit), checks it, and on failure reconnects it and then asks the pool to
48 * reconnect all idle connections. DatabaseBusy, a lost connection and any other
49 * exception are logged (or warned) and followed by an interval sleep, so one bad
50 * cycle cannot spin the loop. The thread reads the atomic stop flag each iteration.
51 * @throws std::logic_error If a monitor thread is already joinable.
52 */
53WatchDog::WatchDog()
54{
55 if (monitor_thread.joinable()) // A fresh object has no thread; this guard prevents accidentally replacing a live one.
56 {
57 throw std::logic_error("WatchDog thread already running!"); // Fail loudly rather than lose track of a running thread.
58 }
59
60 // Start the monitoring thread
61 monitor_thread = std::thread( // Launch the loop on its own thread; it runs until should_monitor becomes false.
62 [this]() // Capture `this` so the loop can read the flag, interval and call members; the destructor joins before `this` dies.
63 {
64 while (should_monitor) // The atomic stop flag is checked every cycle for a prompt, race-free shutdown.
65 {
66 try // Keep one bad iteration from killing the whole monitoring thread.
67 {
68 // RAII lease: the connection is returned to the pool when the handler dies.
69 std::unique_ptr<DatabaseHanndler> dbHandler = std::make_unique<DatabaseHanndler>(); // Heap allocation keeps the lease alive for the whole iteration and returns it automatically.
70 std::shared_ptr<Database> db_ptr = dbHandler->get_connection(); // Reach the actual connection to check.
71
72 if (db_ptr == nullptr) // Defensive: the handler normally throws instead of returning null.
73 {
74 Message::WarningMessage("WatchDog could not acquire a database connection within timeout."); // Warn that this cycle could not run.
75 std::this_thread::sleep_for(check_interval); // Wait before retrying so the loop does not spin.
76 continue; // Skip the rest of this cycle and check the stop flag again.
77 }
78
79 if (!db_ptr->check_connection()) // A failed SELECT 1 probe means the link is dead.
80 {
81 Message::WarningMessage("Database connection lost. Attempting to reconnect..."); // Human-readable notice of the repair attempt.
82 Message::CriticalMessage(fmt::format("Database connection {} link is lost.", static_cast<const void*>(db_ptr.get()))); // Include the connection address for tracing.
83 try // Reconnecting can throw, so isolate it from the outer loop handler.
84 {
85 if (db_ptr->reconnect()) // Rebuild this connection from its cached credentials.
86 {
87 Message::InfoMessage(fmt::format("Database connection {} is re-established successfully", static_cast<void*>(db_ptr.get()))); // Log the repaired link.
88 // Also refresh the rest of the idle pool after this link recovered.
89 dbHandler->reconnect_all(); // Ask the pool to refresh all other idle connections too.
90 }
91 else
92 {
93 Message::ErrorMessage("Failed to reconnect to database"); // Report a non-throwing reconnect failure.
94 }
95 }
96 catch (const std::exception& e) // A driver exception during reconnect should not stop monitoring.
97 {
98 Message::CriticalMessage(fmt::format("Reconnect exception: {}", e.what())); // Record the exception text.
99 }
100 }
101
102 std::this_thread::sleep_for(check_interval); // Pace the loop so health checks happen once per interval.
103 }
104 // A saturated pool is not a fault: log and try again after the interval.
105 catch (const DatabaseBusy&) // Expected under load: all connections were leased when the watchdog tried.
106 {
107 Message::WarningMessage("WatchDog could not acquire a database connection within timeout."); // Note the skipped check without treating it as a crash.
108 std::this_thread::sleep_for(check_interval); // Back off before the next attempt.
109 }
110 catch (const std::exception& e) // Any other error is logged but never ends the thread.
111 {
112 Message::CriticalMessage(fmt::format("Monitoring exception: {}", e.what())); // Include the exception text for diagnosis.
113 std::this_thread::sleep_for(check_interval); // Sleep so a persistent error cannot hot-loop.
114 }
115 }
116 });
117}

119/**
120 * @brief Stops the monitor and waits for its thread to finish.
121 * @details The atomic flag is cleared first so the loop exits at its next iteration; join
122 * then guarantees the thread is no longer using `this` before destruction. This
123 * must happen before the pool the thread borrows from is torn down.
124 */
125WatchDog::~WatchDog()
126{
127 // Ensure proper cleanup
128 should_monitor = false; // Signal the loop to stop at its next check (atomic, so no data race).
129 if (monitor_thread.joinable()) // Only join a thread that is still running.
130 {
131 monitor_thread.join(); // Wait for it to finish so it cannot touch `this` after destruction.
132 }
133}
