1/**
2 * @file watchdog.cpp
3 * @brief Implementation for watchdog: Starts the monitoring thread that
4 * periodically checks a pooled connection.
5 * @details This file implements WatchDog. The constructor guards against a
6 * pre-existing thread and then launches a lambda that loops while the atomic
7 * flag is set. Each cycle leases a connection with an RAII handler, checks it,
8 * and on failure reconnects it and asks the pool to reconnect all idle links.
9 * DatabaseBusy and other exceptions are caught and followed by an interval
10 * sleep. The destructor clears the flag and joins the thread before the
11 * watchdog object disappears.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * It is the server's automatic recovery mechanism for the synchronous database
15 * path. A database restart or a dropped socket is detected here within one
16 * second and repaired without any request failing permanently. Because the
17 * monitor leases its connection through a DatabaseHanndler, it also
18 * participates in the same pool and back-pressure rules as regular requests.
19 * Without this file, stale connections would keep failing until manual
20 * intervention.
21 *
22 * CONCEPTS TO LEARN (for beginners):
23 * - std::thread and lambdas as thread entry points:
24 * https://en.cppreference.com/w/cpp/thread/thread
25 * - std::atomic (the stop flag shared between threads):
26 * https://en.cppreference.com/w/cpp/atomic/atomic
27 * - RAII with std::unique_ptr (automatic cleanup):
28 * https://en.cppreference.com/w/cpp/memory/unique_ptr
29 * - Exceptions and catch blocks:
30 * https://en.cppreference.com/w/cpp/language/try_catch
31 * - std::this_thread::sleep_for (pause a thread):
32 * https://en.cppreference.com/w/cpp/thread/sleep_for
33 *
34 * @author Amr Osman <dr3mro@gmail.com>
35 * @date 2026-11-16
36 */

60/**
61 * @brief Starts the monitoring thread that periodically checks a pooled
62 * connection.
63 * @details The loop leases a connection through a DatabaseHanndler (so it is
64 * returned on scope exit), checks it, and on failure reconnects it and then
65 * asks the pool to reconnect all idle connections. DatabaseBusy, a lost
66 * connection and any other exception are logged (or warned) and followed by an
67 * interval sleep, so one bad cycle cannot spin the loop. The thread reads the
68 * atomic stop flag each iteration.
69 * @throws std::logic_error If a monitor thread is already joinable.
70 */
71WatchDog::WatchDog() {
72 // A fresh object has no thread; this guard
73 // prevents accidentally replacing a live one.
74 if (monitor_thread.joinable()) {
75 throw std::logic_error(
76 "WatchDog thread already running!"); // Fail loudly rather than
77 // lose track of a running
78 // thread.
79 }
80
81 // Start the monitoring thread
82 // Launch the loop on its own thread; it runs
83 // until should_monitor becomes false.
84 // Capture `this` so the loop can read the flag,
85 // interval and call members; the destructor joins
86 // before `this` dies.
87 monitor_thread = std::thread([this]() {
88 // The atomic stop flag is checked
89 // every cycle for a prompt,
90 // race-free shutdown.
91 while (should_monitor) {
92 // Keep one bad iteration from killing the whole
93 // monitoring thread.
94 try {
95 // RAII lease: the connection is returned to the
96 // pool when the handler dies.
97 std::unique_ptr<DatabaseHanndler> dbHandler =
98 std::make_unique<DatabaseHanndler>(); // Heap allocation
99 // keeps the lease
100 // alive for the whole
101 // iteration and
102 // returns it
103 // automatically.
104 std::shared_ptr<Database> db_ptr =
105 dbHandler->get_connection(); // Reach the actual
106 // connection to check.
107
108 // Defensive: the handler normally
109 // throws instead of returning null.
110 if (db_ptr == nullptr) {
111 Message::WarningMessage(
112 "WatchDog could not acquire a database "
113 "connection "
114 "within timeout."); // Warn that this cycle
115 // could not run.
116 std::this_thread::sleep_for(
117 check_interval); // Wait before retrying so
118 // the loop does not spin.
119 continue; // Skip the rest of this cycle and
120 // check the stop flag again.
121 }
122
123 // A failed SELECT 1
124 // probe means the
125 // link is dead.
126 if (!db_ptr->check_connection()) {
127 Message::WarningMessage(
128 "Database connection lost. Attempting to "
129 "reconnect..."); // Human-readable notice
130 // of the repair attempt.
131 Message::CriticalMessage(
132 fmt::format("Database connection {} link is lost.",
133 static_cast<const void*>(
134 db_ptr.get()))); // Include the
135 // connection address
136 // for tracing.
137 // Reconnecting can throw, so isolate it
138 // from the outer loop handler.
139 try {
140 // Rebuild this
141 // connection from
142 // its cached
143 // credentials.
144 if (db_ptr->reconnect()) {
145 Message::InfoMessage(
146 fmt::format("Database connection "
147 "{} is re-established "
148 "successfully",
149 static_cast<void*>(
150 // Log the repaired link. Also
151 // refresh the rest of the idle pool
152 // after this link recovered.
153 db_ptr.get())));
154 // Ask the pool to refresh all other idle
155 // connections too.
156 dbHandler->reconnect_all();
157 } else {
158 Message::ErrorMessage(
159 "Failed to reconnect to "
160 // Report a
161 // non-throwing
162 // reconnect
163 // failure.
164 // A driver exception
165 // during reconnect should
166 // not stop monitoring.
167 "database");
168 }
169 } catch (const std::exception& e) {
170 Message::CriticalMessage(fmt::format("Reconnect exception: {}",
171 e.what())); // Record the
172 // exception text.
173 }
174 }
175
176 std::this_thread::sleep_for(check_interval); // Pace the loop so health
177 // checks happen once per
178 // interval.
179 }
180 // A saturated pool is not a fault: log and try again
181 // after the interval.
182 // Expected under load: all
183 // connections were leased
184 // when the watchdog tried.
185 catch (const DatabaseBusy&) {
186 Message::WarningMessage(
187 "WatchDog could not acquire a database "
188 "connection "
189 "within timeout."); // Note the skipped check
190 // without treating it as a
191 // crash.
192 // Back off before the
193 // next attempt.
194 // Any other error is logged
195 // but never ends the thread.
196 std::this_thread::sleep_for(check_interval);
197 } catch (const std::exception& e) {
198 Message::CriticalMessage(
199 fmt::format("Monitoring exception: {}",
200 e.what())); // Include the exception text
201 // for diagnosis.
202 std::this_thread::sleep_for(check_interval); // Sleep so a persistent
203 // error cannot hot-loop.
204 }
205 }
206 });
207}

209/**
210 * @brief Stops the monitor and waits for its thread to finish.
211 * @details The atomic flag is cleared first so the loop exits at its next
212 * iteration; join then guarantees the thread is no longer using `this` before
213 * destruction. This must happen before the pool the thread borrows from is torn
214 * down.
215 */
216WatchDog::~WatchDog() {
217 // Ensure proper cleanup
218 should_monitor = false; // Signal the loop to stop at its next check
219 // (atomic, so no data race).
220 // Only join a thread that is still running.
221 if (monitor_thread.joinable()) {
222 monitor_thread.join(); // Wait for it to finish so it cannot touch
223 // `this` after destruction.
224 }
225}
