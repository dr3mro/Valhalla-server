1/**
2 * @file http.hpp
3 * @brief HTTP status codes and error payloads shared by the API layer.
4 * @details Defines the Error struct (a numeric code plus a message) and the
5 * Status enum, which lists the HTTP response codes this server uses.
6 * The enum is grouped into the standard families: 1xx informational,
7 * 2xx success, 3xx redirection, 4xx client error and 5xx server error.
8 * A small helper, getDescription, turns a Status value into its standard
9 * human-readable reason phrase (for example "OK" or "Not Found").
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * Every callback in the API reports its outcome as an integer status code, and
13 * this file is the single vocabulary of those codes. Controllers, the error
14 * macros and the response builders all use Status values so a "404" means the
15 * same thing everywhere. getDescription lets the server include a readable
16 * reason phrase when needed. Without this shared list, status codes would be
17 * magic numbers scattered through the code with no consistent meaning.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - Scoped/unscoped enums: https://en.cppreference.com/w/cpp/language/enum
21 * - std::unordered_map: https://en.cppreference.com/w/cpp/container/unordered_map
22 * - Function-local static variables: https://en.cppreference.com/w/cpp/language/storage_duration
23 *
24 * @author Amr Osman <dr3mro@gmail.com>
25 * @date 2026-11-16
26 */

46 /**
47 * @enum Status
48 * @brief HTTP status codes returned as the first CALLBACK_ argument.
49 * @details Enumerators mirror standard HTTP codes, grouped below by class:
50 * 1xx informational, 2xx success, 3xx redirection, 4xx client error
51 * and 5xx server error. INTERNAL_SERVER_ERROR and SERVICE_UNAVAILABLE
52 * are the two values produced by statusForException.
53 * @note Only codes used by this server are listed; getDescription falls back
54 * to a generic string for values outside this set.
55 */
56 enum Status : uint16_t // Stored as a 16-bit unsigned number; codes like 503 fit easily and the type is compact.
57 {
58 // Informational responses
59 CONTINUE = 100, // Server got the request headers; client should go ahead and send the body.
60 SWITCHING_PROTOCOLS = 101, // Server is switching to a different protocol as the client requested.
61 PROCESSING = 102, // Server accepted the request but has not produced a response yet.
62
63 // Successful responses
64 OK = 200, // The general "it worked" answer.
65 CREATED = 201, // The request worked and a new resource was created.
66 ACCEPTED = 202, // Request accepted for processing, but the work is not finished yet.
67 NON_AUTHORITATIVE_INFORMATION = 203, // Returned information may differ from the original source.
68 NO_CONTENT = 204, // Success, but there is no body to send back.
69 RESET_CONTENT = 205, // Success; the client should reset whatever view it is showing.
70 PARTIAL_CONTENT = 206, // Only part of the resource is returned (used for ranged downloads).
71
72 // Redirection messages
73 MULTIPLE_CHOICES = 300, // There are several possible responses; the client should choose.
74 MOVED_PERMANENTLY = 301, // The resource has a new permanent address.
75 FOUND = 302, // The resource is temporarily at a different address.
76 SEE_OTHER = 303, // The answer is at another address; fetch it with GET.
77 NOT_MODIFIED = 304, // The resource has not changed; the client can use its cached copy.
78 TEMPORARY_REDIRECT = 307, // Redirect for now; repeat the same method at the new address.
79 PERMANENT_REDIRECT = 308, // Redirect permanently; repeat the same method at the new address.
80
81 // Client error responses
82 BAD_REQUEST = 400, // The server could not understand the request.
83 UNAUTHORIZED = 401, // The client must log in / authenticate first.
84 PAYMENT_REQUIRED = 402, // Reserved for future use (digital payments).
85 FORBIDDEN = 403, // The server understood but refuses to allow it.
86 NOT_FOUND = 404, // The requested resource does not exist.
87 METHOD_NOT_ALLOWED = 405, // That HTTP method is not allowed for this resource.
88 NOT_ACCEPTABLE = 406, // The server cannot produce a response the client will accept.
89 CONFLICT = 409, // The request conflicts with the resource's current state.
90 GONE = 410, // The resource used to exist but is permanently gone.
91 LENGTH_REQUIRED = 411, // The request must include a Content-Length header.
92 PRECONDITION_FAILED = 412, // Conditions set in the request headers were not met.
93 PAYLOAD_TOO_LARGE = 413, // The request body is bigger than the server allows.
94 UNSUPPORTED_MEDIA_TYPE = 415, // The request's media type is not supported here.
95
96 // Server error responses
97 INTERNAL_SERVER_ERROR = 500, // A generic, unexpected failure on the server.
98 NOT_IMPLEMENTED = 501, // The server does not support the requested feature yet.
99 BAD_GATEWAY = 502, // An upstream server sent back an invalid response.
100 SERVICE_UNAVAILABLE = 503, // The server is temporarily unavailable; the client should retry.
101 GATEWAY_TIMEOUT = 504, // An upstream server took too long to respond.
102 HTTP_VERSION_NOT_SUPPORTED = 505 // The server does not support the HTTP version used.
103 };

106 /**
107 * @brief Returns the standard reason phrase for a status code.
108 * @param[in] code Status code to describe.
109 * @return Reason phrase (e.g. "OK"), or "Unknown Status Code" if `code`
110 * is not present in the lookup table.
111 * @note The lookup table is a function-local static, initialised once and
112 * read-only thereafter.
113 */
114 inline std::string getDescription(Status code) // Inline so the definition can live in the header; takes the code by value.
115 {
116 // Utility: Map status codes to their descriptions
117 static const std::unordered_map<Status, std::string> StatusDescriptions = {{CONTINUE, "Continue"}, {SWITCHING_PROTOCOLS, "Switching Protocols"}, // A read-only table built once, pairing each code with its standard reason phrase (the rows continue below).
118 {PROCESSING, "Processing"}, {OK, "OK"}, {CREATED, "Created"}, {ACCEPTED, "Accepted"},
119 {NON_AUTHORITATIVE_INFORMATION, "Non-Authoritative Information"}, {NO_CONTENT, "No Content"}, {RESET_CONTENT, "Reset Content"},
120 {PARTIAL_CONTENT, "Partial Content"}, {MULTIPLE_CHOICES, "Multiple Choices"}, {MOVED_PERMANENTLY, "Moved Permanently"}, {FOUND, "Found"},
121 {SEE_OTHER, "See Other"}, {NOT_MODIFIED, "Not Modified"}, {TEMPORARY_REDIRECT, "Temporary Redirect"}, {PERMANENT_REDIRECT, "Permanent Redirect"},
122 {BAD_REQUEST, "Bad Request"}, {UNAUTHORIZED, "Unauthorized"}, {PAYMENT_REQUIRED, "Payment Required"}, {FORBIDDEN, "Forbidden"},
123 {NOT_FOUND, "Not Found"}, {METHOD_NOT_ALLOWED, "Method Not Allowed"}, {NOT_ACCEPTABLE, "Not Acceptable"}, {CONFLICT, "Conflict"}, {GONE, "Gone"},
124 {LENGTH_REQUIRED, "Length Required"}, {PRECONDITION_FAILED, "Precondition Failed"}, {PAYLOAD_TOO_LARGE, "Payload Too Large"},
125 {UNSUPPORTED_MEDIA_TYPE, "Unsupported Media Type"}, {INTERNAL_SERVER_ERROR, "Internal Server Error"}, {NOT_IMPLEMENTED, "Not Implemented"},
126 {BAD_GATEWAY, "Bad Gateway"}, {SERVICE_UNAVAILABLE, "Service Unavailable"}, {GATEWAY_TIMEOUT, "Gateway Timeout"},
127 {HTTP_VERSION_NOT_SUPPORTED, "HTTP Version Not Supported"}};
128
129 auto description = StatusDescriptions.find(code); // Look up the code; find returns an iterator (or end() when missing).
