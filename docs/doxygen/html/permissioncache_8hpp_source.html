3 * @brief A short-lived, thread-safe cache of permission data read from the
4 * database. (Originally: "Idle-expiry cache for permission lookups, keyed by
5 * the generated permission.")
6 * @details This header defines a template cache that remembers the answer to a
7 * permission query for a while. Each use renews the entry's lifetime;

50 * @class BasicPermissionCache
51 * @brief Idle-expiry cache for permission lookups, keyed by the generated
52 * permission query (which embeds the entity table and id).
53 * @details Each hit renews the entry's lifetime; an entry that is not used for
54 * the TTL is removed. The TTL comes from Configurator::PermissionCacheConfig
55 * (PERMISSION_CACHE_TTL, default 300 seconds); 0 disables the cache.
56 *
57 * Security trade-off: there is no invalidation on writes, so a revoked
58 * permission can keep being served while its entry stays in use.
59 * invalidateAll() is available for callers that need an explicit flush.
60 *
61 * All public operations are thread-safe via mutex_; generation_ is
62 * atomic so readers can take a token without locking and have a racing put()
63 * rejected. Think of it as a notepad: look up a question, and if the answer is
64 * still fresh use it; otherwise ask the database and jot the new answer down.
65 * @tparam ClockT Clock used for idle expiry; defaults to
66 * std::chrono::steady_clock.
67 */

85 /**
86 * @brief Constructs a cache with the given idle TTL and capacity.
87 * @param[in] ttl Idle lifetime; a value of zero disables the cache
88 * entirely.
89 * @param[in] max_entries Capacity; when reached on put(), the whole map is
90 * cleared.
91 */
92 explicit BasicPermissionCache(std::chrono::seconds ttl,
93 size_t max_entries = 10000)
94 : ttl_(ttl), max_entries_(max_entries) {
95 } // Stores the idle lifetime and capacity; "explicit" prevents accidental
96 // implicit construction.
97
98 /** @brief Whether caching is active (i.e. the TTL is positive). */
99 [[nodiscard]] bool enabled() const {
100 return ttl_.count() > 0;
101 } // Caching is on only when the TTL is a positive number of seconds.
102
103 /**
104 * Token to pass to put(); a put with a stale token is dropped, so a read
105 * that raced with a write cannot store pre-write data.
106 */
107 [[nodiscard]] uint64_t generation() const {
108 return generation_.load(std::memory_order_acquire);
109 } // Reads the current epoch without locking; acquire pairs with the
110 // release in invalidateAll().
111
112 /**
113 * @brief Looks up @p key, renewing the entry's idle timer on a hit.
114 * @param[in] key Generated permission query string.
115 * @return The cached JSON on a live hit; `std::nullopt` when disabled,
116 * absent, or expired (expired entries are erased).
117 */
118 std::optional<jsoncons::json> get(const std::string& key) {
119 // If caching is disabled, never pretend to have a
120 // cached answer.
121 if (!enabled()) {
122 return std::nullopt; // No value to give back.
123 }
124 std::lock_guard<std::mutex> lock(
125 mutex_); // Lock for the whole lookup so no other thread can change
126 // the map mid-read.
127 auto it = map_.find(key); // Look for an entry with this query string.
128 // Not found: a plain cache miss.
129 if (it == map_.end()) {
130 return std::nullopt; // Caller should read from the database.
131 }
132 // Found but too old (idle past its TTL).
133 if (Clock::now() >= it->second.expires) {
134 map_.erase(it); // Remove the stale entry so it cannot be used again.
135 return std::nullopt; // Treat a stale entry as a miss.
136 }
137 it->second.expires =
138 Clock::now() + ttl_; // used: renew the idle timer because this
139 // entry was just used.
140 return it->second.value; // Hand back the cached JSON by value.
141 }
142
143 /**
144 * @brief Stores @p value under @p key if the cache is still in the same
145 * generation.
146 * @details The generation check prevents a read that raced with an
147 * invalidation from re-inserting pre-write data. Before inserting, expired
148 * entries are swept and, if the map is full, all entries are cleared.
149 * @param[in] key Generated permission query string.
150 * @param[in] value Permissions JSON returned by the database.
151 * @param[in] generation_at_read Token from generation() taken before the
152 * read.
153 */
154 void put(const std::string& key,
155 const jsoncons::json& value,
156 uint64_t generation_at_read) {
157 // Nothing to store when caching is off.
158 if (!enabled()) {
159 return; // Do nothing.
160 }
161 std::lock_guard<std::mutex> lock(
162 mutex_); // Guard the map for the whole operation.
163 // A flush happened while the
164 // database read was in flight.
165 if (generation_at_read != generation_.load(std::memory_order_acquire)) {
166 return; // Drop this stale result instead of re-inserting old data.
167 }
168 sweepExpired(); // First try to make room by removing already-expired
169 // entries.
170 // Still full: hit the capacity limit.
171 if (map_.size() >= max_entries_) {
172 map_.clear(); // Simplest policy: throw everything away to bound
173 // memory use.
174 }
175 map_[key] = Entry{value,
176 Clock::now() +
177 ttl_}; // Store the value with a fresh idle deadline.
178 }
179
180 /**
181 * @brief Flushes all entries and bumps the generation so any pending stale
182 * put() is dropped.
183 * @details Callers can use this for an explicit invalidation (e.g. after a
184 * permission write), since the cache otherwise has no write-triggered
185 * invalidation.
186 */
187 void invalidateAll() {
188 std::lock_guard<std::mutex> lock(
189 mutex_); // Serialize the flush with reads and writes.
190 generation_
191 .fetch_add(1,
192 std::memory_order_acq_rel); // Bump the epoch so any
193 // in-flight put() with the old
194 // token is rejected.
195 map_.clear(); // Remove every cached entry.
196 }

198 /** @brief Number of entries currently held (including any not yet swept).
199 */
200 [[nodiscard]] size_t size() {
201 std::lock_guard<std::mutex> lock(
202 mutex_); // Lock so the size is read consistently.
203 return map_.size(); // Number of entries currently held (expired ones
204 // may still be present until swept).
205 }

207 /**
208 * @brief Returns the process-wide singleton, configured from Configurator.
209 * @details The singleton's TTL is fixed at first use from
210 * Configurator::PermissionCacheConfig; thread-safe initialization
211 * is guaranteed by the function-local static.
212 */
213 static BasicPermissionCache& instance() {
214 static BasicPermissionCache cache(
215 Store::getObject<Configurator>()
216 ->get<Configurator::PermissionCacheConfig>()
217 .ttl); // Created once on first use; C++ guarantees thread-safe
218 // initialization of a function-local static.
219 return cache; // Every caller shares this one process-wide cache.
220 }

222 private:
223 /**
224 * @brief Erases entries idle past the TTL, at most once per minute.
225 * @details Rate-limited via next_sweep_ to avoid a full scan on every
226 * put(); expired entries that are looked up are still erased immediately by
227 * get().
228 * @pre Caller holds mutex_.
229 */
230 // Drop entries that have been idle past the TTL (at most once a minute).
231 void sweepExpired() {
232 const auto now = Clock::now(); // One timestamp for the whole sweep so
233 // the cutoff is consistent.
234 // Rate limit: do not scan more than about once
235 // a minute.
236 if (now < next_sweep_) {
237 return; // Not time yet; skip the work.
238 }
239 next_sweep_ =
240 now + std::chrono::seconds{60}; // Schedule the next allowed sweep.
241 for (auto it = map_.begin();
242 // Walk every entry; note there is no ++it in
243 // the header.
244 it != map_.end();) {
245 it = (now >= it->second.expires)
246 ? map_.erase(it)
247 : std::next(it); // Erase expired entries; erase() returns the
248 // next iterator, otherwise advance manually.
249 }
250 }

252 /** @brief A cached permissions value together with its idle-expiry
253 * deadline. */
254 struct Entry {
255 jsoncons::json value; /**< Cached permissions JSON, the actual answer we
256 are storing. */
257 TimePoint expires; /**< Idle deadline; renewed on every hit so busy
258 entries never expire. */
259 };
260
261 std::chrono::seconds ttl_; /**< Idle lifetime; zero disables caching. Set
262 once in the constructor. */
263 size_t
264 max_entries_; /**< Capacity before a full clear; bounds memory usage. */
265 std::mutex mutex_; /**< Guards map_ and next_sweep_ so threads cannot
266 corrupt them. */
267 std::atomic<uint64_t> generation_{
268 0}; /**< Bumped on invalidateAll() to reject stale puts; atomic so
269 readers need no lock. */
270 std::unordered_map<std::string, Entry>
271 map_; /**< Cache contents keyed by query string, the actual hash table.
272 */
273 TimePoint next_sweep_{}; /**< Earliest time of the next sweepExpired() scan;
274 default-constructed to the clock's epoch. */
275};
