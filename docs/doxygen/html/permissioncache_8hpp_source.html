3 * @brief A short-lived, thread-safe cache of permission data read from the database.
4 * (Originally: "Idle-expiry cache for permission lookups, keyed by the generated permission.")
5 * @details This header defines a template cache that remembers the answer to a
6 * permission query for a while. Each use renews the entry's lifetime;
7 * unused entries expire. A generation counter prevents a slow read from
8 * putting stale data back after the cache has been flushed.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * Permission checks happen on almost every request. Reading them from PostgreSQL
12 * each time would be slow, so PermissionManagerPrivate asks this cache first.
13 * Without it, authorization would mean a database round-trip on every API call.
14 *
15 * CONCEPTS TO LEARN (for beginners):
16 * - Caching and time-to-live (TTL): https://en.wikipedia.org/wiki/Cache_(computing)
17 * - Mutexes and thread safety: https://en.cppreference.com/w/cpp/thread/mutex
18 * - std::chrono (clocks and durations): https://en.cppreference.com/w/cpp/chrono
19 * - std::unordered_map (hash table): https://en.cppreference.com/w/cpp/container/unordered_map
20 *

22 * @date 2026-11-16
23 */

40/**
41 * @class BasicPermissionCache
42 * @brief Idle-expiry cache for permission lookups, keyed by the generated permission
43 * query (which embeds the entity table and id).
44 * @details Each hit renews the entry's lifetime; an entry that is not used for the TTL
45 * is removed. The TTL comes from Configurator::PermissionCacheConfig
46 * (PERMISSION_CACHE_TTL, default 300 seconds); 0 disables the cache.
47 *
48 * Security trade-off: there is no invalidation on writes, so a revoked
49 * permission can keep being served while its entry stays in use. invalidateAll()
50 * is available for callers that need an explicit flush.
51 *
52 * All public operations are thread-safe via mutex_; generation_ is atomic so
53 * readers can take a token without locking and have a racing put() rejected.
54 * Think of it as a notepad: look up a question, and if the answer is still
55 * fresh use it; otherwise ask the database and jot the new answer down.
56 * @tparam ClockT Clock used for idle expiry; defaults to std::chrono::steady_clock.
57 */
58template <typename ClockT = std::chrono::steady_clock> // The clock is a template parameter mainly so tests can substitute a fake clock.
59class BasicPermissionCache // "Basic" because a convenience alias with the default clock is provided at the end.
60{
61 public: // Everything below is callable by users of the cache.
62 using Clock = ClockT; /**< Clock type used for expiry timestamps; a local nickname for readability. */
63 using TimePoint = typename Clock::time_point; /**< Timestamp type produced by Clock; a local nickname used for deadlines. */
64
65 /**
66 * @brief Constructs a cache with the given idle TTL and capacity.
67 * @param[in] ttl Idle lifetime; a value of zero disables the cache entirely.
68 * @param[in] max_entries Capacity; when reached on put(), the whole map is cleared.
69 */
70 explicit BasicPermissionCache(std::chrono::seconds ttl, size_t max_entries = 10000) : ttl_(ttl), max_entries_(max_entries) {} // Stores the idle lifetime and capacity; "explicit" prevents accidental implicit construction.
71
72 /** @brief Whether caching is active (i.e. the TTL is positive). */
73 [[nodiscard]] bool enabled() const { return ttl_.count() > 0; } // Caching is on only when the TTL is a positive number of seconds.
74
75 /**
76 * Token to pass to put(); a put with a stale token is dropped, so a read
77 * that raced with a write cannot store pre-write data.
78 */
79 [[nodiscard]] uint64_t generation() const { return generation_.load(std::memory_order_acquire); } // Reads the current epoch without locking; acquire pairs with the release in invalidateAll().
80
81 /**
82 * @brief Looks up @p key, renewing the entry's idle timer on a hit.
83 * @param[in] key Generated permission query string.
84 * @return The cached JSON on a live hit; `std::nullopt` when disabled, absent,
85 * or expired (expired entries are erased).
86 */
87 std::optional<jsoncons::json> get(const std::string& key)
88 {
89 if (!enabled()) // If caching is disabled, never pretend to have a cached answer.
90 {
91 return std::nullopt; // No value to give back.
92 }
93 std::lock_guard<std::mutex> lock(mutex_); // Lock for the whole lookup so no other thread can change the map mid-read.
94 auto it = map_.find(key); // Look for an entry with this query string.
95 if (it == map_.end()) // Not found: a plain cache miss.
96 {
97 return std::nullopt; // Caller should read from the database.
98 }
99 if (Clock::now() >= it->second.expires) // Found but too old (idle past its TTL).
100 {
101 map_.erase(it); // Remove the stale entry so it cannot be used again.
102 return std::nullopt; // Treat a stale entry as a miss.
103 }
104 it->second.expires = Clock::now() + ttl_; // used: renew the idle timer because this entry was just used.
105 return it->second.value; // Hand back the cached JSON by value.
106 }
107
108 /**
109 * @brief Stores @p value under @p key if the cache is still in the same generation.
110 * @details The generation check prevents a read that raced with an invalidation from
111 * re-inserting pre-write data. Before inserting, expired entries are swept and,
112 * if the map is full, all entries are cleared.
113 * @param[in] key Generated permission query string.
114 * @param[in] value Permissions JSON returned by the database.
115 * @param[in] generation_at_read Token from generation() taken before the read.
116 */
117 void put(const std::string& key, const jsoncons::json& value, uint64_t generation_at_read)
118 {
119 if (!enabled()) // Nothing to store when caching is off.
120 {
121 return; // Do nothing.
122 }
123 std::lock_guard<std::mutex> lock(mutex_); // Guard the map for the whole operation.
124 if (generation_at_read != generation_.load(std::memory_order_acquire)) // A flush happened while the database read was in flight.
125 {
126 return; // Drop this stale result instead of re-inserting old data.
127 }
128 sweepExpired(); // First try to make room by removing already-expired entries.
129 if (map_.size() >= max_entries_) // Still full: hit the capacity limit.
130 {
131 map_.clear(); // Simplest policy: throw everything away to bound memory use.
132 }
133 map_[key] = Entry{value, Clock::now() + ttl_}; // Store the value with a fresh idle deadline.
134 }
135
136 /**
137 * @brief Flushes all entries and bumps the generation so any pending stale put() is dropped.
138 * @details Callers can use this for an explicit invalidation (e.g. after a permission
139 * write), since the cache otherwise has no write-triggered invalidation.
140 */
141 void invalidateAll()
142 {
143 std::lock_guard<std::mutex> lock(mutex_); // Serialize the flush with reads and writes.
144 generation_.fetch_add(1, std::memory_order_acq_rel); // Bump the epoch so any in-flight put() with the old token is rejected.
145 map_.clear(); // Remove every cached entry.
146 }
147
148 /** @brief Number of entries currently held (including any not yet swept). */
149 [[nodiscard]] size_t size()
150 {
151 std::lock_guard<std::mutex> lock(mutex_); // Lock so the size is read consistently.
152 return map_.size(); // Number of entries currently held (expired ones may still be present until swept).
153 }
154
155 /**
156 * @brief Returns the process-wide singleton, configured from Configurator.
157 * @details The singleton's TTL is fixed at first use from
158 * Configurator::PermissionCacheConfig; thread-safe initialization is guaranteed
159 * by the function-local static.
160 */
161 static BasicPermissionCache& instance()
162 {
163 static BasicPermissionCache cache(Store::getObject<Configurator>()->get<Configurator::PermissionCacheConfig>().ttl); // Created once on first use; C++ guarantees thread-safe initialization of a function-local static.
164 return cache; // Every caller shares this one process-wide cache.
165 }
166
167 private:
168 /**
169 * @brief Erases entries idle past the TTL, at most once per minute.
170 * @details Rate-limited via next_sweep_ to avoid a full scan on every put(); expired
171 * entries that are looked up are still erased immediately by get().
172 * @pre Caller holds mutex_.
173 */
174 // Drop entries that have been idle past the TTL (at most once a minute).
175 void sweepExpired()
176 {
177 const auto now = Clock::now(); // One timestamp for the whole sweep so the cutoff is consistent.
178 if (now < next_sweep_) // Rate limit: do not scan more than about once a minute.
179 {
180 return; // Not time yet; skip the work.
181 }
182 next_sweep_ = now + std::chrono::seconds{60}; // Schedule the next allowed sweep.
183 for (auto it = map_.begin(); it != map_.end();) // Walk every entry; note there is no ++it in the header.
184 {
185 it = (now >= it->second.expires) ? map_.erase(it) : std::next(it); // Erase expired entries; erase() returns the next iterator, otherwise advance manually.
186 }
187 }

189 /** @brief A cached permissions value together with its idle-expiry deadline. */
190 struct Entry
191 {
192 jsoncons::json value; /**< Cached permissions JSON, the actual answer we are storing. */
193 TimePoint expires; /**< Idle deadline; renewed on every hit so busy entries never expire. */
194 };
195
196 std::chrono::seconds ttl_; /**< Idle lifetime; zero disables caching. Set once in the constructor. */
197 size_t max_entries_; /**< Capacity before a full clear; bounds memory usage. */
198 std::mutex mutex_; /**< Guards map_ and next_sweep_ so threads cannot corrupt them. */
199 std::atomic<uint64_t> generation_{0}; /**< Bumped on invalidateAll() to reject stale puts; atomic so readers need no lock. */
200 std::unordered_map<std::string, Entry> map_; /**< Cache contents keyed by query string, the actual hash table. */
201 TimePoint next_sweep_{}; /**< Earliest time of the next sweepExpired() scan; default-constructed to the clock's epoch. */
202};
