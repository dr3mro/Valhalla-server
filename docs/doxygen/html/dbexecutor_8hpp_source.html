1/**
2 * @file dbexecutor.hpp
3 * @brief BoundedExecutor configured for the synchronous database path.
4 * @details This header declares DbExecutor, a thin subclass of BoundedExecutor that runs
5 * blocking database work off the Drogon IO threads. It has two constructors: one
6 * takes worker count, queue size and fast-path flag from configuration, the other
7 * takes explicit sizes for tests. It also provides runOnDbExecutor(), a helper that
8 * submits work and answers 503 itself if the bounded queue is full.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * Drogon's IO threads must stay free to accept and route HTTP requests; a slow SQL query on
12 * one of them would stall every other request. Routes obtain the shared DbExecutor from the
13 * Store and call runOnDbExecutor() so permission checks, queries and response building run on
14 * dedicated workers. The optional IO-thread fast path (see dbprobe.hpp) lets requests that
15 * need no database finish without ever queueing. Without this executor, blocking database work
16 * would run on IO threads and the server would grind to a halt under load.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - Inheritance / derived classes: https://en.cppreference.com/w/cpp/language/derived_class
20 * - Function templates: https://en.cppreference.com/w/cpp/language/function_template
21 * - Perfect forwarding (std::forward): https://en.cppreference.com/w/cpp/utility/forward
22 * - std::size_t (an unsigned size type): https://en.cppreference.com/w/cpp/types/size_t
23 * - Thread pools and bounded queues (back-pressure)
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

37// The pool that runs blocking database work (permission check, query, response) so a slow query never blocks a Drogon IO
38// thread. Size: DB_WORKERS (default DB_MAX_CONN, so each worker holds at most one connection), queue: DB_QUEUE_MAX.
39/**
40 * @class DbExecutor
41 * @brief BoundedExecutor configured for the synchronous database path.
42 * @details Runs blocking database work (permission checks, queries, response building) off
43 * the Drogon IO threads. Worker count and queue capacity come from configuration in
44 * the default constructor; the other constructor allows explicit sizes for tests.
45 * When configured, requests may first be attempted on the IO thread via a DbProbe
46 * (see ioFastPath()). Thread-safety and shutdown semantics are inherited from
47 * BoundedExecutor.
48 * @see BoundedExecutor, DbProbe, runOnDbExecutor
49 */
50class DbExecutor : public BoundedExecutor // Reuses the generic worker pool; only the database-specific setup is added here.
51{
52 public:
53 /**
54 * @brief Constructs the executor sized and configured from the Configurator.
55 */
56 DbExecutor(); // sizes come from the Configurator
57
58 /**
59 * @brief Constructs an executor with explicit sizes (used by tests).
60 * @param[in] workers Number of worker threads; zero runs tasks inline on the submitter.
61 * @param[in] queue_max Maximum number of queued tasks across all workers.
62 * @param[in] io_fast_path Whether to try requests on the IO thread before queuing.
63 */
64 DbExecutor(std::size_t workers, std::size_t queue_max, bool io_fast_path = false); // The default keeps the fast path off unless explicitly requested.
65
66 // True when requests are first tried on the IO thread (see database/dbprobe.hpp) and only queued if they need the database.
67 /**
68 * @brief Reports whether the IO-thread fast path is active.
69 * @return `true` only when configured and the executor is not in inline mode.
70 */
71 [[nodiscard]] bool ioFastPath() const { return io_fast_path_ && !inlineMode(); } // Inline mode (zero workers) has no IO thread to offload from, so the fast path is pointless there.

80// Runs body(callback) on the DB workers; a full queue answers 503 and body never runs.
81/**
82 * @brief Submits `body` to a DbExecutor, answering 503 if it cannot be accepted.
83 * @details Wraps runOnExecutor with the database-busy message. The callback is shared with
84 * the queued task so it remains available for the 503 reply if submission fails.
85 * Any database work that must not run on an IO thread belongs in `body`.
86 * @tparam Body Callable taking a `CALLBACK_` by value/rvalue reference.
87 * @param[in,out] executor Executor that runs the body.
88 * @param[in,out] callback Completion callback invoked with (status, message).
89 * @param[in,out] body Work to run on a worker; not invoked when the queue is full.
90 * @post Exactly one of `body` (on a worker) or `callback` with 503 (on the caller) runs.
91 */
92template <typename Body>
93void runOnDbExecutor(DbExecutor& executor, CALLBACK_&& callback, Body&& body)
94{
95 runOnExecutor(executor, std::move(callback), "The server is busy, try again shortly.", std::forward<Body>(body)); // Reuse the generic helper; the busy message here tells clients to retry when the DB queue is full.
96}
