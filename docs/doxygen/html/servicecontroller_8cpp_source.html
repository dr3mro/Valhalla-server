1/**
2 * @file servicecontroller.cpp
3 * @brief Definitions for @ref ServiceController and its explicit template
4 * instantiations.
5 * @details The adapter methods are defined here rather than in the header; the
6 * instantiation macro at the bottom makes the concrete specializations linkable
7 * by the route layer.
8 *
9 * In plain words, this file contains the bodies of the small
10 * forwarding methods that were only declared in the header. Each body just
11 * passes its arguments on to @ref EntityController, which does the real work.
12 * The block of macros at the end forces the compiler to build a concrete
13 * version of the class for each service entity type, so the route layer can
14 * link to them. This file has no logic of its own beyond that forwarding.
15 *
16 * WHY THIS FILE MATTERS IN THE SERVER:
17 * Without these definitions and the explicit instantiations, the route
18 * registry's `shared_ptr<ServiceController<T>>` objects would be declared but
19 * never actually compiled, and the program would fail to link (the compiler
20 * would report "undefined reference"). This file is called by the route layer
21 * through @ref ServiceControllerBase and calls into @ref EntityController for
22 * every job. It is what turns the service interface into working, linkable
23 * code.
24 *
25 * CONCEPTS TO LEARN (for beginners):
26 * - Explicit template instantiation (forcing the compiler to build a concrete
27 * version): https://en.cppreference.com/w/cpp/language/template_instantiation
28 * - std::move / std::forward (passing values without copying):
29 * https://en.cppreference.com/w/cpp/utility/move
30 * - C++20 coroutines (functions that pause and resume with co_await/co_return):
31 * https://en.cppreference.com/w/cpp/language/coroutines
32 *
33 * @author Amr Osman <dr3mro@gmail.com>
34 * @date 2026-11-16
35 */

69/**
70 * @brief Creates a service entity; delegates entirely to @ref EntityController.
71 * @details The callback is moved into the entity controller, which performs
72 * JSON parsing, create-schema validation, permission checking, entity
73 * construction and the INSERT. This adapter adds no behaviour and cannot change
74 * the error→HTTP mapping.
75 * @param[in,out] callback Completion handler invoked once by the delegate with
76 * (status, body).
77 * @param[in] requester Authenticated caller forwarded as a const rvalue for
78 * authorisation.
79 * @param[in] data Raw JSON body; must remain valid for the duration of the
80 * call.
81 */
82template <Service_t T> // One definition works for any type T that satisfies
83 // the Service_t concept.
84void ServiceController<T>::Create(CALLBACK_&& callback,
85 const Requester&& requester,
86 std::string_view data) {
87 EntityController<T>::Create(std::move(callback),
88 std::forward<const Requester&&>(requester),
89 data); // Hand everything to the base: std::move
90 // transfers the callback, std::forward
91 // preserves the requester's exact type.
92}

94/**
95 * @brief Coroutine form of @ref Create; co_awaits
96 * `EntityController<T>::CreateAsync`.
97 * @details The awaiting task owns the continuation; the forwarded callback
98 * fires from the delegate's coroutine once its database work completes.
99 * @param[in,out] callback Completion handler invoked once with (status, body).
100 * @param[in] requester Authenticated caller forwarded for authorisation.
101 * @param[in] data Raw JSON body; must remain valid until the task completes.
102 */
103template <Service_t T>
104drogon::Task<void> ServiceController<T>::CreateAsync(
105 CALLBACK_&& callback, const Requester&& requester, std::string_view data) {
106 co_await EntityController<
107 T>::CreateAsync(std::move(callback),
108 std::forward<const Requester&&>(requester),
109 data); // Pause here until the base's async create
110 // finishes; then this task completes.
111}
