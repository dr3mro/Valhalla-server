1/**
2 * @file servicecontroller.cpp
3 * @brief Definitions for @ref ServiceController and its explicit template instantiations.
4 * @details The adapter methods are defined here rather than in the header; the instantiation
5 * macro at the bottom makes the concrete specializations linkable by the route layer.
6 *
7 * In plain words, this file contains the bodies of the small forwarding methods that were
8 * only declared in the header. Each body just passes its arguments on to @ref EntityController,
9 * which does the real work. The block of macros at the end forces the compiler to build a
10 * concrete version of the class for each service entity type, so the route layer can link to
11 * them. This file has no logic of its own beyond that forwarding.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * Without these definitions and the explicit instantiations, the route registry's
15 * `shared_ptr<ServiceController<T>>` objects would be declared but never actually compiled, and the
16 * program would fail to link (the compiler would report "undefined reference"). This file is called by
17 * the route layer through @ref ServiceControllerBase and calls into @ref EntityController for every job.
18 * It is what turns the service interface into working, linkable code.
19 *
20 * CONCEPTS TO LEARN (for beginners):
21 * - Explicit template instantiation (forcing the compiler to build a concrete version): https://en.cppreference.com/w/cpp/language/template_instantiation
22 * - std::move / std::forward (passing values without copying): https://en.cppreference.com/w/cpp/utility/move
23 * - C++20 coroutines (functions that pause and resume with co_await/co_return): https://en.cppreference.com/w/cpp/language/coroutines
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

47/**
48 * @brief Creates a service entity; delegates entirely to @ref EntityController.
49 * @details The callback is moved into the entity controller, which performs JSON parsing,
50 * create-schema validation, permission checking, entity construction and the INSERT.
51 * This adapter adds no behaviour and cannot change the error→HTTP mapping.
52 * @param[in,out] callback Completion handler invoked once by the delegate with (status, body).
53 * @param[in] requester Authenticated caller forwarded as a const rvalue for authorisation.
54 * @param[in] data Raw JSON body; must remain valid for the duration of the call.
55 */
56template <Service_t T> // One definition works for any type T that satisfies the Service_t concept.
57void ServiceController<T>::Create(CALLBACK_&& callback, const Requester&& requester, std::string_view data)
58{
59 EntityController<T>::Create(std::move(callback), std::forward<const Requester&&>(requester), data); // Hand everything to the base: std::move transfers the callback, std::forward preserves the requester's exact type.
60}

62/**
63 * @brief Coroutine form of @ref Create; co_awaits `EntityController<T>::CreateAsync`.
64 * @details The awaiting task owns the continuation; the forwarded callback fires from the
65 * delegate's coroutine once its database work completes.
66 * @param[in,out] callback Completion handler invoked once with (status, body).
67 * @param[in] requester Authenticated caller forwarded for authorisation.
68 * @param[in] data Raw JSON body; must remain valid until the task completes.
69 */
70template <Service_t T>
71drogon::Task<void> ServiceController<T>::CreateAsync(CALLBACK_&& callback, const Requester&& requester, std::string_view data)
72{
73 co_await EntityController<T>::CreateAsync(std::move(callback), std::forward<const Requester&&>(requester), data); // Pause here until the base's async create finishes; then this task completes.
74}
