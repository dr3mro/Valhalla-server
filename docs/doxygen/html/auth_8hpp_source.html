1/**
2 * @file auth.hpp
3 * @brief Drogon filter that validates the Bearer token and identifies the requester.
4 * @details This file defines the Auth filter, the last step of the SECURE filter chain. It reads the
5 * Authorization header, checks that it starts with "Bearer ", and asks GateKeeper whether the
6 * token is valid. It supports two modes: a normal blocking check and an asynchronous coroutine
7 * check chosen by the database configuration. On success it writes "clientID" and
8 * "clientGroup" onto the request so controllers know who is calling; on failure it sends a
9 * JSON 401 (or a 503 when the database is too busy).
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * This is the gatekeeper of the whole protected API. Drogon calls it before any SECURE route handler,
13 * so it is the component that turns an anonymous HTTP request into an identified, permission-checkable
14 * one. Downstream controllers (clients, clinics, staff and so on) read the identity attributes it sets
15 * and would have no idea who the caller was without it. It calls GateKeeper for validation and the
16 * Store/service locator to find shared objects. Without this file every protected route would be open
17 * to anyone.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - HTTP filters/middleware: https://drogon.docsforge.com/master/filter/
21 * - Bearer tokens and the Authorization header: https://developer.mozilla.org/en-US/docs/Web/HTTP/Headers/Authorization
22 * - C++ coroutines (co_await): https://en.cppreference.com/w/cpp/language/coroutines
23 * - HTTP 401 Unauthorized: https://developer.mozilla.org/en-US/docs/Web/HTTP/Status/401
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

52 /**
53 * @class Auth
54 * @brief Drogon filter that validates the Bearer token and identifies the requester.
55 * @details Runs last in the SECURE filter chain, after rate limiting and timing
56 * (see INSECURE/SECURE in basic/common.hpp).
57 * It rejects malformed/missing Authorization headers and invalid tokens with a JSON 401;
58 * a DatabaseBusy from the validator becomes a retryable JSON 503 with Retry-After: 1.
59 * On success it inserts "clientID" and "clientGroup" request attributes that downstream
60 * controllers read to build a Requester for permission checks, then continues the chain.
61 * When DatabaseConfig::async is set, validation runs in a drogon::Task coroutine and the
62 * callbacks are moved into the coroutine frame so they outlive every suspension.
63 * @note All routes that consume the identity attributes must include this filter (SECURE).
64 */
65 class Auth : public drogon::HttpFilter<Auth> // Inherit Drogon's filter template so Drogon can discover and run it.
66 {
67 public:
68 /** @brief Default-constructs the filter; the GateKeeper is obtained from the service locator. */
69 Auth() = default; // No custom setup needed; the member initializer below fetches the GateKeeper.
70 /**
71 * @brief Validate the request's Bearer token, then continue or short-circuit the chain.
72 * @details Extracts the token and peer IP, checks the "Bearer " prefix, and calls GateKeeper
73 * either synchronously or as a coroutine depending on DatabaseConfig::async. Success
74 * stores "clientID"/"clientGroup" on the request and invokes @p fccb; any failure
75 * invokes @p fcb with a JSON error response. This override is invoked by Drogon on
76 * its IO thread; in async mode the remainder runs on the coroutine frame.
77 * @param[in] req Request carrying the Authorization header and peer address.
78 * @param[in,out] fcb Filter callback used to send a rejection response (consumed on reject).
79 * @param[in,out] fccb Chain callback invoked on success to run the next filter/handler.
80 */
81 void doFilter(const drogon::HttpRequestPtr& req, drogon::FilterCallback&& fcb, drogon::FilterChainCallback&& fccb) override // Drogon calls this for every request on the filter chain.
82 {
83 const std::string bearer = "Bearer "; // The exact prefix a valid Authorization header must start with.
84 const size_t bearerMagic = bearer.length(); // Length of the prefix (7); used to split prefix from token.
85
86 // Directly get the authorization header value to avoid
87 // multiple dereferences
88 const auto& auth_header = req->getHeader("Authorization"); // Read the header once into a reference; empty if absent.
89
90 // Check for "Bearer " prefix using a fast substring
91 // comparison
92 if (auth_header.empty() || auth_header.size() <= bearerMagic || auth_header[bearerMagic - 1] != ' ' || auth_header.substr(0, bearerMagic) != bearer) // Reject missing, too-short or wrong-prefix headers before any DB work.
93 {
94 // Malformed or non-Bearer header -> short-circuit with a JSON 401.
95 auto resp = drogon::HttpResponse::newHttpResponse(); // Build an empty HTTP reply object; Drogon sends it when we invoke the callback.
96 resp->setStatusCode(drogon::HttpStatusCode::k401Unauthorized); // 401 means "you did not authenticate".
97 resp->setContentTypeCode(drogon::CT_APPLICATION_JSON); // Mark the body as JSON so clients parse it correctly.
98 resp->setBody(JsonHelper::stringify(JsonHelper::jsonify("failed to extract token, or not a bearer token"))); // Wrap the message in JSON and store it as the body.
99 std::move(fcb)(resp); // Send the rejection and stop the chain (the handler never runs).
100 return; // Leave doFilter; the callback was consumed exactly once.
101 }
102
103 // Extract the token
104 std::optional<Types::ClientLoginData> clientLoginData = Types::ClientLoginData{}; // Start with an empty credentials holder.
105 clientLoginData->token = auth_header.substr(bearerMagic); // Everything after "Bearer " is the raw token.
106 clientLoginData->ip_address = req->getPeerAddr().toIp(); // The caller's IP is needed for IP-bound validation.
107 std::string message; // Filled with a human-readable reason if validation fails.
108 // Validate token. The DB mode is fixed for the process lifetime, so read it once.
109 static const bool asyncDb = Store::cachedObject<Configurator>()->get<Configurator::DatabaseConfig>().async; // Cached so the setting is read only once, not per request.
110 if (asyncDb) // Choose the coroutine path when asynchronous database access is configured.
111 {
112 const size_t bearerMagicLocal = bearerMagic; // Copy by value so the coroutine does not capture a reference to a local.
113 std::shared_ptr<GateKeeper> gateKeeperLocal = gateKeeper; // Copy the shared pointer so the GateKeeper stays alive in the coroutine.
114 // Move the callbacks into storage owned by the coroutine frame: async_run moves the
115 // factory (and its captures) into the coroutine, so they outlive every suspension.
116 drogon::async_run([req, bearerMagicLocal, gateKeeperLocal, fcb = std::move(fcb), fccb = std::move(fccb)]() mutable -> drogon::Task<void> // Capture everything by value; mutable lets us move the callbacks out later.
117 {
118 std::optional<Types::ClientLoginData> clientLoginData = Types::ClientLoginData{}; // Rebuild credentials inside the coroutine (its own stack frame).
119 clientLoginData->token = req->getHeader("Authorization").substr(bearerMagicLocal); // Re-read the token using the saved prefix length.
120 clientLoginData->ip_address = req->getPeerAddr().toIp(); // Re-read the peer IP inside the coroutine.
121 std::string message; // Reason text for a failed validation.
122 bool valid = false; // Result of the validation call.
123 drogon::FilterCallback cb = std::move(fcb); // Move the rejection callback into the coroutine so it survives suspension.
124 drogon::FilterChainCallback chain = std::move(fccb); // Move the success callback in as well.
125 try // Validation may throw when the database is overloaded.
126 {
127 valid = co_await gateKeeperLocal->isAuthenticationValidAsync(clientLoginData, message); // Suspend until the async check finishes; fill valid and message.
128 }
129 catch (const DatabaseBusy& e) // Too many DB requests in flight.
130 {
131 auto resp = drogon::HttpResponse::newHttpResponse(); // Build the retry response.
132 resp->setStatusCode(drogon::HttpStatusCode::k503ServiceUnavailable); // 503 tells the client the server is temporarily unavailable.
133 resp->setContentTypeCode(drogon::CT_APPLICATION_JSON); // Body is JSON.
134 resp->addHeader("Retry-After", "1"); // Ask the client to retry in one second.
135 resp->setBody(JsonHelper::stringify(JsonHelper::jsonify(e.what()))); // Send the exception text as JSON.
136 std::move(cb)(resp); // Deliver the 503 and end this path.
137 co_return; // Exit the coroutine (do not continue the chain).
138 }
139 if (!valid) // The token was rejected.
140 {
141 auto resp = drogon::HttpResponse::newHttpResponse(); // Build the unauthorized response.
142 resp->setStatusCode(drogon::HttpStatusCode::k401Unauthorized); // 401 = not authenticated.
143 resp->setContentTypeCode(drogon::CT_APPLICATION_JSON); // Body is JSON.
144 resp->setBody(JsonHelper::stringify(JsonHelper::jsonify(message))); // Explain why validation failed.
145 std::move(cb)(resp); // Send the 401.
146 co_return; // Stop here; the handler must not run.
147 }
148
149 // Populate the identity attributes consumed by controllers' permission checks.
150 req->attributes()->insert("clientID", clientLoginData->clientId.value()); // Store the authenticated client id on the request.
151 req->attributes()->insert("clientGroup", clientLoginData->group.value()); // Store the client's group used for permission checks.
152
153 // Token is valid, pass control to the next filter/handler
154 std::move(chain)(); // Continue the chain so the actual route handler runs.
155 });
156 return; // The coroutine owns the rest; leave doFilter now.
157 }
158
159 bool valid = false; // Blocking-mode result of the validation call.
160 try // Guard against a busy database.
161 {
162 valid = gateKeeper->isAuthenticationValid(clientLoginData, message); // Blocking check; fills valid and message.
163 }
164 catch (const DatabaseBusy& e) // Connection pool exhausted.
165 {
166 auto resp = drogon::HttpResponse::newHttpResponse(); // Build the retry response.
167 resp->setStatusCode(drogon::HttpStatusCode::k503ServiceUnavailable); // 503 = try again soon.
168 resp->setContentTypeCode(drogon::CT_APPLICATION_JSON); // Body is JSON.
169 resp->addHeader("Retry-After", "1"); // Suggest retrying after one second.
170 resp->setBody(JsonHelper::stringify(JsonHelper::jsonify(e.what()))); // Send the error text as JSON.
171 std::move(fcb)(resp); // Deliver the 503.
172 return; // Stop; no handler runs.
173 }
174 if (!valid) // Token rejected.
175 {
176 auto resp = drogon::HttpResponse::newHttpResponse(); // Build the unauthorized response.
177 resp->setStatusCode(drogon::HttpStatusCode::k401Unauthorized); // 401 = not authenticated.
178 resp->setContentTypeCode(drogon::CT_APPLICATION_JSON); // Body is JSON.
179 resp->setBody(JsonHelper::stringify(JsonHelper::jsonify(message))); // Explain the failure.
180 std::move(fcb)(resp); // Send the 401.
181 return; // Stop; no handler runs.
182 }
183
184 req->attributes()->insert("clientID", clientLoginData->clientId.value()); // Store the authenticated client id for controllers.
185 req->attributes()->insert("clientGroup", clientLoginData->group.value()); // Store the client's group for permission checks.
186
187 // Token is valid, pass control to the next filter/handler
188 std::move(fccb)(); // Continue the chain so the route handler runs.
189 }
190
191 private:
192 std::shared_ptr<GateKeeper> gateKeeper = Store::getObject<GateKeeper>(); // Shared validator fetched once from the service locator.
193 };
