1/**
2 * @file auth.hpp
3 * @brief Drogon filter that validates the Bearer token and identifies the
4 * requester.
5 * @details This file defines the Auth filter, the last step of the SECURE
6 * filter chain. It reads the Authorization header, checks that it starts with
7 * "Bearer ", and asks GateKeeper whether the token is valid. It supports two
8 * modes: a normal blocking check and an asynchronous coroutine check chosen by
9 * the database configuration. On success it writes "clientID" and "clientGroup"
10 * onto the request so controllers know who is calling; on failure it sends a
11 * JSON 401 (or a 503 when the database is too busy).
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * This is the gatekeeper of the whole protected API. Drogon calls it before any
15 * SECURE route handler, so it is the component that turns an anonymous HTTP
16 * request into an identified, permission-checkable one. Downstream controllers
17 * (clients, clinics, staff and so on) read the identity attributes it sets and
18 * would have no idea who the caller was without it. It calls GateKeeper for
19 * validation and the Store/service locator to find shared objects. Without this
20 * file every protected route would be open to anyone.
21 *
22 * CONCEPTS TO LEARN (for beginners):
23 * - HTTP filters/middleware: https://drogon.docsforge.com/master/filter/
24 * - Bearer tokens and the Authorization header:
25 * https://developer.mozilla.org/en-US/docs/Web/HTTP/Headers/Authorization
26 * - C++ coroutines (co_await):
27 * https://en.cppreference.com/w/cpp/language/coroutines
28 * - HTTP 401 Unauthorized:
29 * https://developer.mozilla.org/en-US/docs/Web/HTTP/Status/401
30 *
31 * @author Amr Osman <dr3mro@gmail.com>
32 * @date 2026-11-16
33 */

71 /**
72 * @class Auth
73 * @brief Drogon filter that validates the Bearer token and identifies the
74 * requester.
75 * @details Runs last in the SECURE filter chain, after rate limiting and
76 * timing (see INSECURE/SECURE in basic/common.hpp). It rejects
77 * malformed/missing Authorization headers and invalid tokens with a JSON
78 * 401; a DatabaseBusy from the validator becomes a retryable JSON 503 with
79 * Retry-After: 1. On success it inserts "clientID" and "clientGroup"
80 * request attributes that downstream controllers read to build a Requester
81 * for permission checks, then continues the chain. When
82 * DatabaseConfig::async is set, validation runs in a drogon::Task coroutine
83 * and the callbacks are moved into the coroutine frame so they outlive
84 * every suspension.
85 * @note All routes that consume the identity attributes must include this
86 * filter (SECURE).
87 */
88 class Auth
89 // Inherit Drogon's filter template
90 // so Drogon can discover and run it.
91 : public drogon::HttpFilter<Auth> {
92 public:
93 /** @brief Default-constructs the filter; the GateKeeper is obtained
94 * from the service locator. */
95 Auth() = default; // No custom setup needed; the member initializer
96 // below fetches the GateKeeper.
97 /**
98 * @brief Validate the request's Bearer token, then continue or
99 * short-circuit the chain.
100 * @details Extracts the token and peer IP, checks the "Bearer " prefix,
101 * and calls GateKeeper either synchronously or as a coroutine depending
102 * on DatabaseConfig::async. Success stores "clientID"/"clientGroup" on
103 * the request and invokes @p fccb; any failure invokes @p fcb with a
104 * JSON error response. This override is invoked by Drogon on its IO
105 * thread; in async mode the remainder runs on the coroutine frame.
106 * @param[in] req Request carrying the Authorization header and peer
107 * address.
108 * @param[in,out] fcb Filter callback used to send a rejection response
109 * (consumed on reject).
110 * @param[in,out] fccb Chain callback invoked on success to run the next
111 * filter/handler.
112 */
113 // Drogon calls this for every request on the filter
114 // chain.
115 void doFilter(const drogon::HttpRequestPtr& req,
116 drogon::FilterCallback&& fcb,
117 drogon::FilterChainCallback&& fccb) override {
118 const std::string bearer =
119 "Bearer "; // The exact prefix a valid Authorization header
120 // must start with.
121 const size_t bearerMagic =
122 bearer.length(); // Length of the prefix (7); used to split
123 // prefix from token.
124
125 // Directly get the authorization header value to avoid
126 // multiple dereferences
127 const auto& auth_header =
128 req->getHeader("Authorization"); // Read the header once into a
129 // reference; empty if absent.
130
131 // Check for "Bearer " prefix using a fast substring
132 // comparison
133 // Reject missing, too-short or wrong-prefix
134 // headers before any DB work.
135 if (auth_header.empty() || auth_header.size() <= bearerMagic ||
136 auth_header[bearerMagic - 1] != ' ' ||
137 auth_header.substr(0, bearerMagic) != bearer) {
138 // Malformed or non-Bearer header -> short-circuit with a JSON
139 // 401.
140 auto resp =
141 drogon::HttpResponse::newHttpResponse(); // Build an empty HTTP
142 // reply object; Drogon
143 // sends it when we invoke
144 // the callback.
145 resp->setStatusCode(
146 drogon::HttpStatusCode::k401Unauthorized); // 401 means "you did
147 // not authenticate".
148 resp->setContentTypeCode(
149 drogon::CT_APPLICATION_JSON); // Mark the body as JSON so
150 // clients parse it
151 // correctly.
152 resp->setBody(JsonHelper::stringify(
153 JsonHelper::jsonify("failed to extract token, or not a bearer "
154 "token"))); // Wrap the message in JSON and
155 // store it as the body.
156 std::move(fcb)(resp); // Send the rejection and stop the chain
157 // (the handler never runs).
158 return; // Leave doFilter; the callback was consumed exactly
159 // once.
160 }
161
162 // Extract the token
163 std::optional<Types::ClientLoginData> clientLoginData =
164 Types::ClientLoginData{}; // Start with an empty credentials
165 // holder.
166 clientLoginData->token = auth_header.substr(
167 bearerMagic); // Everything after "Bearer " is the raw token.
168 clientLoginData->ip_address =
169 req->getPeerAddr().toIp(); // The caller's IP is needed for
170 // IP-bound validation.
171 std::string message; // Filled with a human-readable reason if
172 // validation fails.
173 // Validate token. The DB mode is fixed for the process lifetime, so
174 // read it once.
175 static const bool asyncDb = Store::cachedObject<Configurator>()
176 ->get<Configurator::DatabaseConfig>()
177 .async; // Cached so the setting is read
178 // only once, not per request.
179 // Choose the coroutine path when asynchronous
180 // database access is configured.
181 if (asyncDb) {
182 const size_t bearerMagicLocal =
183 bearerMagic; // Copy by value so the coroutine does not
184 // capture a reference to a local.
185 std::shared_ptr<GateKeeper> gateKeeperLocal =
186 gateKeeper; // Copy the shared pointer so the GateKeeper
187 // stays alive in the coroutine.
188 // Move the callbacks into storage owned by the coroutine frame:
189 // async_run moves the factory (and its captures) into the
190 // coroutine, so they outlive every suspension.
191 // Capture everything by value;
192 // mutable lets us move the
193 // callbacks out later.
194 drogon::async_run([req,
195 bearerMagicLocal,
196 gateKeeperLocal,
197 fcb = std::move(fcb),
198 fccb = std::move(
199 fccb)]() mutable -> drogon::Task<void> {
200 std::optional<Types::ClientLoginData> clientLoginData =
201 Types::ClientLoginData{}; // Rebuild credentials
202 // inside the coroutine
203 // (its own stack frame).
204 clientLoginData->token =
205 req->getHeader("Authorization")
206 .substr(bearerMagicLocal); // Re-read the token
207 // using the saved
208 // prefix length.
209 clientLoginData->ip_address =
210 req->getPeerAddr().toIp(); // Re-read the peer IP
211 // inside the coroutine.
212 std::string message; // Reason text for a failed validation.
213 bool valid = false; // Result of the validation call.
214 drogon::FilterCallback cb =
215 std::move(fcb); // Move the rejection callback into the
216 // coroutine so it survives suspension.
217 drogon::FilterChainCallback chain =
218 std::move(fccb); // Move the success callback in as well.
219 // Validation may throw when the database is
220 // overloaded.
221 try {
222 // Suspend until
223 // the async
224 // check
225 // finishes;
226 // fill valid
227 // and message.
228 // Too many DB requests in flight.
229 valid = co_await gateKeeperLocal
230 ->isAuthenticationValidAsync(clientLoginData, message);
231 } catch (const DatabaseBusy& e) {
232 auto resp =
233 drogon::HttpResponse::newHttpResponse(); // Build the retry
234 // response.
235 resp->setStatusCode(drogon::HttpStatusCode::
236 // 503 tells the client the server is
237 // temporarily unavailable.
238 k503ServiceUnavailable);
239 resp->setContentTypeCode(
240 // Body is JSON.
241 drogon::CT_APPLICATION_JSON);
242 resp->addHeader("Retry-After",
243 "1"); // Ask the client to retry in one second.
244 resp->setBody(JsonHelper::stringify(
245 JsonHelper::jsonify(e.what()))); // Send the exception text
246 // as JSON.
247 std::move(cb)(resp); // Deliver the 503 and end this path.
248 co_return; // Exit the coroutine (do not continue
249 // the chain).
250 }
251 // The token was rejected.
252 if (!valid) {
253 auto resp =
254 drogon::HttpResponse::newHttpResponse(); // Build the
255 // unauthorized
256 // response.
257 resp->setStatusCode(
258 drogon::HttpStatusCode::k401Unauthorized); // 401 = not
259 // authenticated.
260 resp->setContentTypeCode(
261 // Body is JSON.
262 drogon::CT_APPLICATION_JSON);
263 resp->setBody(JsonHelper::stringify(
264 JsonHelper::jsonify(message))); // Explain why validation
265 // failed.
266 std::move(cb)(resp); // Send the 401.
267 co_return; // Stop here; the handler must not run.
268 }
269
270 // Populate the identity attributes consumed by
271 // controllers' permission checks.
272 req->attributes()->insert("clientID",
273 clientLoginData->clientId
274 .value()); // Store the authenticated
275 // client id on the request.
276 req->attributes()
277 ->insert("clientGroup",
278 clientLoginData->group
279 .value()); // Store the client's group used for
280 // permission checks.
281
282 // Token is valid, pass control to the next
283 // filter/handler
284 std::move(chain)(); // Continue the chain so the actual
285 // route handler runs.
286 });
287 return; // The coroutine owns the rest; leave doFilter now.
288 }
289
290 bool valid = false; // Blocking-mode result of the validation call.
291 // Guard against a busy database.
292 try {
293 // Blocking check; fills
294 // valid and message.
295 // Connection pool exhausted.
296 valid = gateKeeper->isAuthenticationValid(clientLoginData, message);
297 } catch (const DatabaseBusy& e) {
298 auto resp =
299 // Build the retry response.
300 drogon::HttpResponse::newHttpResponse();
301 resp->setStatusCode(
302 drogon::HttpStatusCode::k503ServiceUnavailable); // 503 = try again
303 // soon.
304 resp->setContentTypeCode(drogon::CT_APPLICATION_JSON); // Body is JSON.
305 resp->addHeader("Retry-After",
306 "1"); // Suggest retrying after one second.
307 resp->setBody(JsonHelper::stringify(
308 JsonHelper::jsonify(e.what()))); // Send the error text as JSON.
309 std::move(fcb)(resp); // Deliver the 503.
310 return; // Stop; no handler runs.
311 }
312 // Token rejected.
313 if (!valid) {
314 auto resp =
315 drogon::HttpResponse::newHttpResponse(); // Build the unauthorized
316 // response.
317 resp->setStatusCode(
318 drogon::HttpStatusCode::k401Unauthorized); // 401 = not
319 // authenticated.
320 resp->setContentTypeCode(drogon::CT_APPLICATION_JSON); // Body is JSON.
321 resp->setBody(JsonHelper::stringify(
322 JsonHelper::jsonify(message))); // Explain the failure.
323 std::move(fcb)(resp); // Send the 401.
324 return; // Stop; no handler runs.
325 }
326
327 req->attributes()->insert("clientID",
328 clientLoginData->clientId
329 .value()); // Store the authenticated
330 // client id for controllers.
331 req->attributes()
332 ->insert("clientGroup",
333 clientLoginData->group.value()); // Store the client's group
334 // for permission checks.
335
336 // Token is valid, pass control to the next filter/handler
337 std::move(fccb)(); // Continue the chain so the route handler runs.
338 }
339
340 private:
341 std::shared_ptr<GateKeeper> gateKeeper =
342 Store::getObject<GateKeeper>(); // Shared validator fetched once
343 // from the service locator.
344 };
