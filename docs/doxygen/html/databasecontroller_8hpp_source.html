1/**
2 * @file databasecontroller.hpp
3 * @brief Single gateway through which controllers execute SQL, in both blocking
4 * and coroutine form.
5 * @details Declares `DatabaseController`, the one class that knows how to run a
6 * SQL statement and map the result to JSON. It offers a blocking family
7 * (executeQuery, executeReadQuery, executeSearchQuery, ...) backed by a
8 * connection pool, and an async family (executeQueryCoro, ...) backed
9 * by Drogon's AsyncDbClient. Every method is virtual so tests can swap in a
10 * fake, and every one returns an `std::optional` that is empty when the
11 * database is unavailable or the query fails.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * It is the single choke point between the application and PostgreSQL. The
15 * Controller class calls these methods, so connection handling, transaction
16 * rules and error mapping (including the special DatabaseBusy signal) live in
17 * one place. Without it, every caller would have to manage connections itself,
18 * and "database down" could not be handled consistently as a 503 instead of a
19 * crash.
20 *
21 * CONCEPTS TO LEARN (for beginners):
22 * - Virtual functions / polymorphism (all methods are virtual for test
23 * injection): https://en.cppreference.com/w/cpp/language/virtual
24 * - Connection pooling and lazy initialization:
25 * https://en.cppreference.com/w/cpp/language/raii
26 * - std::optional (empty means "no result"):
27 * https://en.cppreference.com/w/cpp/utility/optional
28 * - std::atomic (thread-safe flag):
29 * https://en.cppreference.com/w/cpp/atomic/atomic
30 * - std::shared_ptr / std::unique_ptr (ownership of backends and connections):
31 * https://en.cppreference.com/w/cpp/memory/shared_ptr
32 * - Coroutines (co_await / drogon::Task):
33 * https://en.cppreference.com/w/cpp/language/coroutines
34 *
35 * @author Amr Osman <dr3mro@gmail.com>
36 * @date 2026-11-16
37 */

78/**
79 * @class DatabaseController
80 * @brief Single gateway through which controllers execute SQL, in both blocking
81 * and coroutine form.
82 * @details Owns no query logic of its own: it acquires a connection
83 * (synchronously through a `DatabaseConnectionPool` and `DatabaseHanndler`, or
84 * asynchronously through `AsyncDbClient`), invokes the matching `Database`/
85 * `asyncdb` helper, and returns an `std::optional` result that is
86 * empty when the database is unavailable, a connection cannot be acquired, or
87 * the query fails. Every public method is virtual so tests can inject fakes
88 * through `Store::replaceObject`.
89 *
90 * Backend construction is lazy (see ensureBackends): the pool
91 * constructor terminates the process when PostgreSQL is unreachable, which
92 * would otherwise make the class unconstructible in hermetic tests. The
93 * double-checked `backendsReady_` flag makes that init a single atomic
94 * load on the hot path. In hermetic test mode the pool is never created and all
95 * query paths deterministically return `std::nullopt`.
96 *
97 * Query results are returned as JSON objects/arrays or plain scalars.
98 * Identifiers (table/column names) cannot be bound as parameters, so the few
99 * methods that interpolate them accept only fixed, trusted names.
100 * @note Non-copyable and non-movable: the singleton is shared through `Store`,
101 * never duplicated.
102 * @warning Pool exhaustion is not a query failure: `DatabaseBusy` is propagated
103 * so callers can answer 503 with Retry-After instead of 500.
104 * @see Controller, Store, DatabaseBusy
105 */

383 /**
384 * @brief Common implementation for every synchronous query method.
385 * @details `DbProbe::check()` may throw `WouldBlock` to move a request off
386 * the IO thread before this runs. After ensuring backends exist, a missing
387 * pool (test mode) or missing connection yields `nullopt`. The supplied
388 * function is invoked with the acquired `Database*` and forwarded
389 * arguments.
390 * @tparam Result Type wrapped in the returned optional (JSON object/array,
391 * string, bool, or a set).
392 * @tparam Func Callable taking a `Database*` followed by @p args.
393 * @tparam Args Forwarded argument types.
394 * @param[in] func Database helper to invoke.
395 * @param[in] args Arguments forwarded to @p func (query text and
396 * parameters).
397 * @return The helper's result, or `std::nullopt` on unavailable DB/no
398 * connection/query failure.
399 * @throws DatabaseBusy rethrown so callers can answer 503; other exceptions
400 * are logged and swallowed.
401 */
402 template <typename Result, typename Func, typename... Args>
403 std::optional<Result> executer(const Func& func, Args&&... args) {
404 DbProbe::check(); // on the IO-thread fast path this throws WouldBlock:
405 // the request moves to the DB executor
406 try {
407 ensureBackends();
408
409 // Hermetic test mode: no backends were created, so report the
410 // database as unavailable without constructing a handler (which
411 // would create a pool via Store and exit() on init failure).
412 if (!databaseConnectionPool_) {
413 return std::nullopt;
414 }
415
416 std::unique_ptr<DatabaseHanndler> dbHandler =
417 std::make_unique<DatabaseHanndler>(databaseConnectionPool_);
418 std::shared_ptr<Database> db_ptr = dbHandler->get_connection();
419
420 if (db_ptr == nullptr) {
421 return std::nullopt;
422 }
423
424 std::optional<Result> results =
425 std::invoke(func, db_ptr.get(), std::forward<Args>(args)...);
426
427 if (results.has_value()) {
428 return results;
429 }
430
431 return std::nullopt;
432 } catch (const DatabaseBusy&) {
433 throw; // the pool is exhausted: callers answer 503 instead of
434 // treating it as a failed query
435 } catch (const std::exception& e) {
436 Message::CriticalMessage(
437 fmt::format("Exception occurred during query execution: {}",
438 e.what()));
439 return std::nullopt;
440 } catch (...) {
441 Message::CriticalMessage(
442 "Unknown exception occurred during query execution.");
443 return std::nullopt;
444 }
445 }
