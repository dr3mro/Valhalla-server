1/**
2 * @file helper.cpp
3 * @brief Implementations of the shared v2 JSON response helpers.
4 * @details This file contains the bodies of the Helper class functions declared in helper.hpp. Each
5 * public function (failureResponse, errorResponse, successResponse) delegates to the private
6 * reply function with the right status and body. reply creates a Drogon response, sets the
7 * status, JSON content type and body, adds Retry-After for 503, and calls the callback. The
8 * prepare function builds a JSON object with a "Status" tag plus the message.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This is where the API's response format is actually produced. Every JSON success and error sent by
12 * controllers and filters passes through these few functions, so this file guarantees that status
13 * codes, content types and headers are set the same way everywhere. It is called by common.hpp, the
14 * Auth filter and all route controllers. Without it there would be no central place to enforce a
15 * consistent reply format or the retry hint for overloaded servers.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - HTTP status codes: https://developer.mozilla.org/en-US/docs/Web/HTTP/Status
19 * - The Retry-After header: https://developer.mozilla.org/en-US/docs/Web/HTTP/Headers/Retry-After
20 * - std::move (transfer ownership instead of copying): https://en.cppreference.com/w/cpp/utility/move
21 * - CORS (background on HTTP headers): https://developer.mozilla.org/en-US/docs/Web/HTTP/CORS
22 *
23 * @author Amr Osman <dr3mro@gmail.com>
24 * @date 2026-11-16
25 */

67/**
68 * @brief Assemble the response, marking 503 with Retry-After: 1, and deliver it via the callback.
69 * @see Helper::reply
70 */
71void Helper::reply(std::function<void(const drogon::HttpResponsePtr&)>&& callback, const drogon::HttpStatusCode& code, const std::string& message) // The one function that really builds and sends every reply.
72{
73 auto res = drogon::HttpResponse::newHttpResponse(); // Create an empty HTTP response object.
74 res->setStatusCode(code); // Set the HTTP status (200, 400, 500, ...).
75 res->setContentTypeCode(drogon::CT_APPLICATION_JSON); // Tell clients the body is JSON so they parse it.
76 res->setBody(message); // Store the prepared body text.
77 if (code == drogon::k503ServiceUnavailable) // A 503 means "temporarily overloaded".
78 {
79 res->addHeader("Retry-After", "1"); // Advise the client to retry after one second.
80 }
81 std::move(callback)(res); // Deliver the response to Drogon; the callback is consumed exactly once.
82}

84/**
85 * @brief Convert the message to a JSON value, add the "Status" field and stringify it.
86 * @see Helper::prepare
87 */
88std::string Helper::prepare(const std::string& status, const std::string& message) // NOLINT
89{
90 Json::Value object = JsonHelper::jsonify(message); // Parse the message text into a JSON value.
91 object["Status"] = status; // Add the "Failure" or "Error" tag under the "Status" key.
92 return object.toStyledString(); // Return a nicely formatted JSON string for the response body.
93}
