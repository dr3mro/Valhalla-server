1/**
2 * @file helper.cpp
3 * @brief Implementations of the shared v2 JSON response helpers.
4 * @details This file contains the bodies of the Helper class functions declared
5 * in helper.hpp. Each public function (failureResponse, errorResponse,
6 * successResponse) delegates to the private reply function with the right
7 * status and body. reply creates a Drogon response, sets the status, JSON
8 * content type and body, adds Retry-After for 503, and calls the callback. The
9 * prepare function builds a JSON object with a "Status" tag plus the
10 * message.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * This is where the API's response format is actually produced. Every JSON
14 * success and error sent by controllers and filters passes through these few
15 * functions, so this file guarantees that status codes, content types and
16 * headers are set the same way everywhere. It is called by common.hpp, the Auth
17 * filter and all route controllers. Without it there would be no central place
18 * to enforce a consistent reply format or the retry hint for overloaded
19 * servers.
20 *
21 * CONCEPTS TO LEARN (for beginners):
22 * - HTTP status codes: https://developer.mozilla.org/en-US/docs/Web/HTTP/Status
23 * - The Retry-After header:
24 * https://developer.mozilla.org/en-US/docs/Web/HTTP/Headers/Retry-After
25 * - std::move (transfer ownership instead of copying):
26 * https://en.cppreference.com/w/cpp/utility/move
27 * - CORS (background on HTTP headers):
28 * https://developer.mozilla.org/en-US/docs/Web/HTTP/CORS
29 *
30 * @author Amr Osman <dr3mro@gmail.com>
31 * @date 2026-11-16
32 */

51/**
52 * @brief Build a 500 response whose body carries a "Failure" status tag.
53 * @see Helper::failureResponse

55// Unexpected internal error -> status 500.
56void Helper::failureResponse(

96/**
97 * @brief Assemble the response, marking 503 with Retry-After: 1, and deliver it
98 * via the callback.
99 * @see Helper::reply
100 */
101// The one function that really builds and sends every reply.
102void Helper::reply(
103 std::function<void(const drogon::HttpResponsePtr&)>&& callback,
104 const drogon::HttpStatusCode& code,
105 const std::string& message) {
106 auto res = drogon::HttpResponse::newHttpResponse(); // Create an empty HTTP
107 // response object.
108 res->setStatusCode(code); // Set the HTTP status (200, 400, 500, ...).
109 res->setContentTypeCode(
110 drogon::CT_APPLICATION_JSON); // Tell clients the body is JSON so they
111 // parse it.
112 res->setBody(message); // Store the prepared body text.
113 // A 503 means "temporarily
114 // overloaded".
115 if (code == drogon::k503ServiceUnavailable) {
116 res->addHeader("Retry-After",
117 "1"); // Advise the client to retry after one second.
118 }
119 std::move(callback)(res); // Deliver the response to Drogon; the callback
120 // is consumed exactly once.
121}

123/**
124 * @brief Convert the message to a JSON value, add the "Status" field and
125 * stringify it.
126 * @see Helper::prepare
127 */
128std::string Helper::prepare(const std::string& status,
129 const std::string& message) // NOLINT
130{
131 Json::Value object = JsonHelper::jsonify(
132 message); // Parse the message text into a JSON value.
133 object["Status"] =
134 status; // Add the "Failure" or "Error" tag under the "Status" key.
135 return object.toStyledString(); // Return a nicely formatted JSON string
136 // for the response body.
137}
