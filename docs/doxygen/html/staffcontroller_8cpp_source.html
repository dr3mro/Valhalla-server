1/**
2 * @file staffcontroller.cpp
3 * @brief Definitions for @ref StaffController and its explicit template
4 * instantiations.
5 * @details Unlike the other controller adapters, this file contains real logic:
6 * request parsing, staff-management authorisation, and either a staff SQL
7 * statement or an outbound invite POST to the email-sender daemon.
8 *
9 * In plain words, each function here does the same three-step dance:
10 * (1) read the JSON request body, (2) ask GateKeeper whether the caller is
11 * allowed to manage this service's staff, and (3) only if allowed, actually do
12 * the work — run SQL to add/remove a member, or POST an invitation to the email
13 * daemon. If anything goes wrong, the catch block turns the exception into an
14 * HTTP error reply. The macros at the end build a concrete version of the class
15 * for each service entity type so the route layer can link to it.
16 *
17 * WHY THIS FILE MATTERS IN THE SERVER:
18 * This is the implementation behind the staff HTTP routes: it is where
19 * permission is actually enforced before any staff change is made and where
20 * invitations are sent. It is called through
21 * @ref StaffControllerBase by the route layer, and it calls into @ref
22 * Controller (for SQL),
23 * @ref GateKeeper (for permission) and @ref Communicate (for the email POST).
24 * Without it, the staff interface would have no behaviour and no one could add,
25 * remove or invite staff.
26 *
27 * CONCEPTS TO LEARN (for beginners):
28 * - try/catch (handling errors instead of crashing):
29 * https://en.cppreference.com/w/cpp/language/try_catch
30 * - std::optional (a value that may be absent):
31 * https://en.cppreference.com/w/cpp/utility/optional
32 * - C++20 coroutines (functions that pause and resume with co_await/co_return):
33 * https://en.cppreference.com/w/cpp/language/coroutines
34 *
35 * @author Amr Osman <dr3mro@gmail.com>
36 * @date 2026-11-16
37 */

131/**
132 * Adds a staff member described by @p data to the service entity it names.
133 * Parses a JSON body with `payload` and `id`, builds @ref StaffData and `T`,
134 * checks staff-management permission against the id, and finally delegates the
135 * SQL work to `Controller::addStaff`. On denial the GateKeeper's code and
136 * message are returned verbatim; any thrown exception is mapped by
137 * `CRITICALMESSAGERESPONSE` (500, or 503 for a busy database). Note: There is
138 * no pre-parse guard: malformed JSON or a missing key is an exceptional 500
139 * path here, unlike the 400 responses of @ref EntityController. callback:
140 * Completion handler invoked once with (status, body). requester: Authenticated
141 * caller used for the staff-management permission check. data: Raw JSON body
142 * containing `id` and `payload`; must outlive the call.
143 */
144// Works for whichever entity type T this controller was
145// instantiated with.
146// Synchronous add: parse,
147// authorize, then run SQL.
148template <typename T>
149void StaffController<T>::AddStaffToEntity(CALLBACK_&& callback,
150 const Requester&& requester,
151 std::string_view data) {
152 jsoncons::json
153 staff_j; // Will hold the parsed JSON request body; jsoncons::json can
154 // hold objects, arrays, strings, numbers, etc.
155 HttpError error; // Will be filled with {code, message} if the permission
156 // check fails.
157 // Guard the whole operation: bad JSON or a missing key throws and is
158 // handled below.
159 try {
160 // A parse failure or missing key throws and is handled as 500 by
161 // CRITICALMESSAGERESPONSE.
162 staff_j = jsoncons::json::parse(
163 data); // Turn the raw text into a JSON value we can query.
164 jsoncons::json payload =
165 staff_j.at("payload"); // Pull out the "payload" object holding the
166 // new staff member's fields.
167 StaffData staffData(payload); // Convert that JSON object into a typed
168 // StaffData struct.
169
170 T staff(staffData); // Build the concrete entity (e.g. a Clinic) from
171 // the staff data.
172
173 uint64_t _id =
174 staff_j.at("id").as<uint64_t>(); // Read the target service id as a
175 // 64-bit unsigned number.
176
177 // Permission decides before any database side effect; denial is
178 // reported as-is.
179 // The "!" means "if NOT allowed":
180 // stop when permission is denied.
181 if (!gateKeeper->canManageStaff<T>(requester, _id, error)) {
182 std::move(callback)(error.code,
183 error.message); // Report the GateKeeper's error
184 // straight back to the caller.
185 return; // Stop here — no SQL should run when permission was
186 // denied.
187 }
188
189 // Controller builds and runs the entity's add-staff SQL statement.
190 // Hand off the SQL work;
191 // std::move transfers the
192 // callback so no copy is made.
193 // Runs only if something above threw an exception.
194 Controller::addStaff(staff, std::move(callback));
195 } catch (const std::exception& e) {
196 CRITICALMESSAGERESPONSE // Macro: build an HTTP 500 (or 503 if the DB
197 // is busy) error reply and return it.
198 }
199}

253 * invite JSON could be built — POSTs that JSON to the configured email daemon.
254 * The daemon's body is relayed with 200 on success; 400 is returned when invite
255 * construction or delivery fails. Note: `nominee_id` is the value passed to
256 * `canManageStaff` as the permission-check target; note also that the email
257 * request is a synchronous cURL call. callback: Completion handler invoked once

262template <typename T>
263void StaffController<T>::InviteStaffToEntity(
264 CALLBACK_&& callback,
265 const Requester&& requester,
266 std::string_view
267 // Synchronous invite: parse, authorize, then POST an email.
268 data) {
269 // Guard against parse errors and network problems.
270 try {
271 HttpError error; // Filled if permission is denied.
272 jsoncons::json staff_j =
273 jsoncons::json::parse(data); // Parse the whole body as JSON.
274 // The nominee id doubles as the permission-check target for this
275 // operation.
276 uint64_t _id = staff_j.at("nominee_id")
277 .as<uint64_t>(); // Id of the person being invited; also
278 // used for the permission check.
279 StaffData staffData(staff_j); // Build typed invite fields from the JSON.
280 std::optional<std::string> response; // Will hold the daemon's reply text;
281 // empty means delivery failed.
282 T staff(staffData); // Build the concrete entity from the invite data.
283
284 // If NOT allowed to invite for
285 // this service...
286 if (!gateKeeper->canManageStaff<T>(requester, _id, error)) {
287 std::move(
288 callback)(error.code,
289 error.message); // Report the permission error unchanged.
290 return; // Stop; do not send any email.
291 }
292
293 // parse_status records whether StaffData found all required fields;
294 // toInviteJson adds the mail template and invitation link to the JSON
295 // that is sent downstream.
296 // Both must succeed: fields
297 // parsed AND invite JSON built.
298 if (staffData.parse_status && staffData.toInviteJson(staff_j)) {
299 // Synchronous cURL POST; an empty optional denotes delivery
300 // failure.
301 response = Communicate::
302 sendRequest(email_sender_daemon_config_.host,
303 email_sender_daemon_config_
304 .port, // Send the invite to the configured email
305 // daemon (host + port).
306 email_sender_daemon_config_.message_queue_path,
307 drogon::HttpMethod::Post,
308 staff_j.to_string()); // Use the configured queue path,
309 // an HTTP POST, and the JSON as
310 // the body.
311
312 // A value means the daemon replied
313 // successfully.
314 if (response.has_value()) {
315 // Relay the daemon's response body to the HTTP client.
316 // Reply 200 with the daemon's body;
317 // .value() unwraps the optional.
318 // No value means the send failed.
319 std::move(callback)(api::v2::Http::Status::OK, response.value());
320 } else {
321 // Reply 400 with a
322 // diagnostic message.
323 // Parsing or invite-JSON construction failed.
324 std::move(callback)(api::v2::Http::Status::BAD_REQUEST,
325 "Failed to send invite.");
326 }
327 } else {
328 // Reply 400 explaining
329 // the invite could not
330 // be built.
331 // Any unexpected exception is turned into an error response.
332 std::move(callback)(api::v2::Http::Status::BAD_REQUEST,
333 "Failed to create invite json.");
334 }
335 } catch (const std::exception& e) {
336 CRITICALMESSAGERESPONSE // Macro: reply with 500 (or 503 for a busy
337 // database).

343 * Mirrors the synchronous flow but co_awaits `canManageStaffAsync` and

345 * resumed coroutine; callers must co_await the returned task (the route layer

349 * completes.

352template <typename T>
353drogon::Task<void> StaffController<T>::AddStaffToEntityAsync(
354 CALLBACK_&& callback, const Requester&& requester, std::string_view data) {
355 jsoncons::json staff_j; // Will hold the parsed JSON body.
356 HttpError error; // Filled if permission is denied.
357 // Guard the whole coroutine body against exceptions.
358 try {
359 staff_j = jsoncons::json::parse(data); // Parse the raw body into JSON.
360 jsoncons::json payload =
361 staff_j.at("payload"); // Extract the new staff member's fields.
362 StaffData staffData(payload); // Convert them into a typed struct.
363
364 T staff(staffData); // Build the concrete entity.
365
366 uint64_t _id =
367 staff_j.at("id")
368 .as<uint64_t>(); // Target service id for the permission check.
369
370 // co_await pauses until the async permission answer
371 // arrives.
372 if (!co_await gateKeeper->canManageStaffAsync<T>(requester, _id, error)) {
373 std::move(callback)(error.code,
374 error.message); // Denied: report the error through
375 // the callback.
376 // Denial ends the coroutine here; no staff SQL is submitted.
377 co_return; // End this coroutine early without running SQL.
378 }
379
380 // Pause until the async
381 // SQL add finishes.
382 // Exceptions are turned into an error response.
383 co_await Controller::addStaffAsync(staff, std::move(callback));
384 } catch (const std::exception& e) {
385 CRITICALMESSAGERESPONSE // Macro: reply with 500 (or 503 for a busy
386 // database).
387 }
388 co_return; // Every coroutine must explicitly finish with co_return.
389}
