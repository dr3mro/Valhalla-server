1/**
2 * @file staffcontroller.cpp
3 * @brief Definitions for @ref StaffController and its explicit template instantiations.
4 * @details Unlike the other controller adapters, this file contains real logic: request parsing,
5 * staff-management authorisation, and either a staff SQL statement or an outbound invite
6 * POST to the email-sender daemon.
7 *
8 * In plain words, each function here does the same three-step dance: (1) read the JSON
9 * request body, (2) ask GateKeeper whether the caller is allowed to manage this service's
10 * staff, and (3) only if allowed, actually do the work — run SQL to add/remove a member, or
11 * POST an invitation to the email daemon. If anything goes wrong, the catch block turns the
12 * exception into an HTTP error reply. The macros at the end build a concrete version of the
13 * class for each service entity type so the route layer can link to it.
14 *
15 * WHY THIS FILE MATTERS IN THE SERVER:
16 * This is the implementation behind the staff HTTP routes: it is where permission is actually enforced
17 * before any staff change is made and where invitations are sent. It is called through
18 * @ref StaffControllerBase by the route layer, and it calls into @ref Controller (for SQL),
19 * @ref GateKeeper (for permission) and @ref Communicate (for the email POST). Without it, the staff
20 * interface would have no behaviour and no one could add, remove or invite staff.
21 *
22 * CONCEPTS TO LEARN (for beginners):
23 * - try/catch (handling errors instead of crashing): https://en.cppreference.com/w/cpp/language/try_catch
24 * - std::optional (a value that may be absent): https://en.cppreference.com/w/cpp/utility/optional
25 * - C++20 coroutines (functions that pause and resume with co_await/co_return): https://en.cppreference.com/w/cpp/language/coroutines
26 *
27 * @author Amr Osman <dr3mro@gmail.com>
28 * @date 2026-11-16
29 */

78/**
79 * Adds a staff member described by @p data to the service entity it names.
80 * Parses a JSON body with `payload` and `id`, builds @ref StaffData and `T`, checks
81 * staff-management permission against the id, and finally delegates the SQL work to
82 * `Controller::addStaff`. On denial the GateKeeper's code and message are returned
83 * verbatim; any thrown exception is mapped by `CRITICALMESSAGERESPONSE` (500, or 503 for
84 * a busy database).
85 * Note: There is no pre-parse guard: malformed JSON or a missing key is an exceptional 500 path
86 * here, unlike the 400 responses of @ref EntityController.
87 * callback: Completion handler invoked once with (status, body).
88 * requester: Authenticated caller used for the staff-management permission check.
89 * data: Raw JSON body containing `id` and `payload`; must outlive the call.
90 */
91template <typename T> // Works for whichever entity type T this controller was instantiated with.
92void StaffController<T>::AddStaffToEntity(CALLBACK_&& callback, const Requester&& requester, std::string_view data) // Synchronous add: parse, authorize, then run SQL.
93{
94 jsoncons::json staff_j; // Will hold the parsed JSON request body; jsoncons::json can hold objects, arrays, strings, numbers, etc.
95 HttpError error; // Will be filled with {code, message} if the permission check fails.
96 try // Guard the whole operation: bad JSON or a missing key throws and is handled below.
97 {
98 // A parse failure or missing key throws and is handled as 500 by CRITICALMESSAGERESPONSE.
99 staff_j = jsoncons::json::parse(data); // Turn the raw text into a JSON value we can query.
100 jsoncons::json payload = staff_j.at("payload"); // Pull out the "payload" object holding the new staff member's fields.
101 StaffData staffData(payload); // Convert that JSON object into a typed StaffData struct.
102
103 T staff(staffData); // Build the concrete entity (e.g. a Clinic) from the staff data.
104
105 uint64_t _id = staff_j.at("id").as<uint64_t>(); // Read the target service id as a 64-bit unsigned number.
106
107 // Permission decides before any database side effect; denial is reported as-is.
108 if (!gateKeeper->canManageStaff<T>(requester, _id, error)) // The "!" means "if NOT allowed": stop when permission is denied.
109 {
110 std::move(callback)(error.code, error.message); // Report the GateKeeper's error straight back to the caller.
111 return; // Stop here — no SQL should run when permission was denied.
112 }
113
114 // Controller builds and runs the entity's add-staff SQL statement.
115 Controller::addStaff(staff, std::move(callback)); // Hand off the SQL work; std::move transfers the callback so no copy is made.
116 }
117 catch (const std::exception& e) // Runs only if something above threw an exception.
118 {
119 CRITICALMESSAGERESPONSE // Macro: build an HTTP 500 (or 503 if the DB is busy) error reply and return it.
120 }
121}

131template <typename T>
132void StaffController<T>::RemoveStaffFromEntity(CALLBACK_&& callback, const Requester&& requester, std::string_view data) // Synchronous remove: parse, authorize, then run SQL.
133{
134 try // Guard against malformed JSON, just like AddStaffToEntity.
135 {
136 HttpError error; // Filled with the error if permission is denied.
137 jsoncons::json staff_j = jsoncons::json::parse(data); // Parse the raw body into JSON.
138 jsoncons::json payload = staff_j.at("payload"); // Get the staff member's fields.
139 // 'id' names the service whose permissions are checked, not the staff member.
140 uint64_t _id = staff_j.at("id").as<uint64_t>(); // The service id used for the permission check.

156/**
157 * Invites a staff member by handing an invite message to the email-sender daemon.
158 * Parses `nominee_id` and the @ref StaffData fields from the whole body, checks
159 * staff-management permission, and — if the fields parsed and the invite JSON could be
160 * built — POSTs that JSON to the configured email daemon. The daemon's body is relayed
161 * with 200 on success; 400 is returned when invite construction or delivery fails.
162 * Note: `nominee_id` is the value passed to `canManageStaff` as the permission-check target;
163 * note also that the email request is a synchronous cURL call.
164 * callback: Completion handler invoked once with (status, body).
165 * requester: Authenticated caller used for the staff-management permission check.
166 * data: Raw JSON body containing `nominee_id` and the invite fields.
167 */
168template <typename T>
169void StaffController<T>::InviteStaffToEntity(CALLBACK_&& callback, const Requester&& requester, std::string_view data) // Synchronous invite: parse, authorize, then POST an email.
170{
171 try // Guard against parse errors and network problems.
172 {
173 HttpError error; // Filled if permission is denied.
174 jsoncons::json staff_j = jsoncons::json::parse(data); // Parse the whole body as JSON.
175 // The nominee id doubles as the permission-check target for this operation.
176 uint64_t _id = staff_j.at("nominee_id").as<uint64_t>(); // Id of the person being invited; also used for the permission check.
177 StaffData staffData(staff_j); // Build typed invite fields from the JSON.
178 std::optional<std::string> response; // Will hold the daemon's reply text; empty means delivery failed.
179 T staff(staffData); // Build the concrete entity from the invite data.
180
181 if (!gateKeeper->canManageStaff<T>(requester, _id, error)) // If NOT allowed to invite for this service...
182 {
183 std::move(callback)(error.code, error.message); // Report the permission error unchanged.
184 return; // Stop; do not send any email.
185 }
186
187 // parse_status records whether StaffData found all required fields; toInviteJson adds the
188 // mail template and invitation link to the JSON that is sent downstream.
189 if (staffData.parse_status && staffData.toInviteJson(staff_j)) // Both must succeed: fields parsed AND invite JSON built.
190 {
191 // Synchronous cURL POST; an empty optional denotes delivery failure.
192 response = Communicate::sendRequest(email_sender_daemon_config_.host, email_sender_daemon_config_.port, // Send the invite to the configured email daemon (host + port).
193 email_sender_daemon_config_.message_queue_path, drogon::HttpMethod::Post, staff_j.to_string()); // Use the configured queue path, an HTTP POST, and the JSON as the body.
194
195 if (response.has_value()) // A value means the daemon replied successfully.
196 {
197 // Relay the daemon's response body to the HTTP client.
198 std::move(callback)(api::v2::Http::Status::OK, response.value()); // Reply 200 with the daemon's body; .value() unwraps the optional.
199 }
200 else // No value means the send failed.
201 {
202 std::move(callback)(api::v2::Http::Status::BAD_REQUEST, "Failed to send invite."); // Reply 400 with a diagnostic message.
203 }
204 }
205 else // Parsing or invite-JSON construction failed.
206 {
207 std::move(callback)(api::v2::Http::Status::BAD_REQUEST, "Failed to create invite json."); // Reply 400 explaining the invite could not be built.
208 }
209 }
210 catch (const std::exception& e) // Any unexpected exception is turned into an error response.
211 {
212 CRITICALMESSAGERESPONSE // Macro: reply with 500 (or 503 for a busy database).
213 }
214}

217/**
218 * Coroutine form of @ref AddStaffToEntity.
219 * Mirrors the synchronous flow but co_awaits `canManageStaffAsync` and
220 * `Controller::addStaffAsync`. The callback is invoked from the resumed coroutine;
221 * callers must co_await the returned task (the route layer uses `drogon::async_run`).
222 * callback: Completion handler invoked once with (status, body).
223 * requester: Authenticated caller used for the staff-management permission check.
224 * data: Raw JSON body; must remain valid until the task completes.
225 */
226template <typename T>
227drogon::Task<void> StaffController<T>::AddStaffToEntityAsync(CALLBACK_&& callback, const Requester&& requester, std::string_view data) // Async add: returns a pausable task.
228{
229 jsoncons::json staff_j; // Will hold the parsed JSON body.
230 HttpError error; // Filled if permission is denied.
231 try // Guard the whole coroutine body against exceptions.
232 {
233 staff_j = jsoncons::json::parse(data); // Parse the raw body into JSON.
234 jsoncons::json payload = staff_j.at("payload"); // Extract the new staff member's fields.
235 StaffData staffData(payload); // Convert them into a typed struct.
236
237 T staff(staffData); // Build the concrete entity.
238
239 uint64_t _id = staff_j.at("id").as<uint64_t>(); // Target service id for the permission check.
240
241 if (!co_await gateKeeper->canManageStaffAsync<T>(requester, _id, error)) // co_await pauses until the async permission answer arrives.
242 {
243 std::move(callback)(error.code, error.message); // Denied: report the error through the callback.
244 // Denial ends the coroutine here; no staff SQL is submitted.
245 co_return; // End this coroutine early without running SQL.
246 }
247
248 co_await Controller::addStaffAsync(staff, std::move(callback)); // Pause until the async SQL add finishes.
249 }
250 catch (const std::exception& e) // Exceptions are turned into an error response.
251 {
252 CRITICALMESSAGERESPONSE // Macro: reply with 500 (or 503 for a busy database).
253 }
254 co_return; // Every coroutine must explicitly finish with co_return.
255}
