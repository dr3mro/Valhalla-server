4 * (Originally: "Implementation for passwordcrypt: Ensures libsodium is initialized before any hashing or verification.")

8 * older scrypt format so old accounts are not locked out.

41 // Argon2id at OWASP's minimum recommended cost (19 MiB of memory, 2 passes, 1 lane; libsodium always uses one lane).
42 // "Cost" means how hard the recipe makes a guessing attack: more memory and more passes
43 // make hashes slower to attack but also slower for us to compute.
44 // The work runs on the PasswordWorker threads, never on a request thread, so this cost
45 // cannot stall the web server.
46 constexpr unsigned long long ARGON2ID_OPSLIMIT = 2; // Number of passes over memory; small but meets OWASP's minimum.
47 constexpr std::size_t ARGON2ID_MEMLIMIT = 19UL * 1024UL * 1024UL; // 19 MiB of RAM per hash; 19*1024*1024 bytes.
48
49 constexpr const char* ARGON2_PREFIX = "$argon2"; // $argon2id$... label at the start of new hashes (and $argon2i$... if imported).
50 constexpr const char* SCRYPT_PREFIX = "$7$"; // Label of libsodium's scrypt format, written by older versions of this server.
51} // namespace

68/**
69 * Hashes @p password into a self-describing Argon2id string.
70 * In everyday words: ask libsodium to pick a random salt and do the
71 * slow mixing, writing a ready-to-store string into a fixed-size buffer.
72 * password: Plaintext password.
73 * Returns Encoded hash on success; `std::nullopt` if libsodium could not allocate memory.
74 * See also: PasswordCrypt::hashPassword
75 */
76std::optional<std::string> PasswordCrypt::hashPassword(const std::string& password) // NOLINT Takes the password, returns a hash or nothing.
77{
78 char hashed_password[crypto_pwhash_STRBYTES]; // NOLINT A fixed-size char buffer big enough for any encoded hash.
79
80 // The result is a self-describing string: algorithm, cost, random salt and hash.
81 if (crypto_pwhash_str_alg(static_cast<char*>(hashed_password), password.c_str(), password.length(), ARGON2ID_OPSLIMIT, ARGON2ID_MEMLIMIT,
82 crypto_pwhash_ALG_ARGON2ID13) != 0) // NOLINT Zero means success; anything else means failure.
83 {
84 // out of memory: the required RAM could not be reserved.
85 Message::CriticalMessage("Failed to hash password."); // Leave a serious error in the log.
86 return std::nullopt; // No hash to give back; caller must handle the empty result.
87 }
88
89 return std::string(static_cast<const char*>(hashed_password)); // Copy the buffer into a std::string for the caller.
90}

92/**
93 * Verifies @p password against @p hash, selecting the algorithm from the hash prefix.
94 * In everyday words: look at the label at the front of the stored hash
95 * to decide which recipe was used, then let libsodium redo the slow
96 * mixing and tell us if the result matches.
97 * password: Plaintext password.
98 * hash: Stored self-describing hash (Argon2id or legacy scrypt).
99 * Returns `true` on a match, `false` otherwise (including unknown prefixes).
100 * See also: PasswordCrypt::verifyPassword
101 */
102bool PasswordCrypt::verifyPassword(const std::string& password, const std::string& hash) // NOLINT Returns whether the password is correct.
103{
104 // The stored string says which algorithm made it, so hashes written before the switch to Argon2id keep working.
105 if (hash.starts_with(ARGON2_PREFIX)) // Label says this is an Argon2 hash.
106 {
107 return crypto_pwhash_str_verify(hash.c_str(), password.c_str(), password.length()) == 0; // Zero from libsodium means a match.
108 }
109 if (hash.starts_with(SCRYPT_PREFIX)) // Label says this is an older scrypt hash.
110 {
111 return crypto_pwhash_scryptsalsa208sha256_str_verify(hash.c_str(), password.c_str(), password.length()) == 0; // Same idea for scrypt.
112 }
113 return false; // Unknown/foreign format: fail closed (deny) rather than guessing.
114}
