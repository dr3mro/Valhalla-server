45 // Argon2id at OWASP's minimum recommended cost (19 MiB of memory, 2 passes,
46 // 1 lane; libsodium always uses one lane). "Cost" means how hard the recipe
47 // makes a guessing attack: more memory and more passes make hashes slower
48 // to attack but also slower for us to compute. The work runs on the
49 // PasswordWorker threads, never on a request thread, so this cost cannot
50 // stall the web server.
51 constexpr unsigned long long ARGON2ID_OPSLIMIT =
52 2; // Number of passes over memory; small but meets OWASP's minimum.
53 constexpr std::size_t ARGON2ID_MEMLIMIT =
54 19UL * 1024UL * 1024UL; // 19 MiB of RAM per hash; 19*1024*1024 bytes.
55
56 constexpr const char* ARGON2_PREFIX =
57 "$argon2"; // $argon2id$... label at the start of new hashes (and
58 // $argon2i$... if imported).
59 constexpr const char* SCRYPT_PREFIX =
60 "$7$"; // Label of libsodium's scrypt format, written by older versions
61 // of this server.
62} // namespace

80/**
81 * Hashes @p password into a self-describing Argon2id string.
82 * In everyday words: ask libsodium to pick a random salt and do the
83 * slow mixing, writing a ready-to-store string into a fixed-size
84 * buffer. password: Plaintext password. Returns Encoded hash on success;
85 * `std::nullopt` if libsodium could not allocate memory. See also:
86 * PasswordCrypt::hashPassword
87 */
88std::optional<std::string> PasswordCrypt::hashPassword(
89 const std::string&
90 password) // NOLINT Takes the password, returns a hash or nothing.
91{
92 char hashed_password[crypto_pwhash_STRBYTES]; // NOLINT A fixed-size char
93 // buffer big enough for any
94 // encoded hash.
95
96 // The result is a self-describing string: algorithm, cost, random salt and
97 // hash.
98 if (crypto_pwhash_str_alg(static_cast<char*>(hashed_password),
99 password.c_str(),
100 password.length(),
101 ARGON2ID_OPSLIMIT,
102 ARGON2ID_MEMLIMIT,
103 crypto_pwhash_ALG_ARGON2ID13) !=
104 0) // NOLINT Zero means success; anything else means failure.
105 {
106 // out of memory: the required RAM could not be reserved.
107 Message::CriticalMessage(
108 "Failed to hash password."); // Leave a serious error in the log.
109 return std::nullopt; // No hash to give back; caller must handle the
110 // empty result.
111 }
112
113 return std::string(static_cast<const char*>(
114 hashed_password)); // Copy the buffer into a std::string for the
115 // caller.
116}

118/**
119 * Verifies @p password against @p hash, selecting the algorithm from the hash
120 * prefix. In everyday words: look at the label at the front of the stored hash
121 * to decide which recipe was used, then let libsodium redo the slow
122 * mixing and tell us if the result matches.
123 * password: Plaintext password.
124 * hash: Stored self-describing hash (Argon2id or legacy scrypt).
125 * Returns `true` on a match, `false` otherwise (including unknown prefixes).
126 * See also: PasswordCrypt::verifyPassword
127 */
128bool PasswordCrypt::verifyPassword(
129 const std::string& password,
130 const std::string&
131 hash) // NOLINT Returns whether the password is correct.
132{
133 // The stored string says which algorithm made it, so hashes written before
134 // the switch to Argon2id keep working.
135 // Label says this is an Argon2 hash.
136 if (hash.starts_with(ARGON2_PREFIX)) {
137 return crypto_pwhash_str_verify(hash.c_str(),
138 password.c_str(),
139 password.length()) ==
140 0; // Zero from libsodium means a match.
141 }
142 // Label says this is an older scrypt hash.
143 if (hash.starts_with(SCRYPT_PREFIX)) {
144 return crypto_pwhash_scryptsalsa208sha256_str_verify(hash.c_str(),
145 password.c_str(),
146 password.length()) ==
147 0; // Same idea for scrypt.
148 }
149 return false; // Unknown/foreign format: fail closed (deny) rather than
150 // guessing.
151}
