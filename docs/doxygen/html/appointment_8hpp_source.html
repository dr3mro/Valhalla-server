1/**
2 * @file appointment.hpp
3 * @brief Base entity for appointment tables shared by the service-specific
4 * appointment classes.
5 * @details Appointment extends the generic Entity with two static
6 * permission-query builders. One fetches the owner, admin and staff of a
7 * service (organisation) by its id; the other finds the owning service from an
8 * appointment id by joining the appointment table to its organisation table.
9 * Concrete classes fix the table and organisation names and expose these
10 * helpers to PermissionManager, which uses them to authorise appointment
11 * operations.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * Appointments belong to a service (a clinic, pharmacy, laboratory or radiology
15 * centre), so access must be checked against that service's owner/admin/staff
16 * rather than the appointment alone. Controllers call the queries produced here
17 * before allowing an appointment to be created or read. Without Appointment,
18 * each service-specific appointment class would duplicate this ownership lookup
19 * and permission checks would be inconsistent.
20 *
21 * CONCEPTS TO LEARN (for beginners):
22 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
23 * - Static member function: belongs to the class, not an object, so it is
24 * called as Class::fn().
25 * - SQL JOIN: combines rows from two tables using a related column (here a
26 * foreign key to the service id), letting us find the owning service from an
27 * appointment id.
28 * - SQL parameter binding: put $1, $2, ... markers in the SQL and pass the real
29 * values separately; see https://en.cppreference.com/w/cpp or the libpqxx
30 * documentation. (These permission queries are the exception: they use only
31 * internal, trusted names and ids.)
32 *
33 * @author Amr Osman <dr3mro@gmail.com>
34 * @date 2026-11-16
35 */

94 /**
95 * @brief Builds a permission query for a service row by id.
96 * @param[in] service_name Organisation table name (for example "clinics").
97 * @param[in] service_id Primary key of the service row.
98 * @return SELECT of owner_id, admin_id and staff for the given service.

100 // "Impl" suffix: shared implementation used
101 // by concrete subclasses.
102 static std::optional<std::string> getServicePermissionsQueryImpl(
103 const std::string& service_name, std::uint64_t service_id) {
104 // The SQL selects owner_id, admin_id and staff from the given
105 // organisation table for one service id. PermissionManager runs this to
106 // see who is allowed to act on appointments in that service. Names and
107 // ids are internal and trusted, so they are formatted into the text.
108 return fmt::format(R"(
109 SELECT
110 owner_id,
111 admin_id,
112 staff
113 FROM
114 {}
115 WHERE
116 id = {};)",
117 service_name,
118 service_id); // Fill in the organisation table name and
119 // the service id.
120 }

122 /**
123 * @brief Builds a permission query that resolves a service from an
124 * appointment id.
125 * @details Joins the organisation table to the appointment table on the
126 * singular foreign key column and returns the owning service's
127 * owner/admin/staff for the given appointment.
128 * @param[in] org_name Plural organisation table name (for example
129 * "clinics").
130 * @param[in] appointments_table Appointment table name.
131 * @param[in] appointment_id Primary key of the appointment row.
132 * @return SELECT statement identifying the owning service's permission
133 * fields.
134 */
135 // Finds which service owns an appointment,
136 // then its permission fields.
137 static std::optional<std::string> getPermissionsQueryForReadImpl(
138 const std::string& org_name,
139 const std::string& appointments_table,
140 std::uint64_t appointment_id) {
141 // org_name is plural (e.g. "clinics") but the FK column is singular
142 // (e.g. "clinic_id")
143 std::string fk_col =
144 org_name.back() == 's'
145 ? org_name.substr(0, org_name.size() - 1) + "_id"
146 : org_name + "_id"; // Derive "clinic_id" from "clinics" by
147 // dropping the trailing 's'; handles names
148 // without an 's' too.
149 // The SQL joins organisation s to appointment a on a.<fk_col> = s.id,
150 // then filters to the given appointment id, returning the service's
151 // owner/admin/staff. This is how the server maps "can this person read
152 // appointment X?" to "do they belong to the service that owns X?".
153 return fmt::format(R"(
154 SELECT
155 s.owner_id,
156 s.admin_id,
157 s.staff
158 FROM
159 {} s
160 JOIN {} a ON a.{} = s.id
161 WHERE
162 a.id = {};)",
163 org_name,
164 appointments_table,
165 fk_col,
166 appointment_id); // Fill in the organisation table,
167 // appointment table, foreign-key
168 // column and appointment id.
169 }
170};
