1/**
2 * @file appointment.hpp
3 * @brief Base entity for appointment tables shared by the service-specific appointment classes.
4 * @details Appointment extends the generic Entity with two static permission-query builders. One
5 * fetches the owner, admin and staff of a service (organisation) by its id; the other finds the
6 * owning service from an appointment id by joining the appointment table to its organisation table.
7 * Concrete classes fix the table and organisation names and expose these helpers to
8 * PermissionManager, which uses them to authorise appointment operations.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * Appointments belong to a service (a clinic, pharmacy, laboratory or radiology centre), so access
12 * must be checked against that service's owner/admin/staff rather than the appointment alone.
13 * Controllers call the queries produced here before allowing an appointment to be created or read.
14 * Without Appointment, each service-specific appointment class would duplicate this ownership
15 * lookup and permission checks would be inconsistent.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
19 * - Static member function: belongs to the class, not an object, so it is called as Class::fn().
20 * - SQL JOIN: combines rows from two tables using a related column (here a foreign key to the
21 * service id), letting us find the owning service from an appointment id.
22 * - SQL parameter binding: put $1, $2, ... markers in the SQL and pass the real values separately;
23 * see https://en.cppreference.com/w/cpp or the libpqxx documentation. (These permission queries
24 * are the exception: they use only internal, trusted names and ids.)
25 *
26 * @author Amr Osman <dr3mro@gmail.com>
27 * @date 2026-11-16
28 */

78 /**
79 * @brief Builds a permission query for a service row by id.
80 * @param[in] service_name Organisation table name (for example "clinics").
81 * @param[in] service_id Primary key of the service row.
82 * @return SELECT of owner_id, admin_id and staff for the given service.

84 static std::optional<std::string> getServicePermissionsQueryImpl(const std::string& service_name, std::uint64_t service_id) // "Impl" suffix: shared implementation used by concrete subclasses.
85 {
86 // The SQL selects owner_id, admin_id and staff from the given organisation table for one
87 // service id. PermissionManager runs this to see who is allowed to act on appointments in
88 // that service. Names and ids are internal and trusted, so they are formatted into the text.
89 return fmt::format(R"(
90 SELECT
91 owner_id,
92 admin_id,
93 staff
94 FROM
95 {}
96 WHERE
97 id = {};)",
98 service_name, service_id); // Fill in the organisation table name and the service id.

101 /**
102 * @brief Builds a permission query that resolves a service from an appointment id.
103 * @details Joins the organisation table to the appointment table on the singular foreign key
104 * column and returns the owning service's owner/admin/staff for the given appointment.
105 * @param[in] org_name Plural organisation table name (for example "clinics").
106 * @param[in] appointments_table Appointment table name.
107 * @param[in] appointment_id Primary key of the appointment row.
108 * @return SELECT statement identifying the owning service's permission fields.
109 */
110 static std::optional<std::string> getPermissionsQueryForReadImpl(
111 const std::string& org_name, const std::string& appointments_table, std::uint64_t appointment_id) // Finds which service owns an appointment, then its permission fields.
112 {
113 // org_name is plural (e.g. "clinics") but the FK column is singular (e.g. "clinic_id")
114 std::string fk_col = org_name.back() == 's' ? org_name.substr(0, org_name.size() - 1) + "_id" : org_name + "_id"; // Derive "clinic_id" from "clinics" by dropping the trailing 's'; handles names without an 's' too.
115 // The SQL joins organisation s to appointment a on a.<fk_col> = s.id, then filters to the
116 // given appointment id, returning the service's owner/admin/staff. This is how the server
117 // maps "can this person read appointment X?" to "do they belong to the service that owns X?".
118 return fmt::format(R"(
119 SELECT
120 s.owner_id,
121 s.admin_id,
122 s.staff
123 FROM
124 {} s
125 JOIN {} a ON a.{} = s.id
126 WHERE
127 a.id = {};)",
128 org_name, appointments_table, fk_col, appointment_id); // Fill in the organisation table, appointment table, foreign-key column and appointment id.
129 }
