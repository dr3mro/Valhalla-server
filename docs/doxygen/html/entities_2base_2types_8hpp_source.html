1/**
2 * @file types.hpp
3 * @brief CRUD operation payload types and the EntityType variant used by entity builders.
4 * @details This header declares the small value objects that describe every operation the API can
5 * request: Create_t, Read_t, Update_t, Delete_t, plus Search_t, client payloads, suspend and staff
6 * data. Most of them derive from a shared Entity_t template that stores a payload and an optional
7 * id. At the bottom, EntityType lists them all as one std::variant, which is how an Entity keeps
8 * track of which operation it represents. Several types also validate and sanitise input before
9 * the SQL builders ever see it.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * These are the data carriers between controllers and the entity layer. A controller parses an
13 * HTTP body into one of these types, and the Entity/Client/Service builders read it back to produce
14 * SQL. Validation lives here (for example Search_t rejects unsafe column names), so bad input is
15 * stopped in one place instead of being trusted by every SQL builder. Without this file the
16 * builders would have no typed, validated description of what the client asked for.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - std::variant: https://en.cppreference.com/w/cpp/utility/variant
20 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
21 * - SQL parameter binding: put $1, $2, ... markers in the SQL and pass the real values separately;
22 * see https://en.cppreference.com/w/cpp or the libpqxx documentation.
23 * - A "template" is a blueprint for a class that works with many types. Here the blueprint takes
24 * a type parameter named S, and the compiler generates a separate class for each S that is used.
25 *
26 * @author Amr Osman <dr3mro@gmail.com>
27 * @date 2026-11-16
28 */

60 * @class Entity_t
61 * @brief Immutable payload-plus-optional-id base for the CRUD operation value types.
62 * @details Stores the operation's data (a JSON object for create/update, a column set for
63 * read, a sentinel for delete/data) together with an optional primary key. The payload and id
64 * are const, so an instance is a fixed description of one requested operation. Concrete types
65 * below are empty specialisations that choose S and expose construction appropriate to the
66 * operation.
67 * @tparam S Type of the payload returned by get_data().
68 * @see Create_t, Read_t, Update_t, Delete_t, Data_t
69 */
70 template <typename S>
71 struct Entity_t
72 {
73 public:
74 // Entity_t(const Entity_t &) = delete;
75 // Entity_t(Entity_t &&) = delete;
76 // Entity_t &operator=(const Entity_t &) = delete;
77 // Entity_t &operator=(Entity_t &&) = delete;
78
79 /**
80 * @brief Stores a payload and an optional id.
81 * @param[in] _data Operation payload.
82 * @param[in] _id Primary key, or std::nullopt when the database should assign/omit it.
83 */
84 Entity_t(const S& _data, const std::optional<uint64_t> _id) : data(_data), id(_id) {} // Copy both inputs into const members so the operation description cannot change later.
85
86 /**
87 * @brief Returns the optional primary key associated with the operation.
88 */
89 [[nodiscard]] std::optional<uint64_t> get_id() const { return id; } // Return by value; optional is tiny, and callers can check has_value().
90
91 /**
92 * @brief Returns the immutable operation payload.
93 */
94 [[nodiscard]] const S& get_data() const { return this->data; } // Return a const reference to avoid copying potentially large JSON.
95
96 /**
97 * @brief Virtual destructor so derived operation types can be destroyed polymorphically.
98 */
99 virtual ~Entity_t() = default; // Virtual because concrete types inherit and may be deleted through an Entity_t pointer.
100
101 private:
102 /**
103 * @brief The immutable operation payload.
104 */
105 const S data; // The actual data (JSON, column set or sentinel); const keeps the request fixed.
106
107 /**
108 * @brief The immutable optional primary key.
109 */
110 const std::optional<uint64_t> id; // An id that may be absent; optional avoids using a fake value like 0 to mean "no id".
111 };

243 /**
244 * @class Search_t
245 * @brief Parsed and validated parameters for a paginated, case-insensitive search.
246 * @details Construction from a JSON body both extracts the fields and reports, through the
247 * success flag, whether the body is well formed and safe. filter and order_by end up as SQL
248 * identifiers in the generated statement (they cannot be bound), so they are restricted to
249 * plain identifiers; the remaining values are bound as parameters by the Entity builders.
250 * @see Entity::buildSearchStatement
251 */
252 using Search_t = struct Search_t
253 {
254 /**
255 * @brief Substring to match (wrapped in %...% when bound).
256 */
257 std::string keyword; // The text the user typed; it is a value and will be bound as a parameter.
258
259 /**
260 * @brief Column name used in the WHERE clause; must be a plain identifier.
261 */
262 std::string filter; // Written into SQL as an identifier, so it must pass isPlainIdentifier().
263
264 /**
265 * @brief Column name used in ORDER BY; must be a plain identifier.
266 */
267 std::string order_by; // Also written into SQL, so it must pass isPlainIdentifier() too.
268
269 /**
270 * @brief Sort direction, "ASC" or "DESC", derived from the JSON "direction" value.
271 */
272 std::string direction; // Chosen from a numeric flag in the body, not taken directly from user text.
273
274 /**
275 * @brief Maximum number of rows requested (the builder binds limit + 1 for next-page detection).
276 */
277 size_t limit; // A count, so an unsigned size_t is natural; the extra row indicates more results.
278
279 /**
280 * @brief Number of rows to skip.
281 */
282 size_t offset; // Pagination skip count; also bound as a value.
283
284 /**
285 * @brief Parses a search JSON body and reports whether it can be used.
286 * @param[in] search_j JSON body expected to contain keyword, filter, order_by, direction,
287 * limit and offset.
288 * @param[out] success Set true only when every field is present, correctly typed and validated;
289 * callers use this to return a normal client error instead of throwing. On any problem it is
290 * set false and the remaining fields must not be used.
291 */
292 Search_t(const jsoncons::json& search_j, bool& success) // The bool is an output parameter: parsing never throws at the caller for a bad body.
293 {
294 // A missing/malformed field here (e.g. an empty "{}" body) is a client error, not a
295 // server error. success=false tells the caller (EntityController::Search) to reply with
296 // a normal "Failed to search" response instead of throwing - rethrowing after setting
297 // success=false used to defeat that signal and escape uncaught into the generic 500
298 // handler, leaking the server's file path in the response body.
299 try
300 {
301 const jsoncons::json &kw = search_j.at("keyword"), &fl = search_j.at("filter"), &ob = search_j.at("order_by"), &dr = search_j.at("direction"),
302 &lm = search_j.at("limit"), &of = search_j.at("offset"); // Grab references to all six fields; .at throws if a key is missing.
303 // jsoncons' as<T>() happily coerces across types (a JSON bool/null becomes "true"/""
304 // as a string, for instance) instead of throwing, which let a body that violates the
305 // declared schema (e.g. "order_by": false, "limit": null) sail through this try block
306 // untouched. Check the actual JSON type first so a mismatch is treated the same as any
307 // other malformed search body - success=false, not silently-coerced garbage.
308 if (!kw.is_string() || !fl.is_string() || !ob.is_string() || !dr.is_int64() || !lm.is_int64() || !of.is_int64()) // Reject wrong JSON types before coercion can hide them.
309 {
310 success = false; // Mark the body unusable.
311 return; // Stop early; the member fields are left unset but will not be read.
312 }
313 keyword = kw.as<std::string>(); // Safe now that the type is known to be a string.
314 filter = fl.as<std::string>(); // Safe string conversion.
315 order_by = ob.as<std::string>(); // Safe string conversion.
316 direction = dr.as<short>() == 0 ? "ASC" : "DESC"; // Map a numeric flag to a fixed SQL keyword, never raw user text.
317 limit = lm.as<size_t>(); // Convert the numeric limit to a count.
318 offset = of.as<size_t>(); // Convert the numeric offset to a count.
319 success = validate(search_j); // Final gate: exact key set and safe identifiers.
320 }
321 catch (const std::exception&) // Covers a missing key (.at) or a bad numeric conversion.
322 {
323 success = false; // Any parsing problem means the body must not be used.
324 }
325 }
326 // "filter" and "order_by" are written into the SQL text (identifiers cannot be bound), so they must be plain
327 // identifiers: letters, digits and underscores only, not starting with a digit.
328 /**
329 * @brief Checks that name is a safe SQL identifier (letters/digits/underscore, not digit-first).
330 * @details Enforces PostgreSQL's 63-character limit as well, because the name is
331 * interpolated into the statement rather than bound.
332 * @param[in] name Candidate column identifier.
333 * @return true only when the name is non-empty, within the length limit and plain.
334 */
335 static bool isPlainIdentifier(const std::string& name) // Static because it needs no object state; a free-standing helper on the struct.
336 {
337 constexpr size_t MAX_IDENTIFIER_LENGTH = 63; // PostgreSQL's limit; longer names would be truncated.
338 if (name.empty() || name.size() > MAX_IDENTIFIER_LENGTH || std::isdigit(static_cast<unsigned char>(name.front())) != 0) // Reject empty, too long, or starting with a digit (which would be an invalid SQL identifier).
339 {
340 return false; // Unsafe name.
341 }
342 return std::ranges::all_of(name, [](char c) { return std::isalnum(static_cast<unsigned char>(c)) != 0 || c == '_'; }); // Every character must be a letter, digit or underscore.
343 }
344
345 /**
346 * @brief Verifies that the body has exactly the six search keys and safe identifiers.
347 * @param[in] search_j The original JSON body.
348 * @return true when the key set is exact and filter/order_by are plain identifiers.
349 */
350 bool validate(const jsoncons::json& search_j) // Non-static because it reads the already-parsed member fields.
351 {
352 static const std::unordered_set<std::string> keys = {"keyword", "filter", "order_by", "direction", "limit", "offset"}; // The complete allowed key set, built once.
353 return search_j.size() == keys.size() && std::ranges::all_of(keys, [&search_j](const std::string& key) { return search_j.find(key).has_value(); }) && isPlainIdentifier(filter) &&
354 isPlainIdentifier(order_by); // No extra or missing keys, and both identifiers are safe to paste into SQL.
355 }
356 };

358 /**
359 * @class ClientData_t
360 * @brief Common parsing and field extraction for client-style create/update payloads.
361 * @details Parses a raw JSON body, records parse errors through error/success, and keeps both
362 * the parsed JSON and a flat set of (column, value) pairs used by the Client SQL builders.
363 * Password hashing on the flat set is available to subclasses.
364 * @see CreateClient_t, UpdateClient_t
365 */
366 struct ClientData_t
367 {
368 public:
369 /**
370 * @brief Parses the raw request body into data_j and reports status via error/success.
371 * @param[in] _data Raw JSON text of the request body.
372 * @param[in] _id Optional client id (used by updates).
373 * @param[out] error Receives a BAD_REQUEST status for an unparsable body or an
374 * INTERNAL_SERVER_ERROR for an exception, with a diagnostic message.
375 * @param[out] success Set true only when parsing produced a JSON value.
376 */
377 ClientData_t(std::string_view _data, const std::optional<uint64_t> _id, api::v2::Http::Error& error, bool& success) : id(_id) // string_view avoids copying the body; only the id is stored here initially.
378 {
379 try
380 {
381 data_j = jsoncons::json::parse(_data); // Turn the raw text into a JSON value.
382 if (!data_j.has_value()) // A null result means the text was not valid JSON.
383 {
384 error = {.code = api::v2::Http::Status::BAD_REQUEST, .message = "Failed to parse body."}; // A bad body is the client's fault (400).
385 Message::ErrorMessage(error.message); // Log it for operators.
386 success = false; // Tell the caller not to proceed.
387 return; // Stop before the success = true at the end.
388 }
389 }
390 catch (const std::exception& e) // An unexpected exception while parsing is a server-side problem.
391 {
392 error = {.code = api::v2::Http::Status::INTERNAL_SERVER_ERROR, .message = fmt::format("Failed while parsing client data: {}.", e.what())}; // Report 500 with the exception text.
393 success = false; // Mark failure.
394 Message::CriticalMessage(error.message); // Log at critical level.
395 return; // Stop early.
396 }
397
398 success = true; // Reaching here means parsing produced a usable JSON value.
399 }
400
401 /**
402 * @brief Returns the parsed body, if parsing succeeded.
403 */
404 const std::optional<jsoncons::json>& get_data_json() const { return data_j; } // Const reference avoids copying; callers must check has_value() first.
405
406 /**
407 * @brief Returns the optional client id supplied at construction.
408 */
409 std::optional<uint64_t> get_id() const { return id; } // Return by value; callers check has_value() before using it.
410
411 protected:
412 /**
413 * @brief Flattened (column, value) pairs derived from the validated body.
414 */
415 std::unordered_set<std::pair<std::string, std::string>> data_set; // Each pair is one SQL column and its value; a set prevents duplicates.
416
417 /**
418 * @brief The parsed JSON body, or nullopt if parsing failed.
419 */
420 std::optional<jsoncons::json> data_j; // optional because parsing may fail.
421
422 /**
423 * @brief Optional client id.
424 */
425 std::optional<uint64_t> id; // Present for updates, absent for creates.
426
427 /**
428 * @brief Replaces a plaintext "password" entry in data_set with its hash.
429 * @return true when a password was found and hashed; false when there was no password or
430 * hashing failed. Logs a critical message on a hashing failure in the caller path.
431 * @note Operates on the flat pair set rather than data_j, so it must run after validation
432 * has populated data_set.
433 */
434 bool hashPassword() // Protected so subclasses can choose whether to hash (creates do, updates do not).
435 {
436 auto passwd_itr = std::ranges::find_if(data_set, [&](const auto& item) { return item.first == "password"; }); // Look for a column named "password" in the flat set.
437 if (passwd_itr != data_set.end()) // Only act if such an entry exists.
438 {
439 std::string password_raw = passwd_itr->second; // Copy the plaintext out before erasing it.
440 std::optional<std::string> hashed_password = passwordCrypt->hashPassword(password_raw); // Ask the shared hasher; may fail (nullopt).
441 if (!hashed_password.has_value()) // Hashing failed - do not write a plaintext password.
442 {
443 return false; // Signal failure so the caller can react.
444 }
445 data_set.erase(passwd_itr); // Remove the plaintext pair.
446 data_set.emplace("password", hashed_password.value()); // Insert the hashed pair in its place.
447 return true; // Report success.
448 }
449 return false; // No password field existed; nothing to do.
450 }
451
452 private:
453 /**
454 * @brief Shared password hasher obtained from the global object store.
455 */
456 std::shared_ptr<PasswordCrypt> passwordCrypt = Store::cachedObject<PasswordCrypt>(); // One process-wide hasher; shared_ptr prevents it being destroyed while in use.
457 };

459 /**
460 * @class CreateClient_t
461 * @brief ClientData_t for creating a client: validates a create schema and hashes any password.
462 * @details Rejects a body that already contains "id" so clients cannot pick their own key,
463 * runs regex validation to build data_set, then hashes the password before the INSERT is built.
464 */
465 struct CreateClient_t : public ClientData_t
466 {
467 public:
468 /**
469 * @brief Parses and validates a client-create body, populating data_set.
470 * @param[in] _data Raw JSON request body.
471 * @param[in] tablename Table whose create schema is used for validation.
472 * @param[out] error Receives a diagnostic on validation failure.
473 * @param[out] success Set true only after schema and regex validation succeed.
474 */
475 CreateClient_t(std::string_view _data, const std::string& tablename, api::v2::Http::Error& error, bool& success)
476 : ClientData_t(_data, std::nullopt, error, success) // Creates have no id yet, so pass nullopt to the base.
477 {
478 try
479 {
480 Validator::Rule rule(Validator::Rule::Action::ASSERT_NOT_PRESENT, {"id"}); // Build a rule saying "id must not be in the body".
481 success = Validator::validateDatabaseCreateSchema(tablename, data_j, error, rule); // Check the body against the table's create schema.
482 if (!success) // Schema validation failed.
483 {
484 Message::ErrorMessage(error.message); // Log the validation error.
485 return; // Leave success false and stop.
486 }
487
488 success = Validator::clientRegexValidation(data_j, error, data_set); // Regex-validate fields and fill the (column, value) set.
489 if (!success) // Regex validation failed.
490 {
491 Message::ErrorMessage(error.message); // Log it.
492 return; // Stop with success false.
493 }
494 hashPassword(); // Now that data_set exists, replace any plaintext password with its hash.
495 }
496 catch (const std::exception& e) // Unexpected error during validation.
497 {
498 error = {.code = api::v2::Http::Status::INTERNAL_SERVER_ERROR, .message = fmt::format("Failed while parsing client data: {}.", e.what())}; // Report 500.
499 success = false; // Mark failure.
500 Message::CriticalMessage(error.message); // Log details.
501 return; // Stop.
502 }
503
504 success = true; // Everything passed.
505 }
506 /**
507 * @brief Returns the validated (and password-hashed) column/value pairs for the INSERT.
508 */
509 const std::unordered_set<std::pair<std::string, std::string>>& get_data_set() const { return data_set; } // Const reference so callers cannot mutate the validated data.
510 };

512 /**
513 * @class UpdateClient_t
514 * @brief ClientData_t for updating a client: validates an update schema with a caller-supplied rule.
515 * @details Unlike CreateClient_t it does not hash passwords; the validation rule is provided
516 * by the caller because different update endpoints permit different field sets.
517 */
518 struct UpdateClient_t : public ClientData_t
519 {
520 public:
521 /**
522 * @brief Parses and validates a client-update body, populating data_set.
523 * @param[in] _data Raw JSON request body.
524 * @param[in] _id Client id whose row will be updated.
525 * @param[in] tablename Table whose update schema is used for validation.
526 * @param[out] error Receives a diagnostic on validation failure.
527 * @param[out] success Set true only after schema and regex validation succeed.
528 * @param[in] rule Validation rule (for example which fields may or may not be present).
529 * @note Unlike CreateClient_t this type does not re-hash the password: the generic Update
