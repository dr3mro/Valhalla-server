1/**
2 * @file types.hpp
3 * @brief CRUD operation payload types and the EntityType variant used by entity
4 * builders.
5 * @details This header declares the small value objects that describe every
6 * operation the API can request: Create_t, Read_t, Update_t, Delete_t, plus
7 * Search_t, client payloads, suspend and staff data. Most of them derive from a
8 * shared Entity_t template that stores a payload and an optional id. At the
9 * bottom, EntityType lists them all as one std::variant, which is how an Entity
10 * keeps track of which operation it represents. Several types also validate and
11 * sanitise input before the SQL builders ever see it.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * These are the data carriers between controllers and the entity layer. A
15 * controller parses an HTTP body into one of these types, and the
16 * Entity/Client/Service builders read it back to produce SQL. Validation lives
17 * here (for example Search_t rejects unsafe column names), so bad input is
18 * stopped in one place instead of being trusted by every SQL builder. Without
19 * this file the builders would have no typed, validated description of what the
20 * client asked for.
21 *
22 * CONCEPTS TO LEARN (for beginners):
23 * - std::variant: https://en.cppreference.com/w/cpp/utility/variant
24 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
25 * - SQL parameter binding: put $1, $2, ... markers in the SQL and pass the real
26 * values separately; see https://en.cppreference.com/w/cpp or the libpqxx
27 * documentation.
28 * - A "template" is a blueprint for a class that works with many types. Here
29 * the blueprint takes a type parameter named S, and the compiler generates a
30 * separate class for each S that is used.
31 *
32 * @author Amr Osman <dr3mro@gmail.com>
33 * @date 2026-11-16
34 */

80 * @class Entity_t
81 * @brief Immutable payload-plus-optional-id base for the CRUD operation
82 * value types.
83 * @details Stores the operation's data (a JSON object for create/update, a
84 * column set for read, a sentinel for delete/data) together with an
85 * optional primary key. The payload and id are const, so an instance is a
86 * fixed description of one requested operation. Concrete types below are
87 * empty specialisations that choose S and expose construction appropriate
88 * to the operation.
89 * @tparam S Type of the payload returned by get_data().
90 * @see Create_t, Read_t, Update_t, Delete_t, Data_t
91 */
92 template <typename S>
93 struct Entity_t {
94 public:
95 // Entity_t(const Entity_t &) = delete;
96 // Entity_t(Entity_t &&) = delete;
97 // Entity_t &operator=(const Entity_t &) = delete;

100 /**
101 * @brief Stores a payload and an optional id.
102 * @param[in] _data Operation payload.
103 * @param[in] _id Primary key, or std::nullopt when the database should
104 * assign/omit it.
105 */
106 Entity_t(const S& _data, const std::optional<uint64_t> _id)
107 : data(_data), id(_id) {
108 } // Copy both inputs into const members so the operation description
109 // cannot change later.
110
111 /**
112 * @brief Returns the optional primary key associated with the
113 * operation.
114 */
115 [[nodiscard]] std::optional<uint64_t> get_id() const {
116 return id;
117 } // Return by value; optional is tiny, and callers can check
118 // has_value().

121 * @brief Returns the immutable operation payload.
122 */
123 [[nodiscard]] const S& get_data() const {
124 return this->data;
125 } // Return a const reference to avoid copying potentially large JSON.

132 default; // Virtual because concrete types inherit and may be

135 private:
136 /**
137 * @brief The immutable operation payload.

139 const S data; // The actual data (JSON, column set or sentinel); const

143 * @brief The immutable optional primary key.

304 /**
305 * @class Search_t
306 * @brief Parsed and validated parameters for a paginated, case-insensitive
307 * search.
308 * @details Construction from a JSON body both extracts the fields and
309 * reports, through the success flag, whether the body is well formed and
310 * safe. filter and order_by end up as SQL identifiers in the generated
311 * statement (they cannot be bound), so they are restricted to plain
312 * identifiers; the remaining values are bound as parameters by the Entity
313 * builders.
314 * @see Entity::buildSearchStatement
315 */
316 using Search_t = struct Search_t {
317 /**
318 * @brief Substring to match (wrapped in %...% when bound).
319 */
320 std::string keyword; // The text the user typed; it is a value and will
321 // be bound as a parameter.
322
323 /**
324 * @brief Column name used in the WHERE clause; must be a plain
325 * identifier.
326 */
327 std::string filter; // Written into SQL as an identifier, so it must
328 // pass isPlainIdentifier().
329
330 /**
331 * @brief Column name used in ORDER BY; must be a plain identifier.
332 */
333 std::string order_by; // Also written into SQL, so it must pass
334 // isPlainIdentifier() too.
335
336 /**
337 * @brief Sort direction, "ASC" or "DESC", derived from the JSON
338 * "direction" value.
339 */
340 std::string direction; // Chosen from a numeric flag in the body, not
341 // taken directly from user text.
342
343 /**
344 * @brief Maximum number of rows requested (the builder binds limit + 1
345 * for next-page detection).
346 */
347 size_t limit; // A count, so an unsigned size_t is natural; the extra
348 // row indicates more results.
349
350 /**
351 * @brief Number of rows to skip.
352 */
353 size_t offset; // Pagination skip count; also bound as a value.
354
355 /**
356 * @brief Parses a search JSON body and reports whether it can be used.
357 * @param[in] search_j JSON body expected to contain keyword, filter,
358 * order_by, direction, limit and offset.
359 * @param[out] success Set true only when every field is present,
360 * correctly typed and validated; callers use this to return a normal
361 * client error instead of throwing. On any problem it is set false and
362 * the remaining fields must not be used.
363 */
364 // The bool is an output parameter: parsing never
365 // throws at the caller for a bad body.
366 Search_t(const jsoncons::json& search_j, bool& success) {
367 // A missing/malformed field here (e.g. an empty "{}" body) is a
368 // client error, not a server error. success=false tells the caller
369 // (EntityController::Search) to reply with a normal "Failed to
370 // search" response instead of throwing - rethrowing after setting
371 // success=false used to defeat that signal and escape uncaught into
372 // the generic 500 handler, leaking the server's file path in the
373 // response body.
374 try {
375 const jsoncons::json
376 &kw = search_j.at("keyword"),
377 &fl = search_j.at("filter"), &ob = search_j.at("order_by"),
378 &dr = search_j.at("direction"), &lm = search_j.at("limit"),
379 &of = search_j.at("offset"); // Grab references to all six fields;
380 // .at throws if a key is missing.
381 // jsoncons' as<T>() happily coerces across types (a JSON
382 // bool/null becomes "true"/"" as a string, for instance)
383 // instead of throwing, which let a body that violates the
384 // declared schema (e.g. "order_by": false, "limit": null) sail
385 // through this try block untouched. Check the actual JSON type
386 // first so a mismatch is treated the same as any other
387 // malformed search body - success=false, not silently-coerced
388 // garbage.
389 // Reject wrong JSON types before coercion
390 // can hide them.
391 if (!kw.is_string() || !fl.is_string() || !ob.is_string() ||
392 !dr.is_int64() || !lm.is_int64() || !of.is_int64()) {
393 success = false; // Mark the body unusable.
394 return; // Stop early; the member fields are left unset but
395 // will not be read.
396 }
397 keyword = kw.as<std::string>(); // Safe now that the type is
398 // known to be a string.
399 filter = fl.as<std::string>(); // Safe string conversion.
400 order_by = ob.as<std::string>(); // Safe string conversion.
401 direction = dr.as<short>() == 0
402 ? "ASC"
403 : "DESC"; // Map a numeric flag to a fixed SQL
404 // keyword, never raw user text.
405 limit = lm.as<size_t>(); // Convert the numeric limit to a count.
406 offset = of.as<size_t>(); // Convert the numeric offset to a count.
407 // Final gate: exact key set and
408 // safe identifiers.
409 // Covers a missing key (.at) or a
410 // bad numeric conversion.
411 success = validate(search_j);
412 } catch (const std::exception&) {
413 success = false; // Any parsing problem means the body must not
414 // be used.
415 }
416 }
417 // "filter" and "order_by" are written into the SQL text (identifiers
418 // cannot be bound), so they must be plain identifiers: letters, digits
419 // and underscores only, not starting with a digit.
420 /**
421 * @brief Checks that name is a safe SQL identifier
422 * (letters/digits/underscore, not digit-first).
423 * @details Enforces PostgreSQL's 63-character limit as well, because
424 * the name is interpolated into the statement rather than bound.
425 * @param[in] name Candidate column identifier.
426 * @return true only when the name is non-empty, within the length limit
427 * and plain.
428 */
429 // Static because it needs no object state; a
430 // free-standing helper on the struct.
431 static bool isPlainIdentifier(const std::string& name) {
432 constexpr size_t MAX_IDENTIFIER_LENGTH =
433 63; // PostgreSQL's limit; longer names would be truncated.
434 // Reject empty, too long, or starting with a digit
435 // (which would be an invalid SQL identifier).
436 if (name.empty() || name.size() > MAX_IDENTIFIER_LENGTH ||
437 std::isdigit(static_cast<unsigned char>(name.front())) != 0) {
438 return false; // Unsafe name.
439 }
440 return std::ranges::all_of(name, [](char c) {
441 return std::isalnum(static_cast<unsigned char>(c)) != 0 || c == '_';
442 }); // Every character must be a letter, digit
443 // or underscore.
444 }
445
446 /**
447 * @brief Verifies that the body has exactly the six search keys and
448 * safe identifiers.
449 * @param[in] search_j The original JSON body.
450 * @return true when the key set is exact and filter/order_by are plain
451 * identifiers.
452 */
453 // Non-static because it reads the
454 // already-parsed member fields.
455 bool validate(const jsoncons::json& search_j) {
456 static const std::unordered_set<std::string> keys =
457 {"keyword",
458 "filter",
459 "order_by",
460 "direction",
461 "limit",
462 "offset"}; // The complete allowed key set, built once.
463 return search_j.size() == keys.size() &&
464 std::ranges::all_of(keys,
465 [&search_j](const std::string& key) {
466 return search_j.find(key).has_value();
467 }) &&
468 isPlainIdentifier(filter) &&
469 isPlainIdentifier(
470 order_by); // No extra or missing keys, and both
471 // identifiers are safe to paste into SQL.
472 }
473 };

475 /**
476 * @class ClientData_t
477 * @brief Common parsing and field extraction for client-style create/update
478 * payloads.
479 * @details Parses a raw JSON body, records parse errors through
480 * error/success, and keeps both the parsed JSON and a flat set of (column,
481 * value) pairs used by the Client SQL builders. Password hashing on the
482 * flat set is available to subclasses.
483 * @see CreateClient_t, UpdateClient_t
484 */
485 struct ClientData_t {
486 public:
487 /**
488 * @brief Parses the raw request body into data_j and reports status via
489 * error/success.
490 * @param[in] _data Raw JSON text of the request body.
491 * @param[in] _id Optional client id (used by updates).
492 * @param[out] error Receives a BAD_REQUEST status for an unparsable
493 * body or an INTERNAL_SERVER_ERROR for an exception, with a diagnostic
494 * message.
495 * @param[out] success Set true only when parsing produced a JSON value.
496 */
497 ClientData_t(std::string_view _data,
498 const std::optional<uint64_t> _id,
499 api::v2::Http::Error& error,
500 bool& success)
501 // string_view avoids copying the body; only the id is
502 // stored here initially.
503 : id(_id) {
504 try {
505 data_j = jsoncons::json::parse(
506 _data); // Turn the raw text into a JSON value.
507 // A null result means the text was
508 // not valid JSON.
509 if (!data_j.has_value()) {
510 error = {.code = api::v2::Http::Status::BAD_REQUEST,
511 .message =
512 "Failed to parse body."}; // A bad body is the
513 // client's fault (400).
514 Message::ErrorMessage(error.message); // Log it for operators.
515 success = false; // Tell the caller not to proceed.
516 // Stop before the success = true at the end.
517 // An unexpected exception while
518 // parsing is a server-side problem.
519 return;
520 }
521 } catch (const std::exception& e) {
522 error =
523 {.code = api::v2::Http::Status::INTERNAL_SERVER_ERROR,
524 .message =
525 fmt::format("Failed while parsing client data: {}.",
526 e.what())}; // Report 500 with the exception text.
527 success = false; // Mark failure.
528 Message::CriticalMessage(error.message); // Log at critical level.
529 return; // Stop early.
530 }
531
532 success = true; // Reaching here means parsing produced a usable
533 // JSON value.
534 }

536 /**
537 * @brief Returns the parsed body, if parsing succeeded.
538 */
539 const std::optional<jsoncons::json>& get_data_json() const {
540 return data_j;
541 } // Const reference avoids copying; callers must check has_value()
542 // first.
543
544 /**
545 * @brief Returns the optional client id supplied at construction.
546 */
547 std::optional<uint64_t> get_id() const {
548 return id;
549 } // Return by value; callers check has_value() before using it.
550
551 protected:
552 /**
553 * @brief Flattened (column, value) pairs derived from the validated
554 * body.
555 */
556 std::unordered_set<std::pair<std::string, std::string>>
557 data_set; // Each pair is one SQL column and its value; a set
558 // prevents duplicates.
559
560 /**
561 * @brief The parsed JSON body, or nullopt if parsing failed.
562 */
563 std::optional<jsoncons::json> data_j; // optional because parsing may fail.
564
565 /**
566 * @brief Optional client id.
567 */
568 std::optional<uint64_t> id; // Present for updates, absent for creates.
569
570 /**
571 * @brief Replaces a plaintext "password" entry in data_set with its
572 * hash.
573 * @return true when a password was found and hashed; false when there
574 * was no password or hashing failed. Logs a critical message on a
575 * hashing failure in the caller path.
576 * @note Operates on the flat pair set rather than data_j, so it must
577 * run after validation has populated data_set.
578 */
579 // Protected so subclasses can choose whether to
580 // hash (creates do, updates do not).
581 bool hashPassword() {
582 auto passwd_itr = std::ranges::find_if(data_set, [&](const auto& item) {
583 return item.first == "password";
584 }); // Look for a column named
585 // "password" in the flat set.
586 // Only act if such an entry exists.
587 if (passwd_itr != data_set.end()) {
588 std::string password_raw =
589 passwd_itr->second; // Copy the plaintext out before erasing it.
590 std::optional<std::string> hashed_password =
591 passwordCrypt->hashPassword(
592 password_raw); // Ask the shared hasher; may fail
593 // (nullopt).
594 // Hashing failed - do not write a
595 // plaintext password.
596 if (!hashed_password.has_value()) {
597 return false; // Signal failure so the caller can react.
598 }
599 data_set.erase(passwd_itr); // Remove the plaintext pair.
600 data_set.emplace("password",
601 hashed_password
602 .value()); // Insert the hashed pair in its place.
603 return true; // Report success.
604 }
605 return false; // No password field existed; nothing to do.
606 }
607
608 private:
609 /**
610 * @brief Shared password hasher obtained from the global object store.
611 */
612 std::shared_ptr<PasswordCrypt> passwordCrypt =
613 Store::cachedObject<PasswordCrypt>(); // One process-wide hasher;
614 // shared_ptr prevents it being
615 // destroyed while in use.
616 };

618 /**
619 * @class CreateClient_t
620 * @brief ClientData_t for creating a client: validates a create schema and
621 * hashes any password.
622 * @details Rejects a body that already contains "id" so clients cannot pick
623 * their own key, runs regex validation to build data_set, then hashes the
624 * password before the INSERT is built.
625 */
626 struct CreateClient_t : public ClientData_t {
627 public:
628 /**
629 * @brief Parses and validates a client-create body, populating
630 * data_set.
631 * @param[in] _data Raw JSON request body.
632 * @param[in] tablename Table whose create schema is used for
633 * validation.
634 * @param[out] error Receives a diagnostic on validation failure.
635 * @param[out] success Set true only after schema and regex validation
636 * succeed.
637 */
638 CreateClient_t(std::string_view _data,
639 const std::string& tablename,
640 api::v2::Http::Error& error,
641 bool& success)
642 // Creates have no id yet, so pass nullopt to
643 // the base.
644 : ClientData_t(_data, std::nullopt, error, success) {
645 try {
646 Validator::Rule rule(Validator::Rule::Action::ASSERT_NOT_PRESENT,
647 {"id"}); // Build a rule saying "id must not be in
648 // the body".
649 success = Validator::
650 validateDatabaseCreateSchema(tablename,
651 data_j,
652 error,
653 rule); // Check the body against the
654 // table's create schema.
655 // Schema validation failed.
656 if (!success) {
657 Message::ErrorMessage(error.message); // Log the validation error.
658 return; // Leave success false and stop.
659 }
