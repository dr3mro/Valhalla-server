1/**
2 * @file threadsafequeue.hpp
3 * @brief Mutex-protected FIFO queue with non-blocking accessors.
4 * @details Defines a template ThreadSafeQueue<T> that wraps std::queue with a
5 * single mutex, so several threads can push and pop without corrupting
6 * it. push adds items, pop removes and returns the front item, and
7 * empty/size/clear/front provide inspection. pop and front never wait:
8 * if the queue is empty they return std::nullopt immediately. front
9 * returns a copy, so the caller keeps a valid value after the lock is
10 * released.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * Producers (for example worker threads or network callbacks) and consumers
14 * often run at the same time. This queue lets them hand items over safely
15 * without any locking code at each call site. Because it is a small reusable
16 * template, any part of the server can create a thread-safe queue of any
17 * copyable type. Without it, each producer/consumer pairing would need its own
18 * error-prone locking.
19 *
20 * CONCEPTS TO LEARN (for beginners):
21 * - std::mutex and std::lock_guard:
22 * https://en.cppreference.com/w/cpp/thread/lock_guard
23 * - std::unique_lock: https://en.cppreference.com/w/cpp/thread/unique_lock
24 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
25 * - Perfect forwarding (std::forward):
26 * https://en.cppreference.com/w/cpp/utility/forward
27 *
28 * @author Amr Osman <dr3mro@gmail.com>
29 * @date 2026-11-16
30 */

42/**
43 * @class ThreadSafeQueue
44 * @brief FIFO container safe for concurrent producers and consumers.
45 * @details Every operation takes a single mutex, so the queue can be shared
46 * between threads without external synchronisation. pop() and front()
47 * return std::nullopt immediately when empty: there is no condition
48 * variable, so callers never wait for an element to arrive. front()
49 * returns a copy rather than a reference, keeping the result valid
50 * after the lock is released.
51 * @tparam T Element type; must be copyable (front) and movable (pop, rvalue
52 * push).
53 * @note empty() and size() report a snapshot; another thread can modify the
54 * queue as soon as the call returns.

154template <typename T>

156 // Hold the lock only for the container mutation.

173// Definition of pop.
174// Returns a value that may be empty.
175template <typename T>
176std::optional<T> ThreadSafeQueue<T>::pop() {
177 // unique_lock is used because the mutex is released on every return path.
178 std::unique_lock<std::mutex> lock(
179 mutex_); // A flexible lock that can be released on every return path.
180 // Check whether there is anything to take.
181 if (queue_.empty()) {
182 // Empty is not an error: report it without waiting for a producer.
183 return std::nullopt; // Signals "no item" immediately rather than
184 // blocking.
185 }
186 T value = std::move(queue_.front()); // Move the front element into a local
187 // so it survives after the pop.
188 queue_.pop(); // Remove that now-copied-from element from the queue.
189 return value; // Hands the element back to the caller.
190}
