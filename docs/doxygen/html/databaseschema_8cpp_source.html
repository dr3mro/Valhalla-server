5 * @details This file defines the single shared schema map and the functions

84/**
85 * See DatabaseSchema::populateSchema.
86 * Fetches the column descriptions through DatabaseController and stores them
87 * under
88 * @p tableName in the static map. Failure is treated as fatal: an error is
89 * logged and exit(EXIT_FAILURE) is called rather than leaving a partially
90 * populated cache.
91 */
92void DatabaseSchema::populateSchema(const std::string& tableName) {
93 auto dbctl =
94 Store::getObject<DatabaseController>(); // Get the controller used to
95 // query this table's schema.
96 auto schema = dbctl->getTableSchema(
97 tableName); // Ask for the columns of this table (an optional).
98 // The query succeeded and produced column
99 // descriptions.
100 if (schema.has_value()) {
101 // Store the columns under the
102 // table name in the shared map.
103 // The table's schema could not be read.
104 databaseSchema[tableName] = dbctl->getTableSchema(tableName).value();
105 } else {
106 Message::WarningMessage(
107 fmt::format("Table not found.\n",
108 tableName)); // Log a warning naming the problem.
109 Message::ErrorMessage(
110 "Exiting..."); // Log that startup is being abandoned.
111 exit(EXIT_FAILURE);
112 /*NOLINT*/ // End the process rather than keep a half-filled cache.
113 }
114}
