1/**
2 * @file logger.cpp
3 * @brief Implementation for logger.
4 * @details Implements Logger. The constructor reads the server configuration,
5 * initialises spdlog's async thread pool, builds a coloured console sink
6 * and a rotating file sink with the same pattern, enables only the sinks
7 * the config asks for, then registers the logger and wires it into
8 * Trantor. log() maps a Trantor severity to the matching spdlog call,
9 * and get_color returns the ANSI escape code for each colour name.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * This is where logging is actually set up and driven. Every log call in the
13 * process ultimately reaches these sinks, so the chosen format, file-rotation
14 * limits and non-blocking behaviour are decided here. Doing the setup once in
15 * the constructor keeps configuration in one place and makes logging fast
16 * (messages are queued and written by background threads). Without it,
17 * Drogon/Trantor would log with default settings and there would be no unified
18 * file output.
19 *
20 * CONCEPTS TO LEARN (for beginners):
21 * - spdlog sinks: https://github.com/gabime/spdlog/wiki/3.-Custom-sinks
22 * - std::make_shared: https://en.cppreference.com/w/cpp/memory/make_shared
23 * - std::unordered_map:
24 * https://en.cppreference.com/w/cpp/container/unordered_map
25 * - Exceptions / try-catch: https://en.cppreference.com/w/cpp/error/try_catch
26 *
27 * @author Amr Osman <dr3mro@gmail.com>
28 * @date 2026-11-16
29 */

85// Constructor performs all logger setup and registration.
86Logger::Logger() {
87 try {
88 auto configurator =
89 Store::cachedObject<Configurator>(); // Fetch the process-wide
90 // Configurator from the
91 // service locator.
92 auto server_config =
93 configurator
94 ->get<Configurator::ServerConfig>(); // Read the server settings,
95 // including log options.
96 // The sink receives a full path assembled from two independent
97 // config keys.
98 auto file_path =
99 fmt::format("{}{}",
100 server_config.log_dir,
101 server_config.log_file); // Join directory and
102 // filename into one path.
103
104 // Initialize thread pool for async logging
105 spdlog::init_thread_pool(THREADPOOL,
106 THREADS); // Create the shared queue and
107 // worker threads used by async
108 // loggers.
109
110 // Console sink setup with custom pattern
111 auto console_sink = std::make_shared<
112 spdlog::sinks::stdout_color_sink_mt>(); // A thread-safe
113 // coloured console
114 // sink (the _mt means
115 // multi-threaded).
116 console_sink->set_pattern(
117 fmt::format("{}[%Y-%m-%d %H:%M:%S]{} {}[Project Valhalla]{} "
118 "%^[%l]%$ %v",
119 get_color(Color::Magenta),
120 get_color(Color::Reset),
121 get_color(Color::Red),
122 get_color(Color::Reset))); // Sets the line format:
123 // timestamp, name, severity
124 // (coloured) and message.
125
126 // File sink setup
127 auto file_sink = std::make_shared<
128 spdlog::sinks::rotating_file_sink_mt>(fmt::format("{}", file_path),
129 MAX_LOG_SIZE,
130 MAX_LOG_FILES); // A thread-safe
131 // sink that
132 // rotates the
133 // log file at
134 // MAX_LOG_SIZE,
135 // keeping
136 // MAX_LOG_FILES
137 // files.
138
139 file_sink->set_pattern(
140 fmt::format("{}[%Y-%m-%d %H:%M:%S]{} {}[Project Valhalla]{} "
141 "%^[%l]%$ %v",
142 get_color(Color::Magenta),
143 get_color(Color::Reset),
144 get_color(Color::Red),
145 get_color(Color::Reset))); // Give the file the same
146 // layout as the console for
147 // consistency.
148 std::vector<std::shared_ptr<spdlog::sinks::sink>>
149 sinks; // Holds the sinks that are actually enabled by
150 // configuration.
151 // Only include the console sink when
152 // the config asks for it.
153 if (server_config.log_to_console) {
154 sinks.push_back(
155 console_sink); // Add the console sink to the enabled set.
156 }
157
158 // Only include the file sink when the
159 // config asks for it.
160 if (server_config.log_to_file) {
161 sinks.push_back(
162 file_sink); // Add the rotating file sink to the enabled set.
163 }
164
165 // Create the async logger with the thread pool and sinks.
166 // overrun_oldest keeps logging non-blocking under load: if the
167 // queue is full, the oldest queued message is dropped rather than
168 // stalling the calling thread.
169 logger_ = std::make_shared<
170 spdlog::async_logger>(server_config.name,
171 sinks.begin(),
172 sinks.end(),
173 spdlog::thread_pool(),
174 spdlog::async_overflow_policy::
175 overrun_oldest); // Builds the named
176 // async logger over
177 // the chosen sinks
178 // and shared pool.
179
180 // Register logger
181 spdlog::register_logger(logger_); // Adds the logger to spdlog's global
182 // registry under its name.
183 // Optional: set as the default logger
184 spdlog::set_default_logger(logger_); // Makes this the logger used
185 // by spdlog's free functions.
186 // Tells Trantor to forward its own
187 // records to this logger too.
188 // Catches spdlog setup errors, notably
189 // a duplicate registration.
190 trantor::Logger::enableSpdLog(logger_);
191 } catch (const spdlog::spdlog_ex& e) {
192 // Report via stdout, not the logger being constructed, to avoid
193 // depending on the very object whose initialisation failed.
194 spdlog::error("SPDLog initialization failed: {}",
195 e.what()); // Print the error directly because the
196 // logger is not usable yet.
197 }
198}
199// Receives Trantor log records and forwards them to the spdlog backend.
200// Called for each Trantor
201// record that needs logging.
202void Logger::log(const std::string& message,
203 trantor::Logger::LogLevel level) {
204 // Unknown or out-of-range levels fall through the default branch and
205 // are dropped. Choose the spdlog severity that matches the Trantor
206 // level.
207 switch (level) {
208 case trantor::Logger::LogLevel::kDebug: // Debug level.
209 logger_->debug(message); // Forward as a spdlog debug record.
210 break; // Leave the switch.
211 case trantor::Logger::LogLevel::kInfo: // Info level.
212 logger_->info(message); // Forward as a spdlog info record.
213 break; // Leave the switch.
214 case trantor::Logger::LogLevel::kWarn: // Warning level.
215 logger_->warn(message); // Forward as a spdlog warning record.
216 break; // Leave the switch.
217 case trantor::Logger::LogLevel::kError: // Error level.
218 logger_->error(message); // Forward as a spdlog error record.
219 break; // Leave the switch.
220 case trantor::Logger::LogLevel::kFatal: // Fatal level.
221 logger_->critical(message); // Forward as a spdlog critical record
222 break; // (spdlog's name for fatal).
223 // Leave the switch.
224 default: // Any level not listed above.
225 break; // Ignore it silently.
226 }
227}
