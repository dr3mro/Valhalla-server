1/**
2 * @file logger.cpp
3 * @brief Implementation for logger.
4 * @details Implements Logger. The constructor reads the server configuration,
5 * initialises spdlog's async thread pool, builds a coloured console sink
6 * and a rotating file sink with the same pattern, enables only the sinks
7 * the config asks for, then registers the logger and wires it into
8 * Trantor. log() maps a Trantor severity to the matching spdlog call,
9 * and get_color returns the ANSI escape code for each colour name.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * This is where logging is actually set up and driven. Every log call in the
13 * process ultimately reaches these sinks, so the chosen format, file-rotation
14 * limits and non-blocking behaviour are decided here. Doing the setup once in the
15 * constructor keeps configuration in one place and makes logging fast (messages
16 * are queued and written by background threads). Without it, Drogon/Trantor would
17 * log with default settings and there would be no unified file output.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - spdlog sinks: https://github.com/gabime/spdlog/wiki/3.-Custom-sinks
21 * - std::make_shared: https://en.cppreference.com/w/cpp/memory/make_shared
22 * - std::unordered_map: https://en.cppreference.com/w/cpp/container/unordered_map
23 * - Exceptions / try-catch: https://en.cppreference.com/w/cpp/error/try_catch
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

38Logger::Logger() // Constructor performs all logger setup and registration.
39{
40 try
41 {
42 auto configurator = Store::getObject<Configurator>(); // Fetch the process-wide Configurator from the service locator.
43 auto server_config = configurator->get<Configurator::ServerConfig>(); // Read the server settings, including log options.
44 // The sink receives a full path assembled from two independent config keys.
45 auto file_path = fmt::format("{}{}", server_config.log_dir, server_config.log_file); // Join directory and filename into one path.
46
47 // Initialize thread pool for async logging
48 spdlog::init_thread_pool(THREADPOOL, THREADS); // Create the shared queue and worker threads used by async loggers.
49
50 // Console sink setup with custom pattern
51 auto console_sink = std::make_shared<spdlog::sinks::stdout_color_sink_mt>(); // A thread-safe coloured console sink (the _mt means multi-threaded).
52 console_sink->set_pattern(fmt::format("{}[%Y-%m-%d %H:%M:%S]{} {}[Project Valhalla]{} %^[%l]%$ %v", get_color(Color::Magenta), get_color(Color::Reset),
53 get_color(Color::Red), get_color(Color::Reset))); // Sets the line format: timestamp, name, severity (coloured) and message.
54
55 // File sink setup
56 auto file_sink = std::make_shared<spdlog::sinks::rotating_file_sink_mt>(fmt::format("{}", file_path), MAX_LOG_SIZE, MAX_LOG_FILES); // A thread-safe sink that rotates the log file at MAX_LOG_SIZE, keeping MAX_LOG_FILES files.
57
58 file_sink->set_pattern(fmt::format("{}[%Y-%m-%d %H:%M:%S]{} {}[Project Valhalla]{} %^[%l]%$ %v", get_color(Color::Magenta), get_color(Color::Reset),
59 get_color(Color::Red), get_color(Color::Reset))); // Give the file the same layout as the console for consistency.
60 std::vector<std::shared_ptr<spdlog::sinks::sink>> sinks; // Holds the sinks that are actually enabled by configuration.
61 if (server_config.log_to_console) // Only include the console sink when the config asks for it.
62 {
63 sinks.push_back(console_sink); // Add the console sink to the enabled set.
64 }
65
66 if (server_config.log_to_file) // Only include the file sink when the config asks for it.
67 {
68 sinks.push_back(file_sink); // Add the rotating file sink to the enabled set.
69 }
70
71 // Create the async logger with the thread pool and sinks. overrun_oldest
72 // keeps logging non-blocking under load: if the queue is full, the oldest
73 // queued message is dropped rather than stalling the calling thread.
74 logger_ = std::make_shared<spdlog::async_logger>(
75 "Project Valhalla", sinks.begin(), sinks.end(), spdlog::thread_pool(), spdlog::async_overflow_policy::overrun_oldest); // Builds the named async logger over the chosen sinks and shared pool.
76
77 // Register logger
78 spdlog::register_logger(logger_); // Adds the logger to spdlog's global registry under its name.
79 // Optional: set as the default logger
80 spdlog::set_default_logger(logger_); // Makes this the logger used by spdlog's free functions.
81 trantor::Logger::enableSpdLog(logger_); // Tells Trantor to forward its own records to this logger too.
82 }
83 catch (const spdlog::spdlog_ex& e) // Catches spdlog setup errors, notably a duplicate registration.
84 {
85 // Report via stdout, not the logger being constructed, to avoid
86 // depending on the very object whose initialisation failed.
87 fmt::print("SPDLog initialization failed: {}", e.what()); // Print the error directly because the logger is not usable yet.
88 }
89}

91void Logger::log(const std::string& message, trantor::Logger::LogLevel level) // Called for each Trantor record that needs logging.
92{
93 // Unknown or out-of-range levels fall through the default branch and are dropped.
94 switch (level) // Choose the spdlog severity that matches the Trantor level.
95 {
96 case trantor::Logger::LogLevel::kDebug: // Debug level.
97 logger_->debug(message); // Forward as a spdlog debug record.
98 break; // Leave the switch.
99 case trantor::Logger::LogLevel::kInfo: // Info level.
100 logger_->info(message); // Forward as a spdlog info record.
101 break; // Leave the switch.
102 case trantor::Logger::LogLevel::kWarn: // Warning level.
103 logger_->warn(message); // Forward as a spdlog warning record.
104 break; // Leave the switch.
105 case trantor::Logger::LogLevel::kError: // Error level.
106 logger_->error(message); // Forward as a spdlog error record.
107 break; // Leave the switch.
108 case trantor::Logger::LogLevel::kFatal: // Fatal level.
109 logger_->critical(message); // Forward as a spdlog critical record (spdlog's name for fatal).
110 break; // Leave the switch.
111 default: // Any level not listed above.
112 break; // Ignore it silently.
113 }
114}

116std::string Logger::get_color(Color color) // Looks up the escape sequence for one colour.
117{
118 static const std::unordered_map<Logger::Color, std::string> color_map = { // A read-only map built once from colour names to ANSI codes.
119 {Logger::Color::Red, "\033[1;91m"}, // Bright Red for errors
120 {Logger::Color::Green, "\033[1;92m"}, // Bright Green for success
121 {Logger::Color::Yellow, "\033[1;93m"}, // Bright Yellow for warnings
122 {Logger::Color::Blue, "\033[1;34m"}, // Bold Blue for informational messages
123 {Logger::Color::Magenta, "\033[38;5;140m"}, // Soft Purple for debug messages
124 {Logger::Color::White, "\033[1;97m"}, // Bright White for general or unspecified messages
125 {Logger::Color::Cyan, "\033[1;96m"}, // Bold Cyan for optional info or context
126 {Logger::Color::Reset, "\033[0m"} // Reset
127 };
128 return color_map.at(color); // .at returns the code and throws std::out_of_range if the colour is missing.
