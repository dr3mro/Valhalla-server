1/**
2 * @file logger.cpp
3 * @brief Implementation for logger.
4 * @details Implements Logger. The constructor reads the server configuration,
5 * initialises spdlog's async thread pool, builds a coloured console
6 * sink and a rotating file sink with the same pattern, enables only the sinks
7 * the config asks for, then registers the logger and wires it into
8 * Trantor. log() maps a Trantor severity to the matching spdlog call,
9 * and get_color returns the ANSI escape code for each colour name.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * This is where logging is actually set up and driven. Every log call in the
13 * process ultimately reaches these sinks, so the chosen format, file-rotation
14 * limits and non-blocking behaviour are decided here. Doing the setup once in
15 * the constructor keeps configuration in one place and makes logging fast
16 * (messages are queued and written by background threads). Without it,
17 * Drogon/Trantor would log with default settings and there would be no unified
18 * file output.
19 *
20 * CONCEPTS TO LEARN (for beginners):
21 * - spdlog sinks: https://github.com/gabime/spdlog/wiki/3.-Custom-sinks
22 * - std::make_shared: https://en.cppreference.com/w/cpp/memory/make_shared
23 * - std::unordered_map:
24 * https://en.cppreference.com/w/cpp/container/unordered_map
25 * - Exceptions / try-catch: https://en.cppreference.com/w/cpp/error/try_catch
26 *
27 * @author Amr Osman <dr3mro@gmail.com>
28 * @date 2026-11-16
29 */

44Logger::Logger() {
45 try {
46 auto configurator =
47 Store::getObject<Configurator>(); // Fetch the process-wide
48 // Configurator from the service
49 // locator.
50 auto server_config =
51 configurator
52 ->get<Configurator::ServerConfig>(); // Read the server settings,
53 // including log options.
54 // The sink receives a full path assembled from two independent config
55 // keys.
56 auto file_path =
57 fmt::format("{}{}",
58 server_config.log_dir,
59 server_config.log_file); // Join directory and filename
60 // into one path.
61
62 // Initialize thread pool for async logging
63 spdlog::init_thread_pool(THREADPOOL,
64 THREADS); // Create the shared queue and worker
65 // threads used by async loggers.
66
67 // Console sink setup with custom pattern
68 auto console_sink = std::make_shared<
69 spdlog::sinks::stdout_color_sink_mt>(); // A thread-safe coloured
70 // console sink (the _mt
71 // means multi-threaded).
72 console_sink->set_pattern(
73 fmt::
74 format("{}[%Y-%m-%d %H:%M:%S]{} {}[Project Valhalla]{} %^[%l]%$ %v",
75 get_color(Color::Magenta),
76 get_color(Color::Reset),
77 get_color(Color::Red),
78 get_color(Color::Reset))); // Sets the line format:
79 // timestamp, name, severity
80 // (coloured) and message.
81
82 // File sink setup
83 auto file_sink = std::make_shared<
84 spdlog::sinks::
85 rotating_file_sink_mt>(fmt::format("{}", file_path),
86 MAX_LOG_SIZE,
87 MAX_LOG_FILES); // A thread-safe sink that
88 // rotates the log file at
89 // MAX_LOG_SIZE, keeping
90 // MAX_LOG_FILES files.
91
92 file_sink->set_pattern(
93 fmt::
94 format("{}[%Y-%m-%d %H:%M:%S]{} {}[Project Valhalla]{} %^[%l]%$ %v",
95 get_color(Color::Magenta),
96 get_color(Color::Reset),
97 get_color(Color::Red),
98 get_color(
99 Color::Reset))); // Give the file the same layout as the
100 // console for consistency.
101 std::vector<std::shared_ptr<spdlog::sinks::sink>>
102 sinks; // Holds the sinks that are actually enabled by
103 // configuration.
104 // Only include the console sink when
105 // the config asks for it.
106 if (server_config.log_to_console) {
107 sinks.push_back(
108 console_sink); // Add the console sink to the enabled set.
109 }
110
111 // Only include the file sink when the
112 // config asks for it.
113 if (server_config.log_to_file) {
114 sinks.push_back(
115 file_sink); // Add the rotating file sink to the enabled set.
116 }
117
118 // Create the async logger with the thread pool and sinks.
119 // overrun_oldest keeps logging non-blocking under load: if the queue is
120 // full, the oldest queued message is dropped rather than stalling the
121 // calling thread.
122 logger_ = std::make_shared<
123 spdlog::async_logger>("Project Valhalla",
124 sinks.begin(),
125 sinks.end(),
126 spdlog::thread_pool(),
127 spdlog::async_overflow_policy::
128 overrun_oldest); // Builds the named async
129 // logger over the chosen
130 // sinks and shared pool.
131
132 // Register logger
133 spdlog::register_logger(logger_); // Adds the logger to spdlog's global
134 // registry under its name.
135 // Optional: set as the default logger
136 spdlog::set_default_logger(
137 logger_); // Makes this the logger used by spdlog's free functions.
138 // Tells Trantor to forward its own
139 // records to this logger too.
140 // Catches spdlog setup errors, notably
141 // a duplicate registration.
142 trantor::Logger::enableSpdLog(logger_);
143 } catch (const spdlog::spdlog_ex& e) {
144 // Report via stdout, not the logger being constructed, to avoid
145 // depending on the very object whose initialisation failed.
146 fmt::print("SPDLog initialization failed: {}",
147 e.what()); // Print the error directly because the logger is not
148 // usable yet.
149 }
150}
151// Receives Trantor log records and forwards them to the spdlog backend.
152// Called for each Trantor
153// record that needs logging.
154void Logger::log(const std::string& message, trantor::Logger::LogLevel level) {
155 // Unknown or out-of-range levels fall through the default branch and are
156 // dropped.
157 // Choose the spdlog severity that matches the Trantor level.
158 switch (level) {
159 case trantor::Logger::LogLevel::kDebug: // Debug level.
160 logger_->debug(message); // Forward as a spdlog debug record.
161 break; // Leave the switch.
162 case trantor::Logger::LogLevel::kInfo: // Info level.
163 logger_->info(message); // Forward as a spdlog info record.
164 break; // Leave the switch.
165 case trantor::Logger::LogLevel::kWarn: // Warning level.
166 logger_->warn(message); // Forward as a spdlog warning record.
167 break; // Leave the switch.
168 case trantor::Logger::LogLevel::kError: // Error level.
169 logger_->error(message); // Forward as a spdlog error record.
170 break; // Leave the switch.
171 case trantor::Logger::LogLevel::kFatal: // Fatal level.
172 logger_->critical(message); // Forward as a spdlog critical record
173 // (spdlog's name for fatal).
174 break; // Leave the switch.
175 default: // Any level not listed above.
176 break; // Ignore it silently.
177 }
178}

181std::string Logger::get_color(Color color) {
182 static const std::unordered_map<Logger::Color, std::string> color_map = {
183 // A read-only map built once from colour names to ANSI codes.
184 {Logger::Color::Red, "\033[1;91m"}, // Bright Red for errors
185 {Logger::Color::Green, "\033[1;92m"}, // Bright Green for success
186 {Logger::Color::Yellow, "\033[1;93m"}, // Bright Yellow for warnings
187 {Logger::Color::Blue,
188 "\033[1;34m"}, // Bold Blue for informational messages
189 {Logger::Color::Magenta,
190 "\033[38;5;140m"}, // Soft Purple for debug messages
191 {Logger::Color::White,
192 "\033[1;97m"}, // Bright White for general or unspecified messages
193 {Logger::Color::Cyan,
194 "\033[1;96m"}, // Bold Cyan for optional info or context
195 {Logger::Color::Reset, "\033[0m"} // Reset
196 };
197 return color_map.at(color); // .at returns the code and throws
198 // std::out_of_range if the colour is missing.
199}
