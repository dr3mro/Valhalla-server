1/**
2 * @file asyncdbclient.cpp
3 * @brief Implementation for asyncdbclient: Stores the supplied client, sharing
4 * ownership with the caller.
5 * @details This file turns the declarations in asyncdbclient.hpp into working
6 * code. The constructor simply takes ownership of a ready-made Drogon client;
7 * the factory reads the database settings from the Store, asks Drogon to build
8 * a PostgreSQL client and sets its execution timeout; the last function formats
9 * those settings into the text string that PostgreSQL's client library
10 * understands.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * This is where the asynchronous database client is actually created. During
14 * startup Server::run() calls AsyncDbClient::createClient() when DB_ASYNC is
15 * enabled, so this file decides which host, port, database, user, password and
16 * SSL mode the server connects with, and how large the connection pool is.
17 * Every async query made by the database controllers ultimately goes through
18 * the client built here. Without it, the Store would have no client to hand to
19 * the controllers and every DB_ASYNC request would report the database as
20 * unavailable.
21 *
22 * CONCEPTS TO LEARN (for beginners):
23 * - Drogon DbClient: https://drogon.docsforge.com/master/database/
24 * - std::shared_ptr (ownership shared between callers):
25 * https://en.cppreference.com/w/cpp/memory/shared_ptr
26 * - std::move (transferring ownership instead of copying):
27 * https://en.cppreference.com/w/cpp/utility/move
28 * - fmt string formatting: https://fmt.dev/
29 * - PostgreSQL connection strings:
30 * https://www.postgresql.org/docs/current/libpq-connect.html#LIBPQ-CONNSTRING
31 *
32 * @author Amr Osman <dr3mro@gmail.com>
33 * @date 2026-11-16
34 */

65/**
66 * @brief Creates the Drogon Pg client from the Store's database configuration.
67 * @details `newPgClient` receives the generated connection string and
68 * `DB_MAX_CONN`; the timeout is then set from `DB_ACQUIRE_TIMEOUT_MS`. Any
69 * connection-establishment work is left to Drogon.
70 */
71std::shared_ptr<drogon::orm::DbClient> AsyncDbClient::createClient() {
72 const auto& config =
73 Store::getObject<Configurator>()
74 ->get<Configurator::DatabaseConfig>(); // Fetch the shared
75 // configuration from the
76 // Store and read just the
77 // DB section; `const&`
78 // avoids copying it.
79 auto client = drogon::orm::DbClient::
80 newPgClient(connectionString(config),
81 config.max_conn); // Drogon builds the PostgreSQL pool:
82 // connection text first, maximum number of
83 // connections second.
84 // Drogon's DbClient::setTimeout is an SQL execution timeout: it cancels a
85 // running command. It is not a pool-acquire timeout, despite the
86 // DB_ACQUIRE_TIMEOUT_MS name; Drogon takes seconds. The setting still comes
87 // from the DB_ACQUIRE_TIMEOUT_MS variable, so the name documents intent
88 // while the division below adapts the value to what this Drogon API
89 // actually measures.
90 client->setTimeout(
91 static_cast<double>(config.acquire_timeout_ms) /
92 1000.0); // Convert milliseconds to seconds; the cast forces
93 // floating-point division so 1500 ms becomes 1.5 s, not 1 s.
94 return client; // Hand the shared pointer back; the caller wraps and
95 // registers it in the Store.
96}

98/**
99 * @brief Formats the libpq connection string, mapping `config.ssl` to
100 * `sslmode`.
101 * @details `connect_timeout` is a fixed two seconds, independent of DB_*
102 * tuning; host and SSL settings may be overridden by the container image but
103 * still arrive through this one configuration object.
104 */
105std::string AsyncDbClient::connectionString(
106 const Configurator::DatabaseConfig& config) {
107 // The container image overrides DB_HOST and DB_SSL; both flow through the
108 // same DatabaseConfig. That is why nothing special is done here: whichever
109 // source supplied the values, they are already in `config`.
110 return fmt::format(
111 "host={} port={} dbname={} user={} password={} connect_timeout=2 "
112 "sslmode={}",
113 config.host,
114 config.port,
115 config.name,
116 config.user, // Each `{}` is replaced by the next argument, producing
117 // one libpq keyword/value string; note the password is
118 // included in clear text, so
119 // never log it.
120 config.pass,
121 config.ssl ? "require"
122 : "disable"); // The boolean SSL flag becomes the word
123 // PostgreSQL expects; a fixed 2 s connect
124 // timeout keeps startup attempts short.
125}
