46/**
47 * Requires staff membership plus the @p powerlevel capability bit.
48 * error: FORBIDDEN with a reason when the requester is not staff.
49 */
50bool PermissionManagerPrivate::hasPermission(const Requester& requester, const std::optional<jsoncons::json>& permissions_j, // Definition of the staff-capability check (signature continues next line).
51 const Permissions::PowerLevel& powerlevel, const std::string& service_name, Http::Error& error) // Rest of hasPermission's signature: the required bit, a name for messages, and the error output.
52{
53 auto entityStaffPermissions = isStaffOfService(requester, permissions_j, service_name, error); // First find the caller's staff entry; it also fills in error on failure.
54
55 if (!entityStaffPermissions.has_value()) // No staff entry means the caller is not on this service's team.
56 {
57 error.code = Http::Status::FORBIDDEN; // 403 Forbidden: malformed data is treated as a denial.
58 error.message = "You are not a staff of this service"; // Explain exactly why access was denied.
59 return false; // Deny the request.
60 }
61
62 return ((entityStaffPermissions->power & powerlevel) != PowerLevel::NONE); // AND the staff mask with the required bit; non-zero means they have it.
63}

103/**
104 * @brief Scans the `staff` map for a `"<id>:<mask>"` entry belonging to the requester.
105 * @details Malformed numeric entries throw and are reported as FORBIDDEN; a non-match also
106 * leaves FORBIDDEN in @p error. The mask is decoded into PowerLevel bits.
107 */
108std::optional<Permissions::StaffPermission> PermissionManagerPrivate::isStaffOfService( // Definition of the staff lookup (signature continues next line).
109 const Requester& requester, const std::optional<jsoncons::json>& permissions_j, const std::string& service_name, Http::Error& error) // Rest of the signature: caller, permission document, service name and error output.
110{
111 try // JSON access and number parsing can throw, so guard them.
112 {
113 if (!permissions_j.has_value() || permissions_j->empty()) // No permission document means we cannot decide; treat as a bad request.
114 {
115 error.message = "Failed get service permissions for." + service_name; // Say which service's permissions could not be read.
116 error.code = api::v2::Http::BAD_REQUEST; // 400 Bad Request: the lookup produced no usable data.
117 return std::nullopt; // No usable staff entry; caller should treat this as denial.
118 }
119
120 const auto& staff = permissions_j.value()["staff"]; // Grab the 'staff' object from the permission document.
121
122 for (const auto& [role, members] : staff.object_range()) // Walk every role; structured binding gives the role name and its member list.
123 {
124 for (const auto& member : members.array_range()) // Walk each member string inside that role's list.
125 {
126 if (!member.as<std::string>().empty()) // Skip blank entries so they cannot cause parse errors.
127 {
128 auto raw_str = member.as<std::string>(); // The entry is stored as text in the form '<id>:<mask>'.
129 size_t pos = raw_str.find(':'); // Locate the colon that separates the id from the mask.
130 std::string left = raw_str.substr(0, pos); // Everything before the colon is the staff id as text.
131 std::string right = raw_str.substr(pos + 1); // Everything after the colon is the permission mask as text.
132 if (left == std::to_string(requester.getId())) // Is this entry for the caller? Compare the id strings.
133 {
134 return std::optional<Permissions::StaffPermission>(Permissions::StaffPermission{ // Found a match: build the result object describing this staff member.
135 .staff_id = static_cast<uint64_t>(std::stoull(left)), .power = static_cast<PowerLevel>(std::stoi(right)), .isMember = true}); // Convert id and mask strings to numbers, and flag this as a real match.
136 }
137 }
138 }
139 }
140 }
141 catch (const std::exception& e) // Runs when a staff entry is malformed (bad number or missing colon).
142 {
143 error.code = Http::Status::FORBIDDEN; // 403 Forbidden: malformed data is treated as a denial.
144 error.message = std::string("Invalid staff permission: ") + e.what(); // Report the parsing problem in the denial message.
145 }
146
147 error.code = Http::Status::FORBIDDEN; // 403 Forbidden: malformed data is treated as a denial.
148 error.message = "Failed to find the user permission level in service " + service_name; // Reached the end with no match: explain that too.
149 return std::nullopt; // No usable staff entry; caller should treat this as denial.
150}
