1/**
2 * @file staff.hpp
3 * @brief REST controller for managing staff membership of service entities.
4 * @details This file defines the Staff controller, which invites, adds and removes staff members of a
5 * service entity. The {serviceType} part of the URL (clinics, pharmacies, laboratories or
6 * radiologycenters) selects one concrete StaffController from a std::variant registry. Each
7 * handler translates the controller's numeric status into a standard JSON HTTP response and
8 * builds a Requester from the identity attributes set by Auth. Depending on the configuration
9 * it either awaits an async controller coroutine or uses executeControllerMethod for blocking
10 * database work. All routes are SECURE.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * This controller is the HTTP front door for managing who works at each service provider. Drogon
14 * routes the /api/v2 staff URLs here; the handlers call the shared dispatcher, which reaches the staff
15 * controllers and the database. It depends on the Auth filter for identity and on the service locator
16 * for the executor and controllers. Without it there would be no API way to invite or change the staff
17 * of a clinic, pharmacy, laboratory or radiology center.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - REST routes and HTTP POST: https://developer.mozilla.org/en-US/docs/Web/HTTP/Methods
21 * - std::variant (a type-safe union): https://en.cppreference.com/w/cpp/utility/variant
22 * - HTTP filters/middleware: https://drogon.docsforge.com/master/filter/
23 * - CORS: https://developer.mozilla.org/en-US/docs/Web/HTTP/CORS
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

62 * @class Staff
63 * @brief REST controller for managing staff membership of service entities.
64 * @details A {serviceType} path segment selects a StaffController<T> from a variant registry
65 * (clinics, pharmacies, laboratories, radiologycenters). Handlers use the shared callback
66 * pattern to map controller status codes to JSON HTTP responses, build a Requester from Auth
67 * attributes, and run async coroutines or the DbExecutor-backed dispatch according to
68 * DatabaseConfig::async. All routes are SECURE. Controllers and the executor are shared
69 * service-locator objects.
70 * @note Drogon prepends its controller path (/api/v2/staff) to the METHOD_ADD paths below, so the
71 * effective URLs contain a doubled "staff" segment (e.g. /api/v2/staff/.../staff/invite).

86 {

88 if (async) // Async mode: run the controller coroutine.

90 auto cb = std::move(callback); // Move callback into the coroutine.
91 drogon::async_run([this, req, cb, serviceType]() mutable -> drogon::Task<void> // Start coroutine capturing its state by value.
92 {
93 CALLBACK_ mcb = [cb](int code, const std::string& content) // Adapter: controller status -> HTTP response.
94 {
95 using Reply = std::function<void(const drogon::HttpResponsePtr&)>; // Response callback alias.
96 switch (code) // Map the numeric status to a reply.
97 {
98 case api::v2::Http::Status::OK: // Success.
99 Helper::successResponse(content, Reply(cb)); // Send 200 with the body.
100 break; // Leave the switch.
101 case api::v2::Http::Status::INTERNAL_SERVER_ERROR: // Failure.
102 Helper::failureResponse(content, Reply(cb)); // Send tagged 500.
103 break; // Leave the switch.
104 default: // Other error.
105 Helper::errorResponse(static_cast<drogon::HttpStatusCode>(code), content, Reply(cb)); // Reuse the numeric status.
106 break; // Leave the switch.
107 }
108 };
109
110 Requester requester(req->getAttributes()->get<uint64_t>("clientID"), req->getAttributes()->get<std::string>("clientGroup")); // Build caller identity for permission checks.
111
112 auto ctlIt = staffRegistry.find(serviceType); // Find controller for this URL type.
113 if (ctlIt == staffRegistry.end()) // Unknown type.
114 {
115 std::move(mcb)(api::v2::Http::Status::BAD_REQUEST, fmt::format("Type mapping not found for: {}", serviceType)); // Reply 400.
116 co_return; // End coroutine.
117 }
118
119 try // Guard the controller call.
120 {
121 co_await std::visit([&](auto& controller) -> drogon::Task<void> { // Dispatch to correct controller type.
122 co_await controller->InviteStaffToEntityAsync(std::move(mcb), std::move(requester), req->body()); // Invite staff asynchronously with body.
123 }, ctlIt->second); // The selected variant.
124 }
125 catch (const std::exception& e) // Unexpected error.
126 {
127 Helper::failureResponse(e.what(), std::move(cb)); // Reply 500.
128 }
129 });
130
131 return; // Coroutine owns the rest.
132 }
