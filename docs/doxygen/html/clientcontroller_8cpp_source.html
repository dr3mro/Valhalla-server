2 * @file clientcontroller.cpp
3 * @brief Definitions and explicit instantiations for ClientController<User> and
4 * ClientController<Provider>.
5 * @details The synchronous Create/Update paths offload password hashing to a
6 * bounded PasswordWorker; all other CRUDS operations forward to
7 * EntityController. Session bookkeeping (login/logout, suspend, delete) is
8 * owned by GateKeeper.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This file is where account behaviour is actually assembled. Slow password
12 * work is handed to a worker so the IO loop stays responsive, while session
13 * side effects are attached by wrapping callbacks. The coroutine CRUDS methods
14 * simply reuse EntityController. Explicit instantiation at the bottom makes all
15 * of this linkable for User and Provider, the two supported client types.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - Lambdas and captures (copying the body into a worker job):
19 * https://en.cppreference.com/w/cpp/language/lambda
20 * - Concurrency / worker pools: https://en.cppreference.com/w/cpp/thread/thread
21 * - Coroutines (co_await / co_return):
22 * https://en.cppreference.com/w/cpp/language/coroutines
23 * - Callback wrappers / std::function:
24 * https://en.cppreference.com/w/cpp/utility/functional/function
25 * - Move semantics (std::move):
26 * https://en.cppreference.com/w/cpp/language/move_constructor
27 * - Exceptions and catch blocks:
28 * https://en.cppreference.com/w/cpp/language/exceptions
29 *
30 * @author Amr Osman <dr3mro@gmail.com>
31 * @date 2026-11-16
32 */

78template <Client_t T>
79void ClientController<T>::Create(CALLBACK_&& callback,
80 const Requester&& requester,
81 std::string_view data) {
82 // Copy the principal and body into the job: the queued lambda outlives this
83 // request thread. A full queue makes runOnPasswordWorker answer 503 without
84 // ever invoking createNow().
85 runOnPasswordWorker(*passwordWorker,
86 std::move(callback), // Queue the job; a full queue
87 // answers 503 itself.
88 [this,
89 requester = Requester(requester),
90 body = std::string(data)](CALLBACK_&& cb) mutable {
91 createNow(std::move(cb), requester, body);
92 });
93}

95template <Client_t T>
96void ClientController<T>::createNow(CALLBACK_&& callback,
97 const Requester& requester,
98 std::string_view data) {
99 try {
100 bool success =
101 false; // Validation flips this true only for a usable payload.
102 HttpError error; // Receives a code/message on validation or
103 // authorization failure.
104 jsoncons::json
105 json_data; // Parsed body needed for the GateKeeper policy check.
106 try {
107 // The parsed form is needed for the GateKeeper policy check; bad
108 // JSON is an immediate 400.
109 json_data =
110 jsoncons::json::parse(data); // Turn the body text into a JSON value.
111 } catch (const std::exception&) {
112 callback(api::v2::Http::Status::BAD_REQUEST,
113 "Invalid request body: malformed JSON.");
114 return;
115 }
116
117 // Parses and validates the client-specific payload; `success` reports
118 // whether the body was usable.
119 CreateClient_t
120 client_data(data,
121 T::getTableName(),
122 error,
123 success); // Client-specific parse and validation.
124
125 if (success) {
126 T client(client_data); // Build the entity that will be persisted.
127 // Duplicates are a client error (409), reported before
128 // authorization or any write.
129 if (client.template exists<CreateClient_t>()) {
130 callback(api::v2::Http::Status::CONFLICT,
131 "Client already exists"); // Duplicate id/username is a
132 // client error.
133 return;
134 }
135
136 // Authorization is the last gate before the INSERT; failures answer
137 // with the gate's status.
138 if (!gateKeeper->canCreate<T>(requester, json_data, error)) {
139 callback(error.code, error.message);
140 return;
141 }
142
143 Controller::Create(client,
144 std::move(callback)); // Delegate the INSERT to the
145 // persistence engine.
146 } else {
147 callback(error.code,
148 fmt::format("ClientData parsing error: {}.", error.message));
149 return;
150 }
151 } catch (const std::exception& e) {
152 CRITICALMESSAGERESPONSE // Unexpected error: log it and answer with a
153 // mapped 5xx status.
154 }

184template <Client_t T>
185void ClientController<T>::updateNow(CALLBACK_&& callback,
186 const Requester& requester,
187 std::string_view data,
188 const std::optional<uint64_t> _id) {
189 try {
190 bool success =
191 false; // Validation flips this true only for a usable payload.
192 // A client may not change its identity or password through the generic
193 // update route.
194 static const Validator::Rule rule(Validator::Rule::Action::ASSERT_IMMUTABLE,
195 {"username", "password"});
196
197 HttpError error; // Receives a code/message on validation or
198 // authorization failure.
199 UpdateClient_t client_data(data,
200 _id,
201 T::getTableName(),
202 error,
203 success,
204 rule);
205
206 if (success) {
207 T client(client_data); // Build the entity that will be persisted.
208 // Authorize against the target row before delegating the write.
209 if (!gateKeeper->canUpdate<T>(requester, _id.value(), error)) {
210 callback(error.code, error.message);
211 return;
212 }
213 Controller::Update(client,
214 std::move(callback)); // Delegate the UPDATE to the
215 // persistence engine.
216 } else {
217 callback(error.code,
218 fmt::format("ClientData parsing error: {}.", error.message));
219 return;
220 }

222 CRITICALMESSAGERESPONSE // Unexpected error: log it and answer with a

227template <Client_t T>
228void ClientController<T>::Delete(CALLBACK_&& callback,
229 const Requester&& requester,
230 const std::optional<uint64_t> client_id) {
231 auto cb = std::move(callback); // Take ownership so it can be captured into
232 // the wrapper lambda.
233 // Intercept the result so sessions are dropped only when the delete
234 // actually succeeded; then forward unchanged.
235 CALLBACK_ wrapped = [this,
236 client_id,
237 cb = std::move(cb)](int code,
238 const std::string& body) mutable {
239 if (code == static_cast<int>(api::v2::Http::Status::OK) &&
240 client_id.has_value()) {
241 gateKeeper->removeSession(client_id,
242 T::getTableName()); // A suspended client must
243 // not keep a live session.
244 }
245 std::move(cb)(code, body);
246 };
247 EntityController<T>::Delete(std::move(wrapped),
248 std::move(requester),
249 client_id);
250}

301template <Client_t T>
302void ClientController<T>::Suspend(CALLBACK_&& callback,
303 const Requester&& requester,
304 const std::optional<uint64_t> client_id) {
305 try {
306 // No id means no account can be toggled; reject with 406 before any
307 // authorization or DB work.
308 if (!client_id.has_value()) {
309 callback(api::v2::Http::Status::NOT_ACCEPTABLE, "Invalid id provided");
310 return;
311 }
312
313 SuspendData suspendData(client_id.value());
314 T client(suspendData); // Minimal entity carrying the id to toggle.
315 HttpError error;
316
317 // Suspend and Activate share the toggle-active permission; denial
318 // short-circuits with the gate's status.
319 if (!gateKeeper->canToggleActive<T>(requester, client_id.value(), error)) {
320 callback(error.code, error.message);
321 return;
322 }
323
324 Controller::Suspend(client,
325 std::move(callback)); // Issue the deactivation update.
326 // A suspended client must not keep a usable session; drop it after the
327 // state change is issued.
328 gateKeeper->removeSession(client_id,
329 T::getTableName()); // A suspended client must
330 // not keep a live session.
331 } catch (const std::exception& e) {
332 CRITICALMESSAGERESPONSE // Unexpected error: log it and answer with a
333 // mapped 5xx status.
334 }
335}
