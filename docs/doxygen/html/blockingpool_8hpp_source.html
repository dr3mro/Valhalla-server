1/**
2 * @file blockingpool.hpp
3 * @brief Thread-safe bounded pool of reusable resources with idle iteration.
4 * @details This header defines BlockingPool<T>, a header-only container that
5 * holds a fixed set of reusable items as shared_ptrs. release() adds an item
6 * and wakes a waiting acquirer; acquire() waits up to a timeout for an item and
7 * returns nullptr if none appears; idle() reports the idle count; forEachIdle()
8 * visits every idle item under the lock. A mutex protects the deque and a
9 * condition variable handles the waiting.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * This is the generic engine underneath DatabaseConnectionPool. It knows
13 * nothing about databases, which makes it reusable and easy to unit test with
14 * plain objects. It provides the three properties the server needs: connections
15 * are leased one caller at a time, callers wait rather than fail immediately
16 * when all are busy, and a bounded wait can be reported as a timeout. Without
17 * it the pool would have to reimplement tricky condition-variable logic.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - Condition variable (wait until a resource is available):
21 * https://en.cppreference.com/w/cpp/thread/condition_variable
22 * - std::mutex and std::lock_guard (mutual exclusion / automatic locking):
23 * https://en.cppreference.com/w/cpp/thread/lock_guard
24 * - std::unique_lock (a lock that can be waited on):
25 * https://en.cppreference.com/w/cpp/thread/unique_lock
26 * - std::deque (a double-ended queue):
27 * https://en.cppreference.com/w/cpp/container/deque
28 * - Predicate form of wait_for (also guards against spurious wakeups):
29 * https://en.cppreference.com/w/cpp/thread/condition_variable/wait_for
30 * - RAII (the lock guard releases the mutex at scope exit):
31 * https://en.cppreference.com/w/cpp/language/raii
32 *
33 * @author Amr Osman <dr3mro@gmail.com>
34 * @date 2026-11-16
35 */

57/**
58 * @class BlockingPool
59 * @brief Thread-safe fixed set of reusable items leased one caller at a time.
60 * @details Items are added with release() and leased with acquire(), which
61 * waits up to a timeout and returns `nullptr` if nothing becomes free. The pool
62 * is independent of the database so it can be unit tested with plain objects.
63 * Used by DatabaseConnectionPool to hand out Database connections;
64 * DatabaseHanndler models the matching RAII lease.
65 *
66 * Thread-safety: a single mutex protects the deque, a condition
67 * variable wakes waiting acquirers, and all methods may be called concurrently.
68 * @tparam T Type of the pooled items (held via `std::shared_ptr`).
69 */
70// T is the resource type; items are owned through
71// std::shared_ptr<T>.
72// The mutex + condition variable
73// make it safe for many threads to
74// use at once.
75template <typename T>
76class BlockingPool {
77 public:
78 BlockingPool() = default; // An empty deque needs no special setup.
79 BlockingPool(const BlockingPool&) =
80 delete; // Copying a live pool of shared items would duplicate leases.
81 BlockingPool(BlockingPool&&) = delete; // Moving would leave waiters holding
82 // a stale mutex/condition variable.
83 BlockingPool& operator=(const BlockingPool&) =
84 delete; // Copy-assignment would violate single ownership of the queue.
85 BlockingPool& operator=(BlockingPool&&) =
86 delete; // Move-assignment would violate single ownership of the queue.
87 ~BlockingPool() = default; // The deque's shared_ptrs release any remaining
88 // items automatically.
89
90 // Adds a new item, or gives one back after use.
91 /**
92 * @brief Adds an item to the idle set and wakes one waiting acquirer.
93 * @param[in,out] item Item to add; null pointers are ignored.
94 * @details The notification happens after the lock is released, so a woken
95 * acquirer can take the lock immediately.
96 */
97 // Takes by value so a moved-in item
98 // can be stored directly.
99 void release(std::shared_ptr<T> item) {
100 // A null item is not a usable lease; ignore it silently.
101 if (item == nullptr) {
102 return; // Nothing to add and no reason to wake anyone.
103 }
104 {
105 std::lock_guard<std::mutex> lock(
106 mutex_); // Lock the deque briefly; the guard unlocks at the
107 // end of this block.
108 items_.push_back(
109 std::move(item)); // Put the item at the back of the idle queue
110 // without copying the shared_ptr.
111 }
112 cv_.notify_one(); // Wake exactly one waiter now that an item is
113 // available (outside the lock to avoid a
114 // wake-then-block).
115 }
116
117 /**
118 * @brief Leases an idle item, waiting at most `timeout` for one to be
119 * released.
120 * @param[in] timeout Maximum time to wait.
121 * @return The leased item, or `nullptr` if the timeout elapsed (the
122 * predicate form of `wait_for` also handles spurious wakeups).
123 * @post On success the returned item is no longer in the idle set; the
124 * caller must eventually release it (see DatabaseHanndler).
125 */
126 [[nodiscard]] std::shared_ptr<T> acquire(
127 std::chrono::milliseconds
128 // [[nodiscard]] warns callers that dropping the returned
129 // item would strand it.
130 timeout) {
131 std::unique_lock<std::mutex> lock(
132 mutex_); // A unique_lock can be released while waiting and
133 // reacquired when woken.
134 if (!cv_.wait_for(lock, timeout, [this] {
135 // Sleep until an item exists or the timeout expires;
136 // the predicate also defeats spurious wakeups.
137 return !items_.empty();
138 })) {
139 return nullptr; // Timed out: the caller can report exhaustion
140 // (DatabaseBusy in the server).
141 }
142 std::shared_ptr<T> item =
143 std::move(items_.front()); // Take the oldest idle item (FIFO) by
144 // moving it out.
145 items_.pop_front(); // Remove it from the idle queue so no other caller
146 // can lease it.
147 return item; // Hand the exclusive lease to the caller.
148 }
149
150 /**
151 * @brief Returns the number of currently idle items.
152 * @return Idle count, read under the pool lock.
153 */
154 // A snapshot gauge used in logs and startup progress messages.
155 [[nodiscard]] std::size_t idle() const {
156 std::lock_guard<std::mutex> lock(
157 mutex_); // Lock so the size cannot change while it is read.
158 return items_.size(); // Number of items currently available to lease.
159 }
160
161 // Runs func on every idle item, then finish(), all while holding the pool
162 // lock (used to reconnect idle connections and wait for them, so none is
163 // handed out half way).
164 /**
165 * @brief Visits every idle item with `func`, then calls `finish`, under the
166 * pool lock.
167 * @details Holding the lock across both callbacks prevents acquire() from
168 * handing out an item while it is being operated on (e.g. reconnected) or
169 * before the follow-up work in `finish` has completed.
170 * @tparam Func Callable taking `std::shared_ptr<T>&`.
171 * @tparam Finish Callable invoked once after all items were visited.
172 * @param[in,out] func Per-item operation.
173 * @param[in,out] finish Final operation, typically waiting for async work
174 * started by func.
175 * @note Keeping `finish` inside the lock is intentional; a callback that
176 * tries to acquire from the same pool would deadlock.
177 */
178 // Generic callbacks let callers do any per-item work
179 // without changing this class.
180 // Both callbacks run while the
181 // lock is held, on purpose.
182 template <typename Func, typename Finish>
183 void forEachIdle(Func&& func, Finish&& finish) {
184 std::lock_guard<std::mutex> lock(
185 mutex_); // Hold the lock for the whole visit so nothing idle is
186 // handed out mid-operation.
187 for (auto& item :
188 // One iteration per idle item currently in the deque.
189 items_) {
190 func(item); // Let the caller operate on (e.g. reconnect) this item.
191 }
192 finish(); // Run the final step (usually waiting for async work) before
193 // releasing the lock.
194 }
195
196 private:
197 mutable std::mutex mutex_; // Protects items_. `mutable` lets const methods
198 // like idle() lock it.
199 std::condition_variable cv_; // Blocks acquirers until an item is released.
200 std::deque<std::shared_ptr<T>>
201 items_; // The idle items themselves, in FIFO order.
202};
