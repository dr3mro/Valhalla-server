1/**
2 * @file blockingpool.hpp
3 * @brief Thread-safe bounded pool of reusable resources with idle iteration.
4 * @details This header defines BlockingPool<T>, a header-only container that holds a fixed set
5 * of reusable items as shared_ptrs. release() adds an item and wakes a waiting
6 * acquirer; acquire() waits up to a timeout for an item and returns nullptr if none
7 * appears; idle() reports the idle count; forEachIdle() visits every idle item under
8 * the lock. A mutex protects the deque and a condition variable handles the waiting.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This is the generic engine underneath DatabaseConnectionPool. It knows nothing about
12 * databases, which makes it reusable and easy to unit test with plain objects. It provides
13 * the three properties the server needs: connections are leased one caller at a time, callers
14 * wait rather than fail immediately when all are busy, and a bounded wait can be reported as
15 * a timeout. Without it the pool would have to reimplement tricky condition-variable logic.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - Condition variable (wait until a resource is available): https://en.cppreference.com/w/cpp/thread/condition_variable
19 * - std::mutex and std::lock_guard (mutual exclusion / automatic locking): https://en.cppreference.com/w/cpp/thread/lock_guard
20 * - std::unique_lock (a lock that can be waited on): https://en.cppreference.com/w/cpp/thread/unique_lock
21 * - std::deque (a double-ended queue): https://en.cppreference.com/w/cpp/container/deque
22 * - Predicate form of wait_for (also guards against spurious wakeups): https://en.cppreference.com/w/cpp/thread/condition_variable/wait_for
23 * - RAII (the lock guard releases the mutex at scope exit): https://en.cppreference.com/w/cpp/language/raii
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

41/**
42 * @class BlockingPool
43 * @brief Thread-safe fixed set of reusable items leased one caller at a time.
44 * @details Items are added with release() and leased with acquire(), which waits up to a
45 * timeout and returns `nullptr` if nothing becomes free. The pool is independent
46 * of the database so it can be unit tested with plain objects. Used by
47 * DatabaseConnectionPool to hand out Database connections; DatabaseHanndler
48 * models the matching RAII lease.
49 *
50 * Thread-safety: a single mutex protects the deque, a condition variable wakes
51 * waiting acquirers, and all methods may be called concurrently.
52 * @tparam T Type of the pooled items (held via `std::shared_ptr`).
53 */
54template <typename T> // T is the resource type; items are owned through std::shared_ptr<T>.
55class BlockingPool // The mutex + condition variable make it safe for many threads to use at once.
56{
57 public:
58 BlockingPool() = default; // An empty deque needs no special setup.
59 BlockingPool(const BlockingPool&) = delete; // Copying a live pool of shared items would duplicate leases.
60 BlockingPool(BlockingPool&&) = delete; // Moving would leave waiters holding a stale mutex/condition variable.
61 BlockingPool& operator=(const BlockingPool&) = delete; // Copy-assignment would violate single ownership of the queue.
62 BlockingPool& operator=(BlockingPool&&) = delete; // Move-assignment would violate single ownership of the queue.
63 ~BlockingPool() = default; // The deque's shared_ptrs release any remaining items automatically.
64
65 // Adds a new item, or gives one back after use.
66 /**
67 * @brief Adds an item to the idle set and wakes one waiting acquirer.
68 * @param[in,out] item Item to add; null pointers are ignored.
69 * @details The notification happens after the lock is released, so a woken acquirer
70 * can take the lock immediately.
71 */
72 void release(std::shared_ptr<T> item) // Takes by value so a moved-in item can be stored directly.
73 {
74 if (item == nullptr) // A null item is not a usable lease; ignore it silently.
75 {
76 return; // Nothing to add and no reason to wake anyone.
77 }
78 {
79 std::lock_guard<std::mutex> lock(mutex_); // Lock the deque briefly; the guard unlocks at the end of this block.
80 items_.push_back(std::move(item)); // Put the item at the back of the idle queue without copying the shared_ptr.
81 }
82 cv_.notify_one(); // Wake exactly one waiter now that an item is available (outside the lock to avoid a wake-then-block).
83 }
84
85 /**
86 * @brief Leases an idle item, waiting at most `timeout` for one to be released.
87 * @param[in] timeout Maximum time to wait.
88 * @return The leased item, or `nullptr` if the timeout elapsed (the predicate form of
89 * `wait_for` also handles spurious wakeups).
90 * @post On success the returned item is no longer in the idle set; the caller must
91 * eventually release it (see DatabaseHanndler).
92 */
93 [[nodiscard]] std::shared_ptr<T> acquire(std::chrono::milliseconds timeout) // [[nodiscard]] warns callers that dropping the returned item would strand it.
94 {
95 std::unique_lock<std::mutex> lock(mutex_); // A unique_lock can be released while waiting and reacquired when woken.
96 if (!cv_.wait_for(lock, timeout, [this] { return !items_.empty(); })) // Sleep until an item exists or the timeout expires; the predicate also defeats spurious wakeups.
97 {
98 return nullptr; // Timed out: the caller can report exhaustion (DatabaseBusy in the server).
99 }
100 std::shared_ptr<T> item = std::move(items_.front()); // Take the oldest idle item (FIFO) by moving it out.
101 items_.pop_front(); // Remove it from the idle queue so no other caller can lease it.
102 return item; // Hand the exclusive lease to the caller.
103 }
104
105 /**
106 * @brief Returns the number of currently idle items.
107 * @return Idle count, read under the pool lock.
108 */
109 [[nodiscard]] std::size_t idle() const // A snapshot gauge used in logs and startup progress messages.
110 {
111 std::lock_guard<std::mutex> lock(mutex_); // Lock so the size cannot change while it is read.
112 return items_.size(); // Number of items currently available to lease.
113 }
114
115 // Runs func on every idle item, then finish(), all while holding the pool lock (used to reconnect idle connections and
116 // wait for them, so none is handed out half way).
117 /**
118 * @brief Visits every idle item with `func`, then calls `finish`, under the pool lock.
119 * @details Holding the lock across both callbacks prevents acquire() from handing out
120 * an item while it is being operated on (e.g. reconnected) or before the
121 * follow-up work in `finish` has completed.
122 * @tparam Func Callable taking `std::shared_ptr<T>&`.
123 * @tparam Finish Callable invoked once after all items were visited.
124 * @param[in,out] func Per-item operation.
125 * @param[in,out] finish Final operation, typically waiting for async work started by func.
126 * @note Keeping `finish` inside the lock is intentional; a callback that tries to
127 * acquire from the same pool would deadlock.
128 */
129 template <typename Func, typename Finish> // Generic callbacks let callers do any per-item work without changing this class.
130 void forEachIdle(Func&& func, Finish&& finish) // Both callbacks run while the lock is held, on purpose.
131 {
132 std::lock_guard<std::mutex> lock(mutex_); // Hold the lock for the whole visit so nothing idle is handed out mid-operation.
133 for (auto& item : items_) // One iteration per idle item currently in the deque.
134 {
135 func(item); // Let the caller operate on (e.g. reconnect) this item.
136 }
137 finish(); // Run the final step (usually waiting for async work) before releasing the lock.
138 }
139
140 private:
141 mutable std::mutex mutex_; // Protects items_. `mutable` lets const methods like idle() lock it.
142 std::condition_variable cv_; // Blocks acquirers until an item is released.
143 std::deque<std::shared_ptr<T>> items_; // The idle items themselves, in FIFO order.
144};
