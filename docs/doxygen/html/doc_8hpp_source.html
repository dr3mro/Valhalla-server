1/**
2 * @file doc.hpp
3 * @brief Serves the Swagger UI page that documents the API.
4 * @details This file defines the Doc controller with a single endpoint. That endpoint returns an HTML
5 * page containing the Swagger UI, an interactive documentation viewer for the REST API. The
6 * HTML is embedded into the program at build time from docs/api.html, so the running server
7 * does not need to read any file from disk. The handler simply builds a 200 HTML response and
8 * hands it to the callback; there is no per-request state or authentication.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * This is the human-facing documentation endpoint of the API. Developers and testers open it in a
12 * browser to see every available route, its parameters and its response format. Drogon calls it at
13 * the /api/v2/doc URL; it uses no database and no controllers. Because it goes through the INSECURE
14 * chain it is rate-limited and timed but needs no token, so anyone can read the public docs.
15 *
16 * CONCEPTS TO LEARN (for beginners):
17 * - Swagger / OpenAPI: a standard way to describe REST APIs (https://swagger.io/).
18 * - HTTP GET and status 200: https://developer.mozilla.org/en-US/docs/Web/HTTP/Methods
19 * - CORS: https://developer.mozilla.org/en-US/docs/Web/HTTP/CORS
20 *
21 * @author Amr Osman <dr3mro@gmail.com>
22 * @date 2026-11-16
23 */

52 /**
53 * @brief Respond with the embedded Swagger UI HTML document.
54 * @param[in,out] callback Response callback, consumed exactly once.
55 */
56 void doc(const drogon::HttpRequestPtr& /*req*/, std::function<void(const drogon::HttpResponsePtr&)>&& callback) // NOLINT
57 {
58 drogon::HttpResponsePtr response = drogon::HttpResponse::newHttpResponse(); // Build an empty HTTP reply object.
59
60 response->setBody(std::string(docs::kApiHtml)); // Put the compile-time embedded Swagger HTML into the body.
61 response->setStatusCode(drogon::k200OK); // 200 means the page was found and is being returned.
62 response->setContentTypeCode(drogon::CT_TEXT_HTML); // Mark the body as HTML so the browser renders it.
63 std::move(callback)(response); // Send the response to the client and finish.
64 }
