1/**
2 * @file boundedexecutor.cpp
3 * @brief Implementation of BoundedExecutor: per-worker queues, work stealing and shutdown.
4 * @details This file defines the worker pool's behaviour. The constructor creates all worker objects before starting any
5 * threads (so every worker can safely look at every queue), then each thread runs a loop that takes tasks, runs
6 * them, and sleeps when there is nothing to do. submit() picks a worker without ever waiting for space, and
7 * shutdown() stops new work, drains what is queued and joins the threads.
8 *
9 * WHY THIS FILE MATTERS IN THE SERVER:
10 * This is where slow, blocking jobs (database calls, password hashing) actually get executed off Drogon's IO threads.
11 * It is called through BoundedExecutor::submit() and runOnExecutor() from the database and gatekeeper code. Because it
12 * refuses work when full and runs tasks inline with zero workers, it also controls how the server behaves under load
13 * and how the unit tests stay deterministic. Without it, slow work would block the web server or grow an unbounded
14 * backlog of queued requests.
15 *
16 * CONCEPTS TO LEARN (for beginners):
17 * - thread_local: a variable with one separate copy per running thread. https://en.cppreference.com/w/cpp/language/storage_duration
18 * - Condition variables: a way for a thread to sleep until another thread wakes it with new work. https://en.cppreference.com/w/cpp/thread/condition_variable
19 * - Work stealing: an idle worker takes jobs from a busier worker's queue. https://en.wikipedia.org/wiki/Work_stealing
20 * - Atomic memory order: rules for when one thread's writes become visible to another. https://en.cppreference.com/w/cpp/atomic/memory_order
21 *
22 * @author Amr Osman <dr3mro@gmail.com>
23 * @date 2026-11-16
24 */

51/**
52 * Allocates the workers, starts their threads and waits until each reports itself idle.
53 *
54 * queue_max is clamped to at least 1. All Worker objects are created before any thread starts, because a
55 * worker may steal from every other queue. Starting threads only after that avoids a race where a worker sees a
56 * not-yet-constructed peer. The final spin (with yield) removes the startup race with an immediate submit().
57 * workers: Number of worker threads; 0 gives inline mode.
58 * queue_max: Shared queue capacity, clamped to at least 1.
59 * name: Log label for this pool.
60 */
61BoundedExecutor::BoundedExecutor(std::size_t workers, std::size_t queue_max, std::string name)
62 : queue_max_(std::max<std::size_t>(queue_max, 1)), name_(std::move(name))
63{
64 workers_.reserve(workers); // reserve once so push_back cannot reallocate and move the Worker objects
65 for (std::size_t i = 0; i < workers; ++i) // create one Worker record per requested thread
66 {
67 workers_.push_back(std::make_unique<Worker>()); // allocate each Worker; no thread is started yet
68 }
69 // Start the threads only now: a worker looks at every queue, so all of them must exist first.
70 for (std::size_t i = 0; i < workers; ++i)
71 {
72 workers_[i]->thread = std::thread([this, i] { run(i); }); // launch the worker loop, capturing this pool and its index
73 }
74 // Wait until every worker is ready (idle) before returning, so callers that submit
75 // immediately after construction do not race with thread startup and trigger stealing.
76 for (std::size_t i = 0; i < workers; ++i)
77 {
78 while (!workers_[i]->idle.load(std::memory_order_acquire)) // spin until this worker says it is waiting for work
79 {
80 std::this_thread::yield(); // politely let the OS run the worker instead of burning CPU in a tight loop
81 }
82 }
83}

90/**
91 * @brief Stops accepting work, drains queued tasks and joins the workers.
92 *
93 * @details The stopping flag is published with release semantics; the empty-lock/notify pair then wakes a worker that
94 * is either waiting or about to wait, and each worker checks the flag under its own mutex before sleeping.
95 * Join is skipped for the calling thread so a task may shut down its own pool. Statistics are logged only when
96 * there is something to report, and the counters are cleared so a second call does not log them again.
97 * ("Joining" a thread means waiting here until that thread's code has finished.)
98 */
99void BoundedExecutor::shutdown()
100{
101 stopping_.store(true, std::memory_order_release); // publish "no new work" before waking anyone
102 for (auto& worker : workers_) // wake every worker so it can notice the flag
103 {
104 {
105 std::lock_guard<std::mutex> lock(worker->mutex); // a sleeping worker either sees the flag or is already waiting
106 }
107 worker->cv.notify_all(); // the short lock above closes the race with a worker going to sleep
108 }
109 for (auto& worker : workers_)
110 {
111 if (worker->thread.joinable() && worker->thread.get_id() != std::this_thread::get_id()) // never try to join ourselves
112 {
113 worker->thread.join(); // wait until that worker has finished draining its queue
114 }
115 }
116
117 const Stats s = stats(); // take a snapshot of the counters before clearing them
118 if (s.submitted > 0 || s.rejected > 0) // only log when something actually happened
119 {
120 // Log one summary line built from the counters captured above.
121 Message::InfoMessage(
122 fmt::format("{}: {} tasks, {} rejected, deepest queue {}, average wait {:.1f} us.", name_, s.submitted, s.rejected, s.max_queue, s.avg_wait_us));
123 submitted_ = 0; // report once
124 rejected_ = 0; // clear the counter so a second shutdown() does not log the same numbers
125 }
126}

128/**
129 * @brief Samples the tuning counters with relaxed atomic loads.
130 *
131 * @return Stats snapshot; avg_wait_us is total wait time divided by waited tasks, in microseconds, or 0.0 before any
132 * task has been taken by a worker.
133 * @note Values are approximate while the pool is running; they are exact after shutdown().
134 */
135BoundedExecutor::Stats BoundedExecutor::stats() const
136{
137 const auto waited = waited_tasks_.load(std::memory_order_relaxed); // how many tasks went into the total wait time
138 // The last value is total nanoseconds divided by the sample count, then converted to microseconds.
139 return {submitted_.load(std::memory_order_relaxed), rejected_.load(std::memory_order_relaxed), max_queue_.load(std::memory_order_relaxed),
140 waited == 0 ? 0.0 : static_cast<double>(wait_ns_.load(std::memory_order_relaxed)) / static_cast<double>(waited) / 1000.0};
141}

143/**
144 * Enqueues a task without blocking, preferring an idle worker and enforcing a shared queue limit.
145 *
146 * Inline mode runs the task immediately on the caller. Otherwise the shared queued_ counter is reserved with
147 * acq_rel semantics and released if the pool turns out to be full or stopping. The target worker is chosen from
148 * a round-robin start: the first idle one, else the shortest queue. The stopping flag is checked again while
149 * holding that worker's mutex, matching the worker's own exit check. Only a worker that was idle is notified;
150 * a busy worker rechecks its queue when it finishes a task.
151 * task: Nullary callable to execute (takes no arguments).
152 * Returns true if the task ran inline or was queued; false if it was rejected.
153 */
154bool BoundedExecutor::submit(std::function<void()> task)
155{
156 if (inlineMode()) // no workers: run on the calling thread instead of queueing
157 {
158 task(); // do the work right now
159 return true; // inline execution always counts as accepted
160 }
161 if (stopping_.load(std::memory_order_acquire)) // the pool is shutting down
162 {
163 rejected_.fetch_add(1, std::memory_order_relaxed); // count the refusal for diagnostics
164 return false; // caller should answer 503
165 }
166
167 // Reserve a place in the shared limit first; give it back if the pool turns out to be stopping.
168 const std::size_t depth = queued_.fetch_add(1, std::memory_order_acq_rel) + 1; // atomically claim one slot and read the new total
169 if (depth > queue_max_) // we are over the shared limit
170 {
171 queued_.fetch_sub(1, std::memory_order_acq_rel); // give the unused slot back
172 rejected_.fetch_add(1, std::memory_order_relaxed); // count the rejection
173 return false; // refuse quickly instead of waiting for space
174 }
175
176 // From a round-robin start: the first idle worker; with none idle, the one with the shortest queue (the start worker wins ties).
177 const std::size_t n = workers_.size(); // how many workers exist
178 const std::size_t start = next_.fetch_add(1, std::memory_order_relaxed) % n; // rotate the starting point on each call
179 std::size_t target = start; // default pick if nobody is idle
180 std::size_t shortest = std::numeric_limits<std::size_t>::max(); // "infinity", so the first real depth wins
181 for (std::size_t i = 0; i < n; ++i) // scan every worker once
182 {
183 const std::size_t candidate = (start + i) % n; // wrap around the list
184 if (workers_[candidate]->idle.load(std::memory_order_acquire)) // found a sleeping worker
185 {
186 target = candidate; // give it the job
187 break; // stop scanning
188 }
189 const std::size_t depth = workers_[candidate]->depth.load(std::memory_order_relaxed); // otherwise read its queue length
190 if (depth < shortest) // shortest queue seen so far
191 {
192 shortest = depth; // remember the new best
193 target = candidate; // and which worker owns it
194 }
195 }
196
197 Worker& worker = *workers_[target]; // the chosen worker
198 bool wake = false; // whether a wake-up signal is needed
199 {
200 std::lock_guard<std::mutex> lock(worker.mutex); // protect the queue while we add to it
201 if (stopping_.load(std::memory_order_acquire)) // checked under the lock: the worker's exit check is under it too
202 {
203 queued_.fetch_sub(1, std::memory_order_acq_rel); // undo our earlier reservation
204 rejected_.fetch_add(1, std::memory_order_relaxed);
205 return false;
206 }
207 worker.queue.push_back({std::move(task), std::chrono::steady_clock::now()}); // store the job together with its arrival time
208 worker.depth.store(worker.queue.size(), std::memory_order_relaxed); // keep the lock-free length estimate current
209 // An idle worker is claimed (flag cleared) so the next submit() picks another one instead of queueing behind this task.
210 // A busy worker finds its task itself; it needs no wake-up (system call).
211 wake = worker.idle.exchange(false, std::memory_order_acq_rel); // claim the idle flag; exchange returns its old value
212 }
213 submitted_.fetch_add(1, std::memory_order_relaxed); // count the accepted task
214 if (depth > max_queue_.load(std::memory_order_relaxed)) // a new high-water mark
215 {
216 max_queue_.store(depth, std::memory_order_relaxed); // remember it
217 }
218 if (wake) // only a sleeping worker needs a wake-up
219 {
220 worker.cv.notify_one(); // system call to wake it
221 }
222 return true; // tell the caller the job is queued
223}

231/**
232 * Takes a task from this worker's queue, or steals one from a peer if necessary.
233 *
234 * Tries the worker's own queue under its mutex first. On a miss it scans peers using try_lock, so it never
235 * waits for a peer that may be busy with a slow task. Each successful take decrements the shared queued_
236 * counter with acq_rel semantics.
237 * self: Index of the calling worker.
238 * item: Receives the task and its queueing timestamp.
239 * Returns true if a task was taken; false if all queues were empty or locked.
240 */
241bool BoundedExecutor::take(std::size_t self, Item& item)
242{
243 {
244 Worker& own = *workers_[self]; // our own worker record
245 std::lock_guard<std::mutex> lock(own.mutex); // protect our queue while we look at it
246 if (!own.queue.empty()) // we have work of our own
247 {
248 item = std::move(own.queue.front()); // take the oldest task
249 own.queue.pop_front(); // remove it from the queue
250 own.depth.store(own.queue.size(), std::memory_order_relaxed); // update the lock-free estimate
251 queued_.fetch_sub(1, std::memory_order_acq_rel); // one fewer task waiting in total
252 return true; // success
253 }
254 }
255 // Nothing of our own: help a worker that is stuck on a slow task. try_lock keeps this from ever waiting on someone.
256 const std::size_t n = workers_.size(); // total number of workers
257 for (std::size_t i = 1; i < n; ++i) // start at 1 so we do not bother re-checking ourselves
258 {
259 Worker& other = *workers_[(self + i) % n]; // next worker in a ring
260 std::unique_lock<std::mutex> lock(other.mutex, std::try_to_lock); // never block; skip this peer if it is busy
261 if (lock.owns_lock() && !other.queue.empty()) // we got the lock and it has work
262 {
263 item = std::move(other.queue.front()); // steal its oldest task
264 other.queue.pop_front(); // remove it from that queue
265 other.depth.store(other.queue.size(), std::memory_order_relaxed);
266 queued_.fetch_sub(1, std::memory_order_acq_rel);
267 return true; // stolen successfully
268 }
269 }
270 return false; // nobody had any work to give
271}

273/**
274 * Accumulates a task's queue wait time and the number of sampled tasks.
275 *
276 * Uses a monotonic steady_clock timestamp and relaxed atomics: these counters are tuning gauges, not
277 * synchronization points. "Monotonic" means the clock never jumps backwards, so subtracting two timestamps
278 * always gives a sensible elapsed time.
279 * item: Task whose queued_at timestamp is subtracted from now.
280 */
281void BoundedExecutor::account(const Item& item)
282{
283 // Add "now minus queued time" (converted to nanoseconds) to the running total.
284 wait_ns_.fetch_add(
285 static_cast<std::uint64_t>(std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now() - item.queued_at).count()),
286 std::memory_order_relaxed);
287 waited_tasks_.fetch_add(1, std::memory_order_relaxed); // one more sample for the average
288}

290/**
291 * Worker loop: take tasks, run them, and sleep only when every queue is empty.
292 *
293 * Marks the thread with the thread_local executor pointer, then loops taking and running tasks. A task that
294 * throws is logged and swallowed so one misbehaving task cannot end the worker thread. When no work is
295 * available the worker re-arms its idle flag on every pass before waiting, because a submit() may have claimed
296 * it while a thief took the task, leaving a wake-up with an empty queue; sleeping with idle cleared would hide
297 * the worker from every later submit(). The loop exits only when stopping is set and the worker's own queue is
298 * empty, at which point no new task can arrive.
299 * self: Index of this worker.
300 */
301void BoundedExecutor::run(std::size_t self)
302{
303 currentExecutor = this; // mark this thread as one of our workers
304 Worker& own = *workers_[self]; // our own worker record
305 for (;;) // keep running until a return inside the loop
306 {
307 Item item; // empty slot to receive the next task
308 if (!take(self, item)) // try our queue, then stealing; false means nothing to do
309 {
310 std::unique_lock<std::mutex> lock(own.mutex); // hold our lock while deciding whether to sleep
311 if (own.queue.empty()) // double-check after the steal attempt
312 {
313 if (stopping_.load(std::memory_order_acquire)) // shutting down and no work left
314 {
315 return; // stopping, and nothing is left in our queue (no submit can add one any more)
316 }
317 // idle is set again on every pass: a submit() that claimed us may have lost its task to a thief, and then
318 // the wake-up finds an empty queue; going back to sleep with idle cleared would hide us from every later submit().
319 while (!stopping_.load(std::memory_order_acquire) && own.queue.empty()) // sleep while no work and no shutdown
320 {
321 own.idle.store(true, std::memory_order_release); // advertise that we are available for work
322 own.cv.wait(lock); // sleep; wait releases the lock and reacquires it on wake-up
323 }
324 own.idle.store(false, std::memory_order_release); // awake again: no longer idle
325 }
326 continue; // look again: a task arrived, or we are stopping
327 }
328
329 account(item); // record how long the task waited in the queue
330
331 try
332 {
333 item.task(); // actually run the job
334 }
335 catch (const std::exception& e) // a normal C++ exception escaped the task
336 {
337 // A task answers its own request; this only keeps a stray exception from ending the thread.
338 Message::CriticalMessage(name_ + " task failed: " + e.what()); // log it and keep the worker alive
339 }
340 catch (...) // any other, unknown exception
341 {
342 Message::CriticalMessage(name_ + " task failed with an unknown exception."); // log it and continue
343 }
344 }
