1/**
2 * @file boundedexecutor.cpp
3 * @brief Implementation of BoundedExecutor: per-worker queues, work stealing
4 * and shutdown.
5 * @details This file defines the worker pool's behaviour. The constructor
6 * creates all worker objects before starting any threads (so every worker can
7 * safely look at every queue), then each thread runs a loop that takes tasks,
8 * runs them, and sleeps when there is nothing to do. submit() picks a worker
9 * without ever waiting for space, and shutdown() stops new work, drains what is
10 * queued and joins the threads.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * This is where slow, blocking jobs (database calls, password hashing) actually
14 * get executed off Drogon's IO threads. It is called through
15 * BoundedExecutor::submit() and runOnExecutor() from the database and
16 * gatekeeper code. Because it refuses work when full and runs tasks inline with
17 * zero workers, it also controls how the server behaves under load and how the
18 * unit tests stay deterministic. Without it, slow work would block the web
19 * server or grow an unbounded backlog of queued requests.
20 *
21 * CONCEPTS TO LEARN (for beginners):
22 * - thread_local: a variable with one separate copy per running thread.
23 * https://en.cppreference.com/w/cpp/language/storage_duration
24 * - Condition variables: a way for a thread to sleep until another thread wakes
25 * it with new work. https://en.cppreference.com/w/cpp/thread/condition_variable
26 * - Work stealing: an idle worker takes jobs from a busier worker's queue.
27 * https://en.wikipedia.org/wiki/Work_stealing
28 * - Atomic memory order: rules for when one thread's writes become visible to
29 * another. https://en.cppreference.com/w/cpp/atomic/memory_order
30 *
31 * @author Amr Osman <dr3mro@gmail.com>
32 * @date 2026-11-16
33 */

70/**
71 * Allocates the workers, starts their threads and waits until each reports
72 * itself idle.
73 *
74 * queue_max is clamped to at least 1. All Worker objects are created before any
75 * thread starts, because a worker may steal from every other queue. Starting
76 * threads only after that avoids a race where a worker sees a
77 * not-yet-constructed peer. The final spin (with yield) removes the
78 * startup race with an immediate submit(). workers: Number of worker threads; 0
79 * gives inline mode. queue_max: Shared queue capacity, clamped to at least 1.
80 * name: Log label for this pool.
81 */
82BoundedExecutor::BoundedExecutor(std::size_t workers, // NOLINT
83 std::size_t queue_max,
84 std::string name)
85 : queue_max_(std::max<std::size_t>(queue_max,
86 1)),
87 name_(std::move(name)) {
88 workers_.reserve(workers); // reserve once so push_back cannot reallocate
89 // and move the Worker objects
90 for (std::size_t i = 0; i < workers; ++i) {
91 // create one Worker record per requested thread
92 // allocate each Worker; no thread is started yet
93 workers_.push_back(std::make_unique<Worker>());
94 }
95 // Start the threads only now: a worker looks at every queue, so all of them
96 // must exist first.
97 for (std::size_t i = 0; i < workers; ++i) {
98 workers_.at(i)->thread = std::thread([this, i] {
99 // launch the worker loop, capturing this pool and its index
100 run(i);
101 });
102 }
103 // Wait until every worker is ready (idle) before returning, so callers that
104 // submit immediately after construction do not race with thread startup and
105 // trigger stealing.
106 for (std::size_t i = 0; i < workers; ++i) {
107 // spin until this worker says it is waiting for work
108 while (!workers_.at(i)->idle.load(std::memory_order_acquire)) {
109 // politely let the OS run the worker instead of burning CPU in a tight
110 // loop
111 std::this_thread::yield();
112 }
113 }
114}

124/**
125 * @brief Stops accepting work, drains queued tasks and joins the workers.
126 *
127 * @details The stopping flag is published with release semantics; the
128 * empty-lock/notify pair then wakes a worker that is either waiting or about to
129 * wait, and each worker checks the flag under its own mutex before sleeping.
130 * Join is skipped for the calling thread so a task may shut down its own pool.
131 * Statistics are logged only when there is something to report, and the
132 * counters are cleared so a second call does not log them again.
133 * ("Joining" a thread means waiting here until that thread's code has
134 * finished.)
135 */
136void BoundedExecutor::shutdown() {
137 stopping_.store(true, std::memory_order_release);
138 // publish "no new work" before waking anyone
139 // wake every worker so it can notice the flag
140 for (auto& worker : workers_) {
141 {
142 // a sleeping worker either sees the flag or is
143 // already waiting
144 std::scoped_lock lock(worker->mutex);
145 }
146 // the short lock above closes the race with a
147 // worker going to sleep
148 worker->cv.notify_all();
149 }
150 for (auto& worker : workers_) {
151 // never try to join ourselves
152 if (worker->thread.joinable() &&
153 worker->thread.get_id() != std::this_thread::get_id()) {
154 worker->thread.join(); // wait until that worker has finished
155 // draining its queue
156 }
157 }
158 // take a snapshot of the counters before clearing them
159 const Stats state = stats();
160
161 // only log when something actually happened
162 if (state.submitted > 0 || state.rejected > 0) {
163 // Log one summary line built from the counters captured above.
164 Message::InfoMessage(
165 fmt::format("{}: {} tasks, {} rejected, deepest queue {}, average "
166 "wait {:.1f} us.",
167 name_,
168 state.submitted,
169 state.rejected,
170 state.max_queue,
171 state.avg_wait_us));
172 submitted_ = 0; // report once
173 rejected_ = 0; // clear the counter so a second shutdown() does not log
174 // the same numbers
175 }
176}

204/**
205 * Enqueues a task without blocking, preferring an idle worker and enforcing a
206 * shared queue limit.
207 *
208 * Inline mode runs the task immediately on the caller. Otherwise the shared
209 * queued_ counter is reserved with acq_rel semantics and released if the pool
210 * turns out to be full or stopping. The target worker is chosen from a
211 * round-robin start: the first idle one, else the shortest queue. The stopping
212 * flag is checked again while holding that worker's mutex, matching the
213 * worker's own exit check. Only a worker that was idle is notified; a busy
214 * worker rechecks its queue when it finishes a task. task: Nullary callable to
215 * execute (takes no arguments). Returns true if the task ran inline or was
216 * queued; false if it was rejected.
217 */
218bool BoundedExecutor::submit(std::function<void()> task) {
219 // no workers: run on the calling thread instead of queueing
220 if (inlineMode()) {
221 task(); // do the work right now
222 return true; // inline execution always counts as accepted
223 }
224
225 // the pool is shutting down
226 if (stopping_.load(std::memory_order_acquire)) {
227 // count the refusal for diagnostics
228 rejected_.fetch_add(1, std::memory_order_relaxed);
229 // caller should answer 503 (Service Unavailable)
230 return false;
231 }
232
233 // Reserve a place in the shared limit first; give it back if the pool turns
234 // out to be stopping.
235 const std::size_t depth = // atomically claim one slot and read the new total
236 queued_.fetch_add(1, std::memory_order_acq_rel) + 1;
237 // are we over the shared limit
238 if (depth > queue_max_) {
239 queued_.fetch_sub(1,
240 std::memory_order_acq_rel); // give the unused slot back
241 rejected_.fetch_add(1, std::memory_order_relaxed); // count the rejection
242 return false; // refuse quickly instead of waiting for space
243 }
244
245 // From a round-robin start: the first idle worker; with none idle, the one
246 // with the shortest queue (the start worker wins ties).
247 const std::size_t numWorkers = // how many workers exist
248 workers_.size();
249
250 const std::size_t start = // rotate the starting point on each call
251 next_.fetch_add(1, std::memory_order_relaxed) % numWorkers;
252
253 std::size_t target = start; // default pick if nobody is idle
254
255 // "infinity", so the first real depth wins
256 std::size_t shortest = std::numeric_limits<std::size_t>::max();
257
258 // scan every worker once
259 for (std::size_t i = 0; i < numWorkers; ++i) {
260 const std::size_t candidate =
261 (start + i) % numWorkers; // wrap around the list
262 // found a sleeping worker
263 if (workers_.at(candidate)->idle.load(std::memory_order_acquire)) {
264 target = candidate; // give it the job
265 break; // stop scanning
266 }
267
268 const std::size_t depth = workers_.at(candidate)->depth.load(
269 std::memory_order_relaxed); // otherwise read its queue length
270 // shortest queue seen so far
271 if (depth < shortest) {
272 shortest = depth; // remember the new best
273 target = candidate; // and which worker owns it
274 }
275 }
276
277 Worker& worker = *workers_.at(target); // the chosen worker
278 // whether a wake-up signal is needed
279 bool wake = false;
280 {
281 std::scoped_lock lock(
282 worker.mutex); // protect the queue while we add to it
283 // checked under the lock: the worker's exit check is under it too
284
285 if (stopping_.load(std::memory_order_acquire)) {
286 queued_.fetch_sub(1, std::memory_order_acq_rel); // undo our earlier
287 // reservation
288 rejected_.fetch_add(1, std::memory_order_relaxed);
289 return false;
290 }
291 worker.queue.push_back({
292 .task = std::move(task),
293 .queued_at = std::chrono::steady_clock::now(),
294 }); // store the job together
295 // with its arrival time
296 worker.depth.store(worker.queue.size(),
297 std::memory_order_relaxed); // keep the lock-free length
298 // estimate current
299 // An idle worker is claimed (flag cleared) so the next submit() picks
300 // another one instead of queueing behind this task. A busy worker finds
301 // its task itself; it needs no wake-up (system call).
302 wake = worker.idle
303 .exchange(false,
304 std::memory_order_acq_rel); // claim the idle flag;
305 // exchange returns its
306 // old value
307 }
308 submitted_.fetch_add(1,
309 std::memory_order_relaxed); // count the accepted task
310 // a new high-water mark
311 if (depth > max_queue_.load(std::memory_order_relaxed)) {
312 // remember it
313 max_queue_.store(depth, std::memory_order_relaxed);
314 }
315 // only a sleeping worker needs a wake-up
316 if (wake) {
317 worker.cv.notify_one(); // system call to wake it
318 }
319 return true; // tell the caller the job is queued
320}

331/**
332 * Takes a task from this worker's queue, or steals one from a peer if
333 * necessary. Tries the worker's own queue under its mutex first. On a miss it
334 * scans peers using try_lock, so it never waits for a peer that may be busy
335 * with a slow task. Each successful take decrements the shared queued_ counter
336 * with acq_rel semantics. self: Index of the calling worker. item: Receives the
337 * task and its queueing timestamp. Returns true if a task was taken; false if
338 * all queues were empty or locked.
339 */
340bool BoundedExecutor::take(std::size_t self,
341 Item& item) {
342 Worker& own = // our own worker record
343 *workers_.at(self);
344
345 std::scoped_lock lock( // protect our queue while we look at it
346 own.mutex); // we have work of our own
347
348 if (!own.queue.empty()) {
349 item = std::move(own.queue.front()); // take the oldest task
350 own.queue.pop_front(); // remove it from the queue
351 own.depth
352 .store(own.queue.size(),
353 std::memory_order_relaxed); // update the lock-free estimate
354 queued_.fetch_sub(1,
355 std::memory_order_acq_rel); // one fewer task waiting
356 // in total
357 return true; // success
358 }
359
360 // Nothing of our own: help a worker that is stuck on a slow task. try_lock
361 // keeps this from ever waiting on someone.
362
363 const std::size_t numWorkers = workers_.size(); // total number of workers
364 for (std::size_t i = 1; i < numWorkers; ++i) {
365 // start at 1 so we do not bother re-checking ourselves
366
367 Worker& other =
368 *workers_.at((self + i) % numWorkers); // next worker in a ring
369
370 std::unique_lock<std::mutex>
371 lock(other.mutex,
372 std::try_to_lock); // never block; skip this peer if it is busy
373 // we got the lock and it has work
374
375 if (lock.owns_lock() && !other.queue.empty()) {
376 item = std::move(other.queue.front()); // steal its oldest task
377 other.queue.pop_front(); // remove it from that queue
378 other.depth.store(other.queue.size(), std::memory_order_relaxed);
379 queued_.fetch_sub(1, std::memory_order_acq_rel);
380 return true; // stolen successfully
381 }
382 }
383 return false; // nobody had any work to give
384}

406/**
407 * Worker loop: take tasks, run them, and sleep only when every queue is empty.
408 *
409 * Marks the thread with the thread_local executor pointer, then loops taking
410 * and running tasks. A task that throws is logged and swallowed so one
411 * misbehaving task cannot end the worker thread. When no work is available the
412 * worker re-arms its idle flag on every pass before waiting, because a submit()
413 * may have claimed it while a thief took the task, leaving a wake-up with an
414 * empty queue; sleeping with idle cleared would hide the worker from every
415 * later submit(). The loop exits only when stopping is set and the worker's own
416 * queue is empty, at which point no new task can arrive. self: Index of this
417 * worker.
418 */
419void BoundedExecutor::run(std::size_t self) {
420 currentExecutor = this; // mark this thread as one of our workers
421 Worker& own = *workers_.at(self); // our own worker record
422 // keep running until a return inside the loop
423 for (;;) {
424 Item item; // empty slot to receive the next task
425 // try our queue, then stealing; false means nothing to
426 // do
427 if (!take(self, item)) {
428 std::unique_lock<std::mutex> lock(
429 own.mutex); // hold our lock while deciding whether to sleep
430 // double-check after the steal attempt
431 if (own.queue.empty()) {
432 // shutting down and no work left
433 if (stopping_.load(std::memory_order_acquire)) {
434 return; // stopping, and nothing is left in our queue (no
435 // submit can add one any more)
436 }
437 // idle is set again on every pass: a submit() that claimed us
438 // may have lost its task to a thief, and then the wake-up finds
439 // an empty queue; going back to sleep with idle cleared would
440 // hide us from every later submit().
441 // sleep while no work and no shutdown
442 while (!stopping_.load(std::memory_order_acquire) &&
443 own.queue.empty()) {
444 own.idle.store(true,
445 std::memory_order_release); // advertise that we are
446 // available for work
447 own.cv.wait(lock); // sleep; wait releases the lock and
448 // reacquires it on wake-up
449 }
450 own.idle
451 .store(false,
452 std::memory_order_release); // awake again: no longer idle
453 }
454 continue; // look again: a task arrived, or we are stopping
455 }
456
457 account(item); // record how long the task waited in the queue
458
459 try {
460 // actually run the job
461 // a normal C++ exception escaped the task
462 item.task();
463 } catch (const std::exception& e) {
464 // A task answers its own request; this only keeps a stray exception
465 // from ending the thread.
466 // log it and keep the worker alive
467 // any other, unknown exception
468 Message::CriticalMessage(name_ + " task failed: " + e.what());
469 } catch (...) {
470 Message::CriticalMessage(name_ +
471 // log it and continue
472 " task failed with an unknown exception.");
473 }
474 }
