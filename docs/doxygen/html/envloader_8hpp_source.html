1/**
2 * @file envloader.hpp
3 * @brief Base for configuration sections that read typed values from the process environment.
4 * @details This header defines EnvLoader, a reusable base class that turns raw environment
5 * strings into the specific types the server uses: integers, strings, booleans, durations
6 * (lengths of time) and sets of comma-separated names. Each conversion has a fallback default
7 * so a missing or badly written variable never stops the server from starting; problems are
8 * logged instead. Configuration sections inherit from this class and call its helpers.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * Every section of Configurator (server, database, DoS limits, tokens, and so on) derives from
12 * EnvLoader, so this is the shared engine behind all configuration loading. It is the middle
13 * layer: EnvVars below it reads the raw text, and the Configurator sections above it call
14 * getEnvironmentVariable() to obtain ready-to-use typed values. Without it, each section would
15 * have to duplicate parsing and error handling, and a single malformed variable could crash
16 * startup.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - Environment variables: https://man7.org/linux/man-pages/man7/environ.7.html
20 * - Function templates: https://en.cppreference.com/w/cpp/language/function_template
21 * - std::chrono durations (lengths of time): https://en.cppreference.com/w/cpp/chrono/duration
22 * - std::unordered_set (a collection of unique values): https://en.cppreference.com/w/cpp/container/unordered_set
23 *
24 * @author Amr Osman <dr3mro@gmail.com>
25 * @date 2026-11-16
26 */

38/**
39 * @class EnvLoader
40 * @brief Base for configuration sections that read typed values from the process environment.
41 *
42 * @details Extends EnvVars with a family of getEnvironmentVariable() overloads that convert
43 * a raw string into the target type and fall back to a caller-supplied default when the
44 * variable is absent (or, for numeric/duration types, malformed). Conversion failures are
45 * reported through Message::ErrorMessage rather than thrown, so a bad environment cannot
46 * prevent startup. Derived classes must implement printValues() and typically call these
47 * accessors from their constructor initializer lists.
48 *
49 * @note Numeric overloads parse through std::stoi, so integer values outside the range of a
50 * signed int are rejected even when the destination type is wider.
51 * @warning A present boolean variable equals true only when it is exactly "true"; any other
52 * present value is interpreted as false (not as the supplied default).
53 * @see EnvVars, Configurator
54 */
55class EnvLoader : public EnvVars // Inherits the raw environment accessors and adds typed, default-aware conversions on top.
56{
57 public: // The conversion helpers below are available to derived sections and their callers.
58 /** @brief Default constructor; no state of its own beyond the inherited helpers. */
59 explicit EnvLoader() = default; // Nothing to initialise; the inherited EnvVars has no data members.
60 /** @brief Virtual destructor so sections can be destroyed through this base. */
61 virtual ~EnvLoader() = default; // Virtual so deleting through an EnvLoader* runs the derived section's destructor too.
62
63 /**
64 * @brief Reads an integer-like environment variable, converting with std::stoi.
65 *
66 * @details When the variable is absent the default is returned directly. When present,
67 * the value is parsed as a signed int and static_cast to T. A parse failure is logged and
68 * the default returned. In plain language: "if the deployment set this variable, try to
69 * read it as a whole number; if that fails for any reason, quietly use the safe default."
70 *
71 * @tparam T Destination type (typically an unsigned integer or small enum); must be
72 * constructible by static_cast from the std::stoi result.
73 * @param[in] key Environment-variable name.
74 * @param[in] defaultValue Value used when the variable is absent or unparsable.
75 * @return The parsed value, or @p defaultValue.
76 * @note Because parsing goes through std::stoi, values greater than INT_MAX (or less than
77 * INT_MIN) fall back to the default even for 64-bit destinations.
78 * @see getEnvironmentVariable(const std::string&, const std::chrono::seconds&)
79 */
80 template <typename T> // T is filled in from the type of defaultValue at each call site (e.g. uint32_t or uint16_t).
81 T getEnvironmentVariable(const std::string& key, T defaultValue) // Returns the parsed number or defaultValue if the variable is missing or invalid.
82 {
83 auto value = get(key); // Fetch the raw text via the inherited helper; an empty optional means "not set".
84 if (!value) // No value stored under this name, so there is nothing to parse.
85 {
86 return defaultValue; // Give the caller the safe fallback immediately.
87 }
88 try // Parsing text can fail (e.g. "abc"), so guard it and recover instead of letting an exception escape.
89 {
90 return static_cast<T>(std::stoi(value.value())); // Convert text to int then to the requested type (uint32_t, uint8_t, etc.).
91 }
92 catch (const std::exception&) // std::stoi throws when the text is not a number or is out of int range.
93 {
94 Message::ErrorMessage("Invalid numeric value for env var " + key + "; using default."); // Log the bad variable name so an operator can spot the mistake.
95 return defaultValue; // Recover with the default so a bad setting cannot prevent startup.
96 }
97 }
