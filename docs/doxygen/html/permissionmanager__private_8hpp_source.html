167 template <typename T> // Generic helper; T is the value type to extract from the JSON.
168 static std::optional<T> extract_json_value_safely( // Reads a key from JSON as T, reporting BAD_REQUEST instead of throwing.
169 const std::optional<jsoncons::json>& service_j, const std::string& key, const std::string& service_name, Http::Error& error)
170 {
171 if (!service_j.has_value() || service_j->empty()) // Reject a missing or empty payload before trying to read it.
172 {
173 error.code = Http::Status::BAD_REQUEST; // The client sent a bad request; this status is returned to them.
174 error.message = "Service data is not provided for " + service_name; // Explain which service payload was missing.
175 return std::nullopt; // No usable value; caller must treat this as failure.
176 }
177
178 T value; // Default-constructed result, filled in only if extraction succeeds.
179 try // Conversion may throw, so it is attempted inside a try block.
180 {
181 value = service_j->at(key).as<T>(); // Look up the key and convert it to the requested type T.
182 }
183 catch (const std::exception& e) // Runs when the key is missing or cannot be converted.
184 {
185 error.message = "Failed to extract " + key; // Tell the client which key could not be read.
186 error.code = Http::Status::BAD_REQUEST; // The client sent a bad request; this status is returned to them.
187 return std::nullopt; // No usable value; caller must treat this as failure.
188 }
189 return value; // Hand back the successfully extracted value.

192 /**
193 * @brief Fetches an entity's permission JSON, serving the idle-expiry cache when possible.
194 * @details @p func is invoked with @p args to generate the cache key/query (which embeds
195 * the entity table and id). On a cache hit no database access occurs. On a miss
196 * the generation token is captured before the read so a concurrent invalidateAll()
197 * makes the subsequent put() a no-op. Empty results are not cached.
198 * @tparam Func Callable returning `std::optional<std::string>` (the query/key).
199 * @tparam Args Argument types forwarded to @p func.
200 * @param[in,out] func Query generator.
201 * @param[in,out] args Arguments for @p func.
202 * @return Cached or freshly read permissions, or `std::nullopt` on no query/no data.
203 * @see BasicPermissionCache::generation
204 */
205 template <typename Func, typename... Args> // Generic helper; Func builds the query and Args are its inputs.
206 static std::optional<jsoncons::json> getPermissionsOfEntity(Func&& func, Args&... args) // Fetches permissions (cache first, database on a miss).
207 {
208 std::optional<std::string> query = std::invoke(std::forward<Func>(func), std::forward<Args>(args)...); // Build the query/key string by calling the supplied generator with its arguments.
209
210 if (!query.has_value() || query->empty()) // No query means there is nothing to look up; fail early.
211 {
212 return std::nullopt; // No usable value; caller must treat this as failure.
213 }
214
215 auto& cache = PermissionCache::instance(); // Get the shared cache object (a process-wide singleton).
216 if (auto cached = cache.get(query.value()); cached.has_value()) // Cache hit: reuse the stored answer and skip the database entirely.
217 {
218 return cached; // Return the cached JSON to the caller.
219 }
220
221 const std::shared_ptr<DatabaseController> db_ctl = Store::cachedObject<DatabaseController>(); // Only needed on a cache miss; cachedObject returns a shared pointer kept for reuse.
222 const uint64_t generation = cache.generation(); // Remember the current epoch before reading, so a concurrent flush can invalidate our result.
223 std::optional<jsoncons::json> permissions_j = db_ctl->getPermissions(query.value()); // Cache miss: read the permissions from PostgreSQL (blocking version).
224
225 if (!permissions_j.has_value() || permissions_j->empty()) // The database had no data; do not cache an empty result.
226 {
227 return std::nullopt; // No usable value; caller must treat this as failure.
228 }
229
230 cache.put(query.value(), permissions_j.value(), generation); // Store the fresh result, but only if no flush happened in the meantime.
231 return permissions_j; // Return the freshly read permissions to the caller.
232 }

234 /**
235 * @brief Coroutine form of getPermissionsOfEntity(); awaits the async database read.
236 * @details Same cache-key generation, hit short-circuit and generation-token protection
237 * as the synchronous version, but does not block a Drogon IO thread.
238 * @tparam Func Callable returning `std::optional<std::string>` (the query/key).
239 * @tparam Args Argument types forwarded to @p func.
240 * @param[in,out] func Query generator.
241 * @param[in,out] args Arguments for @p func.
242 * @return Cached or freshly read permissions, or `std::nullopt` on no query/no data.
243 * @note The caller must drive the returned task; the query string must outlive the awaited read.
244 */
245 template <typename Func, typename... Args> // Generic helper; Func builds the query and Args are its inputs.
246 static drogon::Task<std::optional<jsoncons::json>> getPermissionsOfEntityCoro(Func&& func, Args&... args) // Coroutine version of the fetch; awaits the async database read.
247 {
248 std::optional<std::string> query = std::invoke(std::forward<Func>(func), std::forward<Args>(args)...); // Build the query/key string by calling the supplied generator with its arguments.
249
250 if (!query.has_value() || query->empty()) // No query means there is nothing to look up; fail early.
251 {
252 co_return std::nullopt; // Coroutine return of 'no value' when there is nothing usable.
253 }
254
255 auto& cache = PermissionCache::instance(); // Get the shared cache object (a process-wide singleton).
256 if (auto cached = cache.get(query.value()); cached.has_value()) // Cache hit: reuse the stored answer and skip the database entirely.
257 {
258 co_return cached; // Coroutine return of the cached answer.
259 }
260
261 const std::shared_ptr<DatabaseController> db_ctl = Store::cachedObject<DatabaseController>();
262 const uint64_t generation = cache.generation(); // Remember the current epoch before the async read for the same stale-write protection.
263 std::optional<jsoncons::json> permissions_j = co_await db_ctl->getPermissionsCoro(query.value()); // Cache miss: await the async read without blocking the IO thread.
264
265 if (!permissions_j.has_value() || permissions_j->empty()) // The database had no data; do not cache an empty result.
266 {
267 co_return std::nullopt; // Coroutine return of 'no value' when there is nothing usable.
268 }
269
270 cache.put(query.value(), permissions_j.value(), generation); // Store the fresh result, but only if no flush happened in the meantime.
271 co_return permissions_j; // Coroutine return of the freshly read answer.
272 }
