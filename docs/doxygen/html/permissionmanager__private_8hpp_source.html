227 /**
228 * @brief Extracts and converts @p key from a JSON payload, reporting
229 * BAD_REQUEST on failure.
230 * @details Returns nullopt (with @p error populated) if the payload is
231 * missing/empty or the key is absent or not convertible to T;
232 * conversion exceptions are swallowed.
233 * @tparam T Target value type (jsoncons `as<T>()`).
234 * @param[in] service_j Payload to read.
235 * @param[in] key Key to extract.
236 * @param[in] service_name Included in the "not provided" message.
237 * @param[out] error Receives BAD_REQUEST and a reason on failure.
238 * @return The extracted value, or `std::nullopt` on failure.
239 */
240 template <typename T> // Generic helper; T is the value type to extract
241 // from the JSON.
242 static std::optional<T> extract_json_value_safely( // Reads a key from JSON
243 // as T, reporting
244 // BAD_REQUEST instead
245 // of throwing.
246 const std::optional<jsoncons::json>& service_j,
247 const std::string& key,
248 const std::string& service_name,
249 Http::Error& error) {
250 // Reject a missing or empty payload before
251 // trying to read it.
252 if (!service_j.has_value() || service_j->empty()) {
253 error.code = Http::Status::BAD_REQUEST; // The client sent a bad
254 // request; this status is
255 // returned to them.
256 error.message =
257 "Service data is not provided for " +
258 service_name; // Explain which service payload was missing.
259 return std::nullopt; // No usable value; caller must treat this
260 // as failure.
261 }
262
263 T value; // Default-constructed result, filled in only if
264 // extraction succeeds.
265 // Conversion may throw, so it is attempted inside a try block.
266 try {
267 // Look up the key and convert
268 // it to the requested type T.
269 // Runs when the key is missing or cannot be converted.
270 value = service_j->at(key).as<T>();
271 } catch (const std::exception& e) {
272 error.message = "Failed to extract " +
273 key; // Tell the client which key could not be read.
274 error.code = Http::Status::BAD_REQUEST; // The client sent a bad
275 // request; this status is
276 // returned to them.
277 return std::nullopt; // No usable value; caller must treat this
278 // as failure.
279 }
280 return value; // Hand back the successfully extracted value.
281 }

283 /**
284 * @brief Fetches an entity's permission JSON, serving the idle-expiry
285 * cache when possible.
286 * @details @p func is invoked with @p args to generate the cache
287 * key/query (which embeds the entity table and id). On a cache hit no
288 * database access occurs. On a miss the generation token is captured
289 * before the read so a concurrent invalidateAll() makes the subsequent
290 * put() a no-op. Empty results are not cached.
291 * @tparam Func Callable returning `std::optional<std::string>` (the
292 * query/key).
293 * @tparam Args Argument types forwarded to @p func.
294 * @param[in,out] func Query generator.
295 * @param[in,out] args Arguments for @p func.
296 * @return Cached or freshly read permissions, or `std::nullopt` on no
297 * query/no data.
298 * @see BasicPermissionCache::generation
299 */
300 // Generic helper; Func builds the query and
301 // Args are its inputs.
302 // Fetches permissions
303 // (cache first, database
304 // on a miss).
305 template <typename Func,
306 typename... Args>
307 static std::optional<jsoncons::json> getPermissionsOfEntity(Func&& func,
308 Args&... args) {
309 std::optional<std::string> query =
310 std::invoke(std::forward<Func>(func),
311 std::forward<Args>(
312 args)...); // Build the query/key string by calling
313 // the supplied generator with its
314 // arguments.
315
316 // No query means there is nothing to look up;
317 // fail early.
318 if (!query.has_value() || query->empty()) {
319 return std::nullopt; // No usable value; caller must treat this
320 // as failure.
321 }
322
323 auto& cache = PermissionCache::instance(); // Get the shared cache object
324 // (a process-wide singleton).
325 if (auto cached = cache.get(query.value());
326 // Cache hit: reuse the stored answer and
327 // skip the database entirely.
328 cached.has_value()) {
329 return cached; // Return the cached JSON to the caller.
330 }
331
332 const std::shared_ptr<DatabaseController> db_ctl =
333 Store::cachedObject<DatabaseController>(); // Only needed on a cache
334 // miss; cachedObject
335 // returns a shared
336 // pointer kept for reuse.
337 const uint64_t generation =
338 cache.generation(); // Remember the current epoch before
339 // reading, so a concurrent flush can
340 // invalidate our result.
341 std::optional<jsoncons::json> permissions_j = db_ctl->getPermissions(
342 query.value()); // Cache miss: read the permissions from
343 // PostgreSQL (blocking version).
344
345 // The database had no data; do not
346 // cache an empty result.
347 if (!permissions_j.has_value() || permissions_j->empty()) {
348 return std::nullopt; // No usable value; caller must treat this
349 // as failure.
350 }
351
352 cache.put(query.value(),
353 permissions_j.value(),
354 generation); // Store the fresh result, but only if no flush
355 // happened in the meantime.
356 return permissions_j; // Return the freshly read permissions to the
357 // caller.
358 }

380 // database read.

383 static drogon::Task<std::optional<jsoncons::json>>
384 getPermissionsOfEntityCoro(Func&& func,
385 Args&... args) {
386 std::optional<std::string> query =
387 std::invoke(std::forward<Func>(func),
388 std::forward<Args>(
389 args)...); // Build the query/key string by calling
390 // the supplied generator with its
391 // arguments.
392
393 // No query means there is nothing to look up;
394 // fail early.
395 if (!query.has_value() || query->empty()) {
396 co_return std::nullopt; // Coroutine return of 'no value' when
397 // there is nothing usable.
398 }
399
400 auto& cache = PermissionCache::instance(); // Get the shared cache object
401 // (a process-wide singleton).
402 if (auto cached = cache.get(query.value());
403 // Cache hit: reuse the stored answer and
404 // skip the database entirely.
405 cached.has_value()) {
406 co_return cached; // Coroutine return of the cached answer.
407 }
408
409 const std::shared_ptr<DatabaseController> db_ctl =
410 Store::cachedObject<DatabaseController>();

412 cache.generation(); // Remember the current epoch before the
413 // async read for the same stale-write
414 // protection.
415 std::optional<jsoncons::json> permissions_j =
416 co_await db_ctl->getPermissionsCoro(
417 query.value()); // Cache miss: await the async read without
418 // blocking the IO thread.
419
420 // The database had no data; do not
421 // cache an empty result.
422 if (!permissions_j.has_value() || permissions_j->empty()) {
423 co_return std::nullopt; // Coroutine return of 'no value' when
424 // there is nothing usable.
425 }
426
427 cache.put(query.value(),
428 permissions_j.value(),
429 generation); // Store the fresh result, but only if no flush
430 // happened in the meantime.

432 // answer.
433 }
