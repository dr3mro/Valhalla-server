2 * @file databasecontroller.cpp
3 * @brief Blocking (pqxx pool) and coroutine (AsyncDbClient/Drogon) implementations of DatabaseController.
4 * @details Synchronous methods funnel through the executer template; coroutine methods each obtain the async client,
5 * short-circuit when it is absent, and translate Drogon busy errors into `DatabaseBusy` for 503 responses.
6 *
7 * WHY THIS FILE MATTERS IN THE SERVER:
8 * This is the lowest layer that actually touches PostgreSQL. All the shared machinery lives here: lazy, thread-safe
9 * creation of the connection pool, the single `executer` template that acquires a connection and runs blocking SQL,
10 * and the async family that runs on Drogon's IO loop. By converting "busy" into a dedicated exception, this file lets
11 * the HTTP layer answer 503 with Retry-After instead of treating overload as a broken query.
12 *
13 * CONCEPTS TO LEARN (for beginners):
14 * - RAII (resources released when objects go out of scope): https://en.cppreference.com/w/cpp/language/raii
15 * - Double-checked locking / std::atomic / std::scoped_lock: https://en.cppreference.com/w/cpp/mutex/scoped_lock
16 * - std::optional and std::nullopt: https://en.cppreference.com/w/cpp/utility/optional
17 * - Templates (the `executer` helper): https://en.cppreference.com/w/cpp/language/templates
18 * - Coroutines (co_await / co_return): https://en.cppreference.com/w/cpp/language/coroutines
19 * - Exceptions and catch blocks: https://en.cppreference.com/w/cpp/language/exceptions
20 *
21 * @author Amr Osman <dr3mro@gmail.com>
22 * @date 2026-11-16
23 */

48void DatabaseController::ensureBackends() // Thread-safe one-time creation of pool + watchdog.
49{
50 // Lazily create the connection pool and watchdog on first use instead of
51 // in the constructor: the pool constructor calls exit(EXIT_FAILURE) when
52 // PostgreSQL is unreachable, which made the whole class unconstructible in
53 // hermetic unit tests. The server still connects at startup, because
54 // Server's constructor eagerly creates DatabaseSchema, which constructs
55 // the (singleton) DatabaseController through Store and then performs the
56 // first query (schema population) during boot.
57 //
58 // Every query calls this, so the common case (already created) is one atomic load. The mutex is only taken during the first
59 // calls; the release/acquire pair makes the pointers written below visible to every thread that sees the flag set.
60 if (backendsReady_.load(std::memory_order_acquire)) // Fast path: already initialized, so return immediately.
61 {
62 return;
63 }
64
65 static std::mutex backendInitMutex; // Function-local mutex guards the slow first-init path.
66 std::scoped_lock lock(backendInitMutex); // RAII lock: held only in this scope, released automatically.
67 if (backendsReady_.load(std::memory_order_relaxed)) // Re-check after locking: another thread may have initialized.
68 {
69 return;
70 }
71 if (!databaseConnectionPool_) // Only create backends if they do not exist yet.
72 {
73 // Hermetic unit-test seam: when VALHALLA_TEST_NO_DB is set (the unit
74 // test binary sets it before any Configurator is created), the pool
75 // and watchdog are never created. Every query path then reports
76 // "database unavailable" (std::nullopt) deterministically, with no
77 // sockets opened and no retries. Production never sets the variable,
78 // so behaviour is unchanged there.
79 if (!testing_config_->no_db) // Skip creation entirely in hermetic test mode.
80 {
81 databaseConnectionPool_ = Store::getObject<DatabaseConnectionPool>(); // Resolve the shared pool from the service locator.
82 watchDog_ = Store::getObject<WatchDog>(); // Resolve the shared watchdog.
83 }
84 }
85 backendsReady_.store(true, std::memory_order_release); // Publish the pointers to all threads.
86}

144drogon::Task<std::optional<jsoncons::json>> DatabaseController::executeQueryCoro(const std::string& query, const pqxx::params& params)
145{
146 // All coroutine entry points share this guard: no async client means the DB is unavailable, so report nullopt
147 // rather than throwing. The client is obtained from the Store, not held as controller state.
148 auto asyncClient = Store::getObject<AsyncDbClient>(nullptr); // Look up the async client; nullptr if not configured.
149 if (asyncClient == nullptr || asyncClient->client() == nullptr) // No client means the DB is unavailable.
150 {
151 co_return std::nullopt; // Return an empty result; callers treat this as unavailable or failed.
152 }
153 try
154 {
155 // Drogon transaction: BEGIN is sent on creation, COMMIT on destruction, ROLLBACK on exception -
156 // the asynchronous counterpart of the pqxx::work used by the synchronous path.
157 const auto transaction = co_await asyncClient->client()->newTransactionCoro(); // BEGIN; COMMIT on scope exit, ROLLBACK on error.
158 const auto result = co_await asyncdb::execSqlCoro(*transaction, query, params); // Execute SQL inside the transaction.
159 co_return asyncdb::mapResultToObject(result); // Convert the rows into a JSON object.
160 }
161 catch (const std::exception& e)
162 {
163 Message::CriticalMessage(fmt::format("Exception occurred during async query execution: {}", e.what()));
164 // A busy database is not a failed query: propagate DatabaseBusy so the HTTP layer can answer 503.
165 if (asyncdb::isBusy(e)) // Distinct from a failed query: signals overload.
166 {
167 throw DatabaseBusy(); // Propagate so the HTTP layer can answer 503.
168 }
169 co_return std::nullopt; // Return an empty result; callers treat this as unavailable or failed.
170 }
171}

248{

250 if (asyncClient == nullptr || asyncClient->client() == nullptr) // No client means the DB is unavailable.

287drogon::Task<std::optional<std::unordered_set<api::v2::ColumnInfo>>> DatabaseController::getTableSchemaCoro(const std::string& tableName)
288{
289 auto asyncClient = Store::getObject<AsyncDbClient>(nullptr); // Look up the async client; nullptr if not configured.
290 if (asyncClient == nullptr || asyncClient->client() == nullptr) // No client means the DB is unavailable.
291 {
292 co_return std::nullopt; // Return an empty result; callers treat this as unavailable or failed.
293 }
294 try
295 {
296 const std::string query = "SELECT column_name, data_type, column_default, is_nullable FROM information_schema.columns WHERE table_name = $1 AND column_name != 'id';"; // Metadata query; the id column is skipped.
297 const auto result = co_await asyncdb::execSqlCoro(*asyncClient->client(), query, pqxx::params{tableName});
298
299 std::unordered_set<api::v2::ColumnInfo> schema; // Collected column descriptions.
300 // Map each information_schema row; a NULL default becomes "None" and isNullable follows the "YES" flag.
301 for (drogon::orm::Result::SizeType row = 0; row < result.size(); ++row) // One iteration per returned column.
302 {
303 api::v2::ColumnInfo column; // One column's metadata record.
304 column.Name = std::string(result[row][0].c_str()); // Column name from the first field.
305 column.DataType = std::string(result[row][1].c_str());
306 column.Constraint = result[row][2].isNull() ? "None" : std::string(result[row][2].c_str()); // A NULL default becomes 'None'.
307 column.isNullable = std::string(result[row][3].c_str()) == "YES"; // 'YES' means the column may be NULL.
308 schema.insert(column); // Store the record in the set for the validator.
309 }
310 co_return schema; // Hand the column set back to the caller.
311 }
312 catch (const std::exception& e)
313 {
314 Message::CriticalMessage(fmt::format("Exception occurred during async schema load: {}", e.what()));
315 if (asyncdb::isBusy(e)) // Distinct from a failed query: signals overload.
316 {
317 throw DatabaseBusy(); // Propagate so the HTTP layer can answer 503.
318 }
319 co_return std::nullopt; // Return an empty result; callers treat this as unavailable or failed.
320 }
