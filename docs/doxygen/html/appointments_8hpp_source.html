1/**
2 * @file appointments.hpp
3 * @brief REST controller for appointment records of the appointment-providing entities.
4 * @details This file defines the Appointments controller, which exposes create, read, update, delete
5 * and search endpoints for appointments. A {entityType} part of the URL (clinics, pharmacies,
6 * laboratories or radiologycenters) selects one concrete AppointmentController from a registry
7 * held in a std::variant. Each handler either awaits an async controller coroutine or calls
8 * executeControllerMethod for blocking database work, depending on the configuration. Every
9 * handler first builds a callback that converts the controller's numeric status into a
10 * standard JSON HTTP response and builds a Requester from the identity attributes set by Auth.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * This controller is the HTTP front door for all appointment management. Drogon routes the /api/v2
14 * appointment URLs here; the handlers then call the shared dispatcher in basic/common.hpp, which calls
15 * the appointment controllers and entities and ultimately the database. It depends on the Auth filter
16 * for identity and on the service locator for the executor and controllers. Without it there would be
17 * no way for clients to book, view, change or cancel appointments through the API.
18 *
19 * CONCEPTS TO LEARN (for beginners):
20 * - REST create/read/update/delete (CRUD) routes: https://developer.mozilla.org/en-US/docs/Web/HTTP/Methods
21 * - std::variant (a type-safe union): https://en.cppreference.com/w/cpp/utility/variant
22 * - HTTP filters/middleware: https://drogon.docsforge.com/master/filter/
23 * - C++ coroutines (co_await): https://en.cppreference.com/w/cpp/language/coroutines
24 *
25 * @author Amr Osman <dr3mro@gmail.com>
26 * @date 2026-11-16
27 */

63 * concrete AppointmentController<T> from a std::variant registry. Every handler builds a

83 void Create(const drogon::HttpRequestPtr& req, std::function<void(const drogon::HttpResponsePtr&)>&& callback, const std::string& entityType) // HTTP handler: req is the request, callback answers it, entityType comes from the URL.
84 {

87 {
88 auto cb = std::move(callback); // Move the callback into the coroutine so it outlives this function.
89 drogon::async_run([this, req, cb, entityType]() mutable -> drogon::Task<void> // Start a coroutine capturing everything it needs by value.
90 {
91 CALLBACK_ mcb = [cb](int code, const std::string& content) // Adapter: turn the controller's (code, JSON) reply into an HTTP response.
92 {
93 using Reply = std::function<void(const drogon::HttpResponsePtr&)>; // Short alias for the response callback type.
94 switch (code) // Translate the numeric status into the matching JSON reply.
95 {
96 case api::v2::Http::Status::OK: // Create succeeded.
97 Helper::successResponse(content, Reply(cb)); // Send 200 with the returned JSON body.
98 break; // Leave the switch.
99 case api::v2::Http::Status::INTERNAL_SERVER_ERROR: // Internal failure.
100 Helper::failureResponse(content, Reply(cb)); // Send a tagged 500 body.
101 break; // Leave the switch.
102 default: // Any specific error code.
103 Helper::errorResponse(static_cast<drogon::HttpStatusCode>(code), content, Reply(cb)); // Reuse the number as the HTTP status.
104 break; // Leave the switch.
105 }
106 };

108 Requester requester(req->getAttributes()->get<uint64_t>("clientID"), req->getAttributes()->get<std::string>("clientGroup")); // Build the caller identity from Auth attributes for permission checks.
109
110 auto ctlIt = appointmentRegistry.find(entityType); // Find the controller variant for this URL type.
111 if (ctlIt == appointmentRegistry.end()) // Unknown entity type cannot be handled.
112 {
113 std::move(mcb)(api::v2::Http::Status::BAD_REQUEST, fmt::format("Type mapping not found for: {}", entityType)); // Reply 400 with the offending type.
114 co_return; // End the coroutine; the response is already sent.
115 }
116
117 try // The controller call may throw.
118 {
119 co_await std::visit([&](auto& controller) -> drogon::Task<void> { // Pick the stored controller type and wait for its coroutine.
120 co_await controller->CreateAsync(std::move(mcb), std::move(requester), req->body()); // Create the appointment asynchronously using the JSON body.
121 }, ctlIt->second); // Pass the chosen controller variant to std::visit.
122 }
123 catch (const std::exception& e) // Any unexpected error.
124 {
125 Helper::failureResponse(e.what(), std::move(cb)); // Reply 500 with the error text.
126 }
127 });
128
129 return; // The coroutine owns the rest; leave the handler now.
130 }
