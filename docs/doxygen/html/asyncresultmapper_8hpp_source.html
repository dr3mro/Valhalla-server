1/**
2 * @file asyncresultmapper.hpp
3 * @brief Maps Drogon PostgreSQL result sets to jsoncons values with the outcomes of the synchronous path in mind.
4 * @details This header declares functions that convert query results into JSON. One function converts a single row
5 * into a JSON object, and two helpers wrap that for the common cases of "first row only" and "all rows as
6 * an array". Column names and the value text decide the JSON type: nulls stay null, a boolean flag becomes
7 * true/false, identifier columns become numbers, JSON-looking text is parsed, and everything else is a string.
8 *
9 * WHY THIS FILE MATTERS IN THE SERVER:
10 * DatabaseController's coroutine methods return JSON to the HTTP layer, but Drogon gives them a Result set of rows
11 * and fields. These functions are the translation step in between. They deliberately reproduce the behaviour of the
12 * older synchronous Database::executeQuery so clients cannot tell which database path answered their request. A
13 * malformed number or JSON value falls back to plain text instead of failing the whole request, which keeps one bad
14 * database row from breaking an entire API response. Without this file the async controllers would have no way to
15 * produce the JSON shapes the rest of the server expects.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - Drogon DbClient and result sets: https://drogon.docsforge.com/master/database/
19 * - jsoncons JSON values: https://danielaparker.github.io/jsoncons/
20 * - PostgreSQL data types and type OIDs: https://www.postgresql.org/docs/current/datatype.html
21 * - C++20 coroutines: https://en.cppreference.com/w/cpp/language/coroutines
22 * - std::string_view (a non-owning view of text): https://en.cppreference.com/w/cpp/string/basic_string_view
23 *
24 * @author Amr Osman <dr3mro@gmail.com>
25 * @date 2026-11-16
26 */

47 // Type OIDs mirrored from Database::executeQuery. mapRow infers types from column names and value
48 // text, so these constants are unused in this repository; no reason for retaining them is recorded.
49 // An OID is PostgreSQL's numeric identifier for a data type. These are kept as documentation of the
50 // types the synchronous mapper cares about; the async mapper does not read the OID of a field.
51 /** @brief varchar type OID. */
52 static constexpr unsigned int PG_TEXT_OID = 1043; // varchar (treated as text); `constexpr` fixes the value at compile time, `static` gives it internal linkage.
53 /** @brief int4 (INTEGER) OID. */
54 static constexpr unsigned int PG_INTEGER_OID = 23; // 23 is PostgreSQL's identifier for a 32-bit integer column.
55 /** @brief bool (BOOLEAN) OID. */
56 static constexpr unsigned int PG_BOOLEAN_OID = 16; // 16 is PostgreSQL's identifier for a boolean column.
57 /** @brief json OID. */
58 static constexpr unsigned int PG_JSON_OID = 114; // 114 identifies the JSON text type.

62 /**
63 * @brief Maps one result row to a JSON object keyed by column name.
64 * @details Conversion order: SQL NULL to JSON null; a column named `active` to a boolean (`t`, `true` or `1` mean true);
65 * `id`, names ending in `_id`, `appointment_duration` and `appointment_payment_amount` to an integer when
66 * `std::stoi` succeeds; a non-empty value starting with `{` or `[` to parsed JSON; otherwise a string. Both
67 * integer and JSON parsing failures fall back to the raw string, so a malformed value cannot abort mapping.
68 * A fresh object is built for each call, so no field can leak between rows.
69 * @param[in] result Result set returned by the query.
70 * @param[in] row Zero-based row index into `result`.
71 * @return JSON object with one entry per column, in column order.
72 * @pre `row` is a valid index into `result`.
73 */
74 [[nodiscard]] jsoncons::json mapRow(const drogon::orm::Result& result, drogon::orm::Result::SizeType row); // `[[nodiscard]]` reminds callers that ignoring the returned object is a mistake.
75
76 /**
77 * @brief Maps the first row of a result to a JSON object.
78 * @param[in] result Result set returned by the query.
79 * @return `mapRow(result, 0)` when non-empty; a default-constructed (null) JSON value when empty.
80 * @note This mirrors the single-object outcome of `Database::executeQuery` for a query with no rows.
81 */
82 [[nodiscard]] jsoncons::json mapResultToObject(const drogon::orm::Result& result); // Used when a query is expected to return at most one row, such as a lookup by id.
83
84 /**
85 * @brief Maps every row of a result to a JSON array, preserving row order.
86 * @param[in] result Result set returned by the query.
87 * @return Array with one JSON object per row; empty when the result set is empty.
88 */
89 [[nodiscard]] jsoncons::json::array mapResultToArray(const drogon::orm::Result& result); // Used for search/list queries that must return every matching row.
90} // namespace asyncdb
