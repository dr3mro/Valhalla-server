1/**
2 * @file gatekeeper.cpp
3 * @brief Implementation for gatekeeper.
4 * @details This file turns the GateKeeper declarations into working code. Login parses the
5 * request, looks up the stored password hash, then moves the slow hashing, the
6 * session write and the token creation onto a password-worker thread so the server's
7 * IO threads stay free. Authentication first tries a cached session and only falls
8 * back to verifying the JWT when there is no live session. All permission helpers are
9 * one-line forwarders to PermissionManager, and a set of explicit template
10 * instantiations generates those helpers for every supported entity type.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * This is where the security facade actually does its job. Controllers call login(), logout()
14 * and the can*() helpers; filters call isDosAttack(); the authentication filter calls
15 * isAuthenticationValid(). This file wires those calls to SessionManager, TokenManager,
16 * DOSDetector, PermissionManager and PasswordWorker. If it were missing, the declared
17 * GateKeeper interface would have no implementation and nothing security-related would link
18 * or run. The macro block at the bottom is what makes the permission helpers exist for concrete
19 * types like User, Patient and ClinicAppointment.
20 *
21 * CONCEPTS TO LEARN (for beginners):
22 * - Callback style (call this function when the answer is ready): https://en.cppreference.com/w/cpp/utility/functional/function
23 * - Lambda capture, especially capturing shared_ptrs by value: https://en.cppreference.com/w/cpp/language/lambda
24 * - Thread pool / offloading slow work: https://en.wikipedia.org/wiki/Thread_pool
25 * - std::move (transfer ownership instead of copying): https://en.cppreference.com/w/cpp/utility/move
26 * - Coroutine / co_await: https://en.cppreference.com/w/cpp/language/coroutines
27 * - Macro / preprocessor directive: https://en.cppreference.com/w/cpp/preprocessor
28 * - Explicit template instantiation: https://en.cppreference.com/w/cpp/language/templates
29 * - JWT: https://jwt.io/introduction
30 * - HTTP status codes: https://developer.mozilla.org/en-US/docs/Web/HTTP/Status
31 *
32 * @author Amr Osman <dr3mro@gmail.com>
33 * @date 2026-11-16
34 */

78void GateKeeper::login(CALLBACK_&& callback, std::string_view data, const std::string& ip_address, std::string_view group)
79{
80 try
81 {
82 bool success = false; // Set true only if JSON and both fields parse; starts false ("not proven").
83 std::string message; // Accumulates a human-readable reason if anything fails.
84 std::optional<Types::Credentials> credentials = parse_credentials(data, message, success); // Parse username/password without throwing.
85
86 // Malformed JSON or missing fields fail fast before any database or worker work.
87 if (!success || !credentials.has_value()) // Either the parse failed or no credentials came back: reject immediately.
88 {
89 std::move(callback)(HttpStatus::BAD_REQUEST, "credentials parse error : " + message); // Tell the client its body was bad (400).
90 return; // Stop; nothing else to do.
91 }
92
93 std::optional<Types::ClientLoginData> clientLoginData = Types::ClientLoginData{}; // Fresh identity bag, value-initialized so bools start false.
94 clientLoginData->group = group; // Record which group (selects the session table later).
95 clientLoginData->ip_address = ip_address; // Record the IP so it can be bound into the token.
96
97 // Database reads only, on this thread. Everything slow (password hashing) and everything that writes happens on the worker.
98 std::optional<std::string> password_hash = sessionManager_->fetchPasswordHash(credentials, clientLoginData, message); // Read the stored hash; nullopt means unknown account.
99 // A missing hash means the account is unknown or unusable; report UNAUTHORIZED without hinting which.
100 if (!password_hash.has_value()) // No hash -> do not continue the login.
101 {
102 std::move(callback)(Http::Status::UNAUTHORIZED, "Login failure: " + message); // Deliberately vague so attackers cannot enumerate users.
103 return;
104 }
105
106 // Lambda captures shared_ptrs by value and all mutable data by move: the worker
107 // outlives this call frame, and callback is invoked on the worker thread.
108 runOnPasswordWorker(*passwordWorker_, std::move(callback), // Queue the slow work; if the queue is full the callback gets a 503.
109 [sessionManager = sessionManager_, tokenManager = tokenManager_, credentials = std::move(credentials), password_hash = std::move(password_hash).value(),
110 clientLoginData = std::move(clientLoginData), message = std::move(message)](CALLBACK_&& callback) mutable
111 {
112 try
113 {
114 // Password comparison plus session write happen together on the worker;
115 // failure here is deliberately indistinguishable from a bad password.
116 if (!sessionManager->completeLogin(credentials, password_hash, clientLoginData, message) || !clientLoginData.has_value()) // Wrong password OR no session data -> fail.
117 {
118 callback(Http::Status::UNAUTHORIZED, "Login failure: " + message); // Same vague error as a missing account.
119 return;
120 }
121
122 // Token creation depends on the fully populated clientLoginData above.
123 if (!tokenManager->generateToken(clientLoginData)) // Build and sign the JWT from the identity fields.
124 {
125 callback(Http::Status::INTERNAL_SERVER_ERROR, "failed to generate token"); // Internal problem signing the token (500).
126 return;
127 }
128
129 // Success payload echoes the identity back so the client can cache it.
130 jsoncons::json token_object; // Build the JSON object that will be the response body.
131 token_object["token"] = clientLoginData->token; // The signed JWT the client must send on later requests.
132 token_object["username"] = clientLoginData->username; // Echo the user name.
133 token_object["client_id"] = clientLoginData->clientId; // Echo the numeric id.
134 token_object["group"] = clientLoginData->group; // Echo the group.
135 token_object["ipAddress"] = clientLoginData->ip_address; // Echo the bound IP.
136
137 callback(Http::Status::OK, token_object.as<std::string>()); // Reply 200 with the JSON serialized to a string.
138 }
139 catch (const std::exception& e) // Any unexpected error inside the worker task.
140 {
141 CRITICALMESSAGERESPONSE // Log it and answer through callback with the right HTTP status.
142 }
143 });
144 }
145 catch (const std::exception& e) // Any unexpected error during parsing or hash lookup.
146 {
147 CRITICALMESSAGERESPONSE // Log and reply; the callback is guaranteed to be called exactly once.
148 }
149}

151drogon::Task<void> GateKeeper::loginAsync(CALLBACK_&& callback, std::string_view data, const std::string& ip_address, std::string_view group)
152{
153 try
154 {
155 bool success = false; // Tracks whether JSON and required fields parsed.
156 std::string message; // Collects failure details.
157 std::optional<Types::Credentials> credentials = parse_credentials(data, message, success); // Same parsing as login().
158
159 if (!success || !credentials.has_value()) // Bad body: fail before any awaited or worker work.
160 {
161 std::move(callback)(HttpStatus::BAD_REQUEST, "credentials parse error : " + message); // Reply 400 with the parse error.
162 co_return; // Leave the coroutine (nothing more to do).
163 }
164
165 std::optional<Types::ClientLoginData> clientLoginData = Types::ClientLoginData{}; // Fresh identity accumulator.
166 clientLoginData->group = group; // Group selects the session table.
167 clientLoginData->ip_address = ip_address; // IP is bound into the token.
168
169 // Runs on a Drogon executor: the hash lookup yields cooperatively instead of
170 // blocking the IO thread; the coroutine frame stays alive for the await duration.
171 std::optional<std::string> password_hash = co_await sessionManager_->fetchPasswordHashCoro(credentials, clientLoginData, message); // Await the DB read; nullopt means no account.
172 if (!password_hash.has_value()) // Unknown/unusable account.
173 {
174 std::move(callback)(Http::Status::UNAUTHORIZED, "Login failure: " + message); // Vague 401 reply.
175 co_return;
176 }
177
178 runOnPasswordWorker(*passwordWorker_, std::move(callback), // Hand slow verification + writes to the worker pool.
179 [sessionManager = sessionManager_, tokenManager = tokenManager_, credentials = std::move(credentials), password_hash = std::move(password_hash).value(),
180 clientLoginData = std::move(clientLoginData), message = std::move(message)](CALLBACK_&& callback) mutable
181 {
182 try
183 {
184 // Password comparison plus session write happen together on the worker;
185 // failure here is deliberately indistinguishable from a bad password.
186 if (!sessionManager->completeLogin(credentials, password_hash, clientLoginData, message) || !clientLoginData.has_value()) // Bad password or missing session data.
187 {
188 callback(Http::Status::UNAUTHORIZED, "Login failure: " + message); // Vague 401 reply.
189 return;
190 }
191
192 // Token creation depends on the fully populated clientLoginData above.
193 if (!tokenManager->generateToken(clientLoginData)) // Sign the JWT from the identity fields.
194 {
195 callback(Http::Status::INTERNAL_SERVER_ERROR, "failed to generate token"); // Signing failed (500).
196 return;
197 }
198
199 // Success payload echoes the identity back so the client can cache it.
200 jsoncons::json token_object; // Response body under construction.
201 token_object["token"] = clientLoginData->token; // Signed JWT.
202 token_object["username"] = clientLoginData->username; // User name.
203 token_object["client_id"] = clientLoginData->clientId; // Numeric id.
204 token_object["group"] = clientLoginData->group; // Group.
205 token_object["ipAddress"] = clientLoginData->ip_address; // Bound IP.
206
207 callback(Http::Status::OK, token_object.as<std::string>()); // Reply 200 with JSON text.
208 }
209 catch (const std::exception& e) // Unexpected worker error.
210 {
211 CRITICALMESSAGERESPONSE // Log and answer through callback.
212 }
213 });
214 }
215 catch (const std::exception& e) // Unexpected error before/around the await.
216 {
217 CRITICALMESSAGERESPONSE // Log and reply.
218 }
219 co_return; // Coroutine form always exits with co_return (even after a normal dispatch).
220}

239bool GateKeeper::isAuthenticationValid(std::optional<Types::ClientLoginData>& clientLoginData, std::string& message)
240{
241 // Fast path: a live session avoids re-verifying the JWT signature.
242 if (sessionManager_->clientHasValidSession(clientLoginData, message)) // Is there a cached, still-valid session?
243 {
244 return true; // Yes: authenticated, no need to check the token.
245 }
246
247 // Cold path: validate the token, then rehydrate the session cache from its claims.
248 if (tokenManager_->isTokenValid(clientLoginData, message)) // No session, so check the JWT signature and claims.
249 {
250 return sessionManager_->storeSession(clientLoginData, message); // Token ok: rebuild the session cache and return that result.
251 }
252 return false; // Neither session nor token proved the client: reject.
253}

255drogon::Task<bool> GateKeeper::isAuthenticationValidAsync(std::optional<Types::ClientLoginData>& clientLoginData, std::string& message)
256{
257 if (sessionManager_->clientHasValidSession(clientLoginData, message)) // Cached session still valid? (synchronous)
258 {
259 co_return true; // Fast path success.
260 }
261
262 // Only the token verification is awaited; both session calls remain synchronous.
263 if (co_await tokenManager_->isTokenValidCoro(clientLoginData, message)) // Await the token check without blocking a thread.
264 {
265 co_return sessionManager_->storeSession(clientLoginData, message); // Rebuild the session cache and return that result.
266 }
267 co_return false; // Not authenticated.
268}

280std::optional<jsoncons::json> GateKeeper::parse_data(/* NOLINT(readability-convert-member-functions-to-static)*/
281 std::string_view data, std::string& message, bool& success)
282{
283 success = false; // Assume failure until parsing actually succeeds.
284 jsoncons::json object_j; // Will hold the parsed JSON document.
285 try
286 {
287 object_j = jsoncons::json::parse(data); // Turn the raw body text into a JSON object.
288 }
289 catch (const std::exception& e)
290 {
291 // Accumulate diagnostics into message rather than throwing across the callback boundary.
292 message = fmt::format("Json parsing error : {} - {}", message, e.what()); // Append the parse error to any existing message.
293 return std::nullopt; // Signal failure to the caller.
294 }
295 success = true; // Parsing worked.
296 return object_j; // Hand back the parsed document.
297}

300std::optional<Credentials> GateKeeper::parse_credentials(std::string_view data, std::string& message, bool& success)
301{
302 success = false; // Assume failure until both fields are extracted.
303
304 std::optional<jsoncons::json> credentials_j = parse_data(data, message, success); // Step 1: parse the JSON.
305 if (!success || !credentials_j.has_value()) // If parsing failed there is nothing to read.
306 {
307 return std::nullopt; // Propagate failure.
308 }
309
310 Types::Credentials credentials; // Result object, filled in below.
311 try
312 {
313 // at() throws for a missing key; as<std::string>() throws for the wrong type.
314 credentials.username = credentials_j.value().at("username").as<std::string>(); // Read the "username" member as text.
315 credentials.password = credentials_j.value().at("password").as<std::string>(); // Read the "password" member as text.
316 }
317 catch (const std::exception& e)
318 {
319 message = fmt::format("Credentials parsing error : {} - {}", message, e.what()); // Append why extraction failed.
320 return std::nullopt; // Missing or ill-typed field -> failure.
321 }
322 success = true; // Both fields were present and were strings.
323 return credentials; // Hand back the credentials.
324}
