1/**
2 * @file gatekeeper.cpp
3 * @brief Implementation for gatekeeper.
4 * @details This file turns the GateKeeper declarations into working code. Login
5 * parses the request, looks up the stored password hash, then moves the slow
6 * hashing, the session write and the token creation onto a password-worker
7 * thread so the server's IO threads stay free. Authentication first tries a
8 * cached session and only falls back to verifying the JWT when there is no live
9 * session. All permission helpers are one-line forwarders to PermissionManager,
10 * and a set of explicit template instantiations generates those helpers for
11 * every supported entity type.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * This is where the security facade actually does its job. Controllers call
15 * login(), logout() and the can*() helpers; filters call isDosAttack(); the
16 * authentication filter calls isAuthenticationValid(). This file wires those
17 * calls to SessionManager, TokenManager, DOSDetector, PermissionManager and
18 * PasswordWorker. If it were missing, the declared GateKeeper interface would
19 * have no implementation and nothing security-related would link or run. The
20 * macro block at the bottom is what makes the permission helpers exist for
21 * concrete types like User, Patient and ClinicAppointment.
22 *
23 * CONCEPTS TO LEARN (for beginners):
24 * - Callback style (call this function when the answer is ready):
25 * https://en.cppreference.com/w/cpp/utility/functional/function
26 * - Lambda capture, especially capturing shared_ptrs by value:
27 * https://en.cppreference.com/w/cpp/language/lambda
28 * - Thread pool / offloading slow work:
29 * https://en.wikipedia.org/wiki/Thread_pool
30 * - std::move (transfer ownership instead of copying):
31 * https://en.cppreference.com/w/cpp/utility/move
32 * - Coroutine / co_await: https://en.cppreference.com/w/cpp/language/coroutines
33 * - Macro / preprocessor directive:
34 * https://en.cppreference.com/w/cpp/preprocessor
35 * - Explicit template instantiation:
36 * https://en.cppreference.com/w/cpp/language/templates
37 * - JWT: https://jwt.io/introduction
38 * - HTTP status codes: https://developer.mozilla.org/en-US/docs/Web/HTTP/Status
39 *
40 * @author Amr Osman <dr3mro@gmail.com>
41 * @date 2026-11-16
42 */

116void GateKeeper::login(CALLBACK_&& callback,
117 std::string_view data,
118 const std::string& ip_address,
119 std::string_view group) {
120 try {
121 bool success = false; // Set true only if JSON and both fields parse;
122 // starts false ("not proven").
123 std::string
124 message; // Accumulates a human-readable reason if anything fails.
125 std::optional<Types::Credentials> credentials =
126 parse_credentials(data, message, success); // Parse username/password
127 // without throwing.
128
129 // Malformed JSON or missing fields fail fast before any database or
130 // worker work.
131 // Either the parse failed or no credentials
132 // came back: reject immediately.
133 if (!success || !credentials.has_value()) {
134 std::move(
135 callback)(HttpStatus::BAD_REQUEST,
136 "credentials parse error : " +
137 message); // Tell the client its body was bad (400).
138 return; // Stop; nothing else to do.
139 }
140
141 std::optional<Types::ClientLoginData> clientLoginData =
142 Types::ClientLoginData{}; // Fresh identity bag, value-initialized
143 // so bools start false.
144 clientLoginData->group =
145 group; // Record which group (selects the session table later).
146 clientLoginData->ip_address =
147 ip_address; // Record the IP so it can be bound into the token.
148
149 // Database reads only, on this thread. Everything slow (password
150 // hashing) and everything that writes happens on the worker.
151 std::optional<std::string> password_hash =
152 sessionManager_
153 ->fetchPasswordHash(credentials,
154 clientLoginData,
155 message); // Read the stored hash; nullopt
156 // means unknown account.
157 // A missing hash means the account is unknown or unusable; report
158 // UNAUTHORIZED without hinting which.
159 // No hash -> do not continue the login.
160 if (!password_hash.has_value()) {
161 std::move(callback)(Http::Status::UNAUTHORIZED,
162 "Login failure: " +
163 message); // Deliberately vague so attackers
164 // cannot enumerate users.
165 return;
166 }
167
168 // Lambda captures shared_ptrs by value and all mutable data by move:
169 // the worker outlives this call frame, and callback is invoked on the
170 // worker thread.
171 runOnPasswordWorker(
172 *passwordWorker_,
173 std::move(callback), // Queue the slow work; if the queue is full
174 // the callback gets a 503.
175 [sessionManager = sessionManager_,
176 tokenManager = tokenManager_,
177 credentials = std::move(credentials),
178 password_hash = std::move(password_hash).value(),
179 clientLoginData = std::move(clientLoginData),
180 message = std::move(message)](CALLBACK_&& callback) mutable {
181 try {
182 // Password comparison plus session write happen together on
183 // the worker; failure here is deliberately
184 // indistinguishable from a bad password.
185 // Wrong password OR no
186 // session data -> fail.
187 if (!sessionManager->completeLogin(credentials,
188 password_hash,
189 clientLoginData,
190 message) ||
191 !clientLoginData.has_value()) {
192 callback(Http::Status::UNAUTHORIZED,
193 "Login failure: " + message); // Same vague error as a
194 // missing account.
195 return;
196 }
197
198 // Token creation depends on the fully populated
199 // clientLoginData above.
200 // Build and sign the JWT from
201 // the identity fields.
202 if (!tokenManager->generateToken(clientLoginData)) {
203 callback(Http::Status::INTERNAL_SERVER_ERROR,
204 "failed to generate token"); // Internal problem
205 // signing the token
206 // (500).
207 return;
208 }
209
210 // Success payload echoes the identity back so the client
211 // can cache it.
212 jsoncons::json token_object; // Build the JSON object that
213 // will be the response body.
214 token_object["token"] =
215 clientLoginData->token; // The signed JWT the client
216 // must send on later requests.
217 token_object["username"] =
218 clientLoginData->username; // Echo the user name.
219 token_object["client_id"] =
220 clientLoginData->clientId; // Echo the numeric id.
221 token_object["group"] = clientLoginData->group; // Echo the group.
222 token_object["ipAddress"] =
223 clientLoginData->ip_address; // Echo the bound IP.
224
225 // Reply 200 with the JSON
226 // serialized to a string.
227 // Any unexpected error inside the worker task.
228 callback(Http::Status::OK, token_object.as<std::string>());
229 } catch (const std::exception& e) {
230 // Log it and answer through
231 // callback with the right HTTP
232 // status.
233 // Any unexpected error during parsing or hash lookup.
234 CRITICALMESSAGERESPONSE
235 }
236 });
237 } catch (const std::exception& e) {
238 CRITICALMESSAGERESPONSE // Log and reply; the callback is guaranteed to
239 // be called exactly once.
240 }
241}

243drogon::Task<void> GateKeeper::loginAsync(CALLBACK_&& callback,
244 std::string_view data,
245 const std::string& ip_address,
246 std::string_view group) {
247 try {
248 bool success = false; // Tracks whether JSON and required fields parsed.
249 std::string message; // Collects failure details.
250 std::optional<Types::Credentials> credentials =
251 parse_credentials(data, message, success); // Same parsing as login().
252
253 // Bad body: fail before any
254 // awaited or worker work.
255 if (!success || !credentials.has_value()) {
256 std::move(callback)(HttpStatus::BAD_REQUEST,
257 "credentials parse error : " +
258 message); // Reply 400 with the parse error.
259 co_return; // Leave the coroutine (nothing more to do).
260 }
261
262 std::optional<Types::ClientLoginData> clientLoginData =
263 Types::ClientLoginData{}; // Fresh identity accumulator.
264 clientLoginData->group = group; // Group selects the session table.
265 clientLoginData->ip_address = ip_address; // IP is bound into the token.
266
267 // Runs on a Drogon executor: the hash lookup yields cooperatively
268 // instead of blocking the IO thread; the coroutine frame stays alive
269 // for the await duration.
270 std::optional<std::string> password_hash =
271 co_await sessionManager_
272 ->fetchPasswordHashCoro(credentials,
273 clientLoginData,
274 message); // Await the DB read; nullopt
275 // means no account.
276 // Unknown/unusable account.
277 if (!password_hash.has_value()) {
278 std::move(callback)(Http::Status::UNAUTHORIZED,
279 "Login failure: " + message); // Vague 401 reply.
280 co_return;
281 }
282
283 runOnPasswordWorker(*passwordWorker_,
284 std::move(callback), // Hand slow verification + writes
285 // to the worker pool.
286 [sessionManager = sessionManager_,
287 tokenManager = tokenManager_,
288 credentials = std::move(credentials),
289 password_hash = std::move(password_hash).value(),
290 clientLoginData = std::move(clientLoginData),
291 message =
292 std::move(message)](CALLBACK_&& callback) mutable {
293 try {
294 // Password comparison plus session write happen
295 // together on the worker; failure here is
296 // deliberately indistinguishable from a bad
297 // password.
298 // Bad password or
299 // missing session data.
300 if (!sessionManager->completeLogin(credentials,
301 password_hash,
302 clientLoginData,
303 message) ||
304 !clientLoginData.has_value()) {
305 callback(Http::Status::UNAUTHORIZED,
306 "Login failure: " +
307 message); // Vague 401 reply.
308 return;
309 }
310
311 // Token creation depends on the fully populated
312 // clientLoginData above.
313 // Sign the JWT from the
314 // identity fields.
315 if (!tokenManager->generateToken(clientLoginData)) {
316 callback(Http::Status::INTERNAL_SERVER_ERROR,
317 "failed to generate token"); // Signing
318 // failed
319 // (500).
320 return;
321 }
322
323 // Success payload echoes the identity back so the
324 // client can cache it.
325 jsoncons::json token_object; // Response body under
326 // construction.
327 token_object["token"] =
328 clientLoginData->token; // Signed JWT.
329 token_object["username"] =
330 clientLoginData->username; // User name.
331 token_object["client_id"] =
332 clientLoginData->clientId; // Numeric id.
333 // Group.
334 token_object["group"] = clientLoginData->group;
335 token_object["ipAddress"] =
336 clientLoginData->ip_address; // Bound IP.
337
338 // Reply 200 with
339 // JSON text.
340 // Unexpected worker error.
341 callback(Http::Status::OK,
342 token_object.as<std::string>());
343 } catch (const std::exception& e) {
344 // Log and answer through
345 // callback.
346 // Unexpected error before/around the await.
347 CRITICALMESSAGERESPONSE
348 }
349 });
350 } catch (const std::exception& e) {
351 CRITICALMESSAGERESPONSE // Log and reply.
352 }
353 co_return; // Coroutine form always exits with co_return (even after a
354 // normal dispatch).
355}

380bool GateKeeper::isAuthenticationValid(
381 std::optional<Types::ClientLoginData>& clientLoginData,
382 std::string& message) {
383 // Fast path: a live session avoids re-verifying the JWT signature.
384 // Is there a cached,
385 // still-valid session?
386 if (sessionManager_->clientHasValidSession(clientLoginData, message)) {
387 return true; // Yes: authenticated, no need to check the token.
388 }
389
390 // Cold path: validate the token, then rehydrate the session cache from its
391 // claims.
392 // No session, so check the JWT
393 // signature and claims.
394 if (tokenManager_->isTokenValid(clientLoginData, message)) {
395 return sessionManager_
396 ->storeSession(clientLoginData,
397 message); // Token ok: rebuild the session cache and
398 // return that result.
399 }
400 return false; // Neither session nor token proved the client: reject.
401}

403drogon::Task<bool> GateKeeper::isAuthenticationValidAsync(
404 std::optional<Types::ClientLoginData>& clientLoginData,
405 std::string& message) {
406 // Cached session still
407 // valid? (synchronous)
408 if (sessionManager_->clientHasValidSession(clientLoginData, message)) {
409 co_return true; // Fast path success.
410 }
411
412 // Only the token verification is awaited; both session calls remain
413 // synchronous.
414 // Await the token check without
415 // blocking a thread.
416 if (co_await tokenManager_->isTokenValidCoro(clientLoginData, message)) {
417 co_return sessionManager_
418 ->storeSession(clientLoginData,
419 message); // Rebuild the session cache and return that
420 // result.
421 }
422 co_return false; // Not authenticated.
423}

441std::optional<jsoncons::json> GateKeeper::
442 parse_data(/* NOLINT(readability-convert-member-functions-to-static)*/
443 std::string_view data,
444 std::string& message,
445 bool& success) {
446 success = false; // Assume failure until parsing actually succeeds.
447 jsoncons::json object_j; // Will hold the parsed JSON document.
448 try {
449 object_j = jsoncons::json::parse(
450 data); // Turn the raw body text into a JSON object.
451 } catch (const std::exception& e) {
452 // Accumulate diagnostics into message rather than throwing across the
453 // callback boundary.
454 message = fmt::format("Json parsing error : {} - {}",
455 message,
456 e.what()); // Append the parse error to any existing
457 // message.
458 return std::nullopt; // Signal failure to the caller.
459 }
460 success = true; // Parsing worked.
461 return object_j; // Hand back the parsed document.
462}

464std::optional<Credentials> GateKeeper::parse_credentials(std::string_view data,
465 std::string& message,
466 bool& success) {
467 success = false; // Assume failure until both fields are extracted.
468
469 std::optional<jsoncons::json> credentials_j =
470 parse_data(data, message, success); // Step 1: parse the JSON.
471 // If parsing failed there is nothing to read.
472 if (!success || !credentials_j.has_value()) {
473 return std::nullopt; // Propagate failure.
474 }
475
476 Types::Credentials credentials; // Result object, filled in below.
477 try {
478 // at() throws for a missing key; as<std::string>() throws for the wrong
479 // type.
480 credentials.username =
481 credentials_j.value()
482 .at("username")
483 .as<std::string>(); // Read the "username" member as text.
484 credentials.password =
485 credentials_j.value()
486 .at("password")
487 .as<std::string>(); // Read the "password" member as text.
488 } catch (const std::exception& e) {
489 message = fmt::format("Credentials parsing error : {} - {}",
490 message,
491 e.what()); // Append why extraction failed.
492 return std::nullopt; // Missing or ill-typed field -> failure.
493 }
494 success = true; // Both fields were present and were strings.
495 return credentials; // Hand back the credentials.
496}

642// synchronous form. Why: a template is only turned into real machine code when
643// it is used. Because the controllers live in different .cpp files, this file
644// must force the compiler to emit a version of each helper for every concrete
645// type. Each macro below expands to several "template bool
646// GateKeeper::canX<Type>..." lines. The lines inside a macro end with a
647// backslash (line continuation), so comments are added above the macros rather
648// than on those lines - a trailing comment after a backslash would break the
649// macro. canCreate specializations

654 Http::Error&); \

659 const uint64_t entity_id, \
