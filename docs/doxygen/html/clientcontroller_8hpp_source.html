1/**
2 * @file clientcontroller.hpp
3 * @brief Client-specific handler that layers password hashing, sessions and account state over the generic CRUDS flow.
4 * @details Declares `ClientController<T>`, which inherits storage from EntityController<T> and the account contract
5 * from ClientControllerBase. Simple operations (Read/Search) just forward to the base. Create/Update are
6 * special: they run on a PasswordWorker because hashing is deliberately slow. Login/Logout and account
7 * state go through GateKeeper. The definitions and instantiations live in clientcontroller.cpp.
8 *
9 * WHY THIS FILE MATTERS IN THE SERVER:
10 * Accounts need protections that ordinary records do not: passwords must be hashed off the request thread, and
11 * sessions must be invalidated when an account is deleted or suspended. This class adds exactly those behaviours on
12 * top of the generic entity pipeline. Without it, clients would be treated like any other row and password hashing
13 * would block the server's IO threads.
14 *
15 * CONCEPTS TO LEARN (for beginners):
16 * - Multiple inheritance: https://en.cppreference.com/w/cpp/language/derived_class
17 * - `final` overrides (cannot be overridden further): https://en.cppreference.com/w/cpp/language/virtual
18 * - Lambdas captured by value for worker jobs: https://en.cppreference.com/w/cpp/language/lambda
19 * - Coroutines (co_await / drogon::Task): https://en.cppreference.com/w/cpp/language/coroutines
20 * - Concepts / constrained templates (`Client_t`): https://en.cppreference.com/w/cpp/language/constraints
21 * - std::shared_ptr (sharing services): https://en.cppreference.com/w/cpp/memory/shared_ptr
22 *
23 * @author Amr Osman <dr3mro@gmail.com>
24 * @date 2026-11-16
25 */

49/**
50 * @class ClientController
51 * @brief Client-specific handler that layers password hashing, sessions and account state over the generic CRUDS flow.
52 * @details Inherits persistence from EntityController<T> and the full client contract from ClientControllerBase.
53 * Read/Search and their coroutine forms simply forward to EntityController; Create/Update dispatch their
54 * bodies to a PasswordWorker so password hashing never occupies a Drogon IO thread. Login/Logout are
55 * delegated to GateKeeper, and Suspend/Delete additionally invalidate sessions on success.
56 * @tparam T Entity type satisfying `Client_t` (derives from `api::v2::Client`), e.g. User or Provider.
57 * @note All overrides are `final`. Coroutine variants borrow their arguments and must be awaited while they live.
58 * @see ClientControllerBase, EntityController, PasswordWorker
59 */
60template <Client_t T>
61class ClientController : public EntityController<T>, public ClientControllerBase

66 ClientController(ClientController&&) = default; // Move transfers the service handles.
67 ClientController& operator=(const ClientController&) = default; // Copy assignment shares the handles.
68 ClientController& operator=(ClientController&&) = default; // Move assignment transfers the handles.
69 ~ClientController() override = default; // Virtual destructor; no explicit cleanup needed.
