1/**
2 * @file types.hpp
3 * @brief Shared value types and their hash specialisations.
4 * @details Defines the ColumnInfo struct, which describes one database column,
5 * and teaches std::hash how to hash it (and a pair of strings).
6 * Hashing matters because these types are used as keys in unordered containers
7 * such as unordered_set. Equality and hashing deliberately look at the
8 * same three fields so that two "equal" objects always get the same
9 * hash.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * The schema validator compares database table descriptions. It stores columns
13 * in hash-based sets, which need both an operator== and a std::hash to work.
14 * This file provides both, and deliberately ignores the "Constraint" text so a
15 * column is recognised by its name, data type and nullability alone. Without
16 * these definitions the unordered sets would not compile for ColumnInfo.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - Template specializations:
20 * https://en.cppreference.com/w/cpp/language/template_specialization
21 * - std::hash: https://en.cppreference.com/w/cpp/utility/hash
22 * - operator== :
23 * https://en.cppreference.com/w/cpp/language/operators#Equality_operators
24 * - std::pair: https://en.cppreference.com/w/cpp/utility/pair
25 *
26 * @author Amr Osman <dr3mro@gmail.com>
27 * @date 2026-11-16
28 */

67 // yes/no flag needs only a bool.

69 /**
70 * @brief Compares name, data type and nullability only.
71 * @param[in] other Column to compare against.
72 * @return True when the three keyed fields match; Constraint is
73 * ignored.
74 */
75 bool operator==(const ColumnInfo& other) const {
76 return Name == other.Name && DataType == other.DataType &&
77 isNullable == other.isNullable;
78 } // Only the three "key" fields decide equality; Constraint is
79 // left out so minor SQL differences do not create false
80 // mismatches.

85// We are allowed to specialise std::hash for our own types by
86// opening namespace std.
87namespace std {
88 /**
89 * @brief Hashes ColumnInfo consistently with its operator==.
90 * @details Combines Name, DataType and isNullable; Constraint is excluded
91 * so equal columns always receive equal hashes.
92 */
93 // An explicit specialisation: replace the generic std::hash
94 // specifically for ColumnInfo.
95 // The hash object type for
96 // ColumnInfo keys.
97 template <>
98 struct hash<api::v2::ColumnInfo> {
99 /** @brief Hashes the three keyed fields, matching
100 * ColumnInfo::operator==. */
101 // Called by unordered containers to get a hash for a column.
102 size_t operator()(const api::v2::ColumnInfo& column) const {
103 // Combine hashes of Name, DataType, and isNullable
104 size_t h1 = hash<std::string>()(column.Name); // Hash the name first.
105 size_t h2 =
106 hash<std::string>()(column.DataType); // Hash the data type second.
107 size_t h3 =
108 hash<bool>()(column.isNullable); // Hash the nullable flag third.
109 return h1 ^ (h2 << 1) ^ (h3 << 2); // Mix the three hashes with XOR and
110 // bit-shifts so different fields land
111 // in different bit positions;
112 // Constraint is excluded on purpose.
113 }
114 };
115
116 /**
117 * @brief Hashes a pair of strings by combining the hashes of its members.
118 */
119 // Another explicit specialisation, this time for a pair of
120 // strings.
121 // Lets a (string, string) pair be used
122 // as a key in unordered containers.
123 template <>
124 struct hash<std::pair<std::string, std::string>> {
125 /** @brief Hashes a string pair by combining its two members' hashes. */
126 // Receives the pair and produces one combined hash value.
127 size_t operator()(const std::pair<std::string, std::string>& pair) const {
128 // Combine the hashes of the pair's two strings.
129 size_t h1 = hash<std::string>()(
130 pair.first); // Hash the first string of the pair.
131 size_t h2 = hash<std::string>()(
132 pair.second); // Hash the second string of the pair.
133 return h1 ^ (h2 << 1); // Shift the second hash left so the two
134 // parts do not collide as easily.
135 }
136 };
137} // namespace std
