1/**
2 * @file types.hpp
3 * @brief Shared value types and their hash specialisations.
4 * @details Defines the ColumnInfo struct, which describes one database column,
5 * and teaches std::hash how to hash it (and a pair of strings). Hashing
6 * matters because these types are used as keys in unordered containers
7 * such as unordered_set. Equality and hashing deliberately look at the
8 * same three fields so that two "equal" objects always get the same hash.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * The schema validator compares database table descriptions. It stores columns
12 * in hash-based sets, which need both an operator== and a std::hash to work.
13 * This file provides both, and deliberately ignores the "Constraint" text so a
14 * column is recognised by its name, data type and nullability alone. Without
15 * these definitions the unordered sets would not compile for ColumnInfo.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - Template specializations: https://en.cppreference.com/w/cpp/language/template_specialization
19 * - std::hash: https://en.cppreference.com/w/cpp/utility/hash
20 * - operator== : https://en.cppreference.com/w/cpp/language/operators#Equality_operators
21 * - std::pair: https://en.cppreference.com/w/cpp/utility/pair
22 *
23 * @author Amr Osman <dr3mro@gmail.com>
24 * @date 2026-11-16
25 */

71 /**
72 * @brief Hashes ColumnInfo consistently with its operator==.
73 * @details Combines Name, DataType and isNullable; Constraint is excluded so
74 * equal columns always receive equal hashes.
75 */
76 template <> // An explicit specialisation: replace the generic std::hash specifically for ColumnInfo.
77 struct hash<api::v2::ColumnInfo> // The hash object type for ColumnInfo keys.
78 {
79 /** @brief Hashes the three keyed fields, matching ColumnInfo::operator==. */
80 size_t operator()(const api::v2::ColumnInfo& column) const // Called by unordered containers to get a hash for a column.
81 {
82 // Combine hashes of Name, DataType, and isNullable
83 size_t h1 = hash<std::string>()(column.Name); // Hash the name first.
84 size_t h2 = hash<std::string>()(column.DataType); // Hash the data type second.
85 size_t h3 = hash<bool>()(column.isNullable); // Hash the nullable flag third.
86 return h1 ^ (h2 << 1) ^ (h3 << 2); // Mix the three hashes with XOR and bit-shifts so different fields land in different bit positions; Constraint is excluded on purpose.
87 }
88 };

91 * @brief Hashes a pair of strings by combining the hashes of its members.
92 */
93 template <> // Another explicit specialisation, this time for a pair of strings.
94 struct hash<std::pair<std::string, std::string>> // Lets a (string, string) pair be used as a key in unordered containers.
95 {
96 /** @brief Hashes a string pair by combining its two members' hashes. */
97 size_t operator()(const std::pair<std::string, std::string>& pair) const // Receives the pair and produces one combined hash value.
98 {
99 // Combine the hashes of the pair's two strings.
100 size_t h1 = hash<std::string>()(pair.first); // Hash the first string of the pair.
101 size_t h2 = hash<std::string>()(pair.second); // Hash the second string of the pair.
102 return h1 ^ (h2 << 1); // Shift the second hash left so the two parts do not collide as easily.
103 }
104 };
