5 * @details This header defines the Permissions struct. It lists the individual
6 * rights (like "can read" or "is owner") as single bits, and provides

11 * Authorization questions ("may this person delete this?") are answered using

13 * helpers here to turn a number stored in the database into a clear yes/no.
14 * Without this file there would be no common language for describing rights.

31/**
32 * @struct Permissions
33 * @brief Holds a bitmask of authorization bits and the helpers to combine/test them.
34 * @details Powers are single-bit flags stored in a uint8_t. Two groups of bits are
35 * distinguished by convention: identity bits (USER, PROVIDER, IS_ADMIN, IS_OWNER,
36 * IS_SUPER) and capability bits (CAN_READ, CAN_WRITE, CAN_DELETE). A value is a
37 * bitwise combination, so a single mask can express roles plus capabilities.
38 * @note All the test helpers (hasPower(), hasMatch(), operator&) answer "any bit in
39 * common", not "all requested bits present".
40 *
41 * In everyday words: imagine an 8-switch control panel, one switch per right.
42 * Each switch is a power of two (1, 2, 4, 8, ...). Turning on several switches
43 * and adding the numbers produces a single value that remembers exactly which
44 * switches are on. Testing "is any of these switches on?" is a fast bitwise AND.
45 */
46struct Permissions // A value type; it simply groups the bit helpers together.
47{
48 public:
49 /**
50 * @brief Individual authorization bits. Values are powers of two and may be OR-ed together.
51 * @note Only NONE and CAN_READ/CAN_WRITE/CAN_DELETE are consumed in src/; USER, PROVIDER
52 * and IS_SUPER are declared but identity is enforced via Requester string helpers.
53 */
54 enum class PowerLevel : std::uint8_t // Named on/off switches stored in exactly one byte (8 bits).
55 {
56 NONE = 0x00U, /**< No powers; 00000000. */
57 USER = 0x01U, /**< Identity bit for ordinary users; 00000001. */
58 PROVIDER = 0x02U, /**< Identity bit for providers; 00000010. */
59 CAN_READ = 0x04U, /**< Capability: may read; 00000100. */
60 CAN_WRITE = 0x08U, /**< Capability: may create/update; 00001000. */
61 CAN_DELETE = 0x10U, /**< Capability: may delete; 00010000. */
62 IS_ADMIN = 0x20U, /**< Identity bit for administrators; 00100000. */
63 IS_OWNER = 0x40U, /**< Identity bit for the service owner; 01000000. */
64 IS_SUPER = 0x80U, /**< Identity bit for super-users; 10000000. */
65 }; // 10000000 is the largest single bit; every value above is a power of two.
66
67 /** @brief Constructs a rule wrapping a single (possibly combined) PowerLevel mask. */

70 virtual ~Permissions() = default; // Virtual destructor allows safe use through a base pointer.
71 Permissions() = delete; // No default: a rule is meaningless without a mask.
72 Permissions(const Permissions&) = default; // Copy is fine; the value is just a byte.
73 Permissions& operator=(const Permissions&) = default; // Copy assignment is fine for the same reason.
74 Permissions(Permissions&&) = default; // Move is fine; there is nothing to transfer.
75 Permissions& operator=(Permissions&&) = default; // Move assignment is fine for the same reason.

77 /**
78 * @brief Tests whether any bit in @p check is present in this rule's mask.
79 * @param[in] check Mask of bits to look for.
80 * @return `true` if at least one checked bit is set.
81 */
82 bool hasPower(const PowerLevel& check) { return static_cast<int>(powerLevel & check) != 0; } // ANDs the masks; non-zero means at least one requested bit is present.
83
84 /** @brief Returns the bitwise OR of two masks. */
85 // Overload | operator
86 friend PowerLevel operator|(PowerLevel first, PowerLevel second) // Combines two masks so the result holds every bit from both.
87 {
88 return static_cast<PowerLevel>(static_cast<uint8_t>(first) | static_cast<uint8_t>(second)); // Work on raw bytes, then wrap back into the enum type.
89 }

91 /** @brief Returns the bitwise AND of two masks (non-zero means a shared bit). */
92 // Overload & operator
93 friend PowerLevel operator&(PowerLevel first, PowerLevel second) // Keeps only the bits present in both masks (the overlap).
94 {
95 return static_cast<PowerLevel>(static_cast<uint8_t>(first) & static_cast<uint8_t>(second)); // A non-zero result means they share a right.
96 }

98 /** @brief Adds the bits of @p second to @p first in place. @return Reference to @p first. */
99 // Overload |= operator
100 friend PowerLevel& operator|=(PowerLevel& first, PowerLevel second)
101 {
102 first = first | second; // Add the other mask's bits to this one.
103 return first; // Return by reference so calls can be chained.
104 }

106 /** @brief Keeps only the bits shared by @p first and @p second. @return Reference to @p first. */
107 // Overload &= operator
108 friend PowerLevel& operator&=(PowerLevel& first, PowerLevel second)
109 {
110 first = first & second; // Keep only the bits the two masks share.
111 return first; // Return by reference so calls can be chained.
112 }

114 /** @brief Returns the one's-complement of the 8-bit mask. */
115 // Invert operator
116 friend PowerLevel operator~(PowerLevel value) { return static_cast<PowerLevel>(~static_cast<uint8_t>(value)); } // Flips every bit: on switches become off and vice versa.
117
118 /** @brief Converts to `true` when at least one power bit is set. */
119 // Check if any flag is set
120 explicit operator bool() const { return static_cast<int>(powerLevel) != 0; } // True when at least one switch is on; "explicit" avoids accidental use as a number.

122 /**

124 * @details `staff_id`/`power` are parsed from an `"<id>:<mask>"` string; `isMember`
125 * marks a successful parse.
126 */

132 };
