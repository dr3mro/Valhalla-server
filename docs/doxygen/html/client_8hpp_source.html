1/**
2 * @file client.hpp
3 * @brief Username column name used by Client::exists() and the client validation layer.
4 * @details This header defines the Client entity for "people" tables such as users and providers.
5 * It replaces the generic JSON-based create and update builders from Entity with versions that
6 * consume already-validated (column, value) pairs, so untrusted JSON never reaches SQL generation.
7 * It also builds the UPDATE that toggles the active flag (suspend/activate), and can check whether
8 * a username is already taken in the table.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * User and provider accounts need stricter handling than ordinary records: their input is validated
12 * and password-hashed before insertion, they can be suspended, and usernames must be unique.
13 * Controllers use Client (via User and Provider) to produce those specialised statements while
14 * reusing read/delete/search from Entity. Without it, account creation would bypass validation and
15 * there would be no safe way to suspend a client.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
19 * - C++ concept: a named compile-time requirement on a template type; here it restricts Client
20 * helpers to payload types that really carry client data or an id.
21 * - SQL parameter binding: put $1, $2, ... markers in the SQL and pass the real values separately;
22 * see https://en.cppreference.com/w/cpp or the libpqxx documentation.
23 * - "final" on a virtual function means subclasses may not override it again - a guard that keeps
24 * the validated create/update path from being bypassed.
25 *
26 * @author Amr Osman <dr3mro@gmail.com>
27 * @date 2026-11-16
28 */

69 /**
70 * @class Client
71 * @brief Entity for client-style tables (users, providers) with dedicated create/update builders.
72 * @details Replaces Entity's generic JSON-based create/update builders with versions that
73 * consume the validated (column, value) pairs held by CreateClient_t/UpdateClient_t. Read,
74 * delete and search remain inherited from Entity. It also builds the active-flag toggle
75 * used to suspend/activate a client and can test whether a username is already taken.
76 * Concrete tables (User, Provider) derive from this class and supply the table name.
77 * @note All values in generated statements are bound through Binder; column names come from
78 * the validated data_set. The id used by updates/suspends is cached in client_id.
79 * @see Entity, Types::CreateClient_t, Types::UpdateClient_t

126 /**
127 * @brief Builds the statement that sets the client's active flag to state.
128 * @details Reads the client id from the SuspendData payload into client_id, then binds
129 * the state and id in a fixed order before formatting the UPDATE.
130 * @param[in] state Desired active value: false suspends, true activates.
131 * @param[out] params Cleared and filled with the bound state and id.
132 * @return UPDATE ... RETURNING id,active, or std::nullopt if the payload is not
133 * SuspendData or another exception occurs.
134 */
135 std::optional<std::string> getSqlToggleSuspendStatement(bool state, pqxx::params& params) // One shared builder for both suspend and activate.
136 {
137 try
138 {
139 Types::SuspendData suspenddata = std::get<Types::SuspendData>(getData()); // The payload must be SuspendData, or std::get throws.
140 client_id = suspenddata.client_id; // Cache the id so callers can retrieve it with getClientId().
141 Binder binder(params); // Fresh binder for the two parameters.
142 // Bound in a fixed order (argument evaluation order is unspecified).
143 const std::string state_ph = binder.bind(state); // Bind the boolean first -> $1.
144 const std::string id_ph = binder.bind(client_id.value()); // Bind the id second -> $2.
145 return fmt::format("UPDATE {} SET active={} WHERE id={} RETURNING id,active;", tablename, state_ph, id_ph); // RETURNING confirms the new state.
146 }
147 catch (const std::exception& e) // Wrong payload type or a formatting problem.
148 {
149 Message::ErrorMessage(fmt::format("Failed to create Sql suspend statement for table {}.", tablename)); // Log the table.
150 Message::CriticalMessage(e.what()); // Log details.
151 return std::nullopt; // Signal failure.
152 }
153 }

175 /**
176 * @brief Checks the database for a client with the username in the validated payload.
177 * @tparam T A payload type exposing get_data_set() (CreateClient_t or UpdateClient_t).
178 * @return true only when a username entry exists and the database reports a match;
179 * false when there is no username entry or the lookup finds nothing.
180 * @note Logs nothing on a negative result; database failures collapse to false.
181 * @throws std::bad_variant_access if T is not the held alternative.
182 */
183 template <typename T>
184 bool exists() // Used to reject duplicate usernames before inserting.
185 {
186 auto client_data = std::get<T>(getData()).get_data_set(); // The validated column/value pairs from the payload.
187 auto iterator = std::ranges::find_if(client_data, [&](const auto& item) { return item.first == USERNAME; }); // Find the pair whose column is "username".
188
189 if (iterator != client_data.end()) // Only check when the payload actually contains a username.
190 {
191 std::string username = iterator->second; // The value part of the pair.
192 auto result = databaseController()->checkItemExists(tablename, USERNAME, username); // Ask the database whether that username already exists.
193 return result.value_or(false); // A failed/unset lookup counts as "does not exist".
194 }
195 return false; // No username in the payload means nothing to check.
196 }

204 /**
205 * @brief Builds the client INSERT from validated column/value pairs.
206 * @details Keys come from the validated data_set; values are bound. Returns only the
207 * generated id so callers can look the new row up separately.
208 * @tparam Binder Binder-like type providing bind().
209 * @param[in,out] binder Appends values and returns their placeholders.
210 * @return INSERT ... RETURNING id, or std::nullopt if the payload is not
211 * CreateClient_t or another exception occurs.
212 */
213 template <typename Binder>
214 std::optional<std::string> buildClientCreateStatement(Binder& binder) // Builds the INSERT for a validated account payload.
215 {
216 try
217 {
218 auto clientdata = std::get<Types::CreateClient_t>(getData()); // Must be a create payload; std::get throws otherwise.
219
220 std::vector<std::string> keys_arr; // Column names from the validated set.
221 std::vector<std::string> values_arr; // Matching placeholders in the same order.
222
223 for (const auto& item : clientdata.get_data_set()) // Walk each validated (column, value) pair.
224 {
225 keys_arr.push_back(item.first); // The column name is trusted because it came from validation.
226 values_arr.push_back(binder.bind(item.second)); // The value is bound safely as a parameter.
227 }
228
229 return fmt::format("INSERT INTO {} ({}) VALUES ({}) RETURNING id;", tablename, fmt::join(keys_arr, ","), fmt::join(values_arr, ",")); // RETURNING id gives back the new account id.
230 }
231 catch (const std::exception& e) // Wrong payload type or formatting error.
232 {
233 Message::ErrorMessage(fmt::format("Failed to create Sql create statement for table {}.", tablename)); // Log the table.
234 Message::CriticalMessage(e.what()); // Log details.
235 return std::nullopt; // Signal failure.
236 }
237 }

239 /**
240 * @brief Builds the client UPDATE from validated column/value pairs and the payload id.
241 * @details Caches the id in client_id and aborts when it is missing, so no unscoped
242 * UPDATE is emitted. Each value is bound; column names come from the validated data_set.
243 * @tparam Binder Binder-like type providing bind().
244 * @param[in,out] binder Appends SET values and the id, returning their placeholders.
245 * @return UPDATE ... RETURNING id, or std::nullopt if the id is missing, the payload is
246 * not UpdateClient_t, or another exception occurs.
247 */
248 template <typename Binder>
249 std::optional<std::string> buildClientUpdateStatement(Binder& binder) // Builds the UPDATE for a validated account payload.
250 {
251 try
252 {
253 auto clientdata = std::get<Types::UpdateClient_t>(getData()).get_data_set(); // Validated (column, value) pairs to set.
254 client_id = std::get<Types::UpdateClient_t>(getData()).get_id(); // Cache the target id for callers.
255 if (!client_id.has_value()) // Without an id the UPDATE would hit every row, so stop.
256 {
257 Message::ErrorMessage(fmt::format("Failed to update client data. No id provided.")); // Log the missing id.
258 return std::nullopt; // No statement returned.
259 }
260
261 std::vector<std::string> set_clauses; // One "column = $n" string per field.
262 for (auto it = clientdata.begin(); it != clientdata.end(); ++it) // Walk every validated pair.
263 {
264 set_clauses.push_back(fmt::format("{} = {}", it->first, binder.bind(it->second))); // Column name is trusted; value is bound.
265 }
266
267 return fmt::format("UPDATE {} SET {} WHERE id={} RETURNING id;", tablename, fmt::join(set_clauses, ", "), binder.bind(client_id.value())); // Bind the id last so its placeholder follows the SET values.
268 }
269 catch (const std::exception& e) // Wrong payload type or formatting error.
270 {
271 Message::ErrorMessage(fmt::format("Failed to create Sql update statement for table {}.", tablename)); // Log the table.
272 Message::CriticalMessage(e.what()); // Log details.
273 return std::nullopt; // Signal failure.
274 }
275 }
