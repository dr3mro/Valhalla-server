1/**
2 * @file client.hpp
3 * @brief Username column name used by Client::exists() and the client
4 * validation layer.
5 * @details This header defines the Client entity for "people" tables such as
6 * users and providers. It replaces the generic JSON-based create and update
7 * builders from Entity with versions that consume already-validated (column,
8 * value) pairs, so untrusted JSON never reaches SQL generation. It also builds
9 * the UPDATE that toggles the active flag (suspend/activate), and can check
10 * whether a username is already taken in the table.
11 *
12 * WHY THIS FILE MATTERS IN THE SERVER:
13 * User and provider accounts need stricter handling than ordinary records:
14 * their input is validated and password-hashed before insertion, they can be
15 * suspended, and usernames must be unique. Controllers use Client (via User and
16 * Provider) to produce those specialised statements while reusing
17 * read/delete/search from Entity. Without it, account creation would bypass
18 * validation and there would be no safe way to suspend a client.
19 *
20 * CONCEPTS TO LEARN (for beginners):
21 * - std::optional: https://en.cppreference.com/w/cpp/utility/optional
22 * - C++ concept: a named compile-time requirement on a template type; here it
23 * restricts Client helpers to payload types that really carry client data or an
24 * id.
25 * - SQL parameter binding: put $1, $2, ... markers in the SQL and pass the real
26 * values separately; see https://en.cppreference.com/w/cpp or the libpqxx
27 * documentation.
28 * - "final" on a virtual function means subclasses may not override it again -
29 * a guard that keeps the validated create/update path from being bypassed.
30 *
31 * @author Amr Osman <dr3mro@gmail.com>
32 * @date 2026-11-16
33 */

169 /**
170 * @brief Builds the statement that sets the client's active flag to
171 * state.
172 * @details Reads the client id from the SuspendData payload into
173 * client_id, then binds the state and id in a fixed order before
174 * formatting the UPDATE.
175 * @param[in] state Desired active value: false suspends, true
176 * activates.
177 * @param[out] params Cleared and filled with the bound state and
178 * id.
179 * @return UPDATE ... RETURNING id,active, or std::nullopt if the
180 * payload is not SuspendData or another exception occurs.
181 */
182 // One shared builder for both suspend
183 // and activate.
184 std::optional<std::string> getSqlToggleSuspendStatement(
185 bool state, pqxx::params& params) {
186 try {
187 Types::SuspendData suspenddata = std::get<Types::SuspendData>(
188 getData()); // The payload must be SuspendData, or
189 // std::get throws.
190 client_id = suspenddata.client_id; // Cache the id so callers can
191 // retrieve it with getClientId().
192 Binder binder(params); // Fresh binder for the two parameters.
193 // Bound in a fixed order (argument evaluation order is
194 // unspecified).
195 const std::string state_ph =
196 binder.bind(state); // Bind the boolean first -> $1.
197 const std::string id_ph =
198 binder.bind(client_id.value()); // Bind the id second -> $2.
199 return fmt::format(
200 "UPDATE {} SET active={} WHERE id={} RETURNING "
201 // RETURNING confirms the new state.
202 // Wrong payload type or a formatting problem.
203 "id,active;",
204 tablename,
205 state_ph,
206 id_ph);
207 } catch (const std::exception& e) {
208 Message::ErrorMessage(
209 fmt::
210 format("Failed to create Sql suspend statement for table {}.",
211 tablename)); // Log the table.
212 Message::CriticalMessage(e.what()); // Log details.
213 return std::nullopt; // Signal failure.
214 }
215 }

248 /**
249 * @brief Checks the database for a client with the username in the
250 * validated payload.
251 * @tparam T A payload type exposing get_data_set() (CreateClient_t
252 * or UpdateClient_t).
253 * @return true only when a username entry exists and the database
254 * reports a match; false when there is no username entry or the
255 * lookup finds nothing.
256 * @note Logs nothing on a negative result; database failures
257 * collapse to false.
258 * @throws std::bad_variant_access if T is not the held alternative.
259 */
260 // Used to reject duplicate usernames before inserting.
261 template <typename T>
262 bool exists() {
263 auto client_data = std::get<T>(getData())
264 .get_data_set(); // The validated column/value
265 // pairs from the payload.
266 auto iterator =
267 std::ranges::find_if(client_data, [&](const auto& item) {
268 return item.first == USERNAME;
269 }); // Find the pair whose column
270 // is "username".
271
272 // Only check when the payload
273 // actually contains a username.
274 if (iterator != client_data.end()) {
275 std::string username =
276 iterator->second; // The value part of the pair.
277 auto result =
278 databaseController()
279 ->checkItemExists(tablename,
280 USERNAME,
281 username); // Ask the database whether that
282 // username already exists.
283 return result.value_or(false); // A failed/unset lookup counts as
284 // "does not exist".
285 }
286 return false; // No username in the payload means nothing to
287 // check.
288 }

296 /**
297 * @brief Builds the client INSERT from validated column/value
298 * pairs.
299 * @details Keys come from the validated data_set; values are bound.
300 * Returns only the generated id so callers can look the new row up
301 * separately.
302 * @tparam Binder Binder-like type providing bind().
303 * @param[in,out] binder Appends values and returns their
304 * placeholders.
305 * @return INSERT ... RETURNING id, or std::nullopt if the payload
306 * is not CreateClient_t or another exception occurs.
307 */
308 // Builds the INSERT for a validated account
309 // payload.
310 template <typename Binder>
311 std::optional<std::string> buildClientCreateStatement(Binder& binder) {
312 try {
313 auto clientdata = std::get<Types::CreateClient_t>(
314 getData()); // Must be a create payload; std::get
315 // throws otherwise.
316
317 std::vector<std::string>
318 keys_arr; // Column names from the validated set.
319 std::vector<std::string>
320 values_arr; // Matching placeholders in the same order.
321
322 for (const auto& item :
323 // Walk each validated
324 // (column, value) pair.
325 clientdata.get_data_set()) {
326 keys_arr.push_back(item.first); // The column name is trusted
327 // because it came from validation.
328 values_arr.push_back(
329 binder.bind(item.second)); // The value is bound safely as a
330 // parameter.
331 }
332
333 // RETURNING id gives back the
334 // new account id.
335 // Wrong payload type or formatting error.
336 return fmt::format("INSERT INTO {} ({}) VALUES ({}) RETURNING id;",
337 tablename,
338 fmt::join(keys_arr, ","),
339 fmt::join(values_arr, ","));
340 } catch (const std::exception& e) {
341 Message::ErrorMessage(
342 fmt::format("Failed to create Sql create statement for table {}.",
343 tablename)); // Log the table.
344 Message::CriticalMessage(e.what()); // Log details.
345 return std::nullopt; // Signal failure.
346 }
347 }

349 /**
350 * @brief Builds the client UPDATE from validated column/value pairs
351 * and the payload id.
352 * @details Caches the id in client_id and aborts when it is
353 * missing, so no unscoped UPDATE is emitted. Each value is bound;
354 * column names come from the validated data_set.
355 * @tparam Binder Binder-like type providing bind().
356 * @param[in,out] binder Appends SET values and the id, returning
357 * their placeholders.
358 * @return UPDATE ... RETURNING id, or std::nullopt if the id is
359 * missing, the payload is not UpdateClient_t, or another exception
360 * occurs.
361 */
362 // Builds the UPDATE for a validated account
363 // payload.
364 template <typename Binder>
365 std::optional<std::string> buildClientUpdateStatement(Binder& binder) {
366 try {
367 auto clientdata = std::get<Types::UpdateClient_t>(getData())
368 .get_data_set(); // Validated (column, value)
369 // pairs to set.
370 client_id = std::get<Types::UpdateClient_t>(getData())
371 .get_id(); // Cache the target id for callers.
372 // Without an id the UPDATE would hit
373 // every row, so stop.
374 if (!client_id.has_value()) {
375 Message::ErrorMessage(
376 fmt::format("Failed to update client data. No id "
377 "provided.")); // Log the missing id.
378 return std::nullopt; // No statement returned.
379 }
380
381 std::vector<std::string>
382 set_clauses; // One "column = $n" string per field.
383 for (auto it = clientdata.begin(); it != clientdata.end();
384 // Walk every validated pair.
385 ++it) {
386 set_clauses.push_back(
387 fmt::format("{} = {}",
388 it->first,
389 binder.bind(
390 it->second))); // Column name is trusted; value
391 // is bound.
392 }
393
394 // Bind the id last so its
395 // placeholder follows the SET
396 // values.
397 // Wrong payload type or formatting error.
398 return fmt::format("UPDATE {} SET {} WHERE id={} RETURNING id;",
399 tablename,
400 fmt::join(set_clauses, ", "),
401 binder.bind(client_id.value()));
402 } catch (const std::exception& e) {
403 Message::ErrorMessage(
404 fmt::format("Failed to create Sql update statement for table {}.",
405 tablename)); // Log the table.
406 Message::CriticalMessage(e.what()); // Log details.
407 return std::nullopt; // Signal failure.
408 }
409 }
