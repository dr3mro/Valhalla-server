1/**
2 * @file server.cpp
3 * @brief Implementation of Server: Drogon configuration, request hooks, and
4 * shutdown sequencing.
5 *
6 * @details This file turns the Server declaration into working code. It builds
7 * the Drogon application — listener, thread count, log level, a custom 404 page
8 * — and registers two global hooks: one that answers CORS preflights and
9 * rejects unusual HTTP methods before routing, and one that adds CORS headers
10 * to every response. It also prints the startup banner and, after the event
11 * loops stop, drains the database and password worker pools in a safe order.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * This is the file that actually boots the web server. main() calls
15 * Server::run(), and the code here is what opens the listening port and starts
16 * Drogon's event loops so HTTP requests can be served. It also contains the two
17 * cross-cutting request rules (CORS and allowed methods) that apply to every
18 * endpoint, so individual controllers do not have to repeat them. Finally, it
19 * owns the shutdown order; without it, worker threads could be destroyed while
20 * they still had queued work, causing crashes at exit.
21 *
22 * CONCEPTS TO LEARN (for beginners):
23 * - Lambda expressions (the inline anonymous functions used as callbacks):
24 * https://en.cppreference.com/w/cpp/language/lambda
25 * - Move semantics and std::move:
26 * https://en.cppreference.com/w/cpp/utility/move
27 * - Drogon's application and advice hooks:
28 * https://github.com/drogonframework/drogon
29 *
30 * @author Amr Osman <dr3mro@gmail.com>
31 * @date 2026-11-16
32 */

136/**
137 * @brief Configures and runs the Drogon application, then drains the worker
138 * pools.
139 *
140 * @details Registers the listener, thread count, upload path, log level, a
141 * custom 404 page and two global advice hooks: the pre-routing hook
142 * short-circuits OPTIONS and rejects uncommon methods, and the pre-sending hook
143 * applies CORS headers to every response. When db_config_.async is set, an
144 * AsyncDbClient is registered in the Store before the event loops start.
145 * Control returns from drogon::app().run() only after the event loops stop.
146 * @return EXIT_SUCCESS on a clean run and shutdown, EXIT_FAILURE if setup or
147 * shutdown throws std::exception.
148 * @note The post-run shutdown order is deliberate: DbExecutor is drained before
149 * PasswordWorker because database tasks may enqueue password work. Both happen
150 * here, while the Store singletons still exist, rather than in static
151 * destructors.
152 * @see print_banner(), Cors::apply(), Store::getObject()
153 */
154// Configure Drogon, serve requests, then drain the worker
155// pools and return an exit code.
156int Server::run() {
157 // Catch any exception from setup or teardown so it can be reported
158 // cleanly.
159 try {
160 print_banner(); // Show the effective settings and the logo before the
161 // framework starts logging.
162 // Only asynchronous DB mode needs a separate
163 // client registered up front.
164 if (db_config_.async) {
165 Store::registerObject<AsyncDbClient>(
166 AsyncDbClient::createClient()); // Build an async DB client and
167 // put it in the Store for
168 // controllers to use.
169 }
170 // Get the single global Drogon application; the chain below
171 // configures it before run() is called.
172 // Listen on the configured host
173 // address and TCP port.
174 // Use the configured number of
175 // I/O worker threads.
176 // Tell Drogon where to
177 // store uploaded files.
178 // Let the event loops drive
179 // shutdown instead of
180 // Drogon's own SIGTERM
181 // handler.
182 // Convert the numeric config
183 // value into the framework's
184 // log-level enum.
185 // Install a hook that runs
186 // before routing: answer
187 // preflights and reject odd
188 // methods.
189 // Request + callbacks.
190 drogon::app()
191 .addListener(std::string(config_.host), config_.port)
192 .setThreadNum(config_.threads)
193 .setUploadPath(config_.upload_dir)
194 .disableSigtermHandling()
195 .setLogLevel(
196 static_cast<trantor::Logger::LogLevel>(config_.debug_level))
197 .registerPreRoutingAdvice(
198 [](const drogon::HttpRequestPtr& req,
199 drogon::AdviceCallback&& callback,
200 drogon::AdviceChainCallback&& chainedcallback) {
201 // CORS preflights are answered once, here, with
202 // 204 and no body; they never reach a
203 // controller.
204 // A browser's preflight
205 // check uses OPTIONS and
206 // expects a quick empty
207 // reply.
208 if (req->method() == drogon::Options) {
209 auto resp =
210 drogon::HttpResponse::newHttpResponse(); // Create a fresh,
211 // empty response
212 // object.
213 resp->setStatusCode(drogon::k204NoContent); // 204 means
214 // "success, no
215 // content",
216 // which is
217 // correct for a
218 // preflight.
219 std::move(callback)(resp); // Send this response immediately
220 // and do not continue the chain.
221 }
222 // Uncommon methods are rejected before routing;
223 // Allow advertises what is accepted.
224 // If the method
225 // is unsupported,
226 // stop before any
227 // route is
228 // matched.
229 else if (!isConventionalMethod(req->method())) {
230 auto resp =
231 drogon::HttpResponse::newHttpResponse(); // Create a
232 // response object
233 // to carry the
234 // error status.
235 resp->setStatusCode(
236 // 405 tells the client the method is not
237 // allowed.
238 drogon::k405MethodNotAllowed);
239 resp->addHeader("Allow",
240 Cors::allowedMethods()); // Advertise
241 // which
242 // methods the
243 // server does
244 // accept.
245 // Respond now; the request never
246 // reaches a controller.
247 // The method is acceptable, so let normal
248 // routing continue.
249 std::move(callback)(resp);
250 } else {
251 std::move(chainedcallback)(); // Call the chain
252 // callback to pass
253 // the request to
254 // the next step.
255 }
256 })
257 // The only place CORS headers are written: runs once per
258 // response, after every filter and pre-routing reply.
259 .registerPreSendingAdvice([](const drogon::HttpRequestPtr& req,
260 const drogon::HttpResponsePtr& resp) {
261 Cors::apply(req, resp);
262 }) // CORS headers.
263 .setCustom404Page(drogon::HttpResponse::newHttpJsonResponse(
264 api::v2::JsonHelper::jsonify(
265 "Endpoint not found."))) // JSON body for
266 // unknown routes.
267 .run(); // Start the event loops and block here until
268 // they stop; execution continues below
269 // afterwards.
270
271 // The event loops have stopped. Finish the queued database and password
272 // work and join the workers now, while the connection pool and the
273 // other singletons still exist, instead of leaving it to static
274 // destruction. The database executor goes first because its tasks may
275 // hand work to the password worker.
276 Store::getObject<DbExecutor>()
277 ->shutdown(); // Drain the DB task queue and stop its threads
278 // before the DB singletons disappear.
279 // Then drain password-hashing work, which DB tasks
280 // may have queued.
281 // Reached if any of the setup or teardown
282 // above threw an exception.
283 Store::getObject<PasswordWorker>()->shutdown();
284 } catch (const std::exception& e) {
285 Message::ErrorMessage(
286 fmt::format("Failed to start server on {}:{}.",
287 config_.host,
288 config_.port)); // Report which address failed to start.
289 Message::CriticalMessage(e.what()); // Include the exception's own
290 // description as a critical message.
291 return EXIT_FAILURE; // Signal failure to main(), and from there to the
292 // operating system.
293 }
294 return EXIT_SUCCESS; // The server ran and shut down cleanly.
295}

353/**
354 * Reports whether a method should be routed rather than rejected in
355 * pre-routing.
356 *
357 * method: Method of the incoming request.
358 * Returns true for GET, POST, PUT, DELETE, OPTIONS and HEAD; false for all
359 * others, notably PATCH. Note: PATCH is advertised by Cors::allowedMethods()
360 * but is not accepted here; the two lists differ in the current implementation.
361 */
362// Return true only for the six methods this
363// server is willing to route.
364bool Server::isConventionalMethod(drogon::HttpMethod method) {
365 return method == drogon::Get || method == drogon::Post ||
366 method == drogon::Put || method == drogon::Delete ||
367 method == drogon::Options || // Accepted.
368 method ==
369 drogon::Head; // Also accept HEAD (like GET but without a body);
370 // everything else, such as PATCH, returns false.
371}
