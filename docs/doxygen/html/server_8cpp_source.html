1/**
2 * @file server.cpp
3 * @brief Implementation of Server: Drogon configuration, request hooks, and shutdown sequencing.
4 *
5 * @details This file turns the Server declaration into working code. It builds the Drogon application — listener,
6 * thread count, log level, a custom 404 page — and registers two global hooks: one that answers CORS
7 * preflights and rejects unusual HTTP methods before routing, and one that adds CORS headers to every
8 * response. It also prints the startup banner and, after the event loops stop, drains the database and
9 * password worker pools in a safe order.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * This is the file that actually boots the web server. main() calls Server::run(), and the code here is what opens the
13 * listening port and starts Drogon's event loops so HTTP requests can be served. It also contains the two cross-cutting
14 * request rules (CORS and allowed methods) that apply to every endpoint, so individual controllers do not have to
15 * repeat them. Finally, it owns the shutdown order; without it, worker threads could be destroyed while they still had
16 * queued work, causing crashes at exit.
17 *
18 * CONCEPTS TO LEARN (for beginners):
19 * - Lambda expressions (the inline anonymous functions used as callbacks): https://en.cppreference.com/w/cpp/language/lambda
20 * - Move semantics and std::move: https://en.cppreference.com/w/cpp/utility/move
21 * - Drogon's application and advice hooks: https://github.com/drogonframework/drogon
22 *
23 * @author Amr Osman <dr3mro@gmail.com>
24 * @date 2026-11-16
25 */

79/**
80 * @brief Configures and runs the Drogon application, then drains the worker pools.
81 *
82 * @details Registers the listener, thread count, upload path, log level, a custom 404 page and two global advice hooks:
83 * the pre-routing hook short-circuits OPTIONS and rejects uncommon methods, and the pre-sending hook applies
84 * CORS headers to every response. When db_config_.async is set, an AsyncDbClient is registered in the Store
85 * before the event loops start. Control returns from drogon::app().run() only after the event loops stop.
86 * @return EXIT_SUCCESS on a clean run and shutdown, EXIT_FAILURE if setup or shutdown throws std::exception.
87 * @note The post-run shutdown order is deliberate: DbExecutor is drained before PasswordWorker because database tasks
88 * may enqueue password work. Both happen here, while the Store singletons still exist, rather than in static
89 * destructors.
90 * @see print_banner(), Cors::apply(), Store::getObject()
91 */
92int Server::run() // Configure Drogon, serve requests, then drain the worker pools and return an exit code.
93{
94 try // Catch any exception from setup or teardown so it can be reported cleanly.
95 {
96 print_banner(); // Show the effective settings and the logo before the framework starts logging.
97 if (db_config_.async) // Only asynchronous DB mode needs a separate client registered up front.
98 {
99 Store::registerObject<AsyncDbClient>(AsyncDbClient::createClient()); // Build an async DB client and put it in the Store for controllers to use.
100 }
101 drogon::app() // Get the single global Drogon application; the chain below configures it before run() is called.
102 .addListener(std::string(config_.host), config_.port) // Listen on the configured host address and TCP port.
103 .setThreadNum(config_.threads) // Use the configured number of I/O worker threads.
104 .setUploadPath(config_.upload_dir) // Tell Drogon where to store uploaded files.
105 .disableSigtermHandling() // Let the event loops drive shutdown instead of Drogon's own SIGTERM handler.
106 .setLogLevel(static_cast<trantor::Logger::LogLevel>(config_.debug_level)) // Convert the numeric config value into the framework's log-level enum.
107 .registerPreRoutingAdvice( // Install a hook that runs before routing: answer preflights and reject odd methods.
108 [](const drogon::HttpRequestPtr& req, drogon::AdviceCallback&& callback, drogon::AdviceChainCallback&& chainedcallback) // Request + callbacks.
109 {
110 // CORS preflights are answered once, here, with 204 and no body; they never reach a controller.
111 if (req->method() == drogon::Options) // A browser's preflight check uses OPTIONS and expects a quick empty reply.
112 {
113 auto resp = drogon::HttpResponse::newHttpResponse(); // Create a fresh, empty response object.
114 resp->setStatusCode(drogon::k204NoContent); // 204 means "success, no content", which is correct for a preflight.
115 std::move(callback)(resp); // Send this response immediately and do not continue the chain.
116 }
117 // Uncommon methods are rejected before routing; Allow advertises what is accepted.
118 else if (!isConventionalMethod(req->method())) // If the method is unsupported, stop before any route is matched.
119 {
120 auto resp = drogon::HttpResponse::newHttpResponse(); // Create a response object to carry the error status.
121 resp->setStatusCode(drogon::k405MethodNotAllowed); // 405 tells the client the method is not allowed.
122 resp->addHeader("Allow", Cors::allowedMethods()); // Advertise which methods the server does accept.
123 std::move(callback)(resp); // Respond now; the request never reaches a controller.
124 }
125 else // The method is acceptable, so let normal routing continue.
126 {
127 std::move(chainedcallback)(); // Call the chain callback to pass the request to the next step.
128 }
129 })
130 // The only place CORS headers are written: runs once per response, after every filter and pre-routing reply.
131 .registerPreSendingAdvice([](const drogon::HttpRequestPtr& req, const drogon::HttpResponsePtr& resp) { Cors::apply(req, resp); }) // CORS headers.
132 .setCustom404Page(drogon::HttpResponse::newHttpJsonResponse(api::v2::JsonHelper::jsonify("Endpoint not found."))) // JSON body for unknown routes.
133 .run(); // Start the event loops and block here until they stop; execution continues below afterwards.
134
135 // The event loops have stopped. Finish the queued database and password work and join the workers now, while the
136 // connection pool and the other singletons still exist, instead of leaving it to static destruction. The database
137 // executor goes first because its tasks may hand work to the password worker.
138 Store::getObject<DbExecutor>()->shutdown(); // Drain the DB task queue and stop its threads before the DB singletons disappear.
139 Store::getObject<PasswordWorker>()->shutdown(); // Then drain password-hashing work, which DB tasks may have queued.
140 }
141 catch (const std::exception& e) // Reached if any of the setup or teardown above threw an exception.
142 {
143 Message::ErrorMessage(fmt::format("Failed to start server on {}:{}.", config_.host, config_.port)); // Report which address failed to start.
144 Message::CriticalMessage(e.what()); // Include the exception's own description as a critical message.
145 return EXIT_FAILURE; // Signal failure to main(), and from there to the operating system.
146 }
147 return EXIT_SUCCESS; // The server ran and shut down cleanly.
148}
