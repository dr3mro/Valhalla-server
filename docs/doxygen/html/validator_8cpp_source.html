45/**
46 * See Validator::validateDatabaseCreateSchema.
47 * Processing order: reject null/empty input, resolve the table schema, validate
48 * present columns' types and missing non-nullable columns, then enforce per-key rules.
49 * The duplicated-key check is intentionally disabled because jsoncons parsing already
50 * rejects duplicate keys upstream.
51 */
52bool Validator::validateDatabaseCreateSchema(const std::string& tablename, const jsoncons::json& data, api::v2::Http::Error& error, const Rule& rule)
53{
54 try // Guard the whole body because JSON access can throw on a badly shaped body.
55 {
56 if (nullCheck(data, error)) // Stop early when the client sent null or an empty object.
57 {
58 return false; // nullCheck already filled the error, so just report failure.
59 }
60 // [ ] I found this is never called as json parser already checks for
61 // duplicate keys if (hasDuplicateKeys(data, error))
62 // {
63 // error.message = fmt::format("Duplicate key found: {}",
64 // error.message); return false;
65 // }
66
67 // Get the schema for the table
68 bool found = false; // Set by the lookup helper to say whether the table's schema exists.
69 auto table_schema = getDatabaseSchemaForTable(tablename, error, found); // Fetch the cached columns and update found/error.
70
71 if (!found) // The table has no usable cached schema, so there is nothing to validate against.
72 {
73 return false; // getDatabaseSchemaForTable already set the error; abort.
74 }
75
76 // now lets go through the data and check if the keys are in the schema
77 // check for type and nullable
78
79 if (!checkColumns(data, table_schema, error, rule)) // Check every column's type and required-ness.
80 {
81 return false; // A column failed, so stop before doing more work.
82 }
83
84 // Ensure all keys in the schema are present in the data
85 if (!ensureAllKeysExist(data, table_schema, error, rule)) // Check every supplied key is allowed and known.
86 {
87 return false; // A key was prohibited or unknown; error is already filled.
88 }
89 return true; // All checks passed, so tell the caller the body is valid.
90 }
91 catch (const std::exception& e) // JSON helpers can throw (e.g. "Not an object" for a non-object body).
92 {
93 // Everything in the try block above only ever touches the client-supplied `data` and the
94 // static table schema (no DB/network I/O), so any exception reaching here (jsoncons
95 // throwing "Not an object" for a non-object body, e.g.) is a malformed-input problem on
96 // the client's side, not a server fault - report it as 400, not 500.
97 error.message = e.what(); // Copy the exception text so the client learns what was wrong.
98 error.code = api::v2::Http::Status::BAD_REQUEST; // It is a client mistake, so use HTTP 400.
99 return false; // Report failure to the caller.
100 }
101
102 // Unreachable: every path in the try block and the catch returns; kept defensively.
103 return true; // Satisfies the compiler's need for a return after the try/catch.
104}

156/**
157 * See Validator::validateDatabaseReadSchema.
158 * Iterates the requested keys once, first applying ASSERT_NOT_PRESENT then checking
159 * membership in the cached schema. A key that is both prohibited and unknown is reported as
160 * prohibited; a missing table is reported as 404 rather than 400.
161 */
162bool Validator::validateDatabaseReadSchema(
163 const std::unordered_set<std::string>& keys, const std::string& table_name, api::v2::Http::Error& error, const Rule& rule)
164{
165 bool found = false; // Set by the schema lookup to indicate success or failure.
166
167 // Retrieve the schema for the given table
168 auto columns = getDatabaseSchemaForTable(table_name, error, found); // Get the cached columns for this table.
169 if (!found) // The table has no cached schema, so requested columns cannot be checked.
170 {
171 error.message = "Table not found"; // Tell the caller which problem occurred.
172 error.code = api::v2::Http::Status::NOT_FOUND; // A missing table is a 404 here (unlike the create path).
173 return false; // Stop with the error filled in.
174 }
175
176 // Check if all keys exist in the schema
177 for (const auto& key : keys) // Walk each column name the client asked to read.
178 {
179 if (rule.check(Rule::Action::ASSERT_NOT_PRESENT, key)) // The rule forbids this key from being requested.
180 {
181 error.message = "Key not allowed: " + key; // Name the prohibited key in the message.
182 error.code = api::v2::Http::Status::BAD_REQUEST; // A forbidden key is a bad request.
183 return false; // Stop at the first prohibited key.
184 }
185 // Use std::ranges::find_if to check if a column with the given name
186 // exists
187 auto col = std::ranges::find_if(columns, [&](const api::v2::ColumnInfo& column) { return column.Name == key; }); // Find a column whose name matches the requested key.
188
189 if (col == columns.end()) // No matching column was found in the schema.
190 {
191 error.message = "Key not found in schema: " + key; // Report the unknown key.
192 error.code = api::v2::Http::Status::BAD_REQUEST; // Asking for a non-existent column is a bad request.
193 return false; // Stop at the first unknown key.
194 }
195 }
196
197 return true; // Every requested key was allowed and present in the schema.
198}
199/**
200 * See Validator::clientRegexValidation.
201 * Fields with a registered regex must be non-empty strings matching that pattern;
202 * unrecognised fields are accepted only if they are non-empty strings. The first offending
203 * field stops processing and fills @p error. Accepted pairs are inserted into @p db_data.
204 */
205bool Validator::clientRegexValidation(const jsoncons::json& data, api::v2::Http::Error& error, std::unordered_set<std::pair<std::string, std::string>>& db_data)
206{
207 static const std::unordered_map<std::string, std::string> regex_client_validators = { // Map field names to regex text; static so it is built only once.
208 {"username", "^[a-z][a-z0-9_]*$"}, // Starts with a lowercase letter, then letters/digits/underscores.
209 {"password",
210 "^(?=.*[a-z])(?=.*[A-Z])(?=.*\\d)(?=.*[!@#$%^&*])[A-Za-z\\d!@#$%^&*]{"
211 "8,}$"}, // At least 8 chars with a lowercase, uppercase, digit and special character.
212 {"phone", R"(^\+?(\d{1,3})?[-.\s]?(\‍(?\d{3}\)?)?[-.\s]?\d{3}[-.\s]?\d{4}$)"}, // Optional country code and separators around a phone number.
213 {"email", R"((\w+)(\.\w+)*@(\w+)(\.\w+)+)"}, // Something@something.something style address.
214 {"dob", R"(^(0[1-9]|[12]\d|3[01])-(0[1-9]|1[0-2])-\d{4}$)"}, // Date in DD-MM-YYYY form.
215 {"gender", "^(Male|Female)$"}, // Exactly "Male" or "Female".
216 };
217
218 // Compiling a std::regex is expensive, so it happens once, not for every request and field.
219 static const std::unordered_map<std::string, std::regex> compiled_validators = [] // Immediate lambda builds the compiled map once at first use.
220 {
221 std::unordered_map<std::string, std::regex> compiled; // Empty map that will hold field name -> compiled regex.
222 for (const auto& [name, pattern] : regex_client_validators) // Visit every (field name, pattern text) pair above.
223 {
224 compiled.emplace(name, std::regex(pattern)); // Compile the text into a regex and store it under the field name.
225 }
226 return compiled; // Hand the finished map back to initialise compiled_validators.
227 }();
228
229 for (const auto& item : data.object_range()) // Walk every key/value pair in the client's JSON object.
230 {
231 auto pattern_item = compiled_validators.find(std::string(item.key())); // Look for a regex registered for this field name.
232
233 // Keys without a dedicated regex are only kept when the value is a non-empty string;
234 // other JSON types are silently dropped here rather than passed through.
235 if (pattern_item == compiled_validators.end()) // No regex rule exists for this field name.
236 {
237 if (item.value().is_string()) // Only plain strings are accepted for unvalidated fields.
238 {
239 std::optional<std::string> v = item.value().as<std::string>(); // Try to read the value as a string.
240 if (v.has_value() && !v->empty()) // Keep it only if conversion worked and the text is not empty.
241 {
242 db_data.insert({std::string(item.key()), *v}); // Store the (key, value) pair for the database call.
243 }
244 }
245 continue; // Move on to the next field; no regex check applies here.
246 }
247
248 if (!item.value().is_string()) // A field with a regex must be a string, so reject other types.
249 {
250 error.message = "Value must be a string for key " + std::string(item.key()); // Say which field had the wrong type.
251 error.code = api::v2::Http::Status::BAD_REQUEST; // Wrong type is a client error.
252 return false; // Stop at the first invalid field.
253 }
254 std::optional<std::string> value = item.value().as<std::string>(); // Read the string value now that the type is known.
255 if (value.has_value() && !value->empty()) // Only non-empty strings can be pattern matched.
256 {
257 if (!std::regex_match(value.value(), pattern_item->second)) // The whole value must match the compiled pattern.
258 {
259 error = {.code = api::v2::Http::Status::BAD_REQUEST, .message = fmt::format("Key ({}) value is invalid.", item.key())}; // Fill code and message in one go.
260 return false; // Stop at the first field that fails its pattern.
261 }
262 db_data.insert({item.key(), value.value()}); // The value passed, so store it for the database call.
263 }
264 else // The value is missing or an empty string.
265 {
266 error.message = "Value is empty for key " + item.key(); // Tell the caller which field was empty.
267 error.code = api::v2::Http::Status::BAD_REQUEST; // Empty required value is a client error.
268 return false; // Stop at the first empty field.
269 }
270 }
271 return true; // Every field passed its check (or was safely ignored).
272}

275/**
276 * See Validator::hasDuplicateKeys.
277 * Uses a temporary set of key names; the first repeated name fills @p error and
278 * returns true. Kept for defensive use even though the JSON parser rejects duplicates first.

280bool Validator::hasDuplicateKeys(const jsoncons::json& data, api::v2::Http::Error& error)
281{
282 std::unordered_set<std::string> keys; // Remember key names we have already seen; a set keeps them unique.
283
284 for (const auto& item : data.object_range()) // Look at each key/value pair in the object.
285 {
286 if (!keys.insert(item.key()).second) // insert returns second=false when the key was already present (a duplicate).
287 {
288 error.message = item.key(); // Report the name of the duplicated key.
289 error.code = api::v2::Http::Status::BAD_REQUEST; // Duplicate keys are a client mistake.
290 return true; // Tell the caller a duplicate was found.
291 }
292 }
293 return false; // No key repeated, so the object is fine.
294}

310/**
311 * See Validator::getDatabaseSchemaForTable.
312 * Reads the process-wide schema snapshot via DatabaseSchema::getDatabaseSchema().
313 * An absent table or an entry with no columns is treated as "not found" and sets @p found
314 * to false; the returned set is a copy so callers cannot mutate the cached schema.
315 */
316std::unordered_set<api::v2::ColumnInfo> Validator::getDatabaseSchemaForTable(const std::string& tablename, api::v2::Http::Error& error, bool& found)
317{
318 SCHEMA_t schema = DatabaseSchema::getDatabaseSchema(); // Get a copy of the whole cached schema (table -> columns).
319
320 // Ensure the table schema exists
321 auto table_schema_it = schema.find(tablename); // Look for this table's entry in the schema map.
322 if (table_schema_it == schema.end() || table_schema_it->second.empty()) // Missing entry or an entry with no columns is unusable.
323 {
324 error.message = "Table not found"; // Report that we cannot validate against this table.
325 error.code = api::v2::Http::Status::BAD_REQUEST; // Treat the unknown table as a bad request here.
326 found = false; // Tell the caller the lookup failed.
327 return {}; // Return an empty set so the caller has nothing to iterate.
328 }
329 found = true; // The schema was found and has columns.
330 return table_schema_it->second; // Hand back a copy of this table's column set.
331}
332/**
333 * See Validator::validateType.
334 * "character varying" and all date/time spellings expect JSON strings; "jsonb"
335 * accepts either an object or an array. Date strings are not parsed here, only type-checked.
336 * Unknown schema type strings return false, which surfaces as a type mismatch to the caller.
337 */
338bool Validator::validateType(const jsoncons::json& value, const std::string& expectedType)
339{
340 if (expectedType == "integer") // The schema expects a whole number.
341 {
342 return value.is_int64(); // True only when the JSON value is a 64-bit integer.
343 }
344 if (expectedType == "float" || expectedType == "double" || expectedType == "real") // All three mean a decimal number in PostgreSQL.
345 {
346 return value.is_double(); // True only when the JSON value is a floating-point number.
347 }
348 if (expectedType == "character varying") // PostgreSQL's name for variable-length text.
349 {
350 return value.is_string(); // True only when the JSON value is a string.
351 }
352 if (expectedType == "boolean") // A true/false column.
353 {
354 return value.is_bool(); // True only when the JSON value is a boolean.
355 }
356 if (expectedType == "jsonb") // A column that stores JSON itself.
357 {
358 return (value.is_object() || value.is_array()); // Accept either a JSON object or a JSON array.
359 }
360 if (expectedType == "timestamp with time zone" || expectedType == "time without time zone" || expectedType == "date") // All date/time spellings.
361 {
362 return value.is_string(); // Assuming dates are strings
363 }
364 return false; // Unknown type
365}

367/**
368 * See Validator::checkColumns.
369 * Iterates schema columns (the authoritative source), not the body. A column that is
370 * absent is allowed when it is nullable or covered by IGNORE_IF_NOT_NULLABLE_IN_SCHEMA;
371 * a present column must match the schema's type according to validateType. The first
372 * violation aborts with HTTP 400.
373 */
374bool Validator::checkColumns(
375 const jsoncons::json& data, const std::unordered_set<api::v2::ColumnInfo>& table_schema, api::v2::Http::Error& error, const Rule& rule)
376{
377 for (const auto& column : table_schema) // Walk the schema's columns, which are the source of truth.
378 {
379 bool column_exists = data.contains(column.Name); // Whether the client supplied this column at all.
380
381 // Check for missing non-nullable columns
382 if (!column_exists && !column.isNullable && !rule.check(Rule::Action::IGNORE_IF_NOT_NULLABLE_IN_SCHEMA, column.Name)) // Missing, required, and not excused by the rule.
383 {
384 error.message = "Non-nullable column missing in data: " + column.Name; // Name the missing required column.
385 error.code = api::v2::Http::Status::BAD_REQUEST; // A missing required field is a client error.
386 return false; // Stop at the first missing required column.
387 }
388
389 // Validate data type if column exists
390 if (column_exists) // Only check the type when the client actually sent this column.
391 {
392 const auto& value = data.at(column.Name); // Fetch the value for this column from the JSON body.
393 if (!validateType(value, column.DataType)) // Compare the value's JSON kind to the schema's declared type.
394 {
395 error.message = fmt::format("Data type mismatch for column: {} ,expected: {} ", column.Name, column.DataType); // Build a helpful mismatch message.
396 error.code = api::v2::Http::Status::BAD_REQUEST; // A wrong type is a client error.
397 return false; // Stop at the first type mismatch.
398 }
399 }
400 }
401 return true; // Every column constraint was satisfied.
402}
403/**
404 * See Validator::ensureAllKeysExist.
405 * Builds a name set from the schema for O(1) membership tests, then scans the body.
406 * ASSERT_NOT_PRESENT and ASSERT_IMMUTABLE are checked first so prohibited/immutable keys are
407 * reported specifically; unknown keys are allowed only under IGNORE_IF_MISSING_FROM_SCHEMA.
408 */
409bool Validator::ensureAllKeysExist(
410 const jsoncons::json& data, const std::unordered_set<api::v2::ColumnInfo>& table_schema, api::v2::Http::Error& error, const Rule& rule)
411{
412 // Create a set of column names for fast key lookup
413 std::unordered_set<std::string> schema_columns; // Will hold just the names, for quick membership checks.
414 for (const auto& column : table_schema) // Walk the schema's column descriptions once.
415 {
416 schema_columns.insert(column.Name); // Add each column name to the fast-lookup set.
417 }
418
419 // Ensure all keys in `data` exist in the schema
420 for (const auto& entry : data.object_range()) // Visit every key the client actually supplied.
421 {
422 const auto& key = entry.key(); // The current field name being checked.
423
424 // Check for missing not allowed keys
425 if (rule.check(Rule::Action::ASSERT_NOT_PRESENT, key)) // The rule says this key must never appear.
426 {
427 error.message = "A prohibited key was found: [" + key + "]"; // Name the forbidden key.
428 error.code = api::v2::Http::Status::BAD_REQUEST; // Including it is a client error.
429 return false; // Stop at the first prohibited key.
430 }
431 // check if the key is in the exclude list
432 if (rule.check(Rule::Action::ASSERT_IMMUTABLE, key)) // The rule says this key cannot be changed.
433 {
434 error.message = "Key: [" + key + "] is not allowed to be changed"; // Explain that the key is immutable.
435 error.code = api::v2::Http::Status::BAD_REQUEST; // Trying to change it is a client error.
436 return false; // Stop at the first immutable key.
437 }
438
439 bool ignore_if_missing_from_database_schema = rule.check(Rule::Action::IGNORE_IF_MISSING_FROM_SCHEMA, key); // Whether the rule excuses keys absent from the schema.
440 if (schema_columns.find(key) == schema_columns.end() && !ignore_if_missing_from_database_schema) // Key not in schema
441 {
442 error.message = "Key: [" + key + "] is not found in database schema"; // Report the unknown key.
443 error.code = api::v2::Http::Status::BAD_REQUEST; // An unknown key is a client error.
444 return false; // Stop at the first unknown key.
445 }
446 }
447 return true; // All supplied keys were permitted and known to the schema.
448}
