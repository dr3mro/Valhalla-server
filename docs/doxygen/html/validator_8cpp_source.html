59/**
60 * See Validator::validateDatabaseCreateSchema.
61 * Processing order: reject null/empty input, resolve the table schema, validate
62 * present columns' types and missing non-nullable columns, then enforce per-key
63 * rules. The duplicated-key check is intentionally disabled because jsoncons
64 * parsing already rejects duplicate keys upstream.
65 */
66bool Validator::validateDatabaseCreateSchema(const std::string& tablename,
67 const jsoncons::json& data,
68 api::v2::Http::Error& error,
69 const Rule& rule) {
70 // Guard the whole body because JSON access can throw on a badly shaped
71 // body.
72 try {
73 // Stop early when the client sent null or an empty
74 // object.
75 if (nullCheck(data, error)) {
76 return false; // nullCheck already filled the error, so just report
77 // failure.
78 }
79 // [ ] I found this is never called as json parser already checks for
80 // duplicate keys if (hasDuplicateKeys(data, error))
81 // {
82 // error.message = fmt::format("Duplicate key found: {}",
83 // error.message); return false;
84 // }
85
86 // Get the schema for the table
87 bool found = false; // Set by the lookup helper to say whether the
88 // table's schema exists.
89 auto table_schema =
90 getDatabaseSchemaForTable(tablename,
91 error,
92 found); // Fetch the cached columns and
93 // update found/error.
94
95 // The table has no usable cached schema, so there is
96 // nothing to validate against.
97 if (!found) {
98 return false; // getDatabaseSchemaForTable already set the error;
99 // abort.
100 }
101
102 // now lets go through the data and check if the keys are in the schema
103 // check for type and nullable
104
105 // Check every column's type and required-ness.
106 if (!checkColumns(data, table_schema, error, rule)) {
107 return false; // A column failed, so stop before doing more work.
108 }
109
110 // Ensure all keys in the schema are present in the data
111 // Check every supplied key is allowed and
112 // known.
113 if (!ensureAllKeysExist(data, table_schema, error, rule)) {
114 return false; // A key was prohibited or unknown; error is already
115 // filled.
116 }
117 // All checks passed, so tell the caller the body is
118 // valid.
119 // JSON helpers can throw (e.g. "Not an
120 // object" for a non-object body).
121 return true;
122 } catch (const std::exception& e) {
123 // Everything in the try block above only ever touches the
124 // client-supplied `data` and the static table schema (no DB/network
125 // I/O), so any exception reaching here (jsoncons throwing "Not an
126 // object" for a non-object body, e.g.) is a malformed-input problem on
127 // the client's side, not a server fault - report it as 400, not 500.
128 error.message = e.what(); // Copy the exception text so the client
129 // learns what was wrong.
130 error.code = api::v2::Http::Status::BAD_REQUEST; // It is a client mistake,
131 // so use HTTP 400.
132 return false; // Report failure to the caller.
133 }
134
135 // Unreachable: every path in the try block and the catch returns; kept
136 // defensively.
137 return true; // Satisfies the compiler's need for a return after the
138 // try/catch.
139}

204/**
205 * See Validator::validateDatabaseReadSchema.
206 * Iterates the requested keys once, first applying ASSERT_NOT_PRESENT then
207 * checking membership in the cached schema. A key that is both prohibited and
208 * unknown is reported as prohibited; a missing table is reported as 404 rather
209 * than 400.
210 */
211bool Validator::validateDatabaseReadSchema(
212 const std::unordered_set<std::string>& keys,
213 const std::string& table_name,
214 api::v2::Http::Error& error,
215 const Rule& rule) {
216 bool found =
217 false; // Set by the schema lookup to indicate success or failure.
218
219 // Retrieve the schema for the given table
220 auto columns = getDatabaseSchemaForTable(table_name,
221 error,
222 found); // Get the cached columns
223 // for this table.
224 // The table has no cached schema, so requested columns cannot
225 // be checked.
226 if (!found) {
227 error.message =
228 "Table not found"; // Tell the caller which problem occurred.
229 error.code =
230 api::v2::Http::Status::NOT_FOUND; // A missing table is a 404 here
231 // (unlike the create path).
232 return false; // Stop with the error filled in.
233 }
234
235 // Check if all keys exist in the schema
236 for (const auto& key :
237 // Walk each column name the client asked to read.
238 keys) {
239 // The rule forbids this key from being requested.
240 if (rule.check(Rule::Action::ASSERT_NOT_PRESENT, key)) {
241 error.message =
242 "Key not allowed: " + key; // Name the prohibited key in the message.
243 error.code = api::v2::Http::Status::BAD_REQUEST; // A forbidden key is a
244 // bad request.
245 return false; // Stop at the first prohibited key.
246 }
247 // Use std::ranges::find_if to check if a column with the given name
248 // exists
249 auto col =
250 std::ranges::find_if(columns, [&](const api::v2::ColumnInfo& column) {
251 return column.Name == key;
252 }); // Find a column whose name matches the
253 // requested key.
254
255 // No matching column was found in the schema.
256 if (col == columns.end()) {
257 error.message =
258 "Key not found in schema: " + key; // Report the unknown key.
259 error.code = api::v2::Http::Status::BAD_REQUEST; // Asking for a
260 // non-existent column
261 // is a bad request.
262 return false; // Stop at the first unknown key.
263 }
264 }
265
266 return true; // Every requested key was allowed and present in the schema.

268/**
269 * See Validator::clientRegexValidation.
270 * Fields with a registered regex must be non-empty strings matching that
271 * pattern; unrecognised fields are accepted only if they are non-empty strings.
272 * The first offending field stops processing and fills @p error. Accepted pairs
273 * are inserted into @p db_data.
274 */
275bool Validator::clientRegexValidation(
276 const jsoncons::json& data,
277 api::v2::Http::Error& error,
278 std::unordered_set<std::pair<std::string, std::string>>& db_data) {
279 static const std::unordered_map<std::string, std::string>
280 regex_client_validators = {
281 // Map field names to regex text; static so it is built only once.
282 {"username",
283 "^[a-z][a-z0-9_]*$"}, // Starts with a lowercase letter, then
284 // letters/digits/underscores.
285 {"password",
286 "^(?=.*[a-z])(?=.*[A-Z])(?=.*\\d)(?=.*[!@#$%^&*])[A-Za-z\\d!@#$"
287 "%^&*"
288 "]{"
289 "8,}$"}, // At least 8 chars with a lowercase, uppercase, digit
290 // and special character.
291 {"phone",
292 // Optional country code and separators around a phone number.
293 R"(^\+?(\d{1,3})?[-.\s]?(\‍(?\d{3}\)?)?[-.\s]?\d{3}[-.\s]?\d{4}$)"},
294 // Something@something.something style address.
295 {"email", R"((\w+)(\.\w+)*@(\w+)(\.\w+)+)"},
296 {"dob",
297 // Date in DD-MM-YYYY form.
298 R"(^(0[1-9]|[12]\d|3[01])-(0[1-9]|1[0-2])-\d{4}$)"},
299 {"gender", "^(Male|Female)$"}, // Exactly "Male" or "Female".
300 };
301
302 // Compiling a std::regex is expensive, so it happens once, not for every
303 // request and field.
304 // Immediate lambda builds the compiled map once at first use.
305 static const std::unordered_map<std::string, std::regex> compiled_validators =
306 [] {
307 std::unordered_map<std::string, std::regex>
308 compiled; // Empty map that will hold field name -> compiled regex.
309 for (const auto& [name, pattern] :
310 // Visit every (field name, pattern text)
311 // pair above.
312 regex_client_validators) {
313 compiled.emplace(name,
314 std::regex(
315 pattern)); // Compile the text into a regex and
316 // store it under the field name.
317 }
318 return compiled; // Hand the finished map back to initialise
319 // compiled_validators.
320 }();
321
322 // Walk every key/value pair in
323 // the client's JSON object.
324 for (const auto& item : data.object_range()) {
325 auto pattern_item = compiled_validators.find(std::string(
326 item.key())); // Look for a regex registered for this field name.
327
328 // Keys without a dedicated regex are only kept when the value is a
329 // non-empty string; other JSON types are silently dropped here rather
330 // than passed through.
331 // No regex rule exists for this field name.
332 if (pattern_item == compiled_validators.end()) {
333 // Only plain strings are accepted
334 // for unvalidated fields.
335 if (item.value().is_string()) {
336 std::optional<std::string> v =
337 item.value().as<std::string>(); // Try to read the value as
338 // a string.
339 // Keep it only if conversion worked
340 // and the text is not empty.
341 if (v.has_value() && !v->empty()) {
342 db_data.insert(
343 {std::string(item.key()), *v}); // Store the (key, value) pair
344 // for the database call.
345 }
346 }
347 continue; // Move on to the next field; no regex check applies
348 // here.
349 }
350
351 // A field with a regex must be a
352 // string, so reject other types.
353 if (!item.value().is_string()) {
354 error.message =
355 "Value must be a string for key " +
356 std::string(item.key()); // Say which field had the wrong type.
357 // Wrong type is a client error.
358 error.code = api::v2::Http::Status::BAD_REQUEST;
359 return false; // Stop at the first invalid field.
360 }
361 std::optional<std::string> value =
362 item.value().as<std::string>(); // Read the string value now that
363 // the type is known.
364 // Only non-empty strings can be pattern matched.
365 if (value.has_value() && !value->empty()) {
366 // The whole value must match the
367 // compiled pattern.
368 if (!std::regex_match(value.value(), pattern_item->second)) {
369 error =
370 {.code = api::v2::Http::Status::BAD_REQUEST,
371 .message =
372 fmt::format("Key ({}) value is invalid.",
373 item.key())}; // Fill code and message in one go.
374 return false; // Stop at the first field that fails its
375 // pattern.
376 }
377 // The value passed, so store it
378 // for the database call.
379 // The value is missing or an empty string.
380 db_data.insert({item.key(), value.value()});
381 } else {
382 error.message = "Value is empty for key " +
383 item.key(); // Tell the caller which field was empty.
384 error.code = api::v2::Http::Status::BAD_REQUEST; // Empty required value
385 // is a client error.
386 return false; // Stop at the first empty field.
387 }
388 }
389 return true; // Every field passed its check (or was safely ignored).
390}

434/**
435 * See Validator::getDatabaseSchemaForTable.
436 * Reads the process-wide schema snapshot via
437 * DatabaseSchema::getDatabaseSchema(). An absent table or an entry with no
438 * columns is treated as "not found" and sets @p found to false; the returned
439 * set is a copy so callers cannot mutate the cached schema.
440 */
441std::unordered_set<api::v2::ColumnInfo> Validator::getDatabaseSchemaForTable(
442 const std::string& tablename, api::v2::Http::Error& error, bool& found) {
443 SCHEMA_t schema =
444 DatabaseSchema::getDatabaseSchema(); // Get a copy of the whole cached
445 // schema (table -> columns).
446
447 // Ensure the table schema exists
448 auto table_schema_it =
449 schema.find(tablename); // Look for this table's entry in the schema map.
450 // Missing entry or an entry with no columns is unusable.
451 if (table_schema_it == schema.end() || table_schema_it->second.empty()) {
452 error.message = "Table not found"; // Report that we cannot validate
453 // against this table.
454 error.code = api::v2::Http::Status::BAD_REQUEST; // Treat the unknown table
455 // as a bad request here.
456 found = false; // Tell the caller the lookup failed.
457 return {}; // Return an empty set so the caller has nothing to iterate.
458 }
459 found = true; // The schema was found and has columns.
460 return table_schema_it
461 ->second; // Hand back a copy of this table's column set.
462}
463/**
464 * See Validator::validateType.
465 * "character varying" and all date/time spellings expect JSON strings; "jsonb"
466 * accepts either an object or an array. Date strings are not parsed here, only
467 * type-checked. Unknown schema type strings return false, which surfaces as a
468 * type mismatch to the caller.
469 */
470bool Validator::validateType(const jsoncons::json& value,
471 const std::string& expectedType) {
472 // The schema expects a whole number.
473 if (expectedType == "integer") {
474 return value
475 .is_int64(); // True only when the JSON value is a 64-bit integer.
476 }
477 // All three mean a decimal number in PostgreSQL.
478 if (expectedType == "float" || expectedType == "double" ||
479 expectedType == "real") {
480 return value.is_double(); // True only when the JSON value is a
481 // floating-point number.
482 }
483 // PostgreSQL's name for variable-length text.
484 if (expectedType == "character varying") {
485 return value.is_string(); // True only when the JSON value is a string.
486 }
487 // A true/false column.
488 if (expectedType == "boolean") {
489 return value.is_bool(); // True only when the JSON value is a boolean.
490 }
491 // A column that stores JSON itself.
492 if (expectedType == "jsonb") {
493 return (value.is_object() ||
494 value.is_array()); // Accept either a JSON object or a JSON array.
495 }
496 // All date/time spellings.
497 if (expectedType == "timestamp with time zone" ||
498 expectedType == "time without time zone" || expectedType == "date") {
499 return value.is_string(); // Assuming dates are strings
500 }
501 return false; // Unknown type
502}

504/**
505 * See Validator::checkColumns.
506 * Iterates schema columns (the authoritative source), not the body. A column
507 * that is absent is allowed when it is nullable or covered by
508 * IGNORE_IF_NOT_NULLABLE_IN_SCHEMA; a present column must match the schema's
509 * type according to validateType. The first violation aborts with HTTP 400.
510 */
511bool Validator::checkColumns(
512 const jsoncons::json& data,
513 const std::unordered_set<api::v2::ColumnInfo>& table_schema,
514 api::v2::Http::Error& error,
515 const Rule& rule) {
516 // Walk the schema's columns, which
517 // are the source of truth.
518 for (const auto& column : table_schema) {
519 bool column_exists = data.contains(
520 column.Name); // Whether the client supplied this column at all.
521
522 // Check for missing non-nullable columns
523 // Missing, required, and not excused by the
524 // rule.
525 if (!column_exists && !column.isNullable &&
526 !rule.check(Rule::Action::IGNORE_IF_NOT_NULLABLE_IN_SCHEMA,
527 column.Name)) {
528 error.message = "Non-nullable column missing in data: " +
529 column.Name; // Name the missing required column.
530 error.code =
531 api::v2::Http::Status::BAD_REQUEST; // A missing required field
532 // is a client error.
533 return false; // Stop at the first missing required column.
534 }
535
536 // Validate data type if column exists
537 // Only check the type when the client actually sent
538 // this column.
539 if (column_exists) {
540 const auto& value = data.at(column.Name); // Fetch the value for this
541 // column from the JSON body.
542 // Compare the value's JSON kind to
543 // the schema's declared type.
544 if (!validateType(value, column.DataType)) {
545 error.message =
546 fmt::format("Data type mismatch for column: {} ,expected: {} ",
547 column.Name,
548 column.DataType); // Build a helpful mismatch message.
549 error.code = api::v2::Http::Status::BAD_REQUEST; // A wrong type is a
550 // client error.
551 return false; // Stop at the first type mismatch.
552 }
553 }
554 }
555 return true; // Every column constraint was satisfied.
556}
557/**
558 * See Validator::ensureAllKeysExist.
559 * Builds a name set from the schema for O(1) membership tests, then scans the
560 * body. ASSERT_NOT_PRESENT and ASSERT_IMMUTABLE are checked first so
561 * prohibited/immutable keys are reported specifically; unknown keys are allowed
562 * only under IGNORE_IF_MISSING_FROM_SCHEMA.
563 */
564bool Validator::ensureAllKeysExist(
565 const jsoncons::json& data,
566 const std::unordered_set<api::v2::ColumnInfo>& table_schema,
567 api::v2::Http::Error& error,
568 const Rule& rule) {
569 // Create a set of column names for fast key lookup
570 std::unordered_set<std::string>
571 schema_columns; // Will hold just the names, for quick membership
572 // checks.
573 for (const auto& column :
574 // Walk the schema's column descriptions once.
575 table_schema) {
576 schema_columns.insert(
577 column.Name); // Add each column name to the fast-lookup set.
578 }
579
580 // Ensure all keys in `data` exist in the schema
581 for (const auto& entry :
582 // Visit every key the client actually supplied.
583 data.object_range()) {
584 const auto& key = entry.key(); // The current field name being checked.
585
586 // Check for missing not allowed keys
587 // The rule says this key must never appear.
588 if (rule.check(Rule::Action::ASSERT_NOT_PRESENT, key)) {
589 error.message = "A prohibited key was found: [" + key +
590 "]"; // Name the forbidden key.
591 // Including it is a client error.
592 error.code = api::v2::Http::Status::BAD_REQUEST;
593 return false; // Stop at the first prohibited key.
594 }
595 // check if the key is in the exclude list
596 // The rule says this key cannot be changed.
597 if (rule.check(Rule::Action::ASSERT_IMMUTABLE, key)) {
598 error.message = "Key: [" + key +
599 "] is not allowed to be changed"; // Explain that the key
600 // is immutable.
601 error.code = api::v2::Http::Status::BAD_REQUEST; // Trying to change it
602 // is a client error.
603 return false; // Stop at the first immutable key.
604 }
605
606 bool ignore_if_missing_from_database_schema =
607 rule.check(Rule::Action::IGNORE_IF_MISSING_FROM_SCHEMA,
608 key); // Whether the rule excuses keys absent from the
609 // schema.
610 // Key not in schema
611 if (schema_columns.find(key) == schema_columns.end() &&
612 !ignore_if_missing_from_database_schema) {
613 error.message =
614 "Key: [" + key +
615 "] is not found in database schema"; // Report the unknown key.
616 error.code = api::v2::Http::Status::BAD_REQUEST; // An unknown key is a
617 // client error.
618 return false; // Stop at the first unknown key.
619 }
620 }
621 return true; // All supplied keys were permitted and known to the schema.
622}
