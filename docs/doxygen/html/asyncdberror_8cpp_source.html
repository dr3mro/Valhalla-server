1/**
2 * @file asyncdberror.cpp
3 * @brief Implementation for asyncdberror.
4 * @details This file implements the three error-classification helpers. isBusy() first recognises our own
5 * DatabaseBusy exception and then looks for a few telling words in the error text (such as "timeout" or
6 * "pool"). toHttpStatus() turns that answer into an HTTP code, and throwIfBusy() re-raises the project's
7 * DatabaseBusy type so the shared HTTP error handler can produce a 503 response.
8 *
9 * WHY THIS FILE MATTERS IN THE SERVER:
10 * Every coroutine-based database method in DatabaseController catches a plain std::exception when a query fails.
11 * This file is what lets those methods tell a temporary overload apart from a real bug. Because both the old
12 * synchronous pool and the new async path end up producing the same DatabaseBusy exception, the HTTP layer needs
13 * only one rule for answering 503 with Retry-After. Without this file, async database failures would all look
14 * alike and temporary outages would be reported as permanent server errors.
15 *
16 * CONCEPTS TO LEARN (for beginners):
17 * - Exceptions and catching by base class: https://en.cppreference.com/w/cpp/error/exception
18 * - dynamic_cast (runtime type check): https://en.cppreference.com/w/cpp/language/dynamic_cast
19 * - std::string::find and npos: https://en.cppreference.com/w/cpp/string/basic_string/find
20 * - HTTP 503 status: https://developer.mozilla.org/en-US/docs/Web/HTTP/Status/503
21 * - Drogon DbClient: https://drogon.docsforge.com/master/database/
22 *
23 * @author Amr Osman <dr3mro@gmail.com>
24 * @date 2026-11-16
25 */

35 bool isBusy(const std::exception& error)
36 {
37 // A DatabaseBusy thrown by our own code is authoritative; Drogon failures are recognised by message text.
38 // "Authoritative" means: if our own code raised it, we trust the type and do not need to guess from words.
39 if (dynamic_cast<const DatabaseBusy*>(&error) != nullptr) // dynamic_cast returns a real pointer only when the object actually is a DatabaseBusy.
40 {
41 return true; // Our own busy exception is definitive: no further checks are needed.
42 }
43
44 const std::string message = error.what(); // Copy the human-readable message into a string so we can search it safely; what() comes from std::exception.
45 // The substrings cover pool-acquisition timeouts and the common PostgreSQL "too many clients" error.
46 // The search is case-sensitive, so the words must appear exactly as written here.
47 return message.find("timeout") != std::string::npos || message.find("pool") != std::string::npos || // npos means "not found"; finding any one word is enough to call the error busy.
48 message.find("too many clients") != std::string::npos || message.find("connection") != std::string::npos; // The last two catch PostgreSQL errors and generic connection failures; "connection" is deliberately broad.
49 }

51 drogon::HttpStatusCode toHttpStatus(const std::exception& error)
52 {
53 // Busy is a retryable condition, so it is the only case answered with 503.
54 if (isBusy(error)) // Ask the shared classifier instead of repeating the string checks here.
55 {
56 return drogon::k503ServiceUnavailable; // 503 tells the caller "service temporarily unavailable, try again later".
57 }
58 return drogon::k500InternalServerError; // Anything else is treated as an unexpected internal failure (500).
59 }

61 void throwIfBusy(const std::exception& error)
62 {
63 // Re-classify at the boundary so the HTTP layer can keep its single DatabaseBusy handling path.
64 if (isBusy(error)) // Only busy errors are re-thrown; other failures are left for the caller to handle.
65 {
66 throw DatabaseBusy(); // Throwing unwinds the coroutine so the existing 503 + Retry-After handler takes over.
67 }
68 }
