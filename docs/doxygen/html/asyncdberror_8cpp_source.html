1/**
2 * @file asyncdberror.cpp
3 * @brief Implementation for asyncdberror.
4 * @details This file implements the three error-classification helpers.
5 * isBusy() first recognises our own DatabaseBusy exception and then looks for a
6 * few telling words in the error text (such as "timeout" or "pool").
7 * toHttpStatus() turns that answer into an HTTP code, and throwIfBusy()
8 * re-raises the project's DatabaseBusy type so the shared HTTP error handler
9 * can produce a 503 response.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * Every coroutine-based database method in DatabaseController catches a plain
13 * std::exception when a query fails. This file is what lets those methods tell
14 * a temporary overload apart from a real bug. Because both the old synchronous
15 * pool and the new async path end up producing the same DatabaseBusy exception,
16 * the HTTP layer needs only one rule for answering 503 with Retry-After.
17 * Without this file, async database failures would all look alike and temporary
18 * outages would be reported as permanent server errors.
19 *
20 * CONCEPTS TO LEARN (for beginners):
21 * - Exceptions and catching by base class:
22 * https://en.cppreference.com/w/cpp/error/exception
23 * - dynamic_cast (runtime type check):
24 * https://en.cppreference.com/w/cpp/language/dynamic_cast
25 * - std::string::find and npos:
26 * https://en.cppreference.com/w/cpp/string/basic_string/find
27 * - HTTP 503 status:
28 * https://developer.mozilla.org/en-US/docs/Web/HTTP/Status/503
29 * - Drogon DbClient: https://drogon.docsforge.com/master/database/
30 *
31 * @author Amr Osman <dr3mro@gmail.com>
32 * @date 2026-11-16
33 */

45 bool isBusy(const std::exception& error) {
46 // A DatabaseBusy thrown by our own code is authoritative; Drogon
47 // failures are recognised by message text. "Authoritative" means: if
48 // our own code raised it, we trust the type and do not need to guess
49 // from words.
50 // dynamic_cast returns a real pointer only when the
51 // object actually is a DatabaseBusy.
52 if (dynamic_cast<const DatabaseBusy*>(&error) != nullptr) {
53 return true; // Our own busy exception is definitive: no further
54 // checks are needed.
55 }
56
57 const std::string message =
58 error.what(); // Copy the human-readable message into a string so
59 // we can search it safely; what() comes from
60 // std::exception.
61 // The substrings cover pool-acquisition timeouts and the common
62 // PostgreSQL "too many clients" error. The search is case-sensitive, so
63 // the words must appear exactly as written here.
64 return message.find("timeout") != std::string::npos ||
65 message.find("pool") !=
66 std::string::npos || // npos means "not found"; finding any
67 // one word is enough to call the error
68 // busy.
69 message.find("too many clients") != std::string::npos ||
70 message.find("connection") !=
71 std::string::npos; // The last two catch PostgreSQL errors
72 // and generic connection failures;
73 // "connection" is deliberately broad.
74 }

76 drogon::HttpStatusCode toHttpStatus(const std::exception& error) {
77 // Busy is a retryable condition, so it is the only case answered with
78 // 503.
79 // Ask the shared classifier instead of repeating
80 // the string checks here.
81 if (isBusy(error)) {
82 return drogon::k503ServiceUnavailable; // 503 tells the caller
83 // "service temporarily
84 // unavailable, try again
85 // later".
86 }
87 return drogon::k500InternalServerError; // Anything else is treated as
88 // an unexpected internal
89 // failure (500).
90 }

92 void throwIfBusy(const std::exception& error) {
93 // Re-classify at the boundary so the HTTP layer can keep its single
94 // DatabaseBusy handling path.
95 // Only busy errors are re-thrown; other failures
96 // are left for the caller to handle.
97 if (isBusy(error)) {
98 throw DatabaseBusy(); // Throwing unwinds the coroutine so the
99 // existing 503 + Retry-After handler takes
100 // over.
101 }
102 }
