1/**
2 * @file cors_tokens.hpp
3 * @brief Drogon-free helpers for parsing and merging CORS header token lists.
4 *
5 * @details This header contains small functions for working with comma-separated header values used by CORS. It can
6 * tell whether a character is legal in an HTTP token, trim and split a header value into tokens, compare
7 * strings without regard to ASCII case, sanitize a client-supplied header list, and merge a token into an
8 * existing Vary header without duplicating it. The helpers deliberately avoid depending on Drogon.
9 *
10 * WHY THIS FILE MATTERS IN THE SERVER:
11 * These functions are the security and correctness layer behind the CORS code in cors.hpp. When a browser sends an
12 * Access-Control-Request-Headers value, this file filters out dangerous characters (such as CRLF or quotes) before
13 * they are echoed into a response, preventing header-injection attacks. It also keeps the Vary header tidy so caches
14 * store separate responses for different origins. Because the helpers have no framework dependency, they can be
15 * tested on their own, which is why a dedicated test file exists for them.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - C++20 ranges and views (lazy, composable sequence operations): https://en.cppreference.com/w/cpp/header/ranges
19 * - std::string_view (a non-owning view over text): https://en.cppreference.com/w/cpp/string/basic_string_view
20 * - constexpr (values/functions usable at compile time): https://en.cppreference.com/w/cpp/language/constexpr
21 *
22 * @author Amr Osman <dr3mro@gmail.com>
23 * @date 2026-11-16
24 */

41 /**
42 * Punctuation permitted in an HTTP token, per the token grammar used by the header parsers below.
43 */
44 constexpr std::string_view tokenPunctuation = "!#$%&'*+-.^_`|~"; // Compile-time punctuation allowed in a token.
45
46 /**
47 * @brief Reports whether a character is valid inside an HTTP token.
48 *
49 * @details Alphabetic, digit and the RFC token punctuation characters are accepted; whitespace, comma, quote,
50 * colon and control characters are not. This rejects values that could inject extra headers.
51 * @param[in] c Character to classify.
52 * @return true when c may appear in a token.
53 */
54 constexpr bool isTokenChar(char c) // Decide whether one character is legal inside an HTTP header token.
55 {
56 return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || tokenPunctuation.find(c) != std::string_view::npos; // Valid char.
57 }
58
59 /**
60 * @brief Removes optional whitespace (spaces and tabs) from both ends of a header token.
61 *
62 * @param[in] value View to trim; not modified.
63 * @return A subview of value with leading/trailing whitespace removed, or an empty view if value is all whitespace.
64 */
65 inline std::string_view trim(std::string_view value) // Return a view with leading and trailing spaces/tabs removed.
66 {
67 const std::size_t first = value.find_first_not_of(" \t"); // Find the first character that is not a space or tab.
68 if (first == std::string_view::npos) // If there is none, the entire view is whitespace.
69 {
70 return {}; // Hand back an empty view.
71 }
72 const std::size_t last = value.find_last_not_of(" \t"); // Find the last non-whitespace character.
73 return value.substr(first, last - first + 1); // Return the subview from first to last, inclusive.
74 }

76 /**
77 * @brief Lazily splits a comma-separated header value into trimmed, non-empty tokens.
78 *
79 * @details The returned range is a view: it allocates nothing and borrows from value, so the caller must keep the
80 * underlying string alive while iterating. Empty list elements (including ones made only of whitespace)
81 * are filtered out.
82 * @param[in] value Header value to split.
83 * @return A lazy range of std::string_view tokens.
84 */
85 inline auto tokens(std::string_view value) // Build a lazy range that splits, trims and filters a header value.
86 {
87 return value | std::views::split(',') | // Split at commas into sub-ranges without copying the string.
88 std::views::transform( // Convert each sub-range into a trimmed string_view.
89 [](auto range) // This lambda handles one split piece.
90 {
91 if (range.begin() == range.end()) // An empty piece (e.g. from two adjacent commas) needs no work.
92 {
93 return std::string_view{}; // Return an empty view so the filter below removes it.
94 }
95 // std::ranges::distance returns a signed ptrdiff_t; the range is non-empty here,
96 // so converting it to string_view's unsigned length is safe.
97 return trim(std::string_view(&*range.begin(), static_cast<std::size_t>(std::ranges::distance(range)))); // Trim this piece's whitespace.
98 }) |
99 std::views::filter([](std::string_view token) { return !token.empty(); }); // Drop empty tokens left by extra commas or spaces.
100 }

102 /**
103 * @brief ASCII case-insensitive string comparison.
104 *
105 * @details Only A-Z are folded to a-z, which is sufficient for HTTP header field names; non-ASCII bytes are
106 * compared verbatim.
107 * @param[in] a First string.
108 * @param[in] b Second string.
109 * @return true when the strings have equal length and match after ASCII lowercasing.
110 */
111 inline bool iequals(std::string_view a, std::string_view b) // Compare two strings ignoring the case of ASCII letters.
112 {
113 auto lower = [](char c) { return (c >= 'A' && c <= 'Z') ? static_cast<char>(c + ('a' - 'A')) : c; }; // Fold A-Z to a-z; other bytes unchanged.
114 return a.size() == b.size() && std::ranges::equal(a, b, [lower](char x, char y) { return lower(x) == lower(y); }); // Same length and content.
115 }

117 /**
118 * @brief Filters a client-supplied Access-Control-Request-Headers value down to syntactically valid tokens.
119 *
120 * @details Rejects the whole value when it is empty or longer than 1024 characters, then keeps only comma-separated
121 * tokens made entirely of token characters and rejoins them with ", ". This prevents a preflight request
122 * from echoing CRLF, quotes or arbitrary separators into Access-Control-Allow-Headers.
123 * @param[in] requested Raw header value from the request.
124 * @return A comma-and-space separated list of safe tokens, or an empty string when nothing is acceptable. An empty
125 * result tells Cors::apply() to fall back to the default header list.
126 */
127 inline std::string sanitizeRequestedHeaders(std::string_view requested) // Keep only safe, valid tokens from a client-supplied header value.
128 {
129 if (requested.empty() || requested.size() > 1024) // Reject empty or very long values outright to avoid abuse.
130 {
131 return {}; // Return empty so the caller falls back to its default header list.
132 }
133
134 std::string result; // Accumulates the safe tokens, separated by ", ".
135 for (std::string_view token : tokens(requested)) // Iterate the lazily split and trimmed tokens.
136 {
137 if (!std::ranges::all_of(token, isTokenChar)) // Drop any token containing invalid characters such as CRLF or quotes.
138 {
139 continue; // Skip this token and move on to the next one.
140 }
141 if (!result.empty()) // Add a separator only when a token has already been written.
142 {
143 result += ", "; // Separate tokens with a comma and a space.
144 }
145 result.append(token); // Append the validated token to the result.
146 }
147 return result; // Return the sanitized list, or empty if nothing was acceptable.
148 }

150 /**
151 * @brief Merges a token into an existing Vary header value, avoiding duplicates.
152 *
153 * @details Tokens are compared case-insensitively. A "*" wildcard in the existing value means every response is
154 * already variable, so the value is returned unchanged; likewise when the token is already present.
155 * Otherwise the token is appended. The original string is returned unchanged when no edit is needed, which
156 * lets the caller skip the header write.
157 * @param[in] existing Current Vary header value.
158 * @param[in] value Token to add.
159 * @return The merged Vary value.
160 */
161 inline std::string mergeVary(std::string_view existing, std::string_view value) // Append a token to Vary unless it is redundant.
162 {
163 std::string combined; // Rebuilds the header list with tokens separated by ", ".
164 bool hasWildcard = false; // True if the existing value already contains the "*" wildcard.
165 bool present = false; // True if the requested token is already in the list.
166 for (std::string_view token : tokens(existing)) // Walk each token of the existing header value.
167 {
168 if (token == "*") // A wildcard means every response is already variable.
169 {
170 hasWildcard = true; // Remember that no change is needed.
171 }
172 if (iequals(token, value)) // Compare case-insensitively, as header names are case-insensitive.
173 {
174 present = true; // Remember that the token is already present.
175 }
176 if (!combined.empty()) // Add a separator before all tokens except the first.
177 {
178 combined += ", "; // Keep the list readable and syntactically correct.
179 }
180 combined.append(token); // Copy this token into the rebuilt list.
181 }
182
183 if (hasWildcard || present) // No change is useful if it is already wildcard or already present.
184 {
185 return std::string(existing); // Return a copy of the original value, unchanged.
186 }
187 if (!combined.empty()) // If existing tokens were copied, separate the new one from them.
188 {
189 combined += ", "; // Insert the separator before appending.
190 }
191 combined.append(value); // Add the new token at the end of the list.
192 return combined; // Return the merged value so the caller can write it.
193 }
