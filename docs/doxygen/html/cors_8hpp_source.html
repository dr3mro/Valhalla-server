1/**
2 * @file cors.hpp
3 * @brief CORS header helpers applied to every response by the pre-sending advice in server.cpp.
4 *
5 * @details This header provides small functions that write the Cross-Origin Resource Sharing (CORS) headers onto an
6 * outgoing HTTP response. CORS is the browser rule that decides whether a web page on one domain may read a
7 * response from another domain. The code here reads the configured list of allowed origins, reflects an exact
8 * match back with credentials, or falls back to a wildcard or first configured origin. It also handles
9 * preflight (OPTIONS) requests by echoing a sanitized list of requested headers.
10 *
11 * WHY THIS FILE MATTERS IN THE SERVER:
12 * Without these headers, a browser front end served from a different address than the API would be blocked by its
13 * same-origin policy, even though the API itself works. server.cpp calls Cors::apply() for every single response, so
14 * this file is the one central place that makes the REST API reachable from a web app. It also contains the small token
15 * helpers' entry points used by the preflight logic, so CORS behaviour stays consistent and easy to audit.
16 *
17 * CONCEPTS TO LEARN (for beginners):
18 * - CORS (browser cross-origin rules): https://developer.mozilla.org/en-US/docs/Web/HTTP/CORS
19 * - Inline functions and function-local statics: https://en.cppreference.com/w/cpp/language/inline
20 * - std::string_view (a non-owning view over text): https://en.cppreference.com/w/cpp/string/basic_string_view
21 *
22 * @author Amr Osman <dr3mro@gmail.com>
23 * @date 2026-11-16
24 */

96 /**
97 * @brief Adds a token to the response's Vary header only if it is not already present.
98 *
99 * @details Deduplication is delegated to CorsTokens::mergeVary; comparing before writing avoids touching the
100 * header map on responses that already carry the value.
101 * @param[in,out] resp Response whose Vary header may be updated.
102 * @param[in] value Header name to add, for example "Origin".
103 */
104 inline void addToVary(const drogon::HttpResponsePtr& resp, std::string_view value) // Add one token to Vary only when it is missing.
105 {
106 const std::string& existing = resp->getHeader("vary"); // Read the current Vary header (empty string when absent).
107 const std::string merged = detail::mergeVary(existing, value); // Reuse the helper that appends without duplicates.
108 if (merged != existing) // Only touch the response if the merge actually changed the list.
109 {
110 resp->addHeader("Vary", merged); // Store the updated list back on the response.
111 }
112 }

114 /**
115 * @brief Splits the comma-separated cors_allowed_origins setting into individual origins.
116 *
117 * @details Spaces are skipped and empty tokens discarded, so a trailing comma or whitespace after a comma does
118 * not create a bogus origin. Order is preserved because apply() falls back to the first entry.
119 * @param[in] value Raw configuration value.
120 * @return Origins in configuration order.
121 */
122 inline std::vector<std::string> splitOrigins(const std::string& value) // Split the configured comma-separated string into origins.
123 {
124 std::vector<std::string> out; // Accumulates the origins found, keeping their original order.
125 std::string cur; // Holds the characters of the origin currently being built.
126 for (char c : value) // Examine every character in the configured value.
127 {
128 if (c == ',') // A comma ends the current origin.
129 {
130 if (!cur.empty()) out.push_back(cur); // Save a non-empty origin; ignore empty ones from stray commas.
131 cur.clear(); // Reset the buffer for the next origin.
132 }
133 else if (c != ' ') // Skip spaces but keep every other character.
134 {
135 cur.push_back(c); // Add this character to the origin being built.
136 }
137 }
138 if (!cur.empty()) out.push_back(cur); // Do not forget the final origin; no trailing comma ends it.
139 return out; // Return the list in configuration order for the caller to use.
140 }

142 /**
143 * @brief Decorates a response with CORS headers; invoked once per response from the pre-sending advice.
144 *
145 * @details If the response already carries Access-Control-Allow-Origin: "*", it is left untouched. Otherwise the
146 * current cors_allowed_origins setting is read from the Configurator. An exact match to the request's
147 * Origin is reflected back with Allow-Credentials and Vary: Origin; if origins are configured but none
148 * match, the first configured origin is used without credentials; if the setting is "*", the wildcard is
149 * used. For OPTIONS requests a sanitized echo of Access-Control-Request-Headers replaces the default
150 * header list, and Max-Age plus Vary are added. Every write uses HttpResponse::addHeader, which replaces
151 * an existing value rather than duplicating it.
152 * @param[in] req Request being answered; supplies Origin and, for preflights, the requested headers.
153 * @param[in,out] resp Response to decorate.
154 * @note No shared mutable state is modified here; the configuration is read through Store's thread-local cache.
155 * @see CorsTokens::sanitizeRequestedHeaders(), CorsTokens::mergeVary()
156 */
157 inline void apply(const drogon::HttpRequestPtr& req, const drogon::HttpResponsePtr& resp) // Write the correct CORS headers onto this response.
158 {
159 static const std::string originKey = "access-control-allow-origin"; // Drogon keys headers in lowercase; built once at first use.
160 const auto& headers = resp->headers(); // Reference to the header map so we can inspect what is already set.
161 const auto originIt = headers.find(originKey); // Look for an existing allow-origin header.
162 if (originIt != headers.end() && originIt->second == "*") // If a wildcard was already set, there is nothing to add.
163 {
164 return; // Leave the response untouched; another layer already allowed every origin.
165 }
166
167 const std::string allowed = Store::cachedObject<Configurator>()->get<Configurator::FrontEndConfig>().cors_allowed_origins; // Live CORS config.
168 const std::string requestOrigin = req->getHeader("Origin"); // Get the Origin header the browser sent (empty if none).
169
170 bool useCredentials = false; // Starts false; only an exact origin match will turn it on.
171 std::string allowOrigin = "*"; // Default to wildcard; replaced if specific origins are configured.
172 if (allowed != "*") // A concrete list is configured, so we must choose one origin.
173 {
174 auto origins = splitOrigins(allowed); // Break the configured string into individual origins.
175 for (const auto& o : origins) // Compare each configured origin with the request's Origin.
176 {
177 if (!requestOrigin.empty() && o == requestOrigin) // An exact match is safe to reflect and allow credentials for.
178 {
179 allowOrigin = requestOrigin; // Reflect exactly what the browser sent back to it.
180 useCredentials = true; // Credentials require an exact origin, never a wildcard.
181 break; // Stop searching once a match is found.
182 }
183 }
184 if (allowOrigin == "*" && !origins.empty()) // No exact match, but a list exists: fall back to its first entry.
185 {
186 allowOrigin = origins.front(); // Use the first configured origin, without credentials.
187 }
188 }
189
190 if (originIt == headers.end() || originIt->second != allowOrigin) // Write the header only if missing or different.
191 {
192 resp->addHeader("Access-Control-Allow-Origin", allowOrigin); // Tell the browser which origin may read this response.
193 }
194 if (useCredentials) // Add the credentials flag only for an exact origin match.
195 {
196 resp->addHeader("Access-Control-Allow-Credentials", "true"); // Allow cookies/auth to be sent cross-origin.
197 addToVary(resp, "Origin"); // Vary on Origin so caches keep separate copies per origin.
198 }
199 resp->addHeader("Access-Control-Allow-Methods", allowedMethods()); // Advertise the HTTP methods permitted cross-origin.
200 resp->addHeader("Access-Control-Expose-Headers", exposedHeaders()); // List the response headers scripts may read.
201
202 const std::string& requested = req->getHeader("Access-Control-Request-Headers"); // Preflight field naming the headers the browser wants to send.
203
204 const std::string* allowedHeaders = &defaultAllowedHeaders(); // By default, advertise the standard header list.
205 std::string echoed; // Will hold the sanitized copy of the client's requested headers.
206 if (req->method() == drogon::Options) // Only preflight requests carry Access-Control-Request-Headers.
207 {
208 echoed = detail::sanitizeRequestedHeaders(requested); // Filter the request value down to safe tokens.
209 if (!echoed.empty()) // Use the echo only if some valid tokens survived filtering.
210 {
211 allowedHeaders = &echoed; // Point at the echoed list instead of the default one.
212 }
213 resp->addHeader("Access-Control-Max-Age", preflightMaxAge()); // Tell the browser how long it may cache this preflight.
214 addToVary(resp, "Access-Control-Request-Headers"); // Vary on requested headers so caches do not mix them up.
215 }
216 resp->addHeader("Access-Control-Allow-Headers", *allowedHeaders); // Write the final list: default or sanitized echo.
217 }
