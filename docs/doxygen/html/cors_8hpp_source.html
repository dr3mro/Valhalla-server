1/**
2 * @file cors.hpp
3 * @brief CORS header helpers applied to every response by the pre-sending
4 * advice in server.cpp.
5 *
6 * @details This header provides small functions that write the Cross-Origin
7 * Resource Sharing (CORS) headers onto an outgoing HTTP response. CORS is the
8 * browser rule that decides whether a web page on one domain may read a
9 * response from another domain. The code here reads the configured
10 * list of allowed origins, reflects an exact match back with credentials, or
11 * falls back to a wildcard or first configured origin. It also handles
12 * preflight (OPTIONS) requests by echoing a sanitized list of
13 * requested headers.
14 *
15 * WHY THIS FILE MATTERS IN THE SERVER:
16 * Without these headers, a browser front end served from a different address
17 * than the API would be blocked by its same-origin policy, even though the API
18 * itself works. server.cpp calls Cors::apply() for every single response, so
19 * this file is the one central place that makes the REST API reachable from a
20 * web app. It also contains the small token helpers' entry points used by the
21 * preflight logic, so CORS behaviour stays consistent and easy to audit.
22 *
23 * CONCEPTS TO LEARN (for beginners):
24 * - CORS (browser cross-origin rules):
25 * https://developer.mozilla.org/en-US/docs/Web/HTTP/CORS
26 * - Inline functions and function-local statics:
27 * https://en.cppreference.com/w/cpp/language/inline
28 * - std::string_view (a non-owning view over text):
29 * https://en.cppreference.com/w/cpp/string/basic_string_view
30 *
31 * @author Amr Osman <dr3mro@gmail.com>
32 * @date 2026-11-16
33 */

135 /**
136 * @brief Adds a token to the response's Vary header only if it is not
137 * already present.
138 *
139 * @details Deduplication is delegated to CorsTokens::mergeVary; comparing
140 * before writing avoids touching the header map on responses that already
141 * carry the value.
142 * @param[in,out] resp Response whose Vary header may be updated.
143 * @param[in] value Header name to add, for example "Origin".
144 */
145 // Add one token to Vary only when it is missing.
146 inline void addToVary(const drogon::HttpResponsePtr& resp,
147 std::string_view value) {
148 const std::string& existing =
149 resp->getHeader("vary"); // Read the current Vary header (empty
150 // string when absent).
151 const std::string merged =
152 detail::mergeVary(existing,
153 value); // Reuse the helper that appends without
154 // duplicates.
155 // Only touch the response if the merge
156 // actually changed the list.
157 if (merged != existing) {
158 resp->addHeader("Vary",
159 merged); // Store the updated list back on the response.
160 }
161 }

163 /**
164 * @brief Splits the comma-separated cors_allowed_origins setting into
165 * individual origins.
166 *
167 * @details Spaces are skipped and empty tokens discarded, so a trailing
168 * comma or whitespace after a comma does not create a bogus origin. Order
169 * is preserved because apply() falls back to the first entry.
170 * @param[in] value Raw configuration value.
171 * @return Origins in configuration order.
172 */
173 // Split the configured comma-separated string into origins.
174 inline std::vector<std::string> splitOrigins(const std::string& value) {
175 std::vector<std::string> out; // Accumulates the origins found, keeping
176 // their original order.
177 std::string
178 cur; // Holds the characters of the origin currently being built.
179 // Examine every character in the configured value.
180 for (char c : value) {
181 // A comma ends the current origin.
182 if (c == ',') {
183 if (!cur.empty())
184 out.push_back(cur); // Save a non-empty origin; ignore
185 // empty ones from stray commas.
186 // Reset the buffer for the next origin.
187 // Skip spaces but keep every other character.
188 cur.clear();
189 } else if (c != ' ') {
190 cur.push_back(c); // Add this character to the origin being built.
191 }
192 }
193 if (!cur.empty())
194 out.push_back(cur); // Do not forget the final origin; no trailing
195 // comma ends it.
196 return out; // Return the list in configuration order for the caller to
197 // use.
198 }

200 /**
201 * @brief Decorates a response with CORS headers; invoked once per response
202 * from the pre-sending advice.
203 *
204 * @details If the response already carries Access-Control-Allow-Origin:
205 * "*", it is left untouched. Otherwise the current cors_allowed_origins
206 * setting is read from the Configurator. An exact match to the request's
207 * Origin is reflected back with Allow-Credentials and Vary:
208 * Origin; if origins are configured but none match, the first configured
209 * origin is used without credentials; if the setting is "*", the wildcard
210 * is used. For OPTIONS requests a sanitized echo of
211 * Access-Control-Request-Headers replaces the default header list, and
212 * Max-Age plus Vary are added. Every write uses HttpResponse::addHeader,
213 * which replaces an existing value rather than duplicating it.
214 * @param[in] req Request being answered; supplies Origin and, for
215 * preflights, the requested headers.
216 * @param[in,out] resp Response to decorate.
217 * @note No shared mutable state is modified here; the configuration is read
218 * through Store's thread-local cache.
219 * @see CorsTokens::sanitizeRequestedHeaders(), CorsTokens::mergeVary()
220 */
221 // Write the correct CORS headers onto this response.
222 inline void apply(const drogon::HttpRequestPtr& req,
223 const drogon::HttpResponsePtr& resp) {
224 static const std::string originKey =
225 "access-control-allow-origin"; // Drogon keys headers in lowercase;
226 // built once at first use.
227 const auto& headers = resp->headers(); // Reference to the header map so we
228 // can inspect what is already set.
229 const auto originIt =
230 headers.find(originKey); // Look for an existing allow-origin header.
231 // If a wildcard was already set, there is nothing to add.
232 if (originIt != headers.end() && originIt->second == "*") {
233 return; // Leave the response untouched; another layer already
234 // allowed every origin.
235 }
236
237 const std::string allowed = Store::cachedObject<Configurator>()
238 ->get<Configurator::FrontEndConfig>()
239 .cors_allowed_origins; // Live CORS config.
240 const std::string requestOrigin =
241 req->getHeader("Origin"); // Get the Origin header the browser sent
242 // (empty if none).
243
244 bool useCredentials =
245 false; // Starts false; only an exact origin match will turn it on.
246 std::string allowOrigin = "*"; // Default to wildcard; replaced if
247 // specific origins are configured.
248 // A concrete list is configured, so we must choose
249 // one origin.
250 if (allowed != "*") {
251 auto origins = splitOrigins(allowed); // Break the configured string into
252 // individual origins.
253 // Compare each configured origin
254 // with the request's Origin.
255 for (const auto& o : origins) {
256 // An exact match is safe to reflect
257 // and allow credentials for.
258 if (!requestOrigin.empty() && o == requestOrigin) {
259 allowOrigin = requestOrigin; // Reflect exactly what the
260 // browser sent back to it.
261 useCredentials = true; // Credentials require an exact
262 // origin, never a wildcard.
263 break; // Stop searching once a match is found.
264 }
265 }
266 // No exact match, but a list exists: fall
267 // back to its first entry.
268 if (allowOrigin == "*" && !origins.empty()) {
269 allowOrigin = origins.front(); // Use the first configured
270 // origin, without credentials.
271 }
272 }
273
274 // Write the header only if missing or different.
275 if (originIt == headers.end() || originIt->second != allowOrigin) {
276 resp->addHeader("Access-Control-Allow-Origin",
277 allowOrigin); // Tell the browser which origin may read
278 // this response.
279 }
280 // Add the credentials flag only for an exact
281 // origin match.
282 if (useCredentials) {
283 resp->addHeader("Access-Control-Allow-Credentials",
284 "true"); // Allow cookies/auth to be sent cross-origin.
285 addToVary(resp,
286 "Origin"); // Vary on Origin so caches keep separate copies per
287 // origin.
288 }
289 resp->addHeader("Access-Control-Allow-Methods",
290 allowedMethods()); // Advertise the HTTP methods permitted
291 // cross-origin.
292 resp->addHeader("Access-Control-Expose-Headers",
293 exposedHeaders()); // List the response headers scripts may
294 // read.
295
296 const std::string& requested = req->getHeader(
297 "Access-Control-Request-Headers"); // Preflight field naming the
298 // headers the browser wants to
299 // send.
300
301 const std::string* allowedHeaders =
302 &defaultAllowedHeaders(); // By default, advertise the standard
303 // header list.
304 std::string echoed; // Will hold the sanitized copy of the client's
305 // requested headers.
306 // Only preflight requests carry
307 // Access-Control-Request-Headers.
308 if (req->method() == drogon::Options) {
309 echoed = detail::sanitizeRequestedHeaders(
310 requested); // Filter the request value down to safe tokens.
311 // Use the echo only if some valid tokens
312 // survived filtering.
313 if (!echoed.empty()) {
314 allowedHeaders = &echoed; // Point at the echoed list instead
315 // of the default one.
316 }
317 resp->addHeader("Access-Control-Max-Age",
318 preflightMaxAge()); // Tell the browser how long it may
319 // cache this preflight.
320 addToVary(resp,
321 "Access-Control-Request-Headers"); // Vary on requested headers
322 // so caches do not mix them
323 // up.
324 }
325 resp->addHeader("Access-Control-Allow-Headers",
326 *allowedHeaders); // Write the final list: default or
327 // sanitized echo.
328 }
