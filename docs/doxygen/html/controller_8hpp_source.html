1/**
2 * @file controller.hpp
3 * @brief Reusable persistence engine for the API's CRUDS
4 * (Create/Read/Update/Delete/Search) operations.
5 * @details This header defines the `Controller` class, a template-heavy helper
6 * that turns an entity object into SQL and turns the database's answer into an
7 * HTTP status plus a response body. It offers both ordinary blocking methods
8 * (Create, Read, ...) and coroutine versions (CreateAsync, ReadAsync, ...) for
9 * the async server loop. It is meant to be read from top to bottom: first the
10 * constructor, then the public CRUDS helpers, then the two shared
11 * `cruds`/`crudsAsync` implementations near the bottom.
12 *
13 * WHY THIS FILE MATTERS IN THE SERVER:
14 * It is the storage engine that sits underneath the request-facing handlers.
15 * When an HTTP request arrives, Drogon routes it to an EntityController /
16 * ClientController / ClinicController, and those classes call this Controller
17 * to do the actual work. This class in turn calls DatabaseController (see
18 * databasecontroller.hpp) to run parameterized SQL against PostgreSQL. Without
19 * this file the handlers would have no way to insert, read, update, delete or
20 * search rows, and the callback that finally answers the client would never be
21 * invoked.
22 *
23 * CONCEPTS TO LEARN (for beginners):
24 * - Templates: https://en.cppreference.com/w/cpp/language/templates
25 * - Pointer-to-member function (used to pick an entity's SQL builder at
26 * runtime): https://en.cppreference.com/w/cpp/language/pointer
27 * - std::optional (a value that may be missing):
28 * https://en.cppreference.com/w/cpp/utility/optional
29 * - std::shared_ptr (shared ownership of the database service):
30 * https://en.cppreference.com/w/cpp/memory/shared_ptr
31 * - Move semantics (std::move / std::forward):
32 * https://en.cppreference.com/w/cpp/language/move_constructor
33 * - Coroutines (co_await / drogon::Task):
34 * https://en.cppreference.com/w/cpp/language/coroutines
35 * - Lambda expressions: https://en.cppreference.com/w/cpp/language/lambda
36 * - virtual functions / polymorphism:
37 * https://en.cppreference.com/w/cpp/language/virtual
38 *
39 * @author Amr Osman <dr3mro@gmail.com>
40 * @date 2026-11-16
41 */

97/**
98 * @class Controller
99 * @brief Reusable persistence engine for the API's CRUDS
100 * (Create/Read/Update/Delete/Search) operations.
101 * @details Sits below the request-facing controllers (EntityController and its
102 * derived classes). It owns the single `DatabaseController` obtained from the
103 * `Store` singleton, turns an entity's SQL-builder methods into parameterized
104 * statements, executes them, and maps results or failures onto an HTTP
105 * status/body pair delivered through a `CALLBACK_`. Every operation has a
106 * synchronous variant (which calls the blocking `DatabaseController` methods)
107 * and a `drogon::Task` coroutine variant (which awaits the `*Coro` methods on
108 * the Drogon IO loop).
109 *
110 * Entities are passed by non-const reference because their
111 * `getSql*Statement(pqxx::params&)` methods fill the parameter bag used by the
112 * query; the controller does not take ownership of them. The callback is moved
113 * into the operation and is invoked exactly once on every path that returns
114 * normally, including validation and database failures. `CALLBACK_` is
115 * `std::function<void(int, const std::string&)>`, so the first argument is an
116 * HTTP status code and the second a response body (usually serialized JSON).
117 *
118 * This class is copyable/movable by design: copies share the same
119 * `DatabaseController` through a `std::shared_ptr`. No mutable per-instance
120 * state is touched after construction, so concurrent use of distinct instances
121 * is independent; the thread-safety of the shared `DatabaseController` itself
122 * is not specified here.
123 * @see EntityController, DatabaseController
124 */
125class Controller {
126 public:
127 /**
128 * @brief Resolves the shared `DatabaseController` service from the `Store`
129 * singleton.
130 * @details All operations depend on that service; resolving it here lets
131 * derived request-scoped controllers stay cheap to construct.
132 * `Store::getObject` may create the singleton on first use.
133 * @throws std::runtime_error if the database controller cannot be obtained
134 * (a critical message is logged first).
135 */
136 Controller() {
137 try {
138 databaseController =
139 Store::getObject<DatabaseController>(); // Ask the service locator
140 // for the single shared DB
141 // gateway.
142 } catch (const std::exception& e) {
143 CRITICALMESSAGE // Log a critical, file/line-tagged error before
144 // giving up.
145 throw std::runtime_error(
146 // Without a DB gateway this object is useless, so stop
147 // construction.
148 "Failed to initialize database controller");
149 }
150 }

361 /**
362 * @brief Runs @p entity's search statement and returns a paginated JSON
363 * envelope through @p callback.
364 * @details The response object always contains `more` (bool, initially
365 * false), `offset` (initially 0) and `results` (an array; empty when no
366 * query or no rows). The search query is built from the `Search_t` held by
367 * the entity. When the returned row count exceeds the requested `limit`,
368 * one extra probe row is fetched (the entity's search builder binds `limit
369 * + 1`): it is removed, `more` is set to true and `offset` advances by
370 * `limit`.
371 * @tparam T Entity type exposing `getSqlSearchStatement(pqxx::params&)` and
372 * `getData()` convertible to Search_t.
373 * @param[in,out] entity Borrowed entity carrying limit/offset and the
374 * search criteria.
375 * @param[in,out] callback Invoked once with 200 and the serialized envelope
376 * (or a mapped error status).
377 * @note Exceptions (including `DatabaseBusy`) are turned into an HTTP
378 * status by `CRITICALMESSAGERESPONSE`.
379 * @see executeSearchQuery (DatabaseController)
380 */
381 template <typename T>
382 void Search(T& entity,
383 CALLBACK_&& callback) {
384 jsoncons::json response_json;
385 response_json["more"] = false; // Assume no further page until the
386 // row-count probe proves otherwise.
387 response_json["offset"] = 0; // Start the envelope at offset zero; a
388 // real offset is set when 'more' is true.
389
390 std::optional<jsoncons::json>
391 query_results_json_array; // Holds the rows once the search
392 // returns; empty means no query ran.
393 std::optional<std::string> query; // Holds the built SQL; empty means the
394 // entity produced nothing to run.
395
396 try {
397 pqxx::params params; // The bag of values bound to the SQL
398 // placeholders ($1, $2, ...).
399 // The entity builds both the parameterized statement and its bound
400 // values.
401 query = entity.getSqlSearchStatement(
402 params); // Ask the entity to build the SQL and fill the
403 // parameter bag.
404
405 if (query.has_value()) {
406 query_results_json_array =
407 databaseController
408 ->executeSearchQuery(query.value(),
409 params); // Blocking search that maps rows
410 // to a JSON array.
411
412 if (query_results_json_array.has_value() &&
413 !query_results_json_array->empty()) {
414 size_t results_count =
415 query_results_json_array->size(); // Count rows to detect the
416 // extra probe row (limit + 1).
417 // More rows than requested means an over-fetch: drop the
418 // extra row and advertise the next page.
419 if (results_count > std::get<Search_t>(entity.getData()).limit) {
420 response_json["more"] = true; // An extra row exists, so tell the
421 // client another page is available.
422 response_json["offset"] =
423 std::get<Search_t>(entity.getData()).offset +
424 std::get<Search_t>(entity.getData())
425 .limit; // Next page starts after this page's
426 // limit.
427 query_results_json_array->erase(
428 query_results_json_array->array_range().end() -
429 1); // Remove the probe row so the client sees only
430 // 'limit' rows.
431 }
432 }
433 }
434 // Absent results still yield a well-formed envelope with an empty
435 // array.
436 if (query_results_json_array.has_value()) {
437 response_json["results"] = std::move(query_results_json_array.value());
438 } else {
439 response_json["results"] =
440 jsoncons::json::array(); // Guarantee an empty array
441 // instead of a null when nothing
442 // was found.
443 }
444
445 // Marshal the envelope and answer once.
446 std::move(callback)(api::v2::Http::Status::OK,
447 response_json.as<std::string>());
448 } catch (const std::exception& e) {
449 CRITICALMESSAGERESPONSE // Log a critical error and answer through
450 // the callback with a mapped status.
451 }
452 }

856 /**
857 * @brief Runs one entity statement: builds the parameterized SQL, executes
858 * it and reports the result.
859 * @details Invokes `(entity.*sqlstatement)(params)`; a missing statement
860 * maps to 400. A present result that is non-empty maps to 200 with its JSON
861 * serialization; empty or absent results map to 400 with a diagnostic body.
862 * The chosen `func` decides whether this is a write or a read. Exceptions
863 * are routed through `CRITICALMESSAGERESPONSE`, which answers 503 for
864 * `DatabaseBusy` and 500 otherwise.
865 * @tparam S Pointer-to-member type of the entity's SQL builder (deduced).
866 * @tparam T Entity type (deduced); must outlive this call.
867 * @param[in,out] entity Entity whose builder populates @p params.
868 * @param[in] sqlstatement Pointer to the entity builder method to use.
869 * @param[in] func Pointer to the `DatabaseController` executor method (e.g.
870 * dbexec/dbrexec).
871 * @param[in,out] callback Invoked exactly once on any returning path.
872 * @note Shared implementation for every synchronous CRUDS helper.
873 */
874 template <typename S,
875 typename T>
876 void cruds(T& entity,
877 S& sqlstatement,
878 std::optional<jsoncons::json> (DatabaseController::*func)(
879 const std::string&,
880 const pqxx::params&),
881 CALLBACK_&& callback) {
882 std::optional<jsoncons::json>
883 results_j; // Will hold the single result object returned by the DB
884 // helper.
885 pqxx::params params;
886 try {
887 std::optional<std::string> query =
888 (entity.*sqlstatement)(params); // Call the chosen builder on the
889 // entity; it also fills params.
890 // A nullopt statement means the entity could not describe the
891 // operation: reject before touching the DB.
892 if (!query.has_value()) {
893 std::move(callback)(api::v2::Http::Status::BAD_REQUEST,
894 fmt::format("Failed to get SQL statement for {}.",
895 entity.getTableName()));
896 return;
897 }
898
899 // Dispatch through the selected executor; params are bound
900 // placeholders, not string-interpolated values.
901 results_j =
902 (*databaseController.*
903 func)(query.value(),
904 params); // Run SQL through the chosen read/write executor.
905 if (results_j.has_value()) {
906 if (!results_j->empty()) {
907 std::move(callback)(api::v2::Http::Status::OK,
908 results_j.value().as<std::string>());
909 return;
910 }
911
912 std::move(callback)(api::v2::Http::Status::BAD_REQUEST,
913 "Query returned empty result, please recheck your "
914 "parameters.");
915 return;
916 }
917
918 std::move(callback)(api::v2::Http::Status::BAD_REQUEST,
919 "Failed to create sql query");
920 } catch (const std::exception& e) {
921 CRITICALMESSAGERESPONSE // Log a critical error and answer through
922 // the callback with a mapped status.
923 }
924 }
