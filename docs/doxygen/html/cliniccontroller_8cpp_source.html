38template <typename T>
39template <typename U>
40void ClinicController<T>::CreateImpl(CALLBACK_&& callback,
41 const Requester&& requester,
42 std::string_view data)
43 requires(!std::is_same<U, Patient>::value && !std::is_same<U, Visits>::value)
44{
45 jsoncons::json request_json; // Will hold the parsed create body.
46 try {
47 request_json = jsoncons::json::parse(
48 data); // Turn the raw body into JSON; malformed text throws.
49 } catch (const std::exception&) {
50 callback(api::v2::Http::Status::BAD_REQUEST,
51 "Invalid request body: malformed JSON.");
52 return;
53 }
54 bool success = false; // Validation flips this true only for a usable body.
55
56 // Unlike clients, these entities carry a client-supplied id, so "id" is
57 // exempt from the normal schema rules and checked explicitly below.
58 static const Validator::Rule
59 rule((Validator::Rule::Action::IGNORE_IF_MISSING_FROM_SCHEMA |
60 Validator::Rule::Action::IGNORE_IF_NOT_NULLABLE_IN_SCHEMA),
61 {"id"});
62
63 api::v2::Http::Error
64 error; // Receives a code/message if authorization fails.
65 try {
66 success =
67 Validator::validateDatabaseCreateSchema(T::getTableName(),
68 request_json,
69 error,
70 rule); // Schema check with
71 // 'id' exempted.
72 if (!success) {
73 callback(error.code, error.message);
74 return;
75 }
76
77 // A body that simply omits "id" (e.g. {"reports": [...], "clinic_id":
78 // ...}) is a client mistake, not a server failure - .at("id") below
79 // throws "Key not found" for exactly that case, and
80 // validateDatabaseCreateSchema() above was told to ignore a missing
81 // "id" (see `rule`), so it has to be checked explicitly here before the
82 // .at() call, or it falls through uncaught into the generic 500 handler
83 // below.
84 if (!request_json.contains("id")) {
85 callback(api::v2::Http::Status::BAD_REQUEST, "id not provided.");
86 return;
87 }
88 // A fuzzer-sized "id" (e.g. 33689528311627667466646192128, far past
89 // UINT64_MAX) makes jsoncons' .as<uint64_t>() throw "Integer overflow"
90 // rather than just failing to convert - that's still a
91 // malformed-client-input case, not a server fault, so it needs its own
92 // catch here rather than falling through to the generic 500 handler
93 // below.
94 std::optional<uint64_t> id; // Explicit client-supplied id, filled below.
95 try {
96 id = request_json.at("id").as<uint64_t>(); // Read the id as a 64-bit
97 // unsigned integer.
98 } catch (const std::exception&) {
99 callback(api::v2::Http::Status::BAD_REQUEST,
100 "id must be a non-negative integer that fits in 64 bits.");
101 return;
102 }
103 if (!id.has_value()) {
104 callback(api::v2::Http::Status::BAD_REQUEST, "id not provided.");
105 return;
106 }
107
108 // Authorize before constructing the entity and issuing the write.
109 if (!gateKeeper->canCreate<T>(requester, request_json, error)) {
110 callback(error.code, error.message);
111 return;
112 }
113
114 // Pass the validated id explicitly: this entity type does not rely on a
115 // database-generated key.
116 Types::Create_t clinic_create_data =
117 Types::Create_t(request_json,
118 id.value()); // Wrap body plus explicit id.
119
120 T entity(clinic_create_data); // Build the entity that will be persisted.
121
122 Controller::Create(entity,
123 std::move(callback)); // Delegate the INSERT to the
124 // persistence engine.
125 } catch (const std::exception& e) {
126 CRITICALMESSAGERESPONSE // Unexpected error: log it and answer with a
127 // mapped 5xx status.
128 }
129}
